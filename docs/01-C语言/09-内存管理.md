## 1. 内存区域划分

### 1. C程序内存布局

```
高地址
┌─────────────────────────────────────┐
│           命令行参数与环境变量         │
├─────────────────────────────────────┤
│              栈区 (Stack)            │ ← 向下增长
│         局部变量、函数参数、返回地址     │
│                  ↓                   │
│                                      │
│                  ↑                   │
│              堆区 (Heap)             │ ← 向上增长
│         动态分配的内存 (malloc等)       │
├─────────────────────────────────────┤
│           BSS段 (未初始化数据)         │
│         未初始化的全局变量、静态变量     │
├─────────────────────────────────────┤
│           数据段 (已初始化数据)         │
│         已初始化的全局变量、静态变量     │
├─────────────────────────────────────┤
│           代码段 (Text)              │
│              程序代码                 │
└─────────────────────────────────────┘
低地址
```

### 2. 各区域特点

```c
#include <stdio.h>
#include <stdlib.h>

int global_init = 100;      /* 数据段：已初始化全局变量 */
int global_uninit;          /* BSS段：未初始化全局变量 */

static int static_init = 100;  /* 数据段：已初始化静态变量 */
static int static_uninit;      /* BSS段：未初始化静态变量 */

void memory_demo(void) {
    int local = 100;          /* 栈：局部变量 */
    static int static_local = 100;  /* 数据段：静态局部变量 */
    
    int *heap = malloc(sizeof(int));  /* 堆：动态分配 */
    *heap = 100;
    
    printf("Local (stack): %p\n", (void*)&local);
    printf("Heap: %p\n", (void*)heap);
    printf("Global init: %p\n", (void*)&global_init);
    printf("Global uninit: %p\n", (void*)&global_uninit);
    printf("Function (text): %p\n", (void*)memory_demo);
    
    free(heap);
}

int main(void) {
    memory_demo();
    return 0;
}
```

| 区域  | 存储内容        | 生命周期   | 管理方式        |
| --- | ----------- | ------ | ----------- |
| 栈   | 局部变量、函数参数   | 函数调用期间 | 自动          |
| 堆   | 动态分配的内存     | 手动控制   | malloc/free |
| BSS | 未初始化全局/静态变量 | 程序运行期间 | 自动          |
| 数据段 | 已初始化全局/静态变量 | 程序运行期间 | 自动          |
| 代码段 | 程序指令        | 程序运行期间 | 只读          |

## 2. 动态内存分配

### 1. 为什么需要动态内存分配

```c
#include <stdio.h>

/* 问题1：数组大小固定 */
void fixed_array_problem(void) {
    int arr[100];  /* 编译时确定大小 */
    /* 如果实际需要1000个元素怎么办？ */
    /* 如果只需要10个元素，浪费内存 */
}

/* 问题2：无法返回局部数组 */
int* create_array_wrong(int size) {
    int arr[size];  /* 局部数组（VLA，C99特性，C11中为可选特性，不可移植） */
    return arr;     /* 错误！返回局部变量地址 */
}

/* 解决方案：使用动态内存分配 */
int* create_array_right(int size) {
    int *arr = malloc(size * sizeof(int));
    return arr;  /* 堆内存，函数返回后仍然有效 */
}
```

### 2. 动态内存分配的基本流程

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *ptr = NULL;
    
    /* 1. 分配内存 */
    ptr = (int*)malloc(sizeof(int));
    if (ptr == NULL) {
        printf("Memory allocation failed\n");
        return 1;
    }
    
    /* 2. 使用内存 */
    *ptr = 100;
    printf("Value: %d\n", *ptr);
    
    /* 3. 释放内存 */
    free(ptr);
    ptr = NULL;  /* 避免悬空指针 */
    
    return 0;
}
```

## 3. 内存分配函数详解

### 1. malloc函数

```c
void *malloc(size_t size);
/* 功能：分配指定字节数的内存
 * 参数：size - 要分配的字节数
 * 返回：成功返回内存指针，失败返回NULL
 * 注意：内存内容未初始化（包含垃圾值）
 */
```

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* 分配单个整数 */
    int *p = (int*)malloc(sizeof(int));
    if (p != NULL) {
        *p = 100;
        printf("*p = %d\n", *p);
        free(p);
    }
    
    /* 分配数组 */
    int n = 5;
    int *arr = (int*)malloc(n * sizeof(int));
    if (arr != NULL) {
        for (int i = 0; i < n; i++) {
            arr[i] = i * 10;  /* 需要手动初始化 */
        }
        for (int i = 0; i < n; i++) {
            printf("arr[%d] = %d\n", i, arr[i]);
        }
        free(arr);
    }
    
    return 0;
}
```

### 2. calloc函数

```c
void *calloc(size_t num, size_t size);
/* 功能：分配并初始化为0的内存
 * 参数：num - 元素个数，size - 每个元素大小
 * 返回：成功返回内存指针，失败返回NULL
 * 注意：内存内容自动初始化为0
 */
```

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    
    /* calloc自动初始化为0 */
    int *arr = (int*)calloc(n, sizeof(int));
    if (arr != NULL) {
        printf("After calloc:\n");
        for (int i = 0; i < n; i++) {
            printf("arr[%d] = %d\n", i, arr[i]);  /* 全部为0 */
        }
        free(arr);
    }
    
    return 0;
}
```

### 3. malloc与calloc的区别

| 特性   | malloc    | calloc      |
| ---- | --------- | ----------- |
| 参数   | 总字节数      | 元素个数 × 元素大小 |
| 初始化  | 不初始化（垃圾值） | 初始化为0       |
| 性能   | 稍快        | 稍慢（需要清零）    |
| 适用场景 | 已知会立即赋值   | 需要0初始化      |

### 4. realloc函数

```c
void *realloc(void *ptr, size_t size);
/* 功能：调整已分配内存的大小
 * 参数：ptr - 原内存指针，size - 新大小
 * 返回：成功返回新内存指针（可能改变），失败返回NULL
 * 注意：原内容保留，新增部分未初始化
 */
```

### 5. realloc的关键特性（重要！）

**超级重要的结论：**

- **realloc 成功时：原指针会被系统自动释放，绝对不能再用！**
- **realloc 失败时：原指针完全保留，不会释放，必须自己管！**

#### 1. 成功情况 ✅

```c
void *new_ptr = realloc(old_ptr, new_size);
```

- **realloc 成功**
  - 新内存地址 = `new_ptr`
  - **旧内存（old_ptr）自动被释放**
  - 你**不能再使用 old_ptr**（变成野指针）

#### 2. 失败情况 ❌

```c
void *new_ptr = realloc(old_ptr, new_size);
if (new_ptr == NULL) {
    // 失败！
}
```

- **realloc 失败**
  - 返回 **NULL**
  - **原内存（old_ptr）完全不会被释放！**
  - 原指针依然有效
  - 如果你不保存 old_ptr，会**直接内存泄漏**！

#### 3. 最危险的错误写法（90%的人都写错）

```c
ptr = realloc(ptr, 100);  // ❌ 绝对禁止！
```

**为什么错？**
如果 realloc 失败 → 返回 NULL → 直接把 ptr 覆盖成 NULL
**原来的内存地址丢了 → 内存泄漏，永远无法释放！**

#### 4. 正确写法（必须背下来）

```c
void *new_ptr = realloc(old_ptr, new_size);

if (new_ptr == NULL) {
    // 失败处理：old_ptr 还在！
    perror("realloc failed");
    // 自己决定是否释放 old_ptr
    return;
}

// 成功：old_ptr 已失效，用 new_ptr
old_ptr = new_ptr;
```

#### 5. 一句话终极总结

1. **realloc 成功 → 原指针自动释放，不能再用**
2. **realloc 失败 → 原指针不动，不会释放，必须自己保存**
3. **永远不要直接赋值给原指针：ptr = realloc(ptr, ...) ❌**

### 6. realloc使用示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    int *arr = (int*)malloc(n * sizeof(int));
    
    if (arr != NULL) {
        /* 初始化 */
        for (int i = 0; i < n; i++) {
            arr[i] = i;
        }
        
        printf("Original array:\n");
        for (int i = 0; i < n; i++) {
            printf("%d ", arr[i]);
        }
        printf("\n");
        
        /* 扩展数组 - 正确写法 */
        int new_n = 10;
        int *new_arr = (int*)realloc(arr, new_n * sizeof(int));
        
        if (new_arr != NULL) {
            arr = new_arr;
            
            /* 初始化新增部分 */
            for (int i = n; i < new_n; i++) {
                arr[i] = i;
            }
            
            printf("Expanded array:\n");
            for (int i = 0; i < new_n; i++) {
                printf("%d ", arr[i]);
            }
            printf("\n");
        } else {
            /* 失败处理：原内存依然有效 */
            printf("realloc failed, original memory remains\n");
        }
        
        free(arr);
    }
    
    return 0;
}
```

### 7. free函数

```c
void free(void *ptr);
/* 功能：释放动态分配的内存
 * 参数：ptr - 要释放的内存指针
 * 注意：ptr必须是malloc/calloc/realloc返回的指针
 *       释放NULL指针是安全的（无操作）
 */
```

### 8. 重复释放的危险性（重要！）

**超级重要结论：**
**绝对不能重复释放！`free` 两次 ＝ 直接崩溃 / 内存踩坏**

#### 1. 为什么不能重复释放？

```c
int *p = malloc(4);
free(p);  // 第一次：正常释放
free(p);  // 第二次：重复释放 → 【未定义行为！】
```

**后果：**

1. **程序直接崩溃**
2. **破坏堆内存**，导致其他地方莫名其妙出错
3. 编译器**不会报错**，运行时才炸（最难排查）

#### 2. 为什么 free(NULL) 却安全？

C 标准规定：

```c
free(NULL); // 安全！什么都不做
```

因为 `NULL` 不是合法堆地址，系统直接跳过。

#### 3. 避免重复释放的最佳实践

释放后 **立刻把指针置 NULL**：

```c
free(p);
p = NULL;  // 关键！
```

这样就算不小心写了第二次 `free(p)`
因为 `p = NULL`，也**不会造成伤害**。

#### 4. 终极口诀（背下来）

1. **同一个指针，只能 free 一次**
2. **重复 free = 未定义行为 = 崩溃**
3. **free 完必须 = NULL**
4. **free(NULL) 安全无害**

### 9. free使用示例

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *p = (int*)malloc(sizeof(int));
    
    if (p != NULL) {
        *p = 100;
        printf("Value: %d\n", *p);
        
        free(p);    /* 释放内存 */
        p = NULL;   /* 关键：释放后置NULL */
    }
    
    /* 释放NULL指针是安全的 */
    free(p);  /* 无操作 */
    
    return 0;
}
```

### 10. 内存分配函数安全使用速查表

| 函数          | 安全使用方式                                                                                               | 注意事项                             |
| ----------- | ---------------------------------------------------------------------------------------------------- | -------------------------------- |
| **malloc**  | `void *ptr = malloc(size);if (ptr != NULL) { ... }`                                                  | 检查返回值，使用后free，free后置NULL         |
| **calloc**  | `void *ptr = calloc(num, size);if (ptr != NULL) { ... }`                                             | 自动初始化为0，其他同malloc                |
| **realloc** | `void *new_ptr = realloc(old_ptr, new_size);if (new_ptr == NULL) { ... }else { old_ptr = new_ptr; }` | 成功时原指针自动释放，失败时原指针保留，永远不要直接赋值给原指针 |
| **free**    | `free(ptr);ptr = NULL;`                                                                              | 只释放动态分配的内存，释放后立即置NULL，不要重复释放     |

### 11. 安全使用准则

1. **分配检查**：始终检查malloc/calloc/realloc的返回值
2. **使用检查**：不要越界访问，不要使用已释放的内存
3. **释放检查**：每个malloc对应一个free，free后立即置NULL
4. **realloc安全**：永远使用临时指针接收realloc的返回值
5. **内存泄漏**：确保所有路径都能释放已分配的内存

### 12. 常见错误与解决方案

| 错误          | 原因          | 解决方案                      |
| ----------- | ----------- | ------------------------- |
| 内存泄漏        | 忘记free      | 每个malloc对应free，使用goto统一清理 |
| 悬空指针        | 使用已释放的内存    | free后置NULL                |
| 双重释放        | 重复释放同一内存    | free后置NULL，检查指针是否为NULL    |
| 越界访问        | 访问超出分配范围的内存 | 检查数组索引，使用边界检查             |
| realloc内存泄漏 | 直接赋值给原指针    | 使用临时指针，检查返回值              |

## 4. 内存管理最佳实践

### 1. 分配与释放配对

```c
#include <stdio.h>
#include <stdlib.h>

/* 好的实践：每个malloc对应一个free */
void good_practice(void) {
    int *p = malloc(sizeof(int));
    if (p != NULL) {
        *p = 100;
        printf("%d\n", *p);
        free(p);  /* 配对释放 */
        p = NULL;
    }
}

/* 坏的实践：忘记释放导致内存泄漏 */
void bad_practice(void) {
    int *p = malloc(sizeof(int));
    if (p != NULL) {
        *p = 100;
        printf("%d\n", *p);
        /* 忘记free！内存泄漏 */
    }
}

/* 好的实践：使用goto统一清理 */
int process_file(const char *filename) {
    FILE *fp = NULL;
    char *buffer = NULL;
    int result = -1;
    
    fp = fopen(filename, "r");
    if (fp == NULL) goto cleanup;
    
    buffer = malloc(1024);
    if (buffer == NULL) goto cleanup;
    
    /* 处理文件... */
    result = 0;
    
cleanup:
    if (buffer) free(buffer);
    if (fp) fclose(fp);
    return result;
}
```

### 2. 检查返回值

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* 始终检查malloc返回值 */
    int *p = (int*)malloc(sizeof(int) * 1000000000000LL);
    
    if (p == NULL) {
        printf("Memory allocation failed!\n");
        return 1;
    }
    
    /* 使用内存... */
    
    free(p);
    return 0;
}
```

### 3. 避免重复释放

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *p = malloc(sizeof(int));
    if (p == NULL) return 1;
    
    *p = 100;
    free(p);
    
    /* 坏的实践：重复释放 */
    /* free(p);  未定义行为！ */
    
    /* 好的实践：释放后置NULL */
    p = NULL;
    free(p);  /* 释放NULL是安全的 */
    
    return 0;
}
```

### 4. 内存管理检查清单

```
□ 分配检查
  ✓ 检查malloc/calloc/realloc返回值是否为NULL
  ✓ 使用sizeof运算符计算大小，避免硬编码

□ 使用检查
  ✓ 不要越界访问分配的内存
  ✓ 不要使用已释放的内存

□ 释放检查
  ✓ 每个malloc对应一个free
  ✓ free后立即将指针置NULL
  ✓ 不要重复释放同一内存
  ✓ 不要释放非动态分配的内存
```

## 5. 常见内存问题

### 1. 内存泄漏

```c
#include <stdio.h>
#include <stdlib.h>

/* 内存泄漏示例 */
void memory_leak_example(void) {
    int *p = malloc(sizeof(int));
    *p = 100;
    printf("%d\n", *p);
    /* 忘记free！每次调用泄漏4字节 */
}

/* 修复方案 */
void no_memory_leak(void) {
    int *p = malloc(sizeof(int));
    if (p != NULL) {
        *p = 100;
        printf("%d\n", *p);
        free(p);
    }
}

/* 检测内存泄漏的工具 */
/*
 * Valgrind (Linux):
 *   valgrind --leak-check=full ./program
 *
 * AddressSanitizer (GCC/Clang):
 *   gcc -fsanitize=address -g program.c
 */
```

### 2. 悬空指针

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *p = malloc(sizeof(int));
    if (p == NULL) return 1;
    
    *p = 100;
    free(p);
    
    /* 悬空指针：访问已释放的内存 */
    /* printf("%d\n", *p);  未定义行为！ */
    
    /* 正确做法：free后置NULL */
    p = NULL;
    
    return 0;
}
```

### 3. 越界访问

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 5;
    int *arr = malloc(n * sizeof(int));
    if (arr == NULL) return 1;
    
    /* 初始化 */
    for (int i = 0; i < n; i++) {
        arr[i] = i;
    }
    
    /* 越界访问 */
    /* arr[5] = 100;  未定义行为！ */
    /* arr[-1] = 200;  未定义行为！ */
    
    free(arr);
    return 0;
}
```

### 4. 双重释放

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *p = malloc(sizeof(int));
    int *q = p;  /* q和p指向同一内存 */
    
    if (p == NULL) return 1;
    
    *p = 100;
    
    free(p);
    /* free(q);  双重释放！未定义行为 */
    
    p = NULL;
    q = NULL;  /* 所有指针都要置NULL */
    
    return 0;
}
```

### 5. 内存问题诊断工具

```bash
# Valgrind (Linux/macOS)
valgrind --leak-check=full --show-leak-kinds=all ./program

# AddressSanitizer (GCC/Clang)
gcc -fsanitize=address -g program.c -o program
./program

# Electric Fence
gcc -lefence program.c -o program
```

## 6. 内存池实现 🟡

> 🟡 **进阶级内容**：适合有一定基础、希望提升性能的读者

### 1. 什么是内存池

内存池是一种内存管理技术，通过预分配大块内存并在内部进行管理，减少频繁调用 `malloc/free` 的开销。

**内存池的优势：**

- 减少内存分配的开销
- 降低内存碎片
- 提高内存分配的速度
- 更好的内存使用控制

### 2. 内存池设计原理

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 内存池块结构
typedef struct MemoryBlock {
    size_t size;           /* 块大小 */
    struct MemoryBlock *next;  /* 指向下一个块 */
    char data[];          /* 数据区域（柔性数组）*/
} MemoryBlock;

// 内存池结构
typedef struct MemoryPool {
    size_t block_size;     /* 块大小 */
    size_t pool_size;      /* 池总大小 */
    size_t used_size;      /* 已使用大小 */
    MemoryBlock *free_list;  /* 空闲块链表 */
    char *pool;            /* 内存池起始地址 */
} MemoryPool;
```

### 3. 内存池实现

**创建内存池：**

```c
MemoryPool* pool_create(size_t block_size, size_t block_count) {
    size_t total_size = block_size * block_count + sizeof(MemoryPool);
    
    MemoryPool *pool = (MemoryPool*)malloc(total_size);
    if (!pool) return NULL;
    
    pool->block_size = block_size;
    pool->pool_size = total_size - sizeof(MemoryPool);
    pool->used_size = 0;
    pool->free_list = NULL;
    pool->pool = (char*)pool + sizeof(MemoryPool);
    
    for (size_t i = 0; i < block_count; i++) {
        char *block_ptr = pool->pool + i * block_size;
        MemoryBlock *block = (MemoryBlock*)block_ptr;
        block->size = block_size - sizeof(MemoryBlock);
        block->next = pool->free_list;
        pool->free_list = block;
    }
    
    return pool;
}
```

**分配内存：**

```c
void* pool_alloc(MemoryPool *pool, size_t size) {
    if (!pool || size == 0) return NULL;
    
    MemoryBlock *prev = NULL;
    MemoryBlock *current = pool->free_list;
    
    while (current) {
        if (current->size >= size) {
            if (prev) {
                prev->next = current->next;
            } else {
                pool->free_list = current->next;
            }
            pool->used_size += current->size + sizeof(MemoryBlock);
            return current->data;
        }
        prev = current;
        current = current->next;
    }
    
    return NULL;
}
```

**释放内存：**

```c
void pool_free(MemoryPool *pool, void *ptr) {
    if (!pool || !ptr) return;
    
    MemoryBlock *block = (MemoryBlock*)((char*)ptr - offsetof(MemoryBlock, data));
    
    block->next = pool->free_list;
    pool->free_list = block;
    pool->used_size -= block->size + sizeof(MemoryBlock);
}
```

**销毁内存池：**

```c
void pool_destroy(MemoryPool *pool) {
    if (pool) {
        free(pool);
    }
}
```

### 4. 内存池使用示例

```c
int main(void) {
    MemoryPool *pool = pool_create(128, 16);
    
    void *ptr1 = pool_alloc(pool, 64);
    void *ptr2 = pool_alloc(pool, 32);
    
    strcpy((char*)ptr1, "Hello, Memory Pool!");
    
    pool_free(pool, ptr2);
    
    void *ptr3 = pool_alloc(pool, 48);
    
    pool_destroy(pool);
    
    return 0;
}
```

## 7. 内存对齐 🟡

> 🟡 **进阶级内容**：理解内存布局优化

### 1. 什么是内存对齐

内存对齐是指变量或数据结构在内存中的起始地址必须是某个值的整数倍。

**为什么需要内存对齐？**

- 提高内存访问效率
- 确保硬件兼容性
- 满足某些硬件平台的要求

### 2. 对齐规则

| 数据类型   | 对齐要求（字节） |
| ------ | -------- |
| char   | 1        |
| short  | 2        |
| int    | 4        |
| long   | 4 或 8    |
| float  | 4        |
| double | 8        |
| 指针     | 4 或 8    |

### 3. 内存对齐的影响

**结构体对齐示例：**

```c
// 未优化的结构体
struct BadStruct {
    char c;    // 1字节
    int i;     // 4字节
    short s;   // 2字节
}; // 总大小：12字节（有3字节填充）

// 优化的结构体
struct GoodStruct {
    int i;     // 4字节
    short s;   // 2字节
    char c;    // 1字节
}; // 总大小：8字节（1字节尾部填充）
```

### 4. 内存对齐的控制

**使用 `__attribute__((aligned(n)))` 控制对齐：**

```c
// GCC extension
// 强制16字节对齐
struct __attribute__((aligned(16))) AlignedStruct {
    char c;
    int i;
};

// 强制64字节对齐（缓存行对齐）
int __attribute__((aligned(64))) aligned_array[64];
```

**使用 `#pragma pack` 控制对齐：**

```c
// GCC, Clang, MSVC 均支持
#pragma pack(push, 1)  // 1字节对齐
struct PackedStruct {
    char c;
    int i;
    short s;
};
#pragma pack(pop)  // 恢复默认对齐
```

## 8. 内存碎片 🟡

> 🟡 **进阶级内容**：理解内存分配器行为

### 1. 什么是内存碎片

内存碎片是指内存中存在的无法被有效利用的小内存块。

**内存碎片的类型：**

1. **外部碎片**：内存中存在多个小的空闲块，但这些块的大小都小于请求的内存大小。
2. **内部碎片**：分配的内存块大于实际需要的大小，导致块内部有未使用的空间。

### 2. 内存碎片的影响

- 降低内存利用率
- 导致内存分配失败
- 增加内存管理的开销

### 3. 内存碎片的解决方案

1. **内存池**：预分配固定大小的内存块，减少外部碎片。
2. **伙伴系统**：将内存划分为大小为2的幂的块，通过合并相邻空闲块减少碎片。
3. **垃圾回收**：定期整理内存，合并空闲块。
4. **内存分配策略优化**：选择合适的内存分配算法，如首次适应、最佳适应、最差适应等。

## 9. 内存分配底层原理 🟡

> 🟡 **进阶级内容**：理解malloc底层实现

### 1. 内存分配底层原理总结

**堆内存管理到底在干嘛？**
操作系统不会让你随便用内存，它做 3 件事：

1. **虚拟内存**：给每个进程独立地址空间，互不干扰
2. **分页**：把内存切成 4KB 一页，方便管理
3. **内存分配器（malloc）**：管理堆，给你分配/释放小块内存

**4 种经典分配算法**

1. **首次适应 First Fit**
   - 从头找，**找到第一个够大的就用**
   - 快
   - 碎片多
2. **最佳适应 Best Fit**
   - 找**最接近大小**的块
   - 省空间
   - 慢，碎片多
3. **最差适应 Worst Fit**
   - 找**最大块**，切开用
   - 剩下的块大
   - 很快没大块内存
4. **伙伴系统 Buddy System**
   - 内存都是 **2的幂** 大小
   - 释放时**相邻块可以合并**
   - 碎片少
   - 速度快
   - Linux 内核用它

**malloc 底层原理（glibc）**

malloc 不是直接调用系统！它分 3 级：

1. **小内存** → **线程缓存 + 内存池**
2. **中等内存** → **堆管理（brk）**
3. **大内存** → **直接 mmap 映射**

**真正的系统调用只有两个：**

- **brk()** ：扩大堆顶
- **mmap()**：映射一块新内存

**性能优化**

1. **少调用 malloc / free**
2. **用内存池**
3. **结构体紧凑，减少对齐浪费**
4. **复用内存，不要反复申请释放**

**总结**

- **操作系统管大内存，malloc 管小内存，内存池管频繁小内存！**
- **算法决定速度，池化决定性能。**

## 10. 高级内存管理 🔴

> 🔴 **专家级内容**：面向系统级开发者、性能工程师

### 10.1 jemalloc/tcmalloc 原理概述

现代高性能应用通常使用替代分配器替代默认的 glibc malloc：

| 分配器 | 特点 | 适用场景 |
|--------|------|---------|
| **jemalloc** | 按size-class分仓、减少碎片、支持profiling | Firefox、Redis、FreeBSD默认 |
| **tcmalloc** | 线程缓存、中央堆、小对象无锁分配 | Google内部、Chrome |
| **mimalloc** | 安全聚焦、紧凑布局、高吞吐量 | Microsoft开源 |

**核心设计思想：**
- **线程缓存（Thread Cache）**：每个线程有独立的小对象缓存，无锁分配
- **Size Class**：将内存按固定大小分仓（8, 16, 32, 48, 64...），减少碎片
- **延迟合并**：释放时不立即合并，避免频繁系统调用

### 10.2 自定义内存分配器接口

```c
// 可替换的分配器接口，便于切换不同实现
typedef struct Allocator {
    void* (*alloc)(struct Allocator* self, size_t size);
    void* (*realloc)(struct Allocator* self, void* ptr, size_t old_size, size_t new_size);
    void  (*free)(struct Allocator* self, void* ptr);
    void  (*destroy)(struct Allocator* self);
} Allocator;

// 默认使用系统malloc
void* default_alloc(Allocator* self, size_t size) {
    (void)self;
    return malloc(size);
}

// 使用示例：允许在测试时注入跟踪分配器
void* tracked_alloc(Allocator* self, size_t size) {
    // 记录分配信息用于泄漏检测
    // ...
    return malloc(size);
}
```

### 10.3 内存分配器选择决策树

```
分配大小？
├── < 64KB 且 高频分配 → 内存池 / 线程缓存分配器
├── < 4MB  且 通用场景 → jemalloc / tcmalloc
├── > 4MB  或 大数组   → 直接 mmap（可独立释放）
└── 特殊对齐需求        → posix_memalign / aligned_alloc
```

## 11. 缓存行与NUMA感知 🔴

> 🔴 **专家级内容**：面向高性能计算、服务器后端开发者

### 11.1 CPU缓存行对齐

现代CPU以缓存行（通常64字节）为单位加载内存。跨缓存行的数据会导致**伪共享（False Sharing）**，严重降低多线程性能。

```c
#include <stdio.h>
#include <pthread.h>
#include <stdint.h>

// ❌ 错误：两个线程频繁修改同一缓存行内的不同变量
struct BadCounter {
    int64_t counter_a;  // 线程A修改
    int64_t counter_b;  // 线程B修改（同一缓存行！）
};

// ✅ 正确：每个变量独占一个缓存行
struct __attribute__((aligned(64))) PaddedCounter {
    int64_t value;
    char padding[64 - sizeof(int64_t)];  // 填充到64字节
};

struct GoodCounter {
    struct PaddedCounter a;
    struct PaddedCounter b;
};

// 伪共享导致性能下降10-100倍的典型场景
void* increment(void* arg) {
    int64_t* counter = (int64_t*)arg;
    for (int i = 0; i < 100000000; i++) {
        (*counter)++;
    }
    return NULL;
}

int main() {
    struct GoodCounter counters = {0};
    
    pthread_t t1, t2;
    pthread_create(&t1, NULL, increment, &counters.a.value);
    pthread_create(&t2, NULL, increment, &counters.b.value);
    
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    
    printf("a=%ld, b=%ld\n", counters.a.value, counters.b.value);
    return 0;
}
```

### 11.2 NUMA架构与内存分配

NUMA（Non-Uniform Memory Access）系统中，CPU访问本地内存比远程内存快得多。

```c
#ifdef __linux__
#include <numa.h>

// NUMA感知分配：在指定节点上分配内存
void* numa_alloc_onnode(size_t size, int node);

// 获取当前线程运行的NUMA节点
int numa_node_of_cpu(int cpu);

// 示例：绑定线程到CPU并在本地节点分配内存
void* allocate_local(size_t size) {
    int cpu = sched_getcpu();
    int node = numa_node_of_cpu(cpu);
    return numa_alloc_onnode(size, node);
}
#endif
```

**NUMA优化原则：**
1. **线程绑定**：将线程固定到特定CPU核心，避免迁移
2. **本地分配**：在线程运行的NUMA节点上分配内存
3. **首次访问策略**：让分配者也成为首次访问者，触发本地页分配

### 11.3 大页内存（HugePages）

```c
#ifdef __linux__
#include <sys/mman.h>

// 使用2MB大页分配
void* alloc_hugepage(size_t size) {
    void* ptr = mmap(NULL, size, 
                     PROT_READ | PROT_WRITE,
                     MAP_PRIVATE | MAP_ANONYMOUS | MAP_HUGETLB,
                     -1, 0);
    return ptr == MAP_FAILED ? NULL : ptr;
}
#endif
```

**大页优势：**
- 减少TLB（Translation Lookaside Buffer）未命中
- 减少页表遍历开销
- 适合大内存工作集（数据库、科学计算）
