## 1. 可变参数函数

### 1. 概念与定义

**可变参数函数（Variadic Function）**：参数个数可变的函数，如 `printf()`。通过 `<stdarg.h>` 提供的宏实现。

### 2. 核心宏

| 宏 | 说明 |
|---|------|
| `va_list` | 可变参数列表类型 |
| `va_start(ap, last)` | 初始化ap，last是最后一个固定参数 |
| `va_arg(ap, type)` | 获取下一个参数，指定类型 |
| `va_end(ap)` | 清理ap |

### 3. 基本示例

```c
#include <stdarg.h>
#include <stdio.h>

// 求任意个整数的和
int sum(int count, ...) {
    va_list ap;
    int total = 0;

    va_start(ap, count);
    for (int i = 0; i < count; i++) {
        total += va_arg(ap, int);
    }
    va_end(ap);

    return total;
}

// 求任意个整数中的最大值
int max_int(int count, ...) {
    va_list ap;
    va_start(ap, count);
    int max_val = va_arg(ap, int);
    for (int i = 1; i < count; i++) {
        int val = va_arg(ap, int);
        if (val > max_val) max_val = val;
    }
    va_end(ap);
    return max_val;
}

int main() {
    printf("sum(3, 10, 20, 30) = %d\n", sum(3, 10, 20, 30));
    printf("sum(5, 10, 20, 30, 40, 50) = %d\n", sum(5, 10, 20, 30, 40, 50));
    printf("max(4, 30, 10, 20, 15) = %d\n", max_int(4, 30, 10, 20, 15));
    return 0;
}
```

### 4. 注意事项

1. **必须至少有一个固定参数**，作为 `va_start` 的基准
2. **无法直接知道参数个数**，需要通过固定参数或其他方式传递
3. **类型安全无保障**，`va_arg` 的类型必须与实际类型匹配
4. **不支持浮点数的默认提升**，`float` 会提升为 `double`

## 2. 实现自己的printf

```c
#include <stdarg.h>
#include <stdio.h>

void my_printf(const char* fmt, ...) {
    va_list ap;
    va_start(ap, fmt);

    while (*fmt) {
        if (*fmt != '%') {
            putchar(*fmt++);
            continue;
        }
        fmt++; // 跳过 %
        switch (*fmt) {
            case 'd': {
                int val = va_arg(ap, int);
                printf("%d", val);
                break;
            }
            case 'f': {
                double val = va_arg(ap, double);
                printf("%f", val);
                break;
            }
            case 's': {
                const char* val = va_arg(ap, const char*);
                printf("%s", val);
                break;
            }
            case 'c': {
                int val = va_arg(ap, int);
                putchar(val);
                break;
            }
            case '%':
                putchar('%');
                break;
            default:
                putchar('%');
                putchar(*fmt);
                break;
        }
        fmt++;
    }

    va_end(ap);
}

int main() {
    my_printf("Hello %s, age=%d, score=%.1f, grade=%c\n",
              "Alice", 20, 95.5, 'A');
    return 0;
}
```

## 3. 命令行参数基础

### 1. argc与argv

```c
int main(int argc, char* argv[]) {
    printf("参数个数: %d\n", argc);
    for (int i = 0; i < argc; i++) {
        printf("argv[%d] = %s\n", i, argv[i]);
    }
    return 0;
}
```

```bash
# 运行示例
$ ./program hello world 123
参数个数: 4
argv[0] = ./program
argv[1] = hello
argv[2] = world
argv[3] = 123
```

### 2. 手动解析参数

```c
int main(int argc, char* argv[]) {
    int verbose = 0;
    const char* input_file = NULL;
    const char* output_file = NULL;

    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "-v") == 0 || strcmp(argv[i], "--verbose") == 0) {
            verbose = 1;
        } else if (strcmp(argv[i], "-i") == 0 && i + 1 < argc) {
            input_file = argv[++i];
        } else if (strcmp(argv[i], "-o") == 0 && i + 1 < argc) {
            output_file = argv[++i];
        } else if (strcmp(argv[i], "-h") == 0 || strcmp(argv[i], "--help") == 0) {
            printf("Usage: %s [-v] [-i input] [-o output]\n", argv[0]);
            return 0;
        }
    }

    printf("verbose=%d, input=%s, output=%s\n",
           verbose, input_file, output_file);
    return 0;
}
```

## 4. getopt短选项解析

```c
#include <unistd.h>
#include <stdio.h>

int main(int argc, char* argv[]) {
    int opt;
    const char* input_file = NULL;
    int verbose = 0;

    // "vi:" 表示 -v 无参数，-i 需要参数
    while ((opt = getopt(argc, argv, "vi:")) != -1) {
        switch (opt) {
            case 'v':
                verbose = 1;
                break;
            case 'i':
                input_file = optarg;
                break;
            case '?':
                fprintf(stderr, "Unknown option: -%c\n", optopt);
                return 1;
            case ':':
                fprintf(stderr, "Missing argument for -%c\n", optopt);
                return 1;
        }
    }

    // optind 指向第一个非选项参数
    for (int i = optind; i < argc; i++) {
        printf("Extra arg: %s\n", argv[i]);
    }

    printf("verbose=%d, input=%s\n", verbose, input_file);
    return 0;
}
```

### 1. getopt规则

| 字符串 | 含义 |
|--------|------|
| `"abc"` | -a -b -c 都无参数 |
| `"a:b:c"` | -a需参数，-b需参数，-c无参数 |
| `"a::b"` | -a可选参数，-b无参数 |

## 5. getopt_long长选项解析

```c
#include <getopt.h>
#include <stdio.h>
#include <string.h>

int main(int argc, char* argv[]) {
    int verbose = 0;
    const char* input_file = NULL;
    const char* output_file = NULL;

    static struct option long_options[] = {
        {"verbose", no_argument,       0, 'v'},
        {"input",   required_argument, 0, 'i'},
        {"output",  required_argument, 0, 'o'},
        {"help",    no_argument,       0, 'h'},
        {0, 0, 0, 0}
    };

    int opt;
    int option_index = 0;
    while ((opt = getopt_long(argc, argv, "vi:o:h",
                              long_options, &option_index)) != -1) {
        switch (opt) {
            case 'v': verbose = 1; break;
            case 'i': input_file = optarg; break;
            case 'o': output_file = optarg; break;
            case 'h':
                printf("Usage: %s [options]\n", argv[0]);
                printf("  -v, --verbose    Verbose mode\n");
                printf("  -i, --input FILE Input file\n");
                printf("  -o, --output FILE Output file\n");
                printf("  -h, --help       Show this help\n");
                return 0;
            case '?': return 1;
        }
    }

    printf("verbose=%d, input=%s, output=%s\n",
           verbose, input_file, output_file);
    return 0;
}
```

### 1. option结构体字段

| 字段 | 说明 |
|------|------|
| `name` | 长选项名称（不含--） |
| `has_arg` | `no_argument`/`required_argument`/`optional_argument` |
| `flag` | 设为0则返回val，否则将val存入*flag |
| `val` | 返回值或存入*flag的值 |
