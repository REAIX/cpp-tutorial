## 1. 位运算回顾

| 运算符 | 名称 | 示例 | 结果 |
|--------|------|------|------|
| `&` | 按位与 | `0b1100 & 0b1010` | `0b1000` |
| `\|` | 按位或 | `0b1100 \| 0b1010` | `0b1110` |
| `^` | 按位异或 | `0b1100 ^ 0b1010` | `0b0110` |
| `~` | 按位取反 | `~0b1100` | `0b...0011` |
| `<<` | 左移 | `1 << 3` | `8` |
| `>>` | 右移 | `8 >> 2` | `2` |

### 1. 运算符优先级注意

```c
// 位运算符优先级低于比较运算符，需要加括号
int x = 5;
if (x & 1 == 0) { }       // 错误！等价于 x & (1 == 0) → x & 0 → false
if ((x & 1) == 0) { }     // 正确：先按位与，再比较

// 移位运算符优先级低于加减法
int y = 1 << 2 + 3;       // 错误！等价于 1 << 5 → 32
int z = (1 << 2) + 3;     // 正确：先移位，再加法 → 7
```

### 2. 有符号数与无符号数的移位区别

```c
// 无符号数：逻辑右移（高位补0）
unsigned int ua = 0x80000000;
unsigned int ur = ua >> 1;   // 0x40000000

// 有符号负数：算术右移（高位补符号位）
int sa = -1;                  // 0xFFFFFFFF
int sr = sa >> 1;            // 0xFFFFFFFF → -1

// 左移有符号数可能溢出（未定义行为）
int overflow = 1 << 31;       // 未定义行为（C/C++标准）
```

### 3. 多标志组合

```c
typedef enum {
    PERM_READ   = 1 << 0,
    PERM_WRITE  = 1 << 1,
    PERM_EXEC   = 1 << 2,
    PERM_ADMIN  = 1 << 3,
} Permission;

int check_permission(unsigned int perms, Permission required) {
    return (perms & required) == required;
}
```

### 4. 掩码范围操作

```c
// 创建从低位开始的n位掩码
unsigned int mask_n(int n) {
    return (1U << n) - 1;      // 例如 n=4 → 0b1111
}

// 创建从pos开始的n位掩码
unsigned int mask_range(int pos, int n) {
    return ((1U << n) - 1) << pos;
}

// 提取特定位范围
unsigned int extract_bits(unsigned int value, int pos, int n) {
    return (value >> pos) & ((1U << n) - 1);
}

// 设置特定位范围的值
unsigned int set_bits_range(unsigned int value, int pos, int n, unsigned int new_val) {
    unsigned int mask = ((1U << n) - 1) << pos;
    return (value & ~mask) | ((new_val << pos) & mask);
}
```

## 4. 常用位操作技巧

```c
// 判断奇偶
int is_odd(int n) { return n & 1; }

// 交换两个数（不用临时变量）
void swap_xor(int* a, int* b) {
    *a ^= *b; *b ^= *a; *a ^= *b;
}

// 获取最低位的1
int lowest_bit(int n) { return n & (-n); }

// 统计1的个数
int popcount(unsigned int n) {
    int count = 0;
    while (n) {
        n &= n - 1;  // 清除最低位的1
        count++;
    }
    return count;
}

// 判断是否为2的幂
int is_power_of_two(int n) { return n > 0 && (n & (n - 1)) == 0; }

// 向上对齐到2的幂
unsigned int align_up_pow2(unsigned int n) {
    n--;
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    n++;
    return n;
}

// 反转位
unsigned int reverse_bits(unsigned int n) {
    n = ((n & 0x55555555) << 1) | ((n & 0xAAAAAAAA) >> 1);
    n = ((n & 0x33333333) << 2) | ((n & 0xCCCCCCCC) >> 2);
    n = ((n & 0x0F0F0F0F) << 4) | ((n & 0xF0F0F0F0) >> 4);
    n = ((n & 0x00FF00FF) << 8) | ((n & 0xFF00FF00) >> 8);
    n = (n << 16) | (n >> 16);
    return n;
}
```

### 1. 更多高级技巧

```c
// 计算前导零的个数（GCC内置函数）
int clz_builtin(unsigned int n) {
    if (n == 0) return 32;
    return __builtin_clz(n);  // 返回前导0的个数
}

// 计算尾随零的个数
int ctz_manual(unsigned int n) {
    if (n == 0) return 32;
    return popcount((n & -n) - 1);  // lowest_bit - 1 的 popcount
}

// 循环移位（左循环）
unsigned int rotl(unsigned int n, int bits) {
    bits &= 31;  // 限制在32位内
    return (n << bits) | (n >> (32 - bits));
}

// 循环移位（右循环）
unsigned int rotr(unsigned int n, int bits) {
    bits &= 31;
    return (n >> bits) | (n << (32 - bits));
}

// 只保留最高位的1
unsigned int keep_highest_bit(unsigned int n) {
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
    return n & ~(n >> 1);
}

// 获取下一个具有相同popcount的数（Next Permutation）
unsigned int next_same_popcount(unsigned int n) {
    unsigned int c = n & -n;           // 最低位的1
    unsigned int r = n + c;            // 进位
    return (((r ^ n) >> 2) / c) | r;   // 重排剩余位
}

// 枚举所有大小为k的子集
void enumerate_subsets(int k, int n) {
    int subset = (1 << k) - 1;  // 初始：最小的k个1
    while (subset < (1 << n)) {
        // 处理当前子集
        for (int i = 0; i < n; i++) {
            if (subset & (1 << i)) printf("%d ", i);
        }
        printf("\n");
        
        if (subset == 0) break;
        // Gosper's hack：下一个具有相同1位数的数
        int c = subset & -subset;
        int r = subset + c;
        subset = (((r ^ subset) >> 2) / c) | r;
    }
}
```

### 2. popcount的高效实现（分治法）

```c
// 分治法统计1的个数（O(log n)）
unsigned int popcount_fast(unsigned int n) {
    n = n - ((n >> 1) & 0x55555555);                    // 每2位一组
    n = (n & 0x33333333) + ((n >> 2) & 0x33333333);     // 每4位一组
    n = (n + (n >> 4)) & 0x0F0F0F0F;                    // 每8位一组
    n = n + (n >> 8);                                    // 每16位一组
    n = n + (n >> 16);                                   // 32位总结果
    return n & 0x7F;                                     // 最多32个1，低7位有效
}
```

```c
// 优化1：使用编译器内置函数
int popcount_builtin(unsigned int n) {
    return __builtin_popcount(n);  // 通常编译为单条指令
}

// 优化2：查表法（空间换时间）
static const int popcount_table[256] = {
    0, 1, 1, 2, 1, 2, 2, 3, 1, 2, 2, 3, 2, 3, 3, 4,
    // ... 预计算所有8位值的popcount
};

int popcount_table8(unsigned int n) {
    return popcount_table[n & 0xFF] +
           popcount_table[(n >> 8) & 0xFF] +
           popcount_table[(n >> 16) & 0xFF] +
           popcount_table[(n >> 24) & 0xFF];
}
```

```c
typedef uint32_t Color;

Color make_color(uint8_t r, uint8_t g, uint8_t b, uint8_t a) {
    return (a << 24) | (r << 16) | (g << 8) | b;
}

uint8_t color_r(Color c) { return (c >> 16) & 0xFF; }
uint8_t color_g(Color c) { return (c >> 8) & 0xFF; }

// 颜色混合（Alpha混合简化版）
Color color_blend(Color src, Color dst) {
    uint8_t sa = color_a(src);
    uint8_t sr = color_r(src) * sa / 255 + color_r(dst) * (255 - sa) / 255;
    uint8_t sg = color_g(src) * sa / 255 + color_g(dst) * (255 - sa) / 255;
    uint8_t sb = color_b(src) * sa / 255 + color_b(dst) * (255 - sa) / 255;
    return make_color(sr, sg, sb, 255);
}
```

```c
uint32_t murmur_hash(const void* key, int len, uint32_t seed) {
    const uint8_t* data = (const uint8_t*)key;
    uint32_t h = seed ^ len;
    while (len >= 4) {
        uint32_t k = *(uint32_t*)data;
```

### 3. 快速绝对值与符号判断

```c
// 绝对值（无分支）
int abs_fast(int n) {
    int mask = n >> (sizeof(int) * 8 - 1);  // 符号位传播：正数0，负数-1
    return (n + mask) ^ mask;                // 负数：n + (-1) ^ (-1) = (n-1)^(-1) = -n
}

// 符号判断：返回1（正）、-1（负）、0（零）
int sign_of(int n) {
    return (n > 0) - (n < 0);                // 利用bool值
}

// 无分支最大值
int max_fast(int a, int b) {
    int diff = a - b;
    int mask = diff >> (sizeof(int) * 8 - 1);
    return a - (diff & mask);                // 等价于 a - ((a-b) & mask)
}

// 判断两个数是否异号
int opposite_sign(int a, int b) {
    return (a ^ b) < 0;                      // 符号位不同则异或结果为负
}
```

## 6. 位图（Bitmap）的实现与应用

### 1. 基本位图结构

```c
#include <stdint.h>
#include <string.h>
#include <stdlib.h>

typedef struct {
    uint32_t* bits;
    size_t size;     // 位数
} Bitmap;

Bitmap* bitmap_create(size_t nbits) {
    Bitmap* bm = malloc(sizeof(Bitmap));
    bm->size = nbits;
    size_t word_count = (nbits + 31) / 32;
    bm->bits = calloc(word_count, sizeof(uint32_t));
    return bm;
}

void bitmap_free(Bitmap* bm) {
    free(bm->bits);
    free(bm);
}

void bitmap_set(Bitmap* bm, size_t index) {
    bm->bits[index / 32] |= (1U << (index % 32));
}

void bitmap_clear(Bitmap* bm, size_t index) {
    bm->bits[index / 32] &= ~(1U << (index % 32));
}

int bitmap_test(Bitmap* bm, size_t index) {
    return (bm->bits[index / 32] >> (index % 32)) & 1;
}

void bitmap_reset(Bitmap* bm) {
    size_t word_count = (bm->size + 31) / 32;
    memset(bm->bits, 0, word_count * sizeof(uint32_t));
}
```

```c
    size_t word_count = (bm->size + 31) / 32;
    size_t total = 0;
    for (size_t i = 0; i < word_count; i++) {
        total += popcount_fast(bm->bits[i]);
```

### 3. 位图应用：素数筛法

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

void sieve_of_eratosthenes(int n) {
    Bitmap* is_prime = bitmap_create(n + 1);
    
    // 初始假设所有数都是素数
    for (int i = 0; i <= n; i++) {
        bitmap_set(is_prime, i);
    }
    
    for (int i = 2; i * i <= n; i++) {
        if (bitmap_test(is_prime, i)) {
            // 将i的倍数标记为非素数
            for (int j = i * i; j <= n; j += i) {
                bitmap_clear(is_prime, j);
            }
        }
    }
    
    // 输出素数
    printf("2到%d的素数：", n);
    for (int i = 2; i <= n; i++) {
        if (bitmap_test(is_prime, i)) {
            printf("%d ", i);
        }
    }
    printf("\n");
    
    bitmap_free(is_prime);
}
```

## 7. 位操作在压缩与解压中的应用

### 1. 简单的位级读写

```c
#include <stdint.h>
#include <stddef.h>

typedef struct {
    uint8_t* buffer;
    size_t buffer_size;
    int bit_pos;        // 当前位位置（0=缓冲区开始）
} BitStream;

void bitstream_init(BitStream* bs, uint8_t* buf, size_t size) {
    bs->buffer = buf;
    bs->buffer_size = size;
    bs->bit_pos = 0;
}

// 写入n位数据
void bitstream_write(BitStream* bs, uint32_t data, int nbits) {
    for (int i = 0; i < nbits; i++) {
        int byte_pos = bs->bit_pos / 8;
        int bit_offset = bs->bit_pos % 8;
        
        if (data & (1 << i)) {
            bs->buffer[byte_pos] |= (1 << bit_offset);
        } else {
            bs->buffer[byte_pos] &= ~(1 << bit_offset);
        }
        
        bs->bit_pos++;
    }
}

// 读取n位数据
uint32_t bitstream_read(BitStream* bs, int nbits) {
    uint32_t result = 0;
    for (int i = 0; i < nbits; i++) {
        int byte_pos = bs->bit_pos / 8;
        int bit_offset = bs->bit_pos % 8;
        
        if (bs->buffer[byte_pos] & (1 << bit_offset)) {
            result |= (1 << i);
        }
        
        bs->bit_pos++;
    }
    return result;
}
```

### 2. 游程编码（RLE）的位实现

```c
// 简单的位游程编码
void rle_compress(const uint8_t* input, int input_len, 
                   uint8_t* output, int* output_len) {
    BitStream bs;
    bitstream_init(&bs, output, *output_len);
    
    int i = 0;
    while (i < input_len * 8) {
        int byte_pos = i / 8;
        int bit_offset = i % 8;
        int current_bit = (input[byte_pos] >> bit_offset) & 1;
        
        // 统计连续相同位的长度
        int run_len = 0;
        while (i + run_len < input_len * 8 && run_len < 15) {
            int bp = (i + run_len) / 8;
            int bo = (i + run_len) % 8;
            int bit = (input[bp] >> bo) & 1;
            if (bit != current_bit) break;
            run_len++;
        }
        
        // 写入游程：1位当前值 + 4位长度
        bitstream_write(&bs, current_bit, 1);
        bitstream_write(&bs, run_len - 1, 4);
        
        i += run_len;
    }
    
    *output_len = (bs.bit_pos + 7) / 8;
}
```

```c
        bs->bit_pos++;
    }
}
```

## 8. 算法竞赛中的位操作

### 1. 状态压缩DP

```c
// 旅行商问题（TSP）的状态压缩
// dp[mask][i] = 从0出发经过mask中的城市，最后到达i的最短距离
#define N 15   // 最多15个城市
#define INF 0x7F7F7F7F

int tsp(int dist[N][N], int n) {
    int dp[1 << N][N];
    memset(dp, 0x7F, sizeof(dp));
    
    dp[1][0] = 0;  // 从城市0出发，只访问了城市0
    
    for (int mask = 1; mask < (1 << n); mask++) {
        for (int last = 0; last < n; last++) {
            if (!(mask & (1 << last))) continue;  // last不在mask中
            if (dp[mask][last] >= INF) continue;
            
            for (int next = 0; next < n; next++) {
                if (mask & (1 << next)) continue;  // next已访问
                int new_mask = mask | (1 << next);
                int new_dist = dp[mask][last] + dist[last][next];
                if (new_dist < dp[new_mask][next]) {
                    dp[new_mask][next] = new_dist;
                }
            }
        }
    }
    
    // 回到起点
    int full_mask = (1 << n) - 1;
    int ans = INF;
    for (int i = 1; i < n; i++) {
        int d = dp[full_mask][i] + dist[i][0];
        if (d < ans) ans = d;
    }
    
    return ans;
}
```

### 3. 位操作经典算法竞赛题型

```c
// 1. 只出现一次的数字（其余出现两次）
int single_number(int* nums, int n) {
    int result = 0;
    for (int i = 0; i < n; i++) {
        result ^= nums[i];  // 出现两次的会抵消
    }
    return result;
}

// 2. 数组中两个只出现一次的数字（其余两次）
void two_single_numbers(int* nums, int n, int* res1, int* res2) {
    int xor_all = 0;
    for (int i = 0; i < n; i++) xor_all ^= nums[i];
    
    // 找到最低不同位
    int diff_bit = xor_all & -xor_all;
    
    *res1 = *res2 = 0;
    for (int i = 0; i < n; i++) {
        if (nums[i] & diff_bit) *res1 ^= nums[i];
        else *res2 ^= nums[i];
    }
}

// 3. 求1的个数（LC 191）
int hamming_weight(uint32_t n) {
    int count = 0;
    while (n) {
        count++;
        n &= n - 1;
    }
    return count;
}

// 4. 反转二进制位（LC 190）
uint32_t reverse_bits_32(uint32_t n) {
    n = ((n & 0x55555555) << 1) | ((n & 0xAAAAAAAA) >> 1);
    n = ((n & 0x33333333) << 2) | ((n & 0xCCCCCCCC) >> 2);
    n = ((n & 0x0F0F0F0F) << 4) | ((n & 0xF0F0F0F0) >> 4);
    n = ((n & 0x00FF00FF) << 8) | ((n & 0xFF00FF00) >> 8);
    return (n << 16) | (n >> 16);
}

// 5. 2的幂判断（LC 231）
int is_power_of_two_lc(int n) {
    return n > 0 && (n & (n - 1)) == 0;
}

// 6. 4的幂判断（LC 342）
int is_power_of_four(int n) {
    return n > 0 && (n & (n - 1)) == 0 && (n & 0xAAAAAAAA) == 0;
}
```

### 4. 位优化技巧总结

```c
// 用位运算替代乘除法
// x * 2  →  x << 1
// x / 16  →  x >> 4
// x % 16  →  x & 15
// x % 2   →  x & 1

// 用位运算实现取模优化
int fast_mod(int x, int mod) {
    // 仅当mod是2的幂时有效
    return x & (mod - 1);
}

// 快速舍入到2的幂的倍数的对齐
int align_to(int x, int alignment) {
    // alignment必须是2的幂
    return (x + alignment - 1) & ~(alignment - 1);
}
```
