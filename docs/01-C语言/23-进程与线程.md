## 1. 进程基础

### 1. 概念与定义

**进程（Process）**：程序的一次执行实例，拥有独立的地址空间、文件描述符表等资源。

**PID**：进程ID，操作系统分配的唯一标识。

```c
#include <stdio.h>
#include <unistd.h>

int main() {
    printf("PID: %d\n", getpid());
    printf("PPID: %d\n", getppid());
    printf("UID: %d\n", getuid());
    printf("GID: %d\n", getgid());
    return 0;
}
```

### 2. 进程状态

```
        fork()
创建 ──────→ 就绪 ──→ 运行 ──→ 等待
                              ↑      ↓
                              └── 被唤醒
                                       ↓
                                     结束
```

## 2. fork与exec

### 1. fork创建子进程

```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();

    if (pid < 0) {
        perror("fork failed");
        return 1;
    } else if (pid == 0) {
        // 子进程
        printf("子进程: PID=%d, PPID=%d\n", getpid(), getppid());
        _exit(0);
    } else {
        // 父进程
        int status;
        waitpid(pid, &status, 0);
        printf("父进程: 子进程PID=%d, 退出码=%d\n",
               pid, WEXITSTATUS(status));
    }
    return 0;
}
```

### 2. exec替换进程映像

```c
#include <unistd.h>
#include <stdio.h>

int main() {
    pid_t pid = fork();
    if (pid == 0) {
        // 子进程执行 ls 命令
        execlp("ls", "ls", "-la", NULL);
        // 如果 exec 成功，下面的代码不会执行
        perror("exec failed");
        _exit(1);
    }
    wait(NULL);
    return 0;
}
```

### 3. exec家族

| 函数 | 路径 | 参数方式 | 环境变量 |
|------|------|---------|---------|
| `execl` | 完整路径 | 列表 | 继承 |
| `execlp` | PATH搜索 | 列表 | 继承 |
| `execle` | 完整路径 | 列表 | 自定义 |
| `execv` | 完整路径 | 数组 | 继承 |
| `execvp` | PATH搜索 | 数组 | 继承 |
| `execvpe` | PATH搜索 | 数组 | 自定义 |

## 3. 进程间通信

### 1. 管道（Pipe）

```c
#include <stdio.h>
#include <unistd.h>
#include <string.h>

int main() {
    int fd[2];
    pipe(fd);

    pid_t pid = fork();
    if (pid == 0) {
        close(fd[1]);  // 子进程关闭写端
        char buf[256];
        int n = read(fd[0], buf, sizeof(buf));
        buf[n] = '\0';
        printf("子进程收到: %s\n", buf);
        close(fd[0]);
    } else {
        close(fd[0]);  // 父进程关闭读端
        const char* msg = "Hello from parent!";
        write(fd[1], msg, strlen(msg));
        close(fd[1]);
        wait(NULL);
    }
    return 0;
}
```

### 2. 共享内存

```c
#include <sys/mman.h>
#include <sys/wait.h>
#include <stdio.h>
#include <unistd.h>
#include <string.h>

int main() {
    char* shared = mmap(NULL, 4096, PROT_READ | PROT_WRITE,
                        MAP_SHARED | MAP_ANONYMOUS, -1, 0);

    pid_t pid = fork();
    if (pid == 0) {
        strcpy(shared, "Hello from child!");
        _exit(0);
    } else {
        wait(NULL);
        printf("父进程读到: %s\n", shared);
    }

    munmap(shared, 4096);
    return 0;
}
```

### 3. IPC方式对比

| 方式 | 速度 | 方向 | 适用场景 |
|------|------|------|---------|
| 管道 | 中 | 单向/双向 | 父子进程 |
| 命名管道 | 中 | 双向 | 无亲缘关系 |
| 共享内存 | 快 | 双向 | 大量数据 |
| 消息队列 | 中 | 双向 | 结构化消息 |
| 信号量 | 快 | 同步 | 资源同步 |
| Socket | 慢 | 双向 | 跨网络 |

## 4. POSIX线程

### 1. 线程创建

```c
#include <pthread.h>
#include <stdio.h>

void* thread_func(void* arg) {
    int id = *(int*)arg;
    printf("线程 %d 运行中\n", id);
    return NULL;
}

int main() {
    pthread_t threads[3];
    int ids[3] = {1, 2, 3};

    for (int i = 0; i < 3; i++) {
        pthread_create(&threads[i], NULL, thread_func, &ids[i]);
    }

    for (int i = 0; i < 3; i++) {
        pthread_join(threads[i], NULL);
    }

    return 0;
}
```

### 2. 线程 vs 进程

| 特性 | 进程 | 线程 |
|------|------|------|
| 地址空间 | 独立 | 共享 |
| 创建开销 | 大 | 小 |
| 通信方式 | IPC | 直接访问共享变量 |
| 切换开销 | 大 | 小 |
| 安全性 | 高（隔离） | 低（共享） |

## 5. 线程同步

### 1. 互斥锁

```c
#include <pthread.h>
#include <stdio.h>

int counter = 0;
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

void* increment(void* arg) {
    for (int i = 0; i < 1000000; i++) {
        pthread_mutex_lock(&mutex);
        counter++;
        pthread_mutex_unlock(&mutex);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    printf("counter = %d\n", counter);  // 2000000
    return 0;
}
```

### 2. 读写锁

**读写锁（Read-Write Lock）**：允许多个线程同时读取共享数据，但只允许一个线程写入。适用于"读多写少"的场景。

**核心规则**：
- 读锁（共享锁）：多个线程可以同时持有读锁
- 写锁（独占锁）：只有一个线程可以持有写锁，且与读锁互斥

```
读锁兼容性：

        读锁请求    写锁请求
读锁     ✅ 允许     ❌ 阻塞
写锁     ❌ 阻塞     ❌ 阻塞
```

#### 1. API一览

| 函数 | 说明 |
|------|------|
| `pthread_rwlock_init()` | 初始化读写锁 |
| `pthread_rwlock_destroy()` | 销毁读写锁 |
| `pthread_rwlock_rdlock()` | 获取读锁（阻塞） |
| `pthread_rwlock_wrlock()` | 获取写锁（阻塞） |
| `pthread_rwlock_unlock()` | 释放锁 |
| `pthread_rwlock_tryrdlock()` | 尝试获取读锁（非阻塞） |
| `pthread_rwlock_trywrlock()` | 尝试获取写锁（非阻塞） |

#### 2. 读写锁 vs 互斥锁

| 特性 | 互斥锁 | 读写锁 |
|------|--------|--------|
| 并发读 | ❌ 串行 | ✅ 并行 |
| 写操作 | 独占 | 独占 |
| 适用场景 | 读写均衡 | 读多写少 |
| 开销 | 低 | 略高 |
| 复杂度 | 简单 | 较复杂 |

> **性能提示**：在读多写少的场景下，读写锁的性能远优于互斥锁，因为多个读者可以并行执行，不必排队等待。

#### 3. 完整示例：用读写锁保护共享数据

```c
#include <pthread.h>
#include <stdio.h>
#include <string.h>

#define READER_COUNT 5
#define WRITER_COUNT 2

int shared_data = 0;
pthread_rwlock_t rwlock = PTHREAD_RWLOCK_INITIALIZER;

void* reader(void* arg) {
    int id = *(int*)arg;
    for (int i = 0; i < 5; i++) {
        pthread_rwlock_rdlock(&rwlock);
        printf("读者%d: 读取数据 = %d\n", id, shared_data);
        pthread_rwlock_unlock(&rwlock);
        usleep(100000);
    }
    return NULL;
}

void* writer(void* arg) {
    int id = *(int*)arg;
    for (int i = 0; i < 3; i++) {
        pthread_rwlock_wrlock(&rwlock);
        shared_data += 10;
        printf("写者%d: 写入数据 = %d\n", id, shared_data);
        pthread_rwlock_unlock(&rwlock);
        usleep(200000);
    }
    return NULL;
}

int main() {
    pthread_t readers[READER_COUNT], writers[WRITER_COUNT];
    int reader_ids[READER_COUNT], writer_ids[WRITER_COUNT];

    for (int i = 0; i < READER_COUNT; i++) {
        reader_ids[i] = i + 1;
        pthread_create(&readers[i], NULL, reader, &reader_ids[i]);
    }
    for (int i = 0; i < WRITER_COUNT; i++) {
        writer_ids[i] = i + 1;
        pthread_create(&writers[i], NULL, writer, &writer_ids[i]);
    }

    for (int i = 0; i < READER_COUNT; i++) {
        pthread_join(readers[i], NULL);
    }
    for (int i = 0; i < WRITER_COUNT; i++) {
        pthread_join(writers[i], NULL);
    }

    pthread_rwlock_destroy(&rwlock);
    printf("最终数据 = %d\n", shared_data);
    return 0;
}
```

### 3. 条件变量

```c
#include <pthread.h>
#include <stdio.h>

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
int ready = 0;

void* producer(void* arg) {
    pthread_mutex_lock(&mutex);
    ready = 1;
    printf("生产者: 数据已就绪\n");
    pthread_cond_signal(&cond);
    pthread_mutex_unlock(&mutex);
    return NULL;
}

void* consumer(void* arg) {
    pthread_mutex_lock(&mutex);
    while (!ready) {
        printf("消费者: 等待数据...\n");
        pthread_cond_wait(&cond, &mutex);
    }
    printf("消费者: 处理数据\n");
    pthread_mutex_unlock(&mutex);
    return NULL;
}
```

### 4. 信号量

**信号量（Semaphore）**：一个计数器，用于控制对共享资源的访问。计数器值表示可用资源的数量，当计数器为0时，请求资源的线程将被阻塞等待。

**核心机制**：
- **P操作（wait）**：计数器减1，若结果小于0则阻塞等待
- **V操作（post）**：计数器加1，若有等待线程则唤醒一个

```
信号量工作流程：

计数器 = 3（3个资源）
  ↓ 线程A请求 → 计数器 = 2 ✅
  ↓ 线程B请求 → 计数器 = 1 ✅
  ↓ 线程C请求 → 计数器 = 0 ✅
  ↓ 线程D请求 → 计数器 = 0 ❌ 阻塞等待
  ↓ 线程A释放 → 计数器 = 1，唤醒线程D ✅
```

#### 1. POSIX信号量API

| 函数 | 说明 |
|------|------|
| `sem_init()` | 初始化无名信号量 |
| `sem_destroy()` | 销毁无名信号量 |
| `sem_wait()` | P操作（阻塞） |
| `sem_post()` | V操作（唤醒） |
| `sem_getvalue()` | 获取当前计数器值 |
| `sem_trywait()` | 尝试P操作（非阻塞） |
| `sem_timedwait()` | 限时P操作 |
| `sem_open()` | 创建/打开有名信号量 |
| `sem_close()` | 关闭有名信号量 |
| `sem_unlink()` | 删除有名信号量 |

#### 2. 有名信号量 vs 无名信号量

| 特性 | 无名信号量 | 有名信号量 |
|------|-----------|-----------|
| 作用域 | 同一进程或亲缘进程间 | 任意进程间 |
| 创建方式 | `sem_init()` | `sem_open()` |
| 存储位置 | 内存中（共享内存） | 文件系统中（/dev/shm） |
| 生命周期 | 随进程 | 需显式unlink |
| 适用场景 | 线程间/父子进程间 | 无亲缘关系进程间 |

#### 3. 信号量 vs 互斥锁

| 特性 | 互斥锁 | 信号量 |
|------|--------|--------|
| 计数值 | 0/1（二值） | 0~N（任意非负整数） |
| 所有权 | 加锁者解锁 | 任意线程可post |
| 用途 | 互斥访问 | 互斥 + 同步 |
| 典型场景 | 保护临界区 | 资源池、生产者-消费者 |

> **关键区别**：互斥锁强调"所有权"——谁加锁谁解锁；信号量没有所有权概念，任何线程都可以执行V操作。

#### 4. 经典应用：生产者-消费者问题

```c
#include <pthread.h>
#include <semaphore.h>
#include <stdio.h>
#include <stdlib.h>

#define BUFFER_SIZE 10

int buffer[BUFFER_SIZE];
int in = 0, out = 0;

sem_t empty;   // 空槽位数
sem_t full;    // 已用槽位数
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

void* producer(void* arg) {
    int id = *(int*)arg;
    for (int i = 0; i < 20; i++) {
        int item = rand() % 100;

        sem_wait(&empty);
        pthread_mutex_lock(&mutex);

        buffer[in] = item;
        printf("生产者%d: 生产 item=%d, 位置=%d\n", id, item, in);
        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&full);

        usleep(100000);
    }
    return NULL;
}

void* consumer(void* arg) {
    int id = *(int*)arg;
    for (int i = 0; i < 20; i++) {
        sem_wait(&full);
        pthread_mutex_lock(&mutex);

        int item = buffer[out];
        printf("消费者%d: 消费 item=%d, 位置=%d\n", id, item, out);
        out = (out + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&empty);

        usleep(150000);
    }
    return NULL;
}

int main() {
    sem_init(&empty, 0, BUFFER_SIZE);
    sem_init(&full, 0, 0);

    pthread_t prod[2], cons[2];
    int prod_ids[2] = {1, 2};
    int cons_ids[2] = {1, 2};

    for (int i = 0; i < 2; i++) {
        pthread_create(&prod[i], NULL, producer, &prod_ids[i]);
        pthread_create(&cons[i], NULL, consumer, &cons_ids[i]);
    }

    for (int i = 0; i < 2; i++) {
        pthread_join(prod[i], NULL);
        pthread_join(cons[i], NULL);
    }

    sem_destroy(&empty);
    sem_destroy(&full);
    pthread_mutex_destroy(&mutex);
    return 0;
}
```

> **编译提示**：使用信号量需要链接 pthread 库，编译时加 `-lpthread`。

## 6. 信号处理

### 1. 信号的概念

**信号（Signal）**：操作系统向进程发送的软件中断通知，用于通知进程发生了某个事件。信号是异步的，进程无法预知信号何时到达。

```
信号生命周期：

产生信号 ──→ 内核 ──→ 递送信号 ──→ 进程处理
  │                      │
  │  来源：               │  处理方式：
  │  · 键盘（Ctrl+C）     │  · 默认动作
  │  · 硬件异常           │  · 忽略
  │  · kill命令           │  · 自定义捕获
  │  · 软件条件           │
```

### 2. 常见信号列表

| 信号 | 值 | 说明 | 默认动作 |
|------|---|------|---------|
| `SIGINT` | 2 | 中断（Ctrl+C） | 终止 |
| `SIGTERM` | 15 | 终止请求 | 终止 |
| `SIGKILL` | 9 | 强制终止 | 终止（不可捕获/忽略） |
| `SIGSEGV` | 11 | 段错误（非法内存访问） | 终止+核心转储 |
| `SIGPIPE` | 13 | 管道破裂 | 终止 |
| `SIGCHLD` | 17 | 子进程状态改变 | 忽略 |
| `SIGUSR1` | 10 | 用户自定义信号1 | 终止 |
| `SIGUSR2` | 12 | 用户自定义信号2 | 终止 |
| `SIGHUP` | 1 | 终端挂断 | 终止 |
| `SIGQUIT` | 3 | 退出（Ctrl+\） | 终止+核心转储 |
| `SIGALRM` | 14 | 定时器超时 | 终止 |
| `SIGSTOP` | 19 | 暂停进程 | 暂停（不可捕获/忽略） |
| `SIGCONT` | 18 | 继续执行 | 继续 |

> **重要**：`SIGKILL` 和 `SIGSTOP` 不能被捕获、阻塞或忽略，它们总是执行默认动作。

### 3. signal()函数

`signal()` 是最简单的信号注册方式，但存在局限性：

```c
#include <signal.h>
void (*signal(int sig, void (*handler)(int)))(int);
```

```c
#include <stdio.h>
#include <signal.h>
#include <unistd.h>

void sigint_handler(int sig) {
    printf("\n捕获到 SIGINT (信号编号=%d)\n", sig);
}

int main() {
    signal(SIGINT, sigint_handler);

    printf("进程运行中，PID=%d，按Ctrl+C测试...\n", getpid());
    while (1) {
        sleep(1);
    }
    return 0;
}
```

**signal()的缺陷**：
- 信号处理函数执行期间，该信号会被重置为默认行为（部分实现）
- 不支持获取信号来源等附加信息
- 行为在不同Unix实现间有差异
- 无法在处理期间屏蔽其他信号

### 4. sigaction()函数（推荐）

`sigaction()` 是更可靠、更强大的信号处理接口：

```c
#include <signal.h>
int sigaction(int sig, const struct sigaction *act, struct sigaction *oact);

struct sigaction {
    void     (*sa_handler)(int);      // 简单处理函数
    void     (*sa_sigaction)(int, siginfo_t*, void*);  // 详细处理函数
    sigset_t   sa_mask;               // 处理期间屏蔽的信号集
    int        sa_flags;              // 行为标志
    void     (*sa_restorer)(void);    // 已废弃
};
```

**常用 sa_flags**：

| 标志 | 说明 |
|------|------|
| `SA_SIGINFO` | 使用 sa_sigaction 而非 sa_handler |
| `SA_RESTART` | 被信号中断的系统调用自动重启 |
| `SA_NOCLDSTOP` | 子进程停止时不发送 SIGCHLD |

### 5. 信号处理函数的限制

信号处理函数中**只能调用异步信号安全（async-signal-safe）函数**。以下函数是安全的：

| 安全函数 | 说明 |
|---------|------|
| `write()` | 系统调用，可安全使用 |
| `read()` | 系统调用 |
| `_exit()` | 安全退出 |
| `signal()` | 信号操作 |
| `sigprocmask()` | 信号屏蔽 |

**不安全函数**（绝对不能在信号处理函数中调用）：
- `printf()` / `fprintf()` — 使用了全局缓冲区
- `malloc()` / `free()` — 操作堆，非原子
- `pthread_mutex_lock()` — 可能死锁

> **最佳实践**：信号处理函数中只设置 `volatile sig_atomic_t` 标志位，主循环中检查标志位并处理。

### 6. 信号屏蔽（sigprocmask）

信号屏蔽可以暂时阻止某些信号的递送，被屏蔽的信号会排队等待解除屏蔽后递送。

```c
#include <signal.h>

int sigprocmask(int how, const sigset_t *set, sigset_t *oldset);

// how 的取值：
// SIG_BLOCK   — 将 set 中的信号加入屏蔽集
// SIG_UNBLOCK — 将 set 中的信号从屏蔽集移除
// SIG_SETMASK — 将屏蔽集设为 set
```

### 7. 完整示例：用sigaction实现优雅关闭

```c
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <unistd.h>
#include <stdbool.h>

volatile sig_atomic_t running = 1;
volatile sig_atomic_t reload_config = 0;

void signal_handler(int sig, siginfo_t* info, void* context) {
    if (sig == SIGTERM || sig == SIGINT) {
        running = 0;
    } else if (sig == SIGUSR1) {
        reload_config = 1;
    }
}

void setup_signal_handlers() {
    struct sigaction sa;
    sa.sa_sigaction = signal_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_SIGINFO | SA_RESTART;

    sigaction(SIGTERM, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGUSR1, &sa, NULL);

    sigaction(SIGPIPE, &(struct sigaction){{SIG_IGN}}, NULL);
}

void do_cleanup() {
    printf("正在清理资源...\n");
    printf("关闭文件描述符...\n");
    printf("保存状态数据...\n");
    printf("清理完成，程序退出。\n");
}

int main() {
    setup_signal_handlers();

    printf("服务启动，PID=%d\n", getpid());
    printf("发送 SIGTERM/SIGINT 停止服务\n");
    printf("发送 SIGUSR1 重新加载配置\n");

    while (running) {
        if (reload_config) {
            printf("收到 SIGUSR1，重新加载配置...\n");
            reload_config = 0;
        }
        sleep(1);
    }

    do_cleanup();
    return 0;
}
```

## 7. 守护进程

### 1. 守护进程的概念

**守护进程（Daemon）**：在后台运行、脱离终端控制的长期运行进程。特点：
- 后台运行，不占用终端
- 脱离控制终端，不受终端信号影响
- 独立会话，不受用户登录/注销影响
- 通常以 root 权限运行
- 生命周期长，随系统启动而启动

```
普通进程 vs 守护进程：

普通进程：终端 ──→ 进程（受终端控制）
守护进程：init ──→ 守护进程（脱离终端）
```

### 2. 创建守护进程的5个步骤

| 步骤 | 操作 | 目的 |
|------|------|------|
| 1 | `fork()` + 父进程退出 | 保证子进程不是进程组组长 |
| 2 | `setsid()` | 创建新会话，脱离控制终端 |
| 3 | 再次 `fork()` + 父进程退出 | 防止重新打开控制终端 |
| 4 | `chdir("/")` | 改变工作目录，避免占用挂载点 |
| 5 | 关闭标准文件描述符 + 重定向到 `/dev/null` | 避免读写终端 |

> **补充**：步骤2之后还应调用 `umask(0)` 以消除文件权限继承的影响。

### 3. 完整示例：创建守护进程

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <time.h>
#include <stdbool.h>

volatile sig_atomic_t running = 1;

void sigterm_handler(int sig) {
    running = 0;
}

void daemonize() {
    // 步骤1：创建子进程，父进程退出
    pid_t pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(1);
    }
    if (pid > 0) {
        exit(0);
    }

    // 步骤2：创建新会话
    if (setsid() < 0) {
        perror("setsid");
        exit(1);
    }

    // 步骤3：再次fork，防止重新获取控制终端
    pid = fork();
    if (pid < 0) {
        perror("fork");
        exit(1);
    }
    if (pid > 0) {
        exit(0);
    }

    // 步骤4：改变工作目录，清除文件权限掩码
    chdir("/");
    umask(0);

    // 步骤5：关闭标准文件描述符，重定向到 /dev/null
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);

    int devnull = open("/dev/null", O_RDWR);
    dup(devnull);
    dup(devnull);
}

void write_log(const char* msg) {
    FILE* fp = fopen("/tmp/daemon_example.log", "a");
    if (fp) {
        time_t now = time(NULL);
        fprintf(fp, "[%s] %s\n", ctime(&now), msg);
        fclose(fp);
    }
}

int main() {
    daemonize();

    struct sigaction sa;
    sa.sa_handler = sigterm_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    sigaction(SIGTERM, &sa, NULL);
    sigaction(SIGINT, &sa, NULL);

    write_log("守护进程启动");

    int count = 0;
    while (running) {
        sleep(10);
        count++;
        char buf[128];
        snprintf(buf, sizeof(buf), "守护进程运行中，第%d次心跳", count);
        write_log(buf);
    }

    write_log("守护进程正常退出");
    return 0;
}
```

> **测试方法**：编译后运行，程序会自动进入后台。可通过 `tail -f /tmp/daemon_example.log` 查看日志，`kill <PID>` 发送 SIGTERM 优雅关闭。
