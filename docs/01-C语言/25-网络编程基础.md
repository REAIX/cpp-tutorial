## 1. 网络基础概念

### 1. TCP vs UDP

| 特性 | TCP | UDP |
|------|-----|-----|
| 连接 | 面向连接 | 无连接 |
| 可靠性 | 可靠传输 | 不可靠 |
| 顺序 | 有序 | 无序 |
| 速度 | 较慢 | 较快 |
| 适用 | 文件传输/HTTP | 视频/游戏/DNS |

### 2. 字节序

```c
// POSIX (Linux/macOS)
#include <arpa/inet.h>

uint32_t host_long = 0x12345678;
uint32_t net_long = htonl(host_long);   // 主机序→网络序
uint16_t host_short = 0x1234;
uint16_t net_short = htons(host_short); // 主机序→网络序

// 反向转换
uint32_t h = ntohl(net_long);
uint16_t hs = ntohs(net_short);
```

## 2. Socket基础

### 1. Socket API概览

```
服务端: socket() → bind() → listen() → accept() → recv()/send() → close()
客户端: socket() ────────────────────→ connect() → send()/recv() → close()
```

### 2. 地址结构体

```c
struct sockaddr_in {
    sa_family_t sin_family;  // AF_INET
    in_port_t sin_port;      // 端口号（网络字节序）
    struct in_addr sin_addr; // IP地址
};

// 初始化地址
struct sockaddr_in addr;
memset(&addr, 0, sizeof(addr));
addr.sin_family = AF_INET;
addr.sin_port = htons(8080);
addr.sin_addr.s_addr = htonl(INADDR_ANY);  // 0.0.0.0
// 或 inet_pton(AF_INET, "192.168.0.1", &addr.sin_addr);
```

## 3. TCP客户端

```c
// POSIX (Linux/macOS)
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock < 0) { perror("socket"); return 1; }

    struct sockaddr_in server;
    memset(&server, 0, sizeof(server));
    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    inet_pton(AF_INET, "192.168.0.1", &server.sin_addr);

    if (connect(sock, (struct sockaddr*)&server, sizeof(server)) < 0) {
        perror("connect");
        close(sock);
        return 1;
    }

    const char* msg = "Hello, Server!";
    send(sock, msg, strlen(msg), 0);

    char buf[1024] = {0};
    int n = recv(sock, buf, sizeof(buf) - 1, 0);
    if (n > 0) {
        buf[n] = '\0';
        printf("收到: %s\n", buf);
    }

    close(sock);
    return 0;
}
```

## 4. TCP服务端

```c
// POSIX (Linux/macOS)
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <signal.h>

volatile sig_atomic_t running = 1;
void handle_sigint(int sig) { running = 0; }

int main() {
    signal(SIGINT, handle_sigint);
    signal(SIGPIPE, SIG_IGN);

    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) { perror("socket"); return 1; }

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);

    if (bind(server_fd, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
        perror("bind");
        close(server_fd);
        return 1;
    }

    if (listen(server_fd, 5) < 0) {
        perror("listen");
        close(server_fd);
        return 1;
    }

    printf("服务端启动，监听 8080 端口...\n");

    while (running) {
        struct sockaddr_in client_addr;
        socklen_t client_len = sizeof(client_addr);
        int client_fd = accept(server_fd, (struct sockaddr*)&client_addr, &client_len);
        if (client_fd < 0) continue;

        char client_ip[INET_ADDRSTRLEN];
        inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
        printf("客户端连接: %s:%d\n", client_ip, ntohs(client_addr.sin_port));

        char buf[1024];
        int n = recv(client_fd, buf, sizeof(buf) - 1, 0);
        if (n > 0) {
            buf[n] = '\0';
            printf("收到: %s\n", buf);
            const char* response = "Hello from server!";
            send(client_fd, response, strlen(response), 0);
        }

        close(client_fd);
    }

    close(server_fd);
    printf("服务端关闭\n");
    return 0;
}
```

## 5. UDP编程

```c
// POSIX (Linux/macOS)
// UDP 服务端
int main() {
    int sock = socket(AF_INET, SOCK_DGRAM, 0);

    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);

    bind(sock, (struct sockaddr*)&addr, sizeof(addr));

    char buf[1024];
    struct sockaddr_in client_addr;
    socklen_t client_len = sizeof(client_addr);

    while (1) {
        int n = recvfrom(sock, buf, sizeof(buf) - 1, 0,
                         (struct sockaddr*)&client_addr, &client_len);
        if (n > 0) {
            buf[n] = '\0';
            printf("收到: %s\n", buf);
            sendto(sock, "ACK", 3, 0,
                   (struct sockaddr*)&client_addr, client_len);
        }
    }

    close(sock);
    return 0;
}

// UDP 客户端
int main() {
    int sock = socket(AF_INET, SOCK_DGRAM, 0);

    struct sockaddr_in server;
    memset(&server, 0, sizeof(server));
    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    inet_pton(AF_INET, "192.168.0.1", &server.sin_addr);

    const char* msg = "Hello UDP!";
    sendto(sock, msg, strlen(msg), 0,
           (struct sockaddr*)&server, sizeof(server));

    char buf[1024];
    int n = recvfrom(sock, buf, sizeof(buf) - 1, 0, NULL, NULL);
    if (n > 0) {
        buf[n] = '\0';
        printf("收到: %s\n", buf);
    }

    close(sock);
    return 0;
}
```

void print_local_ip() {
    char hostname[256];
    gethostname(hostname, sizeof(hostname));
    struct hostent* host = gethostbyname(hostname);
    for (int i = 0; host->h_addr_list[i]; i++) {
        printf("IP %d: %s\n", i, inet_ntoa(*(struct in_addr*)host->h_addr_list[i]));
    }
}

## 7. I/O多路复用基础

### 1. 为什么需要I/O多路复用

> 💡 **通俗理解 - 多客户端并发问题**
>
> 想象一个餐厅：
> - **单线程阻塞**：一个服务员只服务一桌客人，其他客人干等
> - **多线程**：每桌配一个服务员（线程开销大，容易打架）
> - **I/O多路复用**：一个服务员同时照看多桌，谁举手就过去

前面的TCP服务端示例一次只能处理一个客户端。如果客户端A正在连接，客户端B必须等A处理完才能被接受。这在实际应用中是不可接受的。

**三种并发方案对比：**

| 方案 | 优点 | 缺点 |
|------|------|------|
| 多进程 | 隔离性好 | 进程创建开销大，资源占用多 |
| 多线程 | 比进程轻量 | 线程安全复杂，大量连接时仍耗资源 |
| I/O多路复用 | 单线程处理大量连接 | 编程模型稍复杂 |

**I/O多路复用**的核心思想：让一个线程同时监控多个文件描述符，当某个fd就绪（可读/可写/异常）时，内核通知应用程序进行处理。

```
┌─────────────────────────────────────────────────┐
│          I/O多路复用工作流程                      │
├─────────────────────────────────────────────────┤
│                                                 │
│   fd1 ──┐                                       │
│   fd2 ──┤     ┌─────────────┐                   │
│   fd3 ──┼────►│   select    │──► 遍历就绪fd     │
│   fd4 ──┤     │  /poll/epoll │    逐个处理       │
│   fd5 ──┘     └─────────────┘                   │
│                                                 │
│   一个线程监控多个fd，谁有数据就处理谁            │
│                                                 │
└─────────────────────────────────────────────────┘
```

### 2. select基础用法

#### 1. fd_set操作

```c
#include <sys/select.h>

// fd_set是一个位图，每一位对应一个文件描述符
fd_set readfds;

// 四个操作宏
FD_ZERO(&readfds);        // 清空集合（必须先清空）
FD_SET(fd, &readfds);     // 将fd加入集合
FD_CLR(fd, &readfds);     // 将fd从集合移除
FD_ISSET(fd, &readfds);   // 检查fd是否在集合中（是否就绪）
```

#### 2. select函数原型

```c
// POSIX (Linux/macOS)
int select(int nfds, fd_set* readfds, fd_set* writefds,
           fd_set* exceptfds, struct timeval* timeout);

// nfds:      监控的最大文件描述符值 + 1
// readfds:   监控可读事件的fd集合（传入传出参数）
// writefds:  监控可写事件的fd集合（可传NULL）
// exceptfds: 监控异常事件的fd集合（可传NULL）
// timeout:   超时时间，NULL表示永远等待
// 返回值:    就绪fd的总数，0表示超时，-1表示出错

// 注意：select返回后，未就绪的fd位会被清零
// 所以每次调用前必须重新设置fd_set
```

#### 3. 超时设置

```c
struct timeval timeout;

// 永远等待（阻塞）
select(nfds, &readfds, NULL, NULL, NULL);

// 等待5秒
timeout.tv_sec = 5;
timeout.tv_usec = 0;
select(nfds, &readfds, NULL, NULL, &timeout);

// 非阻塞（立即返回，用于轮询）
timeout.tv_sec = 0;
timeout.tv_usec = 0;
select(nfds, &readfds, NULL, NULL, &timeout);
```

### 3. select TCP服务器示例

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/select.h>

#define PORT 8080
#define BUF_SIZE 1024
#define MAX_CLIENTS 64

int main() {
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) { perror("socket"); return 1; }

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(PORT);

    if (bind(server_fd, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
        perror("bind"); close(server_fd); return 1;
    }
    if (listen(server_fd, 5) < 0) {
        perror("listen"); close(server_fd); return 1;
    }

    printf("select服务器启动，监听 %d 端口...\n", PORT);

    int client_fds[MAX_CLIENTS];
    int client_count = 0;
    for (int i = 0; i < MAX_CLIENTS; i++) client_fds[i] = -1;

    while (1) {
        fd_set readfds;
        FD_ZERO(&readfds);
        FD_SET(server_fd, &readfds);
        int max_fd = server_fd;

        for (int i = 0; i < client_count; i++) {
            if (client_fds[i] >= 0) {
                FD_SET(client_fds[i], &readfds);
                if (client_fds[i] > max_fd) max_fd = client_fds[i];
            }
        }

        struct timeval timeout;
        timeout.tv_sec = 5;
        timeout.tv_usec = 0;

        int ready = select(max_fd + 1, &readfds, NULL, NULL, &timeout);
        if (ready < 0) {
            perror("select");
            break;
        }
        if (ready == 0) continue;

        if (FD_ISSET(server_fd, &readfds)) {
            struct sockaddr_in client_addr;
            socklen_t client_len = sizeof(client_addr);
            int cfd = accept(server_fd, (struct sockaddr*)&client_addr, &client_len);
            if (cfd >= 0) {
                char ip[INET_ADDRSTRLEN];
                inet_ntop(AF_INET, &client_addr.sin_addr, ip, sizeof(ip));
                printf("新连接: %s:%d (fd=%d)\n", ip, ntohs(client_addr.sin_port), cfd);

                if (client_count < MAX_CLIENTS) {
                    client_fds[client_count++] = cfd;
                } else {
                    printf("连接数已满，拒绝连接\n");
                    close(cfd);
                }
            }
        }

        for (int i = 0; i < client_count; i++) {
            if (client_fds[i] >= 0 && FD_ISSET(client_fds[i], &readfds)) {
                char buf[BUF_SIZE];
                int n = recv(client_fds[i], buf, sizeof(buf) - 1, 0);
                if (n <= 0) {
                    printf("连接关闭: fd=%d\n", client_fds[i]);
                    close(client_fds[i]);
                    client_fds[i] = -1;
                } else {
                    buf[n] = '\0';
                    printf("收到数据(fd=%d): %s", client_fds[i], buf);
                    send(client_fds[i], buf, n, 0);
                }
            }
        }

        for (int i = client_count - 1; i >= 0; i--) {
            if (client_fds[i] == -1) {
                client_fds[i] = client_fds[client_count - 1];
                client_fds[client_count - 1] = -1;
                client_count--;
            }
        }
    }

    for (int i = 0; i < client_count; i++) {
        if (client_fds[i] >= 0) close(client_fds[i]);
    }
    close(server_fd);
    return 0;
}
```

### 4. select的局限性

| 局限 | 说明 |
|------|------|
| **FD_SETSIZE限制** | 默认1024，修改需重新编译，不适合大量连接 |
| **全量拷贝** | 每次调用select都要将fd_set从用户态拷贝到内核态 |
| **线性扫描** | 返回后需遍历所有fd判断哪些就绪，O(n)复杂度 |
| **fd_set不可重用** | select会修改fd_set，每次调用前必须重新构建 |
| **跨平台差异** | 不同平台FD_SETSIZE不同，行为也有差异 |

```c
// FD_SETSIZE限制示例
// 在Linux上默认定义：
#define FD_SETSIZE 1024

// 即使修改这个值也必须重新编译所有依赖的库
// 不如直接使用poll或epoll
```

### 5. epoll简介

epoll是Linux特有的I/O多路复用机制，专为大量并发连接设计：

| 对比项 | select | epoll |
|--------|--------|-------|
| **最大连接数** | 1024（FD_SETSIZE） | 无限制（仅受系统内存限制） |
| **每次调用拷贝** | 全量fd_set拷贝 | 只拷贝就绪事件 |
| **就绪检测** | 遍历所有fd | 只返回就绪fd |
| **性能随连接数** | 线性下降 | 基本不变 |
| **触发模式** | 仅水平触发 | 支持水平触发和边缘触发 |
| **跨平台** | 是（POSIX标准） | 仅Linux |

#### 1. epoll核心API

```c
#include <sys/epoll.h>

// 1. 创建epoll实例
int epfd = epoll_create1(0);

// 2. 注册fd
struct epoll_event ev;
ev.events = EPOLLIN;     // 监控可读事件
ev.data.fd = some_fd;
epoll_ctl(epfd, EPOLL_CTL_ADD, some_fd, &ev);

// 3. 等待事件
struct epoll_event events[64];
int n = epoll_wait(epfd, events, 64, -1);

// 4. 处理就绪事件
for (int i = 0; i < n; i++) {
    if (events[i].events & EPOLLIN) {
        int fd = events[i].data.fd;
        char buf[1024];
        int len = recv(fd, buf, sizeof(buf), 0);
    }
}

// 5. 移除fd
epoll_ctl(epfd, EPOLL_CTL_DEL, some_fd, NULL);

// 6. 关闭epoll
close(epfd);
```

### 6. 何时使用select vs epoll

| 场景 | 推荐 | 原因 |
|------|------|------|
| 客户端程序，少量连接 | select | 简单，跨平台 |
| 需要跨平台（macOS/Windows） | select | epoll仅Linux可用 |
| 服务器，几十个连接 | select | 连接少，select够用 |
| 服务器，数百到数万连接 | epoll | 性能优势明显 |
| 服务器，C10K以上 | epoll | select无法支撑 |

> ⚠️ **选择建议**：
> - 学习阶段：先掌握select，理解I/O多路复用的核心思想
> - 实际项目：Linux服务器优先使用epoll；跨平台可考虑libevent等封装库
> - macOS/iOS上对应的机制是kqueue，Windows上是IOCP
