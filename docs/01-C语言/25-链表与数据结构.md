## 1. 单链表

### 1. 概念与定义

**链表（Linked List）**：一种线性数据结构，元素在内存中非连续存储，通过指针连接。每个节点包含数据域和指针域。

**节点（Node）**：链表的基本单元，包含数据和指向下一个节点的指针。

**头指针（Head）**：指向链表第一个节点的指针，是访问链表的入口。

### 2. 节点定义

```c
typedef int DataType;

typedef struct Node {
    DataType data;
    struct Node* next;
} Node;
```

### 3. 核心操作

```c
// 创建新节点
Node* create_node(DataType data) {
    Node* node = (Node*)malloc(sizeof(Node));
    if (!node) return NULL;
    node->data = data;
    node->next = NULL;
    return node;
}

// 头插法
Node* insert_head(Node* head, DataType data) {
    Node* node = create_node(data);
    node->next = head;
    return node;
}

// 尾插法
Node* insert_tail(Node* head, DataType data) {
    Node* node = create_node(data);
    if (!head) return node;
    Node* cur = head;
    while (cur->next) cur = cur->next;
    cur->next = node;
    return head;
}

// 删除节点
Node* delete_node(Node* head, DataType data) {
    if (!head) return NULL;
    if (head->data == data) {
        Node* temp = head;
        head = head->next;
        free(temp);
        return head;
    }
    Node* cur = head;
    while (cur->next && cur->next->data != data) {
        cur = cur->next;
    }
    if (cur->next) {
        Node* temp = cur->next;
        cur->next = temp->next;
        free(temp);
    }
    return head;
}

// 查找节点
Node* find_node(Node* head, DataType data) {
    Node* cur = head;
    while (cur) {
        if (cur->data == data) return cur;
        cur = cur->next;
    }
    return NULL;
}

// 释放链表
void free_list(Node* head) {
    while (head) {
        Node* temp = head;
        head = head->next;
        free(temp);
    }
}

// 遍历打印
void print_list(Node* head) {
    Node* cur = head;
    while (cur) {
        printf("%d -> ", cur->data);
        cur = cur->next;
    }
    printf("NULL\n");
}
```

### 4. 哨兵节点（虚拟头节点）

```c
// 使用哨兵节点简化插入/删除逻辑
Node* insert_with_sentinel(Node* sentinel, DataType data) {
    Node* node = create_node(data);
    node->next = sentinel->next;
    sentinel->next = node;
    return sentinel;
}

Node* delete_with_sentinel(Node* sentinel, DataType data) {
    Node* prev = sentinel;
    while (prev->next) {
        if (prev->next->data == data) {
            Node* temp = prev->next;
            prev->next = temp->next;
            free(temp);
            return sentinel;
        }
        prev = prev->next;
    }
    return sentinel;
}
```

### 5. 链表反转

```c
// 迭代反转
Node* reverse_list(Node* head) {
    Node* prev = NULL;
    Node* cur = head;
    while (cur) {
        Node* next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    return prev;
}

// 递归反转
Node* reverse_list_recursive(Node* head) {
    if (!head || !head->next) return head;
    Node* new_head = reverse_list_recursive(head->next);
    head->next->next = head;
    head->next = NULL;
    return new_head;
}
```

### 6. 检测环

```c
// 快慢指针法（Floyd算法）
int has_cycle(Node* head) {
    Node* slow = head;
    Node* fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return 1;
    }
    return 0;
}
```

## 2. 双链表

### 1. 节点定义

```c
typedef struct DNode {
    DataType data;
    struct DNode* prev;
    struct DNode* next;
} DNode;
```

### 2. 核心操作

```c
DNode* create_dnode(DataType data) {
    DNode* node = (DNode*)malloc(sizeof(DNode));
    if (!node) return NULL;
    node->data = data;
    node->prev = NULL;
    node->next = NULL;
    return node;
}

// 在节点后插入
void insert_after(DNode* node, DataType data) {
    DNode* new_node = create_dnode(data);
    new_node->next = node->next;
    new_node->prev = node;
    if (node->next) node->next->prev = new_node;
    node->next = new_node;
}

// 删除节点
void delete_dnode(DNode* node) {
    if (node->prev) node->prev->next = node->next;
    if (node->next) node->next->prev = node->prev;
    free(node);
}
```

## 3. 栈

### 1. 概念与定义

**栈（Stack）**：后进先出（LIFO）的线性数据结构。只允许在栈顶进行插入和删除操作。

### 2. 数组实现

```c
#define MAX_SIZE 100

typedef struct {
    DataType data[MAX_SIZE];
    int top;
} ArrayStack;

void stack_init(ArrayStack* s) { s->top = -1; }
int stack_is_empty(ArrayStack* s) { return s->top == -1; }
int stack_is_full(ArrayStack* s) { return s->top == MAX_SIZE - 1; }

int stack_push(ArrayStack* s, DataType val) {
    if (stack_is_full(s)) return -1;
    s->data[++s->top] = val;
    return 0;
}

int stack_pop(ArrayStack* s, DataType* val) {
    if (stack_is_empty(s)) return -1;
    *val = s->data[s->top--];
    return 0;
}

DataType stack_peek(ArrayStack* s) {
    return s->data[s->top];
}
```

### 3. 链表实现

```c
typedef struct StackNode {
    DataType data;
    struct StackNode* next;
} StackNode;

typedef struct {
    StackNode* top;
    int size;
} LinkedStack;

void lstack_init(LinkedStack* s) { s->top = NULL; s->size = 0; }

int lstack_push(LinkedStack* s, DataType val) {
    StackNode* node = (StackNode*)malloc(sizeof(StackNode));
    if (!node) return -1;
    node->data = val;
    node->next = s->top;
    s->top = node;
    s->size++;
    return 0;
}

int lstack_pop(LinkedStack* s, DataType* val) {
    if (!s->top) return -1;
    StackNode* temp = s->top;
    *val = temp->data;
    s->top = temp->next;
    free(temp);
    s->size--;
    return 0;
}
```

### 4. 应用：括号匹配

```c
int is_bracket_matched(const char* str) {
    ArrayStack s;
    stack_init(&s);
    for (int i = 0; str[i]; i++) {
        if (str[i] == '(' || str[i] == '[' || str[i] == '{') {
            stack_push(&s, str[i]);
        } else if (str[i] == ')' || str[i] == ']' || str[i] == '}') {
            DataType top;
            if (stack_pop(&s, &top) != 0) return 0;
            if ((str[i] == ')' && top != '(') ||
                (str[i] == ']' && top != '[') ||
                (str[i] == '}' && top != '{')) return 0;
        }
    }
    return stack_is_empty(&s);
}
```

## 4. 队列

### 1. 概念与定义

**队列（Queue）**：先进先出（FIFO）的线性数据结构。从队尾入队，从队头出队。

### 2. 循环队列（数组实现）

```c
#define QUEUE_SIZE 100

typedef struct {
    DataType data[QUEUE_SIZE];
    int front;
    int rear;
    int count;
} CircularQueue;

void queue_init(CircularQueue* q) { q->front = 0; q->rear = 0; q->count = 0; }
int queue_is_empty(CircularQueue* q) { return q->count == 0; }
int queue_is_full(CircularQueue* q) { return q->count == QUEUE_SIZE; }

int queue_enqueue(CircularQueue* q, DataType val) {
    if (queue_is_full(q)) return -1;
    q->data[q->rear] = val;
    q->rear = (q->rear + 1) % QUEUE_SIZE;
    q->count++;
    return 0;
}

int queue_dequeue(CircularQueue* q, DataType* val) {
    if (queue_is_empty(q)) return -1;
    *val = q->data[q->front];
    q->front = (q->front + 1) % QUEUE_SIZE;
    q->count--;
    return 0;
}
```

### 3. 链式队列

```c
typedef struct QueueNode {
    DataType data;
    struct QueueNode* next;
} QueueNode;

typedef struct {
    QueueNode* front;
    QueueNode* rear;
    int size;
} LinkedQueue;

void lqueue_init(LinkedQueue* q) { q->front = q->rear = NULL; q->size = 0; }

int lqueue_enqueue(LinkedQueue* q, DataType val) {
    QueueNode* node = (QueueNode*)malloc(sizeof(QueueNode));
    if (!node) return -1;
    node->data = val;
    node->next = NULL;
    if (q->rear) q->rear->next = node;
    else q->front = node;
    q->rear = node;
    q->size++;
    return 0;
}

int lqueue_dequeue(LinkedQueue* q, DataType* val) {
    if (!q->front) return -1;
    QueueNode* temp = q->front;
    *val = temp->data;
    q->front = temp->next;
    if (!q->front) q->rear = NULL;
    free(temp);
    q->size--;
    return 0;
}
```

## 5. 哈希表

### 1. 概念与定义

**哈希表（Hash Table）**：通过哈希函数将键映射到数组索引，实现O(1)平均时间复杂度的查找。

**哈希冲突（Collision）**：不同的键映射到相同索引的情况。

### 2. 链地址法实现

```c
#define TABLE_SIZE 128

typedef struct HashNode {
    char* key;
    DataType value;
    struct HashNode* next;
} HashNode;

typedef struct {
    HashNode* buckets[TABLE_SIZE];
    int size;
} HashTable;

// 哈希函数（djb2算法）
unsigned int hash(const char* key) {
    unsigned int hash = 5381;
    int c;
    while ((c = *key++)) {
        hash = ((hash << 5) + hash) + c;
    }
    return hash % TABLE_SIZE;
}

void ht_init(HashTable* ht) {
    memset(ht->buckets, 0, sizeof(ht->buckets));
    ht->size = 0;
}

int ht_insert(HashTable* ht, const char* key, DataType value) {
    unsigned int idx = hash(key);
    HashNode* node = ht->buckets[idx];
    while (node) {
        if (strcmp(node->key, key) == 0) {
            node->value = value;
            return 0;
        }
        node = node->next;
    }
    HashNode* new_node = (HashNode*)malloc(sizeof(HashNode));
    new_node->key = strdup(key);
    new_node->value = value;
    new_node->next = ht->buckets[idx];
    ht->buckets[idx] = new_node;
    ht->size++;
    return 0;
}

int ht_get(HashTable* ht, const char* key, DataType* value) {
    unsigned int idx = hash(key);
    HashNode* node = ht->buckets[idx];
    while (node) {
        if (strcmp(node->key, key) == 0) {
            *value = node->value;
            return 0;
        }
        node = node->next;
    }
    return -1;
}

void ht_free(HashTable* ht) {
    for (int i = 0; i < TABLE_SIZE; i++) {
        HashNode* node = ht->buckets[i];
        while (node) {
            HashNode* temp = node;
            node = node->next;
            free(temp->key);
            free(temp);
        }
    }
}
```

## 6. 数据结构选型指南

| 数据结构 | 查找 | 插入 | 删除 | 适用场景 |
|---------|------|------|------|---------|
| 数组 | O(1)/O(n) | O(n) | O(n) | 随机访问、数据量固定 |
| 单链表 | O(n) | O(1) | O(1) | 频繁插入删除 |
| 栈 | O(n) | O(1) | O(1) | 撤销操作、递归模拟 |
| 队列 | O(n) | O(1) | O(1) | 任务调度、BFS |
| 哈希表 | O(1)均 | O(1)均 | O(1)均 | 快速查找、键值映射 |
| 树（BST/AVL/红黑树） | O(log n) | O(log n) | O(log n) | 有序存储和快速查找（见 [算法与数据结构](../04-工程实践/09-算法与数据结构.md)） |
| 图（邻接矩阵/邻接表） | O(V+E) | O(1)/O(V) | O(V)/O(V) | 表示实体间关系（见 [算法与数据结构](../04-工程实践/09-算法与数据结构.md)） |
| 堆/优先队列 | O(n) | O(log n) | O(log n) | 快速获取最大/最小元素（见 [算法与数据结构](../04-工程实践/09-算法与数据结构.md)） |

> 💡 **进阶指引**：树、图、堆等数据结构涉及更复杂的组织方式和算法思想，本章仅作选型引导。如需深入了解二叉搜索树、AVL树、红黑树的实现，图的遍历与最短路径算法，以及堆的建堆与调整操作，请参阅 [算法与数据结构](../04-工程实践/09-算法与数据结构.md)。
