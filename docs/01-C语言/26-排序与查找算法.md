## 2. 冒泡排序

```c
void bubble_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = 1;
            }
        }
        if (!swapped) break;
    }
}
```

## 3. 选择排序

```c
void selection_sort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int min_idx = i;
        for (int j = i + 1; j < n; j++) {
            if (arr[j] < arr[min_idx]) min_idx = j;
        }
        if (min_idx != i) {
            int temp = arr[i];
            arr[i] = arr[min_idx];
            arr[min_idx] = temp;
        }
    }
}
```

## 4. 插入排序

```c
void insertion_sort(int arr[], int n) {
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}
```

## 5. 快速排序

```c
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            int temp = arr[i]; arr[i] = arr[j]; arr[j] = temp;
        }
    }
    int temp = arr[i + 1]; arr[i + 1] = arr[high]; arr[high] = temp;
    return i + 1;
}

void quick_sort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quick_sort(arr, low, pi - 1);
        quick_sort(arr, pi + 1, high);
    }
}
```

## 6. 归并排序

```c
void merge(int arr[], int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;
    int* L = (int*)malloc(n1 * sizeof(int));
    int* R = (int*)malloc(n2 * sizeof(int));

    for (int i = 0; i < n1; i++) L[i] = arr[left + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = left;
    while (i < n1 && j < n2) {
        arr[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];

    free(L);
    free(R);
}

void merge_sort(int arr[], int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        merge_sort(arr, left, mid);
        merge_sort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}
```

## 7. 堆排序

```c
void heapify(int arr[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest]) largest = left;
    if (right < n && arr[right] > arr[largest]) largest = right;

    if (largest != i) {
        int temp = arr[i]; arr[i] = arr[largest]; arr[largest] = temp;
        heapify(arr, n, largest);
    }
}

void heap_sort(int arr[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) heapify(arr, n, i);
    for (int i = n - 1; i > 0; i--) {
        int temp = arr[0]; arr[0] = arr[i]; arr[i] = temp;
        heapify(arr, i, 0);
    }
}
```

## 8. 计数排序

> **核心思想：** 不比较元素大小，而是统计每个值出现的次数，然后按次数依次输出。

### 1. 原理

1. 找出数组中的最大值和最小值，确定值域范围 `k = max - min + 1`
2. 创建计数数组 `count[k]`，统计每个元素出现的次数
3. 对计数数组做前缀和，使 `count[i]` 表示「小于等于 i 的元素个数」
4. 从后往前遍历原数组，根据计数数组将元素放入正确位置（保证稳定性）

### 2. 适用条件

- 元素必须是非负整数（或可映射为整数）
- 值域范围 `k` 不能太大，否则空间开销过高

### 3. 复杂度

| 指标 | 复杂度 |
|------|--------|
| 时间 | O(n + k) |
| 空间 | O(n + k) |
| 稳定性 | **稳定** |

### 4. 代码实现

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void counting_sort(int arr[], int n) {
    if (n <= 1) return;

    /* 找最大值和最小值 */
    int min_val = arr[0], max_val = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] < min_val) min_val = arr[i];
        if (arr[i] > max_val) max_val = arr[i];
    }

    int range = max_val - min_val + 1;

    /* 分配计数数组并初始化为0 */
    int* count = (int*)calloc(range, sizeof(int));
    int* output = (int*)malloc(n * sizeof(int));
    if (!count || !output) { free(count); free(output); return; }

    /* 统计每个元素出现的次数 */
    for (int i = 0; i < n; i++) {
        count[arr[i] - min_val]++;
    }

    /* 前缀和：count[i] 变为小于等于 i+min_val 的元素个数 */
    for (int i = 1; i < range; i++) {
        count[i] += count[i - 1];
    }

    /* 从后往前遍历，保证稳定性 */
    for (int i = n - 1; i >= 0; i--) {
        int idx = arr[i] - min_val;
        output[count[idx] - 1] = arr[i];
        count[idx]--;
    }

    /* 将结果复制回原数组 */
    memcpy(arr, output, n * sizeof(int));

    free(count);
    free(output);
}

int main(void) {
    int arr[] = {4, 2, 2, 1, 3, 3, 1};
    int n = sizeof(arr) / sizeof(arr[0]);

    printf("排序前：");
    for (int i = 0; i < n; i++) printf("%d ", arr[i]);
    printf("\n");

    counting_sort(arr, n);

    printf("排序后：");
    for (int i = 0; i < n; i++) printf("%d ", arr[i]);
    printf("\n");

    return 0;
}
```

### 5. 应用场景

- **成绩排序**：分数范围 0~100，k=101，非常高效
- **年龄排序**：年龄范围 0~150，k=151，线性时间完成
- **字符频率统计**：ASCII 字符范围 0~127

## 9. 基数排序

> **核心思想：** 按位排序，从最低位到最高位（LSD），每位的排序必须是稳定的。

### 1. 原理

1. 找到数组中的最大值，确定最大位数 `d`
2. 从最低位（个位）开始，对每一位执行稳定排序（通常用计数排序）
3. 逐位处理直到最高位，排序完成

### 2. LSD vs MSD

| 方式 | 全称 | 处理顺序 | 特点 |
|------|------|---------|------|
| LSD | 最低位优先 | 个位→十位→百位 | 实现简单，无需递归 |
| MSD | 最高位优先 | 百位→十位→个位 | 需要递归处理子区间 |

> 本章采用 LSD 版本实现。

### 3. 复杂度

| 指标 | 复杂度 |
|------|--------|
| 时间 | O(d × (n + k))，d 为位数，k 为基数（通常为10） |
| 空间 | O(n + k) |
| 稳定性 | **稳定** |

### 4. 代码实现

```c
#include <stdio.h>
#include <stdlib.h>

/* 获取数组中的最大值 */
int get_max(int arr[], int n) {
    int max = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > max) max = arr[i];
    }
    return max;
}

/* 对数组按第 exp 位进行计数排序 */
void counting_sort_by_digit(int arr[], int n, int exp) {
    int* output = (int*)malloc(n * sizeof(int));
    int count[10] = {0};

    /* 统计当前位上各数字出现的次数 */
    for (int i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }

    /* 前缀和 */
    for (int i = 1; i < 10; i++) {
        count[i] += count[i - 1];
    }

    /* 从后往前遍历，保证稳定性 */
    for (int i = n - 1; i >= 0; i--) {
        int digit = (arr[i] / exp) % 10;
        output[count[digit] - 1] = arr[i];
        count[digit]--;
    }

    /* 复制回原数组 */
    for (int i = 0; i < n; i++) {
        arr[i] = output[i];
    }

    free(output);
}

void radix_sort(int arr[], int n) {
    if (n <= 1) return;

    int max = get_max(arr, n);

    /* 从个位开始，逐位排序 */
    for (int exp = 1; max / exp > 0; exp *= 10) {
        counting_sort_by_digit(arr, n, exp);
    }
}

int main(void) {
    int arr[] = {100, 45, 25, 10, 100, 24, 2, 20};
    int n = sizeof(arr) / sizeof(arr[0]);

    radix_sort(arr, n);

    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
```

### 5. 应用场景

- **电话号码排序**：11位数字，d=11，远小于 n log n
- **身份证号排序**：18位数字，d=18
- **日期排序**：按年月日拆分后逐位排序

## 10. 桶排序

> **核心思想：** 将元素分配到有限数量的桶中，每个桶单独排序，最后合并。

### 1. 原理

1. 确定桶的数量和映射函数
2. 遍历数组，将每个元素分配到对应的桶中
3. 对每个非空桶内部进行排序（通常用插入排序）
4. 按桶的顺序依次取出元素，得到有序序列

### 2. 复杂度

| 指标 | 复杂度 |
|------|--------|
| 平均时间 | O(n + k) |
| 最坏时间 | O(n²)（所有元素分到同一个桶） |
| 空间 | O(n + k) |
| 稳定性 | **稳定**（桶内排序稳定时） |

### 3. 代码实现

```c
#include <stdio.h>
#include <stdlib.h>

/* 桶内节点 */
typedef struct Node {
    double data;
    struct Node* next;
} Node;

/* 在桶内按升序插入（插入排序） */
Node* insert_sorted(Node* head, double value) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    new_node->data = value;
    new_node->next = NULL;

    if (head == NULL || head->data >= value) {
        new_node->next = head;
        return new_node;
    }

    Node* curr = head;
    while (curr->next != NULL && curr->next->data < value) {
        curr = curr->next;
    }
    new_node->next = curr->next;
    curr->next = new_node;

    return head;
}

void bucket_sort(double arr[], int n) {
    if (n <= 1) return;

    /* 创建 n 个桶 */
    Node** buckets = (Node**)calloc(n, sizeof(Node*));

    /* 将元素分配到桶中 */
    for (int i = 0; i < n; i++) {
        int bucket_idx = (int)(arr[i] * n);
        if (bucket_idx >= n) bucket_idx = n - 1;
        buckets[bucket_idx] = insert_sorted(buckets[bucket_idx], arr[i]);
    }

    /* 按顺序收集所有桶中的元素 */
    int idx = 0;
    for (int i = 0; i < n; i++) {
        Node* curr = buckets[i];
        while (curr != NULL) {
            arr[idx++] = curr->data;
            Node* temp = curr;
            curr = curr->next;
            free(temp);
        }
    }

    free(buckets);
}

int main(void) {
    double arr[] = {0.42, 0.32, 0.33, 0.52, 0.32, 0.42, 0.51};
    int n = sizeof(arr) / sizeof(arr[0]);

    bucket_sort(arr, n);

    for (int i = 0; i < n; i++) {
        printf("%.2f ", arr[i]);
    }
    printf("\n");

    return 0;
}
```

### 4. 应用场景

- **浮点数排序**：0~1 范围内的浮点数，均匀分布时接近线性时间
- **均匀分布数据**：数据均匀分散时，每个桶内元素极少，效率极高
- **大规模数据分片**：先分桶再各桶并行排序

## 11. 非比较排序 vs 比较排序

### 1. 对比表

| 对比项 | 比较排序 | 非比较排序 |
|--------|---------|-----------|
| 核心原理 | 元素两两比较 | 利用元素的固有属性（值、位、范围） |
| 时间下界 | O(n log n) | O(n)（理想情况） |
| 适用范围 | 任意可比较类型 | 仅限整数或可映射为整数的数据 |
| 空间开销 | 通常 O(1)~O(n) | 通常 O(n + k)，k 为值域范围 |
| 稳定性 | 取决于具体实现 | 通常为稳定排序 |
| 典型算法 | 快排、归并、堆排 | 计数排序、基数排序、桶排序 |

### 2. 为什么非比较排序能突破 O(n log n) 下界？

> **决策树模型**：比较排序的每一次比较相当于决策树的一个分支，n 个元素的所有排列共 n! 种，决策树至少需要 n! 个叶子节点，因此树高至少为 log₂(n!) = Ω(n log n)。

非比较排序**不依赖元素间的比较**，而是直接利用元素的值来确定位置，因此不受决策树模型的约束，可以突破 O(n log n) 的理论下界。

但这一突破是有代价的：
- 需要额外的空间（用空间换时间）
- 对数据类型和分布有要求（不是通用排序）

### 3. 选择建议

| 场景 | 推荐算法 |
|------|---------|
| 整数且值域较小（k ≈ n） | 计数排序 |
| 多位数整数（如电话号码） | 基数排序 |
| 0~1 均匀分布浮点数 | 桶排序 |
| 通用排序、数据类型未知 | 快速排序 / 归并排序 |
| 内存受限 | 堆排序 |

## 12. 顺序查找

int linear_search(int arr[], int n, int target) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == target) return i;
    }
    return -1;
}

## 13. 二分查找

```c
// 迭代版
int binary_search(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// 递归版
int binary_search_recursive(int arr[], int left, int right, int target) {
    if (left > right) return -1;
    int mid = left + (right - left) / 2;
    if (arr[mid] == target) return mid;
    if (arr[mid] > target) return binary_search_recursive(arr, left, mid - 1, target);
    return binary_search_recursive(arr, mid + 1, right, target);
}

// 查找左边界（第一个等于target的位置）
int lower_bound(int arr[], int n, int target) {
    int left = 0, right = n;
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] < target) left = mid + 1;
        else right = mid;
    }
    return left;
}
```

## 14. 排序算法对比

| 算法 | 平均时间 | 最坏时间 | 空间 | 稳定性 |
|------|---------|---------|------|--------|
| 冒泡排序 | O(n²) | O(n²) | O(1) | 稳定 |
| 选择排序 | O(n²) | O(n²) | O(1) | 不稳定 |
| 插入排序 | O(n²) | O(n²) | O(1) | 稳定 |
| 快速排序 | O(n log n) | O(n²) | O(log n) | 不稳定 |
| 归并排序 | O(n log n) | O(n log n) | O(n) | 稳定 |
| 堆排序 | O(n log n) | O(n log n) | O(1) | 不稳定 |
| 计数排序 | O(n + k) | O(n + k) | O(n + k) | 稳定 |
| 基数排序 | O(d(n + k)) | O(d(n + k)) | O(n + k) | 稳定 |
| 桶排序 | O(n + k) | O(n²) | O(n + k) | 稳定 |

> 计数排序中 k 为值域范围；基数排序中 d 为位数，k 为基数；桶排序中 k 为桶数量。

### 1. 选择建议

- **小数据量（n < 50）**：插入排序
- **大数据量，要求稳定**：归并排序
- **大数据量，不要求稳定**：快速排序
- **内存受限**：堆排序
- **整数且值域小（k ≈ n）**：计数排序
- **多位数整数（电话号码等）**：基数排序
- **均匀分布浮点数**：桶排序
