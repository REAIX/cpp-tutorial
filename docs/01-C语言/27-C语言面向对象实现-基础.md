### 2. 概念与定义

**面向对象编程（Object-Oriented Programming, OOP）**：一种编程范式，将数据和操作绑定在一起，形成对象。面向对象编程的三大特性是封装、继承和多态。

**封装（Encapsulation）**：将数据和操作绑定在一起，隐藏实现细节，只对外提供接口。

**继承（Inheritance）**：复用已有类型的属性和方法，扩展新的功能。

**多态（Polymorphism）**：同一接口，不同实现。多态可以提高代码的灵活性和可扩展性。

**构造函数（Constructor）**：用于初始化对象的函数。构造函数在对象创建时自动调用。

**析构函数（Destructor）**：用于清理对象的函数。析构函数在对象销毁时自动调用。

## 2. 封装的实现

### 1. 基本封装模式

```c
/*
 * ============================================================
 * 文件: shape.h
 * 描述: 形状基类 - 演示封装实现
 * 功能: 定义形状接口和基本操作
 * ============================================================
 */

#ifndef SHAPE_H
#define SHAPE_H

/* ==================== 前向声明 ==================== */
typedef struct Shape Shape;

/* ==================== 虚函数表结构 ==================== */
typedef struct {
    double (*area)(const Shape *self);       /* 计算面积 */
    double (*perimeter)(const Shape *self);  /* 计算周长 */
    void (*draw)(const Shape *self);         /* 绘制形状 */
    void (*destroy)(Shape *self);            /* 析构函数 */
} ShapeVTable;

/* ==================== Shape基类结构 ==================== */
struct Shape {
    const ShapeVTable *vtable;  /* 虚函数表指针 - 实现多态的关键 */
    int id;                     /* 形状ID */
    char name[32];              /* 形状名称 */
};

/* ==================== 公共接口函数 ==================== */

/**
 * @brief 计算形状面积（多态调用）
 */
static inline double shape_area(const Shape *shape) {
    return shape->vtable->area(shape);
}

/**
 * @brief 计算形状周长（多态调用）
 */
static inline double shape_perimeter(const Shape *shape) {
    return shape->vtable->perimeter(shape);
}

/**
 * @brief 绘制形状（多态调用）
 */
static inline void shape_draw(const Shape *shape) {
    shape->vtable->draw(shape);
}

/**
 * @brief 销毁形状（多态调用）
 */
static inline void shape_destroy(Shape *shape) {
    if (shape) {
        shape->vtable->destroy(shape);
    }
}

#endif /* SHAPE_H */
```

### 2. 封装实现要点

```c
/*
 * 封装实现的核心要点：
 * 
 * 1. 数据隐藏：使用不透明指针（Opaque Pointer）
 *    - 头文件只声明 typedef struct Shape Shape;
 *    - 实现细节在.c文件中，用户无法直接访问成员
 * 
 * 2. 访问控制：通过函数访问数据
 *    - 提供 get_xxx() 和 set_xxx() 函数
 *    - 内部数据通过 self 指针访问
 * 
 * 3. 接口稳定：公共API不变
 *    - 用户只通过公共函数操作对象
 *    - 内部实现可以自由修改
 */

/* 不透明指针示例 - 头文件中 */
typedef struct Person Person;  /* 只声明类型名，不暴露内部结构 */

/* 公共接口 - 头文件中 */
Person* person_create(const char *name, int age);
void person_destroy(Person *p);
const char* person_get_name(const Person *p);
int person_get_age(const Person *p);
void person_set_age(Person *p, int age);
void person_introduce(const Person *p);

/* 实现文件 - .c文件中 */
struct Person {
    char name[64];    /* 私有数据 */
    int age;           /* 私有数据 */
    char secret[256];  /* 完全隐藏的数据 */
};

Person* person_create(const char *name, int age) {
    Person *p = malloc(sizeof(Person));
    if (p) {
        strncpy(p->name, name, sizeof(p->name) - 1);
        p->name[sizeof(p->name) - 1] = '\0';
        p->age = age;
        p->secret[0] = '\0';  /* 初始化私有数据 */
    }
    return p;
}
```

```c
 * ============================================================
 * 文件: rectangle.c
 * 描述: 矩形类实现
 * ============================================================

#include "rectangle.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 虚函数实现 ==================== */

static double rectangle_area_impl(const Shape *shape) {
    /* 从基类指针获取派生类指针 */
    const Rectangle *rect = (const Rectangle *)shape;
    return rect->width * rect->height;
}

static double rectangle_perimeter_impl(const Shape *shape) {
    const Rectangle *rect = (const Rectangle *)shape;
    return 2 * (rect->width + rect->height);
}

static void rectangle_draw_impl(const Shape *shape) {
    const Rectangle *rect = (const Rectangle *)shape;
    printf("绘制矩形: %.2fx%.2f\n", rect->width, rect->height);
}

static void rectangle_destroy_impl(Shape *shape) {
    free(shape);  /* 释放整个结构体 */
}

/* ==================== 虚函数表 ==================== */

static const ShapeVTable rectangle_vtable = {
    .area = rectangle_area_impl,
    .perimeter = rectangle_perimeter_impl,
    .draw = rectangle_draw_impl,
    .destroy = rectangle_destroy_impl
};

/* ==================== 公共接口实现 ==================== */

Rectangle* rectangle_create(double width, double height) {
    Rectangle *rect = malloc(sizeof(Rectangle));
    if (rect) {
        /* 初始化基类部分 */
        rect->base.vtable = &rectangle_vtable;
        rect->base.id = 1;
        strcpy(rect->base.name, "Rectangle");
        
        /* 初始化派生类部分 */
        rect->width = width;
        rect->height = height;
    }
    return rect;
}

void rectangle_destroy(Rectangle *rect) {
    shape_destroy((Shape *)rect);
}

void rectangle_set_size(Rectangle *rect, double width, double height) {
    if (rect) {
        rect->width = width;
        rect->height = height;
    }
}

void rectangle_get_size(const Rectangle *rect, double *width, double *height) {
    if (rect) {
        *width = rect->width;
        *height = rect->height;
    }
}
```

## 4. 多态的实现

### 1. 虚函数表机制

```c
/*
 * ============================================================
 * 多态实现的核心机制：虚函数表
 * 
 * 原理：
 * 1. 每个类有一个静态的虚函数表（vtable）
 * 2. 每个对象有一个指向虚函数表的指针
 * 3. 调用方法时通过vtable查找实际函数
 * 
 * 优势：
 * - 运行时决定调用哪个函数
 * - 支持向上转型（基类指针操作派生类对象）
 * - 实现接口统一
 * ============================================================
 */

/* 虚函数表定义 */
typedef struct {
    void (*method1)(void *self);
    int (*method2)(void *self, int param);
    void (*method3)(void *self, const char *str);
} VTable;

/* 基类结构 */
typedef struct {
    const VTable *vtable;  /* 指向虚函数表 */
    int data;
} Base;

/* 多态调用示例 */
void call_method1(Base *obj) {
    /* 通过虚函数表调用 - 运行时决定调用哪个实现 */
    obj->vtable->method1(obj);
}

int call_method2(Base *obj, int param) {
    return obj->vtable->method2(obj, param);
}
```

### 2. 多态使用示例

```c
/*
 * ============================================================
 * 文件: polymorphism_demo.c
 * 描述: 多态完整示例
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>

/* ==================== 动物基类 ==================== */
typedef struct Animal Animal;

typedef struct {
    void (*speak)(const Animal *self);
    void (*move)(const Animal *self);
    void (*destroy)(Animal *self);
} AnimalVTable;

struct Animal {
    const AnimalVTable *vtable;
    char name[32];
};

/* ==================== 狗类 ==================== */
typedef struct {
    Animal base;
    char breed[32];
} Dog;

static void dog_speak(const Animal *animal) {
    const Dog *dog = (const Dog *)animal;
    printf("%s (狗, 品种: %s) 说: 汪汪汪!\n", dog->base.name, dog->breed);
}

static void dog_move(const Animal *animal) {
    printf("狗在奔跑...\n");
}

static void dog_destroy(Animal *animal) {
    free(animal);
}

static const AnimalVTable dog_vtable = {
    .speak = dog_speak,
    .move = dog_move,
    .destroy = dog_destroy
};

Dog* dog_create(const char *name, const char *breed) {
    Dog *dog = malloc(sizeof(Dog));
    if (dog) {
        dog->base.vtable = &dog_vtable;
        strncpy(dog->base.name, name, sizeof(dog->base.name) - 1);
        strncpy(dog->breed, breed, sizeof(dog->breed) - 1);
    }
    return dog;
}

/* ==================== 猫类 ==================== */
typedef struct {
    Animal base;
    char color[32];

static void cat_speak(const Animal *animal) {

static void cat_move(const Animal *animal) {
    printf("猫在跳跃...\n");

Cat* cat_create(const char *name, const char *color) {

/* ==================== 多态演示 ==================== */

void animal_speak(const Animal *animal) {
    animal->vtable->speak(animal);  /* 多态调用 */
}

void animal_move(const Animal *animal) {
```

## 5. 构造与析构

### 1. 对象生命周期管理

```c
/*
 * ============================================================
 * 对象生命周期管理
 * 
 * C语言没有自动构造/析构，需要手动管理
 * 
 * 最佳实践：
 * 1. 每个类型提供 create 和 destroy 函数
 * 2. create 负责分配内存和初始化
 * 3. destroy 负责清理资源和释放内存
 * 4. 使用 RAII 思想（资源获取即初始化）
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 文件资源类 ==================== */
typedef struct {
    FILE *fp;
    char path[256];
    int is_open;
} FileResource;

/* 构造函数：创建并打开文件 */
FileResource* file_create(const char *path, const char *mode) {
    FileResource *file = malloc(sizeof(FileResource));
    if (!file) return NULL;
    
    /* 初始化成员 */
    strncpy(file->path, path, sizeof(file->path) - 1);
    file->fp = fopen(path, mode);
    file->is_open = (file->fp != NULL);
    
    if (!file->is_open) {
        printf("警告: 无法打开文件 %s\n", path);
    }
    
    return file;
}

/* 析构函数：关闭文件并释放内存 */
void file_destroy(FileResource *file) {
    if (!file) return;
    
    /* 清理资源 */
    if (file->is_open && file->fp) {
        fclose(file->fp);
        file->is_open = 0;
        printf("文件 %s 已关闭\n", file->path);
    }
    
    /* 释放内存 */
    free(file);
}

/* 使用示例 */
void demo_lifecycle(void) {
    printf("=== 对象生命周期演示 ===\n\n");
    
    /* 构造：创建对象 */
    FileResource *file = file_create("test.txt", "w");
    if (!file) {
        printf("创建失败\n");
        return;
    }
    
    /* 使用对象 */
    if (file->is_open) {
        fprintf(file->fp, "Hello, C OOP!\n");
        printf("已写入数据到 %s\n", file->path);
    }
    
    /* 析构：销毁对象 */
    file_destroy(file);
    /* 此时 file 指针已无效，不应再使用 */
}
```

### 2. 资源管理宏 (模拟RAII)

#### 1. AUTO\_FREE 宏

`AUTO_FREE` 宏利用 GCC 的 `__attribute__((cleanup))` 特性，实现函数退出时自动释放内存。

```c
/*
 * ============================================================
 * AUTO_FREE 宏 - 自动内存释放
 * ============================================================
 */

/* 辅助函数：实际执行 free */
static inline void free_auto(void *ptr) {
    // 需要将 隐式转换的void*指针 还原为 void**指针
    void **p = (void **)ptr; 
    if (*p) {
        free(*p);
        *p = NULL;
    }
}

/* 核心宏定义：声明一个自动释放的指针 */
#define AUTO_FREE(ptr) \
    __attribute__((cleanup(free_auto))) void *ptr

/* ✅ 正确使用宏的示例 */
void demo_auto_free(void) {
    /* 使用宏声明变量，函数退出时自动释放 */
    AUTO_FREE(data) = malloc(100 * sizeof(int));
    
    /* 使用数据 */
    for (int i = 0; i < 100; i++) {
        ((int*)data)[i] = i;
    }
    
    /* 函数结束时自动调用 free_auto(&data) */ /* 传递的是 void** 指针*/
}
```

**宏展开说明：**

当你写：

```c
AUTO_FREE(data) = malloc(100 * sizeof(int));
```

编译器会展开为：

```c
__attribute__((cleanup(free_auto))) void *data = malloc(100 * sizeof(int));
```

**🔥 最核心规则（必须背下来）**

> **是的！`cleanup`** **会自动加** **`&`！自动取地址！**

**GCC 官方规定：**

当变量离开作用域时，编译器会**强制、自动、偷偷**生成代码：

```c
清理函数(&变量);  // 自动取地址！你不用写 &！
```

**举例说明：**

你写：

```c
FILE *fp __attribute__((cleanup(fclose_auto))) = fopen(...);
```

函数退出时，编译器偷偷插入：

```c
fclose_auto(&fp);  // 自动加 &！
```

**所以：**

- 你写的是：`fp`
- 编译器实际传的是：`&fp`

**为什么清理函数参数是指针的指针？**

因为传进去的是 `&fp`（地址的地址），所以接收必须用指针的指针：

```c
void fclose_auto(FILE **fp)  // 接收的是 &fp，所以是 FILE**
{
    if (*fp) {
        fclose(*fp);  // *fp 才是真正的文件指针
    }
}
```

**两种写法对比：**

| 方式     | 写法                                                             | 说明      |
| ------ | -------------------------------------------------------------- | ------- |
| 直接使用属性 | `int *data __attribute__((cleanup(free_auto))) = malloc(...);` | 冗长，容易写错 |
| 使用宏    | `AUTO_FREE(data) = malloc(...);`                               | 简洁，一目了然 |

#### 2. SCOPE\_GUARD 宏

通用的作用域守卫宏，可用于任意资源的自动清理：

```c
/*
 * ============================================================
 * SCOPE_GUARD 宏 - 通用资源清理
 * ============================================================
 */

/* 通用作用域守卫宏 */
#define SCOPE_GUARD(var, cleanup_func) \
    __attribute__((cleanup(cleanup_func))) typeof(var) var

/* 文件自动关闭示例 */
static void fclose_auto(FILE **fp) {
    if (*fp) {
        fclose(*fp);
        *fp = NULL;
    }
}

void demo_scope_guard(void) {
    /* 文件指针，函数退出时自动关闭 */
    SCOPE_GUARD(fp, fclose_auto) = fopen("test.txt", "r");
    if (!fp) return;
    
    /* 使用文件... */
    
    /* 函数退出时自动调用 fclose_auto(&fp) */
}
```

#### 3. 完整可运行示例

```c
#include <stdio.h>
#include <stdlib.h>

static inline void free_auto(void *ptr) {
    void **p = ptr;
    if (*p) {
        free(*p);
        printf("✅ 内存自动释放！\n");
    }
}

#define AUTO_FREE(ptr) \
    __attribute__((cleanup(free_auto))) void *ptr

void demo_auto_free(void) {
    /* ✅ 使用宏声明自动释放的指针 */
    AUTO_FREE(data) = malloc(100 * sizeof(int));
    
    /* 使用内存 */
    for (int i = 0; i < 10; i++) {
        ((int*)data)[i] = i;
    }
    
    printf("使用内存完成\n");
    
    /* 不用手动 free！自动释放 */
}

int main() {
    demo_auto_free();
    return 0;
}
```

```
✅ 内存自动释放！
```

```c
typedef struct {
    int ref_count;
    /* 其他数据... */
} RefCounted;

RefCounted* ref_counted_create(void) {
    RefCounted *obj = malloc(sizeof(RefCounted));
    if (obj) obj->ref_count = 1;
    return obj;
}

void ref_counted_add_ref(RefCounted *obj) {
    if (obj) obj->ref_count++;
}

void ref_counted_release(RefCounted *obj) {
    if (obj && --obj->ref_count == 0) {
        free(obj);
    }
}

/* 3. 内存池（性能优化） */
typedef struct {
    char *buffer;
    size_t capacity;
    size_t used;
} MemoryPool;

MemoryPool* pool_create(size_t capacity) {
    MemoryPool *pool = malloc(sizeof(MemoryPool));
    if (pool) {
        pool->buffer = malloc(capacity);
        pool->capacity = capacity;
        pool->used = 0;
    }
    return pool;
}

void* pool_alloc(MemoryPool *pool, size_t size) {
    if (pool->used + size > pool->capacity) return NULL;
    void *ptr = pool->buffer + pool->used;
    pool->used += size;
    return ptr;
}

void pool_destroy(MemoryPool *pool) {
    if (pool) {
        free(pool->buffer);
        free(pool);
    }
}
```
