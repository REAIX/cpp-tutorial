## 1. 接口与抽象类

### 1.1 通俗理解

> 💡 **接口就像"合同"，抽象类就像"半成品"**

- **接口**：只规定"能做什么"，不规定"怎么做"。就像合同只写"乙方需交付产品"，不写"乙方用什么工艺"。
- **抽象类**：规定"能做什么"的同时，还提供部分"怎么做"。就像半成品家具，框架做好了，细节你自己定。

**C语言中的实现方式：**

- 接口 = 只包含函数指针的结构体（纯虚函数表）
- 抽象类 = 包含函数指针 + 部分数据成员的结构体

### 1.2 接口定义与实现

```c
/*
 * ============================================================
 * 接口定义：可序列化接口
 * 任何实现此接口的类型都可以被序列化和反序列化
 * ============================================================
 */

/* ==================== 接口定义（纯虚函数表） ==================== */

typedef struct Serializable Serializable;

/* 纯虚函数表 - 只有函数指针，没有数据 */
typedef struct {
    /* 序列化为字符串，返回字符串长度 */
    int (*serialize)(const Serializable *self, char *buf, size_t buf_size);

    /* 从字符串反序列化 */
    int (*deserialize)(Serializable *self, const char *buf);

    /* 获取序列化格式的名称 */
    const char* (*format_name)(void);
} SerializableVTable;

/* 接口基类 - 只包含虚函数表指针 */
struct Serializable {
    const SerializableVTable *vtable;
};

/* 接口调用函数 */
static inline int serializable_to_string(const Serializable *obj,
                                          char *buf, size_t size) {
    return obj->vtable->serialize(obj, buf, size);
}

static inline int serializable_from_string(Serializable *obj, const char *buf) {
    return obj->vtable->deserialize(obj, buf);
}

static inline const char* serializable_format(void) {
    /* 格式名不需要self，但保持接口一致性 */
    return "Serializable";
}
```

### 1.3 一个类实现多个接口

```c
/*
 * ============================================================
 * 一个类同时实现多个接口
 * 就像一个人可以同时是"员工"和"父母"
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 接口1：可打印 ==================== */
typedef struct Printable Printable;

typedef struct {
    void (*print)(const Printable *self);
    const char* (*to_string)(const Printable *self, char *buf, size_t size);
} PrintableVTable;

struct Printable {
    const PrintableVTable *vtable;
};

/* ==================== 接口2：可比较 ==================== */
typedef struct Comparable Comparable;

typedef struct {
    int (*compare)(const Comparable *self, const Comparable *other);
} ComparableVTable;

struct Comparable {
    const ComparableVTable *vtable;
};

/* ==================== 具体类：Student ==================== */
typedef struct {
    /* 实现多个接口 - 每个接口一个虚函数表 */
    Printable  printable;     /* 可打印接口 */
    Comparable comparable;    /* 可比较接口 */

    /* 自身数据 */
    char name[64];
    int  score;
} Student;

/* --- Printable 接口实现 --- */
static void student_print(const Printable *self) {
    /* 从接口指针反推对象指针 */
    Student *stu = (Student *)((char *)self - offsetof(Student, printable));
    printf("学生: %s, 成绩: %d\n", stu->name, stu->score);
}

static const char* student_to_string(const Printable *self, char *buf, size_t size) {
    Student *stu = (Student *)((char *)self - offsetof(Student, printable));
    snprintf(buf, size, "Student(%s, %d)", stu->name, stu->score);
    return buf;
}

static const PrintableVTable student_printable_vtable = {
    .print = student_print,
    .to_string = student_to_string
};

/* --- Comparable 接口实现 --- */
static int student_compare(const Comparable *self, const Comparable *other) {
    Student *a = (Student *)((char *)self - offsetof(Student, comparable));
    Student *b = (Student *)((char *)other - offsetof(Student, comparable));
    if (a->score > b->score) return 1;
    if (a->score < b->score) return -1;
    return 0;
}

static const ComparableVTable student_comparable_vtable = {
    .compare = student_compare
};

/* --- 构造与析构 --- */
Student* student_create(const char *name, int score) {
    Student *stu = malloc(sizeof(Student));
    if (stu) {
        /* 绑定接口虚函数表 */
        stu->printable.vtable = &student_printable_vtable;
        stu->comparable.vtable = &student_comparable_vtable;

        strncpy(stu->name, name, sizeof(stu->name) - 1);
        stu->score = score;
    }
    return stu;
}

void student_destroy(Student *stu) {
    free(stu);
}

/* ==================== 使用示例 ==================== */
int main(void) {
    Student *s1 = student_create("张三", 90);
    Student *s2 = student_create("李四", 85);

    /* 通过 Printable 接口使用 */
    s1->printable.vtable->print(&s1->printable);

    char buf[128];
    s2->printable.vtable->to_string(&s2->printable, buf, sizeof(buf));
    printf("字符串: %s\n", buf);

    /* 通过 Comparable 接口使用 */
    int result = s1->comparable.vtable->compare(&s1->comparable, &s2->comparable);
    printf("比较结果: %s\n", result > 0 ? "s1 > s2" : result < 0 ? "s1 < s2" : "s1 == s2");

    student_destroy(s1);
    student_destroy(s2);
    return 0;
}
```

### 1.4 抽象类实现

```c
/*
 * ============================================================
 * 抽象类：提供部分默认实现
 * 子类可以覆写，也可以直接使用默认实现
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 抽象类：Logger ==================== */

typedef struct Logger Logger;

typedef struct {
    void (*log)(Logger *self, const char *message);
    void (*log_info)(Logger *self, const char *message);
    void (*log_error)(Logger *self, const char *message);
    void (*destroy)(Logger *self);
} LoggerVTable;

struct Logger {
    const LoggerVTable *vtable;
    int  level;          /* 日志级别：0=ALL, 1=INFO, 2=ERROR */
    char prefix[32];     /* 日志前缀 */
};

/* 默认实现：log_info 调用 log */
static void logger_default_log_info(Logger *self, const char *message) {
    if (self->level <= 1) {
        char buf[512];
        snprintf(buf, sizeof(buf), "[INFO] %s", message);
        self->vtable->log(self, buf);
    }
}

/* 默认实现：log_error 调用 log */
static void logger_default_log_error(Logger *self, const char *message) {
    if (self->level <= 2) {
        char buf[512];
        snprintf(buf, sizeof(buf), "[ERROR] %s", message);
        self->vtable->log(self, buf);
    }
}

/* ==================== 具体类：ConsoleLogger ==================== */

typedef struct {
    Logger base;
    int color_enabled;
} ConsoleLogger;

static void console_log(Logger *self, const char *message) {
    ConsoleLogger *cl = (ConsoleLogger *)self;
    if (cl->color_enabled) {
        printf("\033[1;37m%s\033[0m: %s\n", self->prefix, message);
    } else {
        printf("%s: %s\n", self->prefix, message);
    }
}

static void console_destroy(Logger *self) {
    free(self);
}

/* 虚函数表：log 是自己的，log_info/log_error 用默认实现 */
static const LoggerVTable console_logger_vtable = {
    .log = console_log,
    .log_info = logger_default_log_info,
    .log_error = logger_default_log_error,
    .destroy = console_destroy
};

ConsoleLogger* console_logger_create(const char *prefix, int color) {
    ConsoleLogger *cl = malloc(sizeof(ConsoleLogger));
    if (cl) {
        cl->base.vtable = &console_logger_vtable;
        cl->base.level = 0;
        strncpy(cl->base.prefix, prefix, sizeof(cl->base.prefix) - 1);
        cl->color_enabled = color;
    }
    return cl;
}

/* ==================== 具体类：FileLogger ==================== */

typedef struct {
    Logger base;
    FILE *fp;
} FileLogger;

static void file_log(Logger *self, const char *message) {
    FileLogger *fl = (FileLogger *)self;
    if (fl->fp) {
        fprintf(fl->fp, "%s: %s\n", self->prefix, message);
        fflush(fl->fp);
    }
}

static void file_destroy(Logger *self) {
    FileLogger *fl = (FileLogger *)self;
    if (fl->fp) fclose(fl->fp);
    free(self);
}

static const LoggerVTable file_logger_vtable = {
    .log = file_log,
    .log_info = logger_default_log_info,
    .log_error = logger_default_log_error,
    .destroy = file_destroy
};

FileLogger* file_logger_create(const char *prefix, const char *filepath) {
    FileLogger *fl = malloc(sizeof(FileLogger));
    if (fl) {
        fl->base.vtable = &file_logger_vtable;
        fl->base.level = 0;
        strncpy(fl->base.prefix, prefix, sizeof(fl->base.prefix) - 1);
        fl->fp = fopen(filepath, "a");
    }
    return fl;
}

/* ==================== 使用示例 ==================== */
int main(void) {
    /* 控制台日志 */
    ConsoleLogger *cl = console_logger_create("APP", 1);
    Logger *logger1 = (Logger *)cl;

    logger1->vtable->log_info(logger1, "程序启动");
    logger1->vtable->log_error(logger1, "连接超时");

    /* 文件日志 */
    FileLogger *fl = file_logger_create("SYSTEM", "app.log");
    Logger *logger2 = (Logger *)fl;

    logger2->vtable->log_info(logger2, "系统初始化完成");
    logger2->vtable->log_error(logger2, "磁盘空间不足");

    /* 清理 */
    logger1->vtable->destroy(logger1);
    logger2->vtable->destroy(logger2);

    return 0;
}
```

### 1.5 注意事项

| 要点 | 说明 |
| --- | --- |
| **接口应尽量小** | 一个接口只定义一组相关操作，避免"胖接口" |
| **offsetof 反推指针** | 多接口时从接口指针反推对象指针，注意内存对齐 |
| **默认实现复用** | 抽象类的默认实现可减少子类重复代码 |
| **纯虚函数表用NULL** | 如果接口方法必须实现，可在纯虚函数表中填NULL，运行时检查 |

## 2. 混入(Mixin)模式

### 2.1 通俗理解

> 💡 **Mixin就像"技能卡"**

想象一个RPG游戏角色：
- 继承是"种族"——精灵天生会射箭
- Mixin是"技能卡"——任何种族都可以装备"火球术"技能卡

**Mixin的核心思想：**
- 不改变继承体系
- 给现有类"混入"额外能力
- 比继承更灵活，避免深层继承链

### 2.2 Mixin实现

```c
/*
 * ============================================================
 * Mixin模式：给类动态添加能力
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== Mixin1：可计数能力 ==================== */

typedef struct {
    int count;
    void (*increment)(void *self);
    void (*decrement)(void *self);
    int (*get_count)(const void *self);
} CountableMixin;

/* Mixin的初始化函数 */
static void countable_init(CountableMixin *mixin) {
    mixin->count = 0;
}

static void countable_increment(void *self) {
    /* 由使用者在具体类中实现绑定 */
    CountableMixin *mixin = (CountableMixin *)self;
    mixin->count++;
}

static void countable_decrement(void *self) {
    CountableMixin *mixin = (CountableMixin *)self;
    mixin->count--;
}

static int countable_get_count(const void *self) {
    const CountableMixin *mixin = (const CountableMixin *)self;
    return mixin->count;
}

/* ==================== Mixin2：可命名能力 ==================== */

typedef struct {
    char name[64];
    void (*set_name)(void *self, const char *name);
    const char* (*get_name)(const void *self);
} NameableMixin;

static void nameable_init(NameableMixin *mixin, const char *name) {
    strncpy(mixin->name, name, sizeof(mixin->name) - 1);
    mixin->name[sizeof(mixin->name) - 1] = '\0';
}

static void nameable_set_name(void *self, const char *name) {
    NameableMixin *mixin = (NameableMixin *)self;
    strncpy(mixin->name, name, sizeof(mixin->name) - 1);
}

static const char* nameable_get_name(const void *self) {
    const NameableMixin *mixin = (const NameableMixin *)self;
    return mixin->name;
}

/* ==================== Mixin3：可序列化能力 ==================== */

typedef struct {
    char (*serialize)(void *self, char *buf, size_t size);
} SerializableMixin;

/* ==================== 具体类：Task（混入多个能力） ==================== */

typedef struct {
    /* 混入各种能力 */
    CountableMixin   countable;
    NameableMixin    nameable;
    SerializableMixin serializable;

    /* 自身属性 */
    int priority;
    int completed;
} Task;

/* Task的序列化实现 */
static char task_serialize(void *self, char *buf, size_t size) {
    Task *task = (Task *)self;
    snprintf(buf, size, "Task{name=%s, count=%d, priority=%d, done=%d}",
             task->nameable.name,
             task->countable.count,
             task->priority,
             task->completed);
    return 0;
}

/* 构造函数 */
Task* task_create(const char *name, int priority) {
    Task *task = calloc(1, sizeof(Task));
    if (task) {
        /* 初始化各个Mixin */
        countable_init(&task->countable);
        nameable_init(&task->nameable, name);
        task->serializable.serialize = task_serialize;

        /* 初始化自身属性 */
        task->priority = priority;
        task->completed = 0;
    }
    return task;
}

void task_destroy(Task *task) {
    free(task);
}

/* ==================== 使用示例 ==================== */
int main(void) {
    Task *task = task_create("完成报告", 1);

    /* 使用 Countable Mixin */
    task->countable.increment(&task->countable);
    task->countable.increment(&task->countable);
    printf("计数: %d\n", task->countable.get_count(&task->countable));

    /* 使用 Nameable Mixin */
    printf("名称: %s\n", task->nameable.get_name(&task->nameable));
    task->nameable.set_name(&task->nameable, "完成月度报告");
    printf("新名称: %s\n", task->nameable.get_name(&task->nameable));

    /* 使用 Serializable Mixin */
    char buf[256];
    task->serializable.serialize(task, buf, sizeof(buf));
    printf("序列化: %s\n", buf);

    task_destroy(task);
    return 0;
}
```

### 2.3 注意事项

| 要点 | 说明 |
| --- | --- |
| **Mixin不是继承** | Mixin是组合，不形成"is-a"关系 |
| **避免Mixin之间耦合** | 每个Mixin应独立，不依赖其他Mixin |
| **Mixin大小要适中** | 太小不值得做Mixin，太大会让类臃肿 |
| **初始化要完整** | 每个Mixin需要单独初始化，容易遗漏 |

## 3. 组合优于继承

### 3.1 通俗理解

> 💡 **"有一个"比"是一个"更灵活**

- 继承："汽车**是一个**交通工具" → 强耦合
- 组合："汽车**有一个**引擎" → 松耦合，引擎可以换

**为什么组合更好？**

- 继承层次太深 → 代码难以理解
- 修改基类影响所有子类 → 牵一发动全身
- C语言只能单继承 → 无法同时继承多个基类
- 组合可以运行时更换 → 更灵活

### 3.2 继承的问题示例

```c
/*
 * ============================================================
 * 继承的陷阱：深层继承链
 * ============================================================
 */

/* ❌ 不好的设计：深层继承 */
typedef struct {
    int x, y;
} Point;

typedef struct {
    Point base;          /* 第1层 */
    int radius;
} Circle;

typedef struct {
    Circle base;         /* 第2层 */
    int border_width;
} CircleWithBorder;

typedef struct {
    CircleWithBorder base; /* 第3层 */
    int shadow_depth;
} CircleWithBorderAndShadow;

/* 问题：
 * 1. 访问 x 需要写 obj.base.base.base.x —— 太深了！
 * 2. 修改 Point 影响所有层
 * 3. 类型转换容易出错
 */
```

### 3.3 组合方式重写

```c
/*
 * ============================================================
 * ✅ 好的设计：组合代替继承
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 各个组件 ==================== */

typedef struct {
    int x, y;
} Position;

typedef struct {
    int radius;
} CircleShape;

typedef struct {
    int width;
    int color;
} Border;

typedef struct {
    int depth;
    int direction;
} Shadow;

/* ==================== 组合类 ==================== */

typedef struct {
    Position    position;     /* 位置组件 */
    CircleShape shape;        /* 形状组件 */
    Border     *border;       /* 边框组件（可选） */
    Shadow     *shadow;       /* 阴影组件（可选） */
} GraphicCircle;

/* 构造函数 - 只有必要组件 */
GraphicCircle* graphic_circle_create(int x, int y, int radius) {
    GraphicCircle *gc = calloc(1, sizeof(GraphicCircle));
    if (gc) {
        gc->position.x = x;
        gc->position.y = y;
        gc->shape.radius = radius;
        gc->border = NULL;   /* 可选，暂不添加 */
        gc->shadow = NULL;   /* 可选，暂不添加 */
    }
    return gc;
}

/* 动态添加边框 */
void graphic_circle_add_border(GraphicCircle *gc, int width, int color) {
    if (!gc) return;
    if (!gc->border) {
        gc->border = malloc(sizeof(Border));
    }
    if (gc->border) {
        gc->border->width = width;
        gc->border->color = color;
    }
}

/* 动态添加阴影 */
void graphic_circle_add_shadow(GraphicCircle *gc, int depth, int direction) {
    if (!gc) return;
    if (!gc->shadow) {
        gc->shadow = malloc(sizeof(Shadow));
    }
    if (gc->shadow) {
        gc->shadow->depth = depth;
        gc->shadow->direction = direction;
    }
}

/* 绘制 */
void graphic_circle_draw(const GraphicCircle *gc) {
    if (!gc) return;
    printf("圆形: 位置(%d,%d), 半径=%d",
           gc->position.x, gc->position.y, gc->shape.radius);
    if (gc->border) {
        printf(", 边框[宽=%d, 色=%d]", gc->border->width, gc->border->color);
    }
    if (gc->shadow) {
        printf(", 阴影[深=%d, 向=%d]", gc->shadow->depth, gc->shadow->direction);
    }
    printf("\n");
}

/* 析构函数 */
void graphic_circle_destroy(GraphicCircle *gc) {
    if (gc) {
        free(gc->border);
        free(gc->shadow);
        free(gc);
    }
}

/* ==================== 使用示例 ==================== */
int main(void) {
    /* 基本圆形 */
    GraphicCircle *gc1 = graphic_circle_create(10, 20, 5);
    graphic_circle_draw(gc1);

    /* 带边框的圆形 */
    GraphicCircle *gc2 = graphic_circle_create(30, 40, 8);
    graphic_circle_add_border(gc2, 2, 0xFF0000);
    graphic_circle_draw(gc2);

    /* 带边框和阴影的圆形 */
    GraphicCircle *gc3 = graphic_circle_create(50, 60, 12);
    graphic_circle_add_border(gc3, 3, 0x00FF00);
    graphic_circle_add_shadow(gc3, 5, 135);
    graphic_circle_draw(gc3);

    /* 清理 */
    graphic_circle_destroy(gc1);
    graphic_circle_destroy(gc2);
    graphic_circle_destroy(gc3);

    return 0;
}
```

### 3.4 组合与继承的选择

```
┌─────────────────────────────────────────────────────────────────────────┐
│                    继承 vs 组合 选择指南                                │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  ✅ 使用继承：                                                          │
│     • 确实是"is-a"关系（狗是动物）                                     │
│     • 需要多态（基类指针操作派生类）                                   │
│     • 继承层次不超过2-3层                                              │
│                                                                         │
│  ✅ 使用组合：                                                          │
│     • "has-a"关系（汽车有引擎）                                        │
│     • 需要运行时更换组件                                               │
│     • 组件可选（有些对象有，有些没有）                                 │
│     • 需要独立复用组件                                                 │
│                                                                         │
│  💡 原则：优先使用组合，只在确实需要多态时使用继承                     │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

## 4. 观察者模式

### 4.1 通俗理解

> 💡 **就像"订阅公众号"**

- 你关注了一个公众号（注册观察者）
- 公众号发新文章时，所有关注者都收到通知（通知观察者）
- 你取消关注，就不再收到通知（注销观察者）

**核心角色：**
- **Subject（主题）**：被观察的对象，维护观察者列表
- **Observer（观察者）**：订阅主题，接收通知

### 4.2 C语言实现

```c
/*
 * ============================================================
 * 观察者模式：对象间的一对多依赖关系
 * 当一个对象状态改变时，所有依赖它的对象都会收到通知
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 观察者接口 ==================== */

typedef struct Observer Observer;

typedef struct {
    void (*on_notify)(Observer *self, const char *event, const void *data);
} ObserverVTable;

struct Observer {
    const ObserverVTable *vtable;
};

/* ==================== 主题（被观察者） ==================== */

#define MAX_OBSERVERS 32

typedef struct {
    Observer *observers[MAX_OBSERVERS];  /* 观察者列表 */
    int observer_count;                   /* 观察者数量 */
    char name[64];                        /* 主题名称 */
} Subject;

/* 初始化主题 */
void subject_init(Subject *sub, const char *name) {
    memset(sub, 0, sizeof(Subject));
    strncpy(sub->name, name, sizeof(sub->name) - 1);
}

/* 注册观察者 */
int subject_attach(Subject *sub, Observer *obs) {
    if (sub->observer_count >= MAX_OBSERVERS) {
        printf("错误: 观察者数量已达上限\n");
        return -1;
    }
    /* 检查是否已注册 */
    for (int i = 0; i < sub->observer_count; i++) {
        if (sub->observers[i] == obs) {
            return 0;  /* 已注册 */
        }
    }
    sub->observers[sub->observer_count++] = obs;
    return 0;
}

/* 注销观察者 */
void subject_detach(Subject *sub, Observer *obs) {
    for (int i = 0; i < sub->observer_count; i++) {
        if (sub->observers[i] == obs) {
            /* 移动后面的观察者填补空位 */
            for (int j = i; j < sub->observer_count - 1; j++) {
                sub->observers[j] = sub->observers[j + 1];
            }
            sub->observer_count--;
            return;
        }
    }
}

/* 通知所有观察者 */
void subject_notify(Subject *sub, const char *event, const void *data) {
    for (int i = 0; i < sub->observer_count; i++) {
        if (sub->observers[i] && sub->observers[i]->vtable) {
            sub->observers[i]->vtable->on_notify(sub->observers[i], event, data);
        }
    }
}

/* ==================== 具体观察者1：日志观察者 ==================== */

typedef struct {
    Observer base;
    char name[32];
} LogObserver;

static void log_observer_on_notify(Observer *self, const char *event, const void *data) {
    LogObserver *lo = (LogObserver *)self;
    printf("[日志观察者 %s] 收到事件: %s, 数据: %s\n",
           lo->name, event, data ? (const char *)data : "无");
}

static const ObserverVTable log_observer_vtable = {
    .on_notify = log_observer_on_notify
};

LogObserver* log_observer_create(const char *name) {
    LogObserver *lo = malloc(sizeof(LogObserver));
    if (lo) {
        lo->base.vtable = &log_observer_vtable;
        strncpy(lo->name, name, sizeof(lo->name) - 1);
    }
    return lo;
}

/* ==================== 具体观察者2：统计观察者 ==================== */

typedef struct {
    Observer base;
    int event_count;
} StatsObserver;

static void stats_observer_on_notify(Observer *self, const char *event, const void *data) {
    StatsObserver *so = (StatsObserver *)self;
    so->event_count++;
    printf("[统计观察者] 累计收到 %d 个事件 (本次: %s)\n",
           so->event_count, event);
}

static const ObserverVTable stats_observer_vtable = {
    .on_notify = stats_observer_on_notify
};

StatsObserver* stats_observer_create(void) {
    StatsObserver *so = calloc(1, sizeof(StatsObserver));
    if (so) {
        so->base.vtable = &stats_observer_vtable;
    }
    return so;
}

/* ==================== 使用示例：温度传感器 ==================== */

int main(void) {
    printf("========== 观察者模式演示 ==========\n\n");

    /* 创建主题 */
    Subject sensor;
    subject_init(&sensor, "温度传感器");

    /* 创建观察者 */
    LogObserver *logger = log_observer_create("温度日志");
    StatsObserver *stats = stats_observer_create();

    /* 注册观察者 */
    subject_attach(&sensor, (Observer *)logger);
    subject_attach(&sensor, (Observer *)stats);

    /* 模拟温度变化 */
    printf("--- 温度上升 ---\n");
    subject_notify(&sensor, "TEMP_CHANGED", "25.5°C");

    printf("\n--- 温度继续上升 ---\n");
    subject_notify(&sensor, "TEMP_CHANGED", "30.2°C");

    printf("\n--- 温度报警 ---\n");
    subject_notify(&sensor, "TEMP_ALARM", "40.0°C");

    /* 注销日志观察者 */
    printf("\n--- 注销日志观察者 ---\n");
    subject_detach(&sensor, (Observer *)logger);

    printf("\n--- 再次温度变化 ---\n");
    subject_notify(&sensor, "TEMP_CHANGED", "28.0°C");

    /* 清理 */
    free(logger);
    free(stats);

    return 0;
}
```

### 4.3 注意事项

| 要点 | 说明 |
| --- | --- |
| **避免循环引用** | A观察B，B又观察A，通知会无限循环 |
| **通知顺序不确定** | 不要假设观察者接收通知的顺序 |
| **观察者中不要注销自己** | 通知过程中修改列表可能导致崩溃 |
| **内存管理** | 主题不拥有观察者，只持有引用 |

## 5. 策略模式

### 5.1 通俗理解

> 💡 **就像"出行策略"**

去北京出差，你可以选择：
- 坐飞机（快但贵）
- 坐高铁（适中）
- 自驾（慢但自由）

**策略模式的核心：**
- 把算法封装成独立的策略对象
- 运行时可以切换策略
- 新增策略不影响已有代码

### 5.2 C语言实现

```c
/*
 * ============================================================
 * 策略模式：将算法封装为可互换的策略
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 策略接口：排序策略 ==================== */

typedef struct SortStrategy SortStrategy;

typedef struct {
    void (*sort)(int *arr, int n);
    const char* (*name)(void);
} SortStrategyVTable;

struct SortStrategy {
    const SortStrategyVTable *vtable;
};

/* ==================== 具体策略1：冒泡排序 ==================== */

static void bubble_sort(int *arr, int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int tmp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = tmp;
            }
        }
    }
}

static const char* bubble_sort_name(void) { return "冒泡排序"; }

static const SortStrategyVTable bubble_sort_vtable = {
    .sort = bubble_sort,
    .name = bubble_sort_name
};

static SortStrategy bubble_strategy = { .vtable = &bubble_sort_vtable };

/* ==================== 具体策略2：选择排序 ==================== */

static void selection_sort(int *arr, int n) {
    for (int i = 0; i < n - 1; i++) {
        int min_idx = i;
        for (int j = i + 1; j < n; j++) {
            if (arr[j] < arr[min_idx]) {
                min_idx = j;
            }
        }
        if (min_idx != i) {
            int tmp = arr[i];
            arr[i] = arr[min_idx];
            arr[min_idx] = tmp;
        }
    }
}

static const char* selection_sort_name(void) { return "选择排序"; }

static const SortStrategyVTable selection_sort_vtable = {
    .sort = selection_sort,
    .name = selection_sort_name
};

static SortStrategy selection_strategy = { .vtable = &selection_sort_vtable };

/* ==================== 具体策略3：插入排序 ==================== */

static void insertion_sort(int *arr, int n) {
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

static const char* insertion_sort_name(void) { return "插入排序"; }

static const SortStrategyVTable insertion_sort_vtable = {
    .sort = insertion_sort,
    .name = insertion_sort_name
};

static SortStrategy insertion_strategy = { .vtable = &insertion_sort_vtable };

/* ==================== 上下文：排序器 ==================== */

typedef struct {
    SortStrategy *strategy;  /* 当前使用的策略 */
} Sorter;

void sorter_set_strategy(Sorter *sorter, SortStrategy *strategy) {
    sorter->strategy = strategy;
}

void sorter_sort(Sorter *sorter, int *arr, int n) {
    if (sorter->strategy) {
        printf("使用策略: %s\n", sorter->strategy->vtable->name());
        sorter->strategy->vtable->sort(arr, n);
    }
}

/* ==================== 辅助函数 ==================== */

void print_array(const int *arr, int n) {
    printf("[");
    for (int i = 0; i < n; i++) {
        printf("%d", arr[i]);
        if (i < n - 1) printf(", ");
    }
    printf("]\n");
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 策略模式演示 ==========\n\n");

    int data[] = {64, 25, 12, 22, 11, 90, 45};
    int n = sizeof(data) / sizeof(data[0]);

    Sorter sorter = {0};

    /* 策略1：冒泡排序 */
    int arr1[7];
    memcpy(arr1, data, sizeof(data));
    printf("原始数据: ");
    print_array(arr1, n);

    sorter_set_strategy(&sorter, &bubble_strategy);
    sorter_sort(&sorter, arr1, n);
    printf("排序结果: ");
    print_array(arr1, n);
    printf("\n");

    /* 策略2：选择排序 */
    int arr2[7];
    memcpy(arr2, data, sizeof(data));
    sorter_set_strategy(&sorter, &selection_strategy);
    sorter_sort(&sorter, arr2, n);
    printf("排序结果: ");
    print_array(arr2, n);
    printf("\n");

    /* 策略3：插入排序 */
    int arr3[7];
    memcpy(arr3, data, sizeof(data));
    sorter_set_strategy(&sorter, &insertion_strategy);
    sorter_sort(&sorter, arr3, n);
    printf("排序结果: ");
    print_array(arr3, n);

    return 0;
}
```

### 5.3 注意事项

| 要点 | 说明 |
| --- | --- |
| **策略无状态** | 策略对象通常不需要保存状态，可以共享实例 |
| **运行时切换** | 策略模式的核心优势是运行时切换算法 |
| **开闭原则** | 新增策略只需新增代码，不修改现有代码 |
| **策略过多时** | 如果策略太多，考虑用函数指针数组代替 |

## 6. 工厂模式

### 6.1 通俗理解

> 💡 **就像"点外卖"**

- 你不需要知道厨房怎么做菜（不需要知道对象怎么创建）
- 你只需要说"我要一份宫保鸡丁"（告诉工厂要什么类型）
- 厨房做好后给你（工厂返回创建好的对象）

**工厂模式的好处：**
- 创建逻辑集中管理
- 调用者不需要知道具体类名
- 新增类型只需修改工厂

### 6.2 简单工厂

```c
/*
 * ============================================================
 * 简单工厂模式：根据类型创建对象
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 产品接口 ==================== */

typedef struct Document Document;

typedef struct {
    void (*open)(Document *self);
    void (*save)(Document *self);
    void (*close)(Document *self);
    void (*destroy)(Document *self);
} DocumentVTable;

struct Document {
    const DocumentVTable *vtable;
    char filepath[256];
};

/* ==================== 具体产品：文本文档 ==================== */

typedef struct {
    Document base;
    char *content;
    size_t content_size;
} TextDocument;

static void text_doc_open(Document *self) {
    TextDocument *td = (TextDocument *)self;
    printf("打开文本文档: %s\n", self->filepath);
    /* 模拟读取内容 */
    td->content = strdup("这是文本内容...");
    td->content_size = strlen(td->content);
}

static void text_doc_save(Document *self) {
    printf("保存文本文档: %s (%zu 字节)\n",
           self->filepath,
           ((TextDocument *)self)->content_size);
}

static void text_doc_close(Document *self) {
    TextDocument *td = (TextDocument *)self;
    printf("关闭文本文档: %s\n", self->filepath);
    free(td->content);
    td->content = NULL;
}

static void text_doc_destroy(Document *self) {
    TextDocument *td = (TextDocument *)self;
    free(td->content);
    free(td);
}

static const DocumentVTable text_doc_vtable = {
    .open = text_doc_open,
    .save = text_doc_save,
    .close = text_doc_close,
    .destroy = text_doc_destroy
};

/* ==================== 具体产品：表格文档 ==================== */

typedef struct {
    Document base;
    int rows;
    int cols;
} SpreadsheetDocument;

static void sheet_doc_open(Document *self) {
    SpreadsheetDocument *sd = (SpreadsheetDocument *)self;
    printf("打开表格文档: %s (%dx%d)\n", self->filepath, sd->rows, sd->cols);
}

static void sheet_doc_save(Document *self) {
    printf("保存表格文档: %s\n", self->filepath);
}

static void sheet_doc_close(Document *self) {
    printf("关闭表格文档: %s\n", self->filepath);
}

static void sheet_doc_destroy(Document *self) {
    free(self);
}

static const DocumentVTable sheet_doc_vtable = {
    .open = sheet_doc_open,
    .save = sheet_doc_save,
    .close = sheet_doc_close,
    .destroy = sheet_doc_destroy
};

/* ==================== 简单工厂 ==================== */

typedef enum {
    DOC_TEXT,
    DOC_SPREADSHEET
} DocType;

Document* document_factory_create(DocType type, const char *filepath) {
    Document *doc = NULL;

    switch (type) {
    case DOC_TEXT: {
        TextDocument *td = calloc(1, sizeof(TextDocument));
        if (td) {
            td->base.vtable = &text_doc_vtable;
            strncpy(td->base.filepath, filepath, sizeof(td->base.filepath) - 1);
            doc = (Document *)td;
        }
        break;
    }
    case DOC_SPREADSHEET: {
        SpreadsheetDocument *sd = calloc(1, sizeof(SpreadsheetDocument));
        if (sd) {
            sd->base.vtable = &sheet_doc_vtable;
            strncpy(sd->base.filepath, filepath, sizeof(sd->base.filepath) - 1);
            sd->rows = 100;
            sd->cols = 26;
            doc = (Document *)sd;
        }
        break;
    }
    default:
        printf("错误: 未知的文档类型 %d\n", type);
        break;
    }

    return doc;
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 工厂模式演示 ==========\n\n");

    /* 通过工厂创建不同类型的文档 */
    Document *docs[2];

    docs[0] = document_factory_create(DOC_TEXT, "report.txt");
    docs[1] = document_factory_create(DOC_SPREADSHEET, "data.csv");

    /* 统一操作 */
    for (int i = 0; i < 2; i++) {
        if (docs[i]) {
            docs[i]->vtable->open(docs[i]);
            docs[i]->vtable->save(docs[i]);
            docs[i]->vtable->close(docs[i]);
            docs[i]->vtable->destroy(docs[i]);
            printf("\n");
        }
    }

    return 0;
}
```

```c
#include <string.h>

struct Transport {

static void truck_deliver(const Transport *self) {
    (void)self;
    printf("卡车运输：通过公路配送货物\n");
}

static const char* truck_name(void) { return "卡车"; }

static const TransportVTable truck_vtable = {
    .deliver = truck_deliver,
    .name = truck_name
};

/* 具体产品：轮船 */
static void ship_deliver(const Transport *self) {
    (void)self;
    printf("轮船运输：通过海运配送货物\n");
}

static const char* ship_name(void) { return "轮船"; }

static const TransportVTable ship_vtable = {
    .deliver = ship_deliver,
    .name = ship_name
};

/* ==================== 工厂接口 ==================== */

typedef struct Logistics Logistics;

typedef struct {
    Transport* (*create_transport)(void);
    void (*plan_delivery)(Logistics *self);
} LogisticsVTable;

struct Logistics {
    const LogisticsVTable *vtable;
};

/* 默认的配送计划 */
static void logistics_plan_delivery(Logistics *self) {
    Transport *t = self->vtable->create_transport();
    if (t) {
        printf("使用 %s 进行配送:\n", t->vtable->name());
        t->vtable->deliver(t);
        free(t);
    }
}

/* 具体工厂：公路物流 */
static Transport* road_create_transport(void) {
    Transport *t = malloc(sizeof(Transport));
    if (t) t->vtable = &truck_vtable;
    return t;
}

static const LogisticsVTable road_logistics_vtable = {
    .create_transport = road_create_transport,
    .plan_delivery = logistics_plan_delivery
};

/* 具体工厂：海运物流 */
static Transport* sea_create_transport(void) {
    Transport *t = malloc(sizeof(Transport));
    if (t) t->vtable = &ship_vtable;
    return t;
}

static const LogisticsVTable sea_logistics_vtable = {
    .create_transport = sea_create_transport,
    .plan_delivery = logistics_plan_delivery
};

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 工厂方法模式演示 ==========\n\n");

    /* 公路物流 */
    Logistics road = { .vtable = &road_logistics_vtable };
    printf("--- 公路物流 ---\n");
    road.vtable->plan_delivery(&road);
    printf("\n");

    /* 海运物流 */
    Logistics sea = { .vtable = &sea_logistics_vtable };
    printf("--- 海运物流 ---\n");
    sea.vtable->plan_delivery(&sea);

    return 0;
}
```

### 6.4 注意事项

| 要点 | 说明 |
| --- | --- |
| **简单工厂 vs 工厂方法** | 简单工厂用switch，工厂方法用虚函数 |
| **新增产品类型** | 简单工厂需修改工厂代码，工厂方法只需新增子类 |
| **工厂通常是单例** | 工厂对象本身不需要多个实例 |
| **返回值检查** | 工厂可能返回NULL，调用者必须检查 |

## 7. 对象生命周期管理（引用计数）

### 7.1 通俗理解

> 💡 **就像"共享自行车"**

- 有人借了一辆车 → 计数+1
- 有人还了一辆车 → 计数-1
- 计数为0 → 没人用了，回收车辆

**为什么需要引用计数？**

- 多个对象共享同一资源时，谁负责释放？
- 释放早了 → 其他对象访问已释放的内存（悬空指针）
- 释放晚了 → 内存泄漏
- 引用计数自动解决这个难题

### 7.2 引用计数实现

```c
/*
 * ============================================================
 * 引用计数：自动管理共享对象的生命周期
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 引用计数基类 ==================== */

typedef struct RefObject RefObject;

typedef struct {
    void (*destroy)(RefObject *self);  /* 析构函数 */
} RefObjectVTable;

struct RefObject {
    const RefObjectVTable *vtable;
    int ref_count;           /* 引用计数 */
    char type_name[32];      /* 类型名（调试用） */
};

/* 初始化引用计数对象 */
static void ref_object_init(RefObject *obj, const RefObjectVTable *vtable,
                             const char *type_name) {
    obj->vtable = vtable;
    obj->ref_count = 1;  /* 创建时引用计数为1 */
    strncpy(obj->type_name, type_name, sizeof(obj->type_name) - 1);
}

/* 增加引用 */
static RefObject* ref_object_retain(RefObject *obj) {
    if (obj) {
        obj->ref_count++;
        printf("  [引用] %s ref_count=%d\n", obj->type_name, obj->ref_count);
    }
    return obj;
}

/* 释放引用 */
static void ref_object_release(RefObject *obj) {
    if (!obj) return;

    obj->ref_count--;
    printf("  [释放] %s ref_count=%d\n", obj->type_name, obj->ref_count);

    if (obj->ref_count <= 0) {
        printf("  [销毁] %s 引用计数归零，释放资源\n", obj->type_name);
        if (obj->vtable && obj->vtable->destroy) {
            obj->vtable->destroy(obj);
        }
    }
}

/* ==================== 具体类：共享缓冲区 ==================== */

typedef struct {
    RefObject base;       /* 引用计数基类 */
    char *data;           /* 缓冲区数据 */
    size_t size;          /* 缓冲区大小 */
} SharedBuffer;

static void shared_buffer_destroy(RefObject *self) {
    SharedBuffer *buf = (SharedBuffer *)self;
    printf("  释放缓冲区数据 (%zu 字节)\n", buf->size);
    free(buf->data);
    free(buf);
}

static const RefObjectVTable shared_buffer_vtable = {
    .destroy = shared_buffer_destroy
};

/* 创建共享缓冲区 */
SharedBuffer* shared_buffer_create(size_t size) {
    SharedBuffer *buf = malloc(sizeof(SharedBuffer));
    if (!buf) return NULL;

    ref_object_init(&buf->base, &shared_buffer_vtable, "SharedBuffer");
    buf->data = malloc(size);
    buf->size = size;

    if (!buf->data) {
        free(buf);
        return NULL;
    }

    memset(buf->data, 0, size);
    printf("创建共享缓冲区 (%zu 字节), ref_count=1\n", size);
    return buf;
}

/* 便捷宏 */
#define RETAIN(obj) ref_object_retain((RefObject *)(obj))
#define RELEASE(obj) ref_object_release((RefObject *)(obj))

/* ==================== 使用示例：多个消费者共享缓冲区 ==================== */

typedef struct {
    const char *name;
    SharedBuffer *buffer;  /* 共享的缓冲区 */
} Consumer;

Consumer* consumer_create(const char *name, SharedBuffer *buffer) {
    Consumer *c = malloc(sizeof(Consumer));
    if (c) {
        c->name = name;
        c->buffer = buffer;
        /* 持有缓冲区引用 */
        RETAIN(buffer);
        printf("消费者 [%s] 创建，持有缓冲区\n", name);
    }
    return c;
}

void consumer_destroy(Consumer *c) {
    if (c) {
        printf("消费者 [%s] 销毁，释放缓冲区引用\n", c->name);
        RELEASE(c->buffer);
        free(c);
    }
}

void consumer_use_buffer(Consumer *c) {
    if (c && c->buffer) {
        printf("消费者 [%s] 使用缓冲区 (size=%zu)\n",
               c->name, c->buffer->size);
    }
}

/* ==================== 演示 ==================== */

int main(void) {
    printf("========== 引用计数演示 ==========\n\n");

    /* 创建共享缓冲区 (ref_count=1) */
    printf("--- 创建共享缓冲区 ---\n");
    SharedBuffer *buffer = shared_buffer_create(1024);

    /* 消费者1获取缓冲区 (ref_count=2) */
    printf("\n--- 消费者1获取缓冲区 ---\n");
    Consumer *c1 = consumer_create("读取器A", buffer);

    /* 消费者2获取缓冲区 (ref_count=3) */
    printf("\n--- 消费者2获取缓冲区 ---\n");
    Consumer *c2 = consumer_create("读取器B", buffer);

    /* 使用缓冲区 */
    printf("\n--- 使用缓冲区 ---\n");
    consumer_use_buffer(c1);
    consumer_use_buffer(c2);

    /* 消费者1销毁 (ref_count=2) */
    printf("\n--- 消费者1销毁 ---\n");
    consumer_destroy(c1);

    /* 缓冲区仍然有效，消费者2可以继续使用 */
    printf("\n--- 消费者2继续使用 ---\n");
    consumer_use_buffer(c2);

    /* 消费者2销毁 (ref_count=1) */
    printf("\n--- 消费者2销毁 ---\n");
    consumer_destroy(c2);

    /* 释放原始引用 (ref_count=0 → 自动销毁) */
    printf("\n--- 释放原始引用 ---\n");
    RELEASE(buffer);

    printf("\n========== 演示完成 ==========\n");
    return 0;
}
```

### 7.3 引用计数的陷阱与解决

```c
/*
 * ============================================================
 * 引用计数的常见陷阱
 * ============================================================
 */

/*
 * ❌ 陷阱1：循环引用
 *
 *   A → B → A  （互相引用，计数永远不为0）
 *
 * 解决方案：
 *   - 使用"弱引用"（不增加计数）
 *   - 明确所有权（一个用强引用，一个用弱引用）
 */

/* 弱引用示例 */
typedef struct {
    RefObject *target;   /* 目标对象 */
    int is_weak;         /* 是否为弱引用 */
} Ref;

static Ref ref_strong(RefObject *obj) {
    Ref r = { .target = ref_object_retain(obj), .is_weak = 0 };
    return r;
}

static Ref ref_weak(RefObject *obj) {
    Ref r = { .target = obj, .is_weak = 1 };  /* 不增加计数 */
    return r;
}

static void ref_release(Ref *r) {
    if (r->is_weak) {
        r->target = NULL;  /* 弱引用只清空指针 */
    } else {
        ref_object_release(r->target);
        r->target = NULL;
    }
}

/*
 * ❌ 陷阱2：线程安全
 *
 *   多线程同时 retain/release 会导致计数错误
 *
 * 解决方案：
 *   - 使用原子操作（C11 stdatomic.h）
 *   - 或使用互斥锁保护
 */

/* C11 原子操作版本 */
#ifdef __STDC_NO_ATOMICS__
/* 如果不支持原子操作，使用互斥锁 */
#else
#include <stdatomic.h>

typedef struct {
    const RefObjectVTable *vtable;
    _Atomic int ref_count;  /* 原子引用计数 */
    char type_name[32];
} AtomicRefObject;

static AtomicRefObject* atomic_retain(AtomicRefObject *obj) {
    if (obj) {
        atomic_fetch_add(&obj->ref_count, 1);
    }
    return obj;
}

static void atomic_release(AtomicRefObject *obj) {
    if (obj && atomic_fetch_sub(&obj->ref_count, 1) == 1) {
        /* 计数归零，释放资源 */
        if (obj->vtable && obj->vtable->destroy) {
            obj->vtable->destroy((RefObject *)obj);
        }
    }
}
#endif
```

### 7.4 注意事项

| 要点 | 说明 |
| --- | --- |
| **创建即持有** | 对象创建时 ref_count=1，创建者负责最终 release |
| **retain/release 配对** | 每个 retain 必须对应一个 release |
| **循环引用** | 两个对象互相强引用会导致内存泄漏，用弱引用打破 |
| **线程安全** | 多线程环境必须使用原子操作或锁保护计数 |
| **调试技巧** | 可以记录 retain/release 的调用位置，帮助排查泄漏 |

## 8. C语言面向对象的局限性与最佳实践

### 8.1 局限性

```
┌─────────────────────────────────────────────────────────────────────────┐
│                    C语言面向对象的局限性                                 │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  1. 没有编译期类型检查                                                  │
│     • 函数指针转换没有类型安全保证                                      │
│     • 错误的类型转换只能在运行时发现                                    │
│                                                                         │
│  2. 没有自动内存管理                                                    │
│     • 没有 RAII，没有析构函数自动调用                                   │
│     • 每个对象必须手动 create/destroy                                   │
│                                                                         │
│  3. 没有访问控制                                                        │
│     • struct 成员默认全部公开                                           │
│     • 只能通过不透明指针模拟 private                                    │
│                                                                         │
│  4. 没有模板/泛型                                                       │
│     • 无法实现类型安全的容器                                            │
│     • 只能用 void* 或宏模拟                                            │
│                                                                         │
│  5. 没有异常处理                                                        │
│     • 错误只能通过返回值传递                                            │
│     • 构造函数失败只能返回 NULL                                         │
│                                                                         │
│  6. 代码量较大                                                          │
│     • 同样的功能，C需要更多代码                                         │
│     • 大量样板代码（vtable、create、destroy等）                         │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

### 8.2 最佳实践

```c
/*
 * ============================================================
 * C语言面向对象最佳实践总结
 * ============================================================
 */

/* ==================== 1. 头文件设计 ==================== */

/*
 * ✅ 好的头文件设计：
 *    - 只暴露必要的类型声明和函数声明
 *    - 内部结构体在 .c 文件中定义
 *    - 使用前向声明实现不透明指针
 */

/* widget.h - 公共接口 */
#ifndef WIDGET_H
#define WIDGET_H

typedef struct Widget Widget;  /* 不透明类型 */

Widget* widget_create(int width, int height);
void    widget_destroy(Widget *w);
void    widget_resize(Widget *w, int width, int height);
int     widget_get_width(const Widget *w);
int     widget_get_height(const Widget *w);

#endif

/*
 * widget.c - 实现细节（用户看不到）
 *
 * struct Widget {
 *     const WidgetVTable *vtable;
 *     int width, height;
 *     void *internal_data;   // 私有数据
 * };
 */

/* ==================== 2. 错误处理策略 ==================== */

/* 策略1：返回错误码 */
typedef enum {
    WIDGET_OK = 0,
    WIDGET_ERR_NULL = -1,
    WIDGET_ERR_PARAM = -2,
    WIDGET_ERR_MEMORY = -3
} WidgetError;

WidgetError widget_safe_create(Widget **out, int w, int h) {
    if (!out) return WIDGET_ERR_NULL;
    if (w <= 0 || h <= 0) return WIDGET_ERR_PARAM;

    *out = widget_create(w, h);
    return *out ? WIDGET_OK : WIDGET_ERR_MEMORY;
}

/* 策略2：返回NULL + 日志 */
Widget* widget_create_or_null(int w, int h) {
    if (w <= 0 || h <= 0) {
        fprintf(stderr, "widget_create: 无效参数 w=%d h=%d\n", w, h);
        return NULL;
    }
    return widget_create(w, h);
}

/* ==================== 3. 宏辅助减少样板代码 ==================== */

/* 通用对象头部宏 - 减少重复定义 */
#define OBJECT_HEADER(Type)       \
    const Type##VTable *vtable;   \
    int ref_count

/* 通用创建/销毁宏 */
#define OBJECT_CREATE(Type, ...)  \
    Type *obj = calloc(1, sizeof(Type)); \
    if (obj) {                           \
        obj->vtable = &Type##_vtable;    \
        obj->ref_count = 1;              \
        Type##_init(obj, ##__VA_ARGS__); \
    }                                    \
    return obj

#define OBJECT_DESTROY(Type, obj) \
    do {                          \
        Type##_cleanup(obj);      \
        free(obj);                \
    } while(0)

/* ==================== 4. 内存安全检查 ==================== */

/* 调试模式下的内存跟踪 */
#ifdef DEBUG_MEMORY

typedef struct {
    const char *file;
    int line;
    size_t size;
    void *ptr;
} MemRecord;

#define TRACKED_MALLOC(size) tracked_malloc(size, __FILE__, __LINE__)
#define TRACKED_FREE(ptr) tracked_free(ptr)

static void* tracked_malloc(size_t size, const char *file, int line) {
    void *ptr = malloc(size);
    printf("[MEM] 分配 %zu 字节 at %s:%d → %p\n", size, file, line, ptr);
    return ptr;
}

static void tracked_free(void *ptr) {
    printf("[MEM] 释放 %p\n", ptr);
    free(ptr);
}

#else
#define TRACKED_MALLOC(size) malloc(size)
#define TRACKED_FREE(ptr) free(ptr)
#endif
```
