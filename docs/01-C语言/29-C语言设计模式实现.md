## 1. 单例模式

### 问题场景

> 💡 系统中某些对象只需要一个实例：配置管理器、日志系统、数据库连接池。如果创建多个实例，会导致资源浪费或状态不一致。

### C语言实现

```c
/*
 * ============================================================
 * 单例模式：确保一个类只有一个实例
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 方式1：静态全局变量（最简单） ==================== */

typedef struct {
    char app_name[64];
    int  log_level;
    int  max_connections;
    char db_path[256];
} Config;

/* 静态全局变量 - 进程内唯一 */
static Config *g_config = NULL;

Config* config_get_instance(void) {
    if (g_config == NULL) {
        g_config = calloc(1, sizeof(Config));
        if (g_config) {
            /* 默认配置 */
            strcpy(g_config->app_name, "MyApp");
            g_config->log_level = 1;
            g_config->max_connections = 100;
            strcpy(g_config->db_path, "/tmp/app.db");
            printf("[Config] 初始化配置单例\n");
        }
    }
    return g_config;
}

void config_set_app_name(const char *name) {
    Config *cfg = config_get_instance();
    if (cfg) {
        strncpy(cfg->app_name, name, sizeof(cfg->app_name) - 1);
    }
}

void config_print(const Config *cfg) {
    if (!cfg) return;
    printf("配置: app=%s, level=%d, max_conn=%d, db=%s\n",
           cfg->app_name, cfg->log_level,
           cfg->max_connections, cfg->db_path);
}

/* ==================== 方式2：线程安全版本 ==================== */

#include <pthread.h>

typedef struct {
    char server_name[64];
    int  port;
    int  is_running;
} ServerSingleton;

static ServerSingleton *g_server = NULL;
static pthread_mutex_t server_mutex = PTHREAD_MUTEX_INITIALIZER;

ServerSingleton* server_get_instance(void) {
    if (g_server == NULL) {
        pthread_mutex_lock(&server_mutex);
        /* 双重检查：加锁后再检查一次 */
        if (g_server == NULL) {
            g_server = calloc(1, sizeof(ServerSingleton));
            if (g_server) {
                strcpy(g_server->server_name, "DefaultServer");
                g_server->port = 8080;
                g_server->is_running = 0;
                printf("[Server] 初始化服务器单例\n");
            }
        }
        pthread_mutex_unlock(&server_mutex);
    }
    return g_server;
}

void server_start(ServerSingleton *srv) {
    if (srv && !srv->is_running) {
        srv->is_running = 1;
        printf("[Server] %s 启动于端口 %d\n", srv->server_name, srv->port);
    }
}

void server_stop(ServerSingleton *srv) {
    if (srv && srv->is_running) {
        srv->is_running = 0;
        printf("[Server] %s 已停止\n", srv->server_name);
    }
}

/* ==================== 方式3：编译期单例（嵌入式常用） ==================== */

/* 全局唯一实例，编译期分配 */
static Config g_app_config = {
    .app_name = "EmbeddedApp",
    .log_level = 0,
    .max_connections = 10,
    .db_path = "/flash/app.db"
};

Config* config_get_global(void) {
    return &g_app_config;  /* 返回静态实例的指针 */
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 单例模式演示 ==========\n\n");

    /* 方式1：懒加载单例 */
    printf("--- 配置单例 ---\n");
    Config *cfg1 = config_get_instance();
    Config *cfg2 = config_get_instance();
    printf("cfg1 == cfg2 ? %s\n", cfg1 == cfg2 ? "是" : "否");
    config_set_app_name("TestApp");
    config_print(cfg2);  /* cfg2 也能看到修改 */

    /* 方式2：线程安全单例 */
    printf("\n--- 服务器单例 ---\n");
    ServerSingleton *srv = server_get_instance();
    server_start(srv);
    server_stop(srv);

    /* 方式3：编译期单例 */
    printf("\n--- 全局配置 ---\n");
    Config *gcfg = config_get_global();
    config_print(gcfg);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 全局唯一实例，避免重复创建 | 全局状态，增加耦合 |
| 延迟初始化，节省资源 | 不易测试（难以 mock） |
| 提供全局访问点 | 多线程需额外同步 |
| 嵌入式可用编译期单例 | 生命周期难控制 |

## 2. 工厂方法模式

### 问题场景

> 💡 日志系统需要支持多种输出方式（文件、控制台、网络），但调用者不应该关心具体创建哪种日志器。新增日志类型时，不应修改调用者代码。

### C语言实现

```c
/*
 * ============================================================
 * 工厂方法模式：让子类决定创建哪个产品
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 产品接口 ==================== */

typedef struct Logger Logger;

typedef struct {
    void (*log)(Logger *self, const char *msg);
    void (*destroy)(Logger *self);
} LoggerVTable;

struct Logger {
    const LoggerVTable *vtable;
    int level;
};

/* ==================== 具体产品：文件日志 ==================== */

typedef struct {
    Logger base;
    FILE *fp;
    char filepath[256];
} FileLogger;

static void file_logger_log(Logger *self, const char *msg) {
    FileLogger *fl = (FileLogger *)self;
    if (fl->fp) {
        fprintf(fl->fp, "[FILE] %s\n", msg);
        fflush(fl->fp);
    }
}

static void file_logger_destroy(Logger *self) {
    FileLogger *fl = (FileLogger *)self;
    if (fl->fp) fclose(fl->fp);
    free(fl);
}

static const LoggerVTable file_logger_vtable = {
    .log = file_logger_log,
    .destroy = file_logger_destroy
};

/* ==================== 具体产品：控制台日志 ==================== */

typedef struct {
    Logger base;
    int use_color;
} ConsoleLogger;

static void console_logger_log(Logger *self, const char *msg) {
    ConsoleLogger *cl = (ConsoleLogger *)self;
    if (cl->use_color) {
        printf("\033[32m[CONSOLE] %s\033[0m\n", msg);
    } else {
        printf("[CONSOLE] %s\n", msg);
    }
}

static void console_logger_destroy(Logger *self) {
    free(self);
}

static const LoggerVTable console_logger_vtable = {
    .log = console_logger_log,
    .destroy = console_logger_destroy
};

/* ==================== 具体产品：网络日志 ==================== */

typedef struct {
    Logger base;
    char server[128];
    int port;
} NetworkLogger;

static void network_logger_log(Logger *self, const char *msg) {
    NetworkLogger *nl = (NetworkLogger *)self;
    printf("[NETWORK → %s:%d] %s\n", nl->server, nl->port, msg);
    /* 实际实现会使用socket发送 */
}

static void network_logger_destroy(Logger *self) {
    free(self);
}

static const LoggerVTable network_logger_vtable = {
    .log = network_logger_log,
    .destroy = network_logger_destroy
};

/* ==================== 工厂接口 ==================== */

typedef struct LoggerFactory LoggerFactory;

typedef struct {
    Logger* (*create_logger)(LoggerFactory *self);
    void (*destroy)(LoggerFactory *self);
} LoggerFactoryVTable;

struct LoggerFactory {
    const LoggerFactoryVTable *vtable;
};

/* 具体工厂：文件日志工厂 */
typedef struct {
    LoggerFactory base;
    char filepath[256];
} FileLoggerFactory;

static Logger* file_factory_create(LoggerFactory *self) {
    FileLoggerFactory *ff = (FileLoggerFactory *)self;
    FileLogger *fl = calloc(1, sizeof(FileLogger));
    if (fl) {
        fl->base.vtable = &file_logger_vtable;
        strncpy(fl->filepath, ff->filepath, sizeof(fl->filepath) - 1);
        fl->fp = fopen(fl->filepath, "a");
    }
    return (Logger *)fl;
}

static const LoggerFactoryVTable file_factory_vtable = {
    .create_logger = file_factory_create,
    .destroy = NULL
};

/* 具体工厂：控制台日志工厂 */
typedef struct {
    LoggerFactory base;
    int use_color;
} ConsoleLoggerFactory;

static Logger* console_factory_create(LoggerFactory *self) {
    ConsoleLoggerFactory *cf = (ConsoleLoggerFactory *)self;
    ConsoleLogger *cl = calloc(1, sizeof(ConsoleLogger));
    if (cl) {
        cl->base.vtable = &console_logger_vtable;
        cl->use_color = cf->use_color;
    }
    return (Logger *)cl;
}

static const LoggerFactoryVTable console_factory_vtable = {
    .create_logger = console_factory_create,
    .destroy = NULL
};

/* 具体工厂：网络日志工厂 */
typedef struct {
    LoggerFactory base;
    char server[128];
    int port;
} NetworkLoggerFactory;

static Logger* network_factory_create(LoggerFactory *self) {
    NetworkLoggerFactory *nf = (NetworkLoggerFactory *)self;
    NetworkLogger *nl = calloc(1, sizeof(NetworkLogger));
    if (nl) {
        nl->base.vtable = &network_logger_vtable;
        strncpy(nl->server, nf->server, sizeof(nl->server) - 1);
        nl->port = nf->port;
    }
    return (Logger *)nl;
}

static const LoggerFactoryVTable network_factory_vtable = {
    .create_logger = network_factory_create,
    .destroy = NULL
};

/* ==================== 使用示例 ==================== */

void use_logger(LoggerFactory *factory) {
    Logger *logger = factory->vtable->create_logger(factory);
    if (logger) {
        logger->vtable->log(logger, "系统启动");
        logger->vtable->log(logger, "执行任务...");
        logger->vtable->log(logger, "系统关闭");
        logger->vtable->destroy(logger);
    }
}

int main(void) {
    printf("========== 工厂方法模式演示 ==========\n\n");

    /* 文件日志工厂 */
    FileLoggerFactory file_factory = {
        .base.vtable = &file_factory_vtable,
        .filepath = "app.log"
    };
    printf("--- 文件日志 ---\n");
    use_logger((LoggerFactory *)&file_factory);

    /* 控制台日志工厂 */
    ConsoleLoggerFactory console_factory = {
        .base.vtable = &console_factory_vtable,
        .use_color = 1
    };
    printf("\n--- 控制台日志 ---\n");
    use_logger((LoggerFactory *)&console_factory);

    /* 网络日志工厂 */
    NetworkLoggerFactory network_factory = {
        .base.vtable = &network_factory_vtable,
        .server = "log.example.com",
        .port = 514
    };
    printf("\n--- 网络日志 ---\n");
    use_logger((LoggerFactory *)&network_factory);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 符合开闭原则 | 每个产品需要一个工厂类 |
| 调用者不需要知道具体类 | 代码量增加 |
| 新增产品只需新增工厂 | 工厂类可能很相似 |

## 3. 抽象工厂模式

### 问题场景

> 💡 跨平台UI库需要为不同操作系统创建按钮、文本框等控件。Windows和Linux的控件外观不同，但接口一致。需要确保同一平台的所有控件风格统一。

### C语言实现

```c
/*
 * ============================================================
 * 抽象工厂模式：创建一系列相关对象
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 抽象产品：按钮 ==================== */

typedef struct Button Button;

typedef struct {
    void (*render)(const Button *self);
    void (*set_text)(Button *self, const char *text);
    void (*destroy)(Button *self);
} ButtonVTable;

struct Button {
    const ButtonVTable *vtable;
    char text[64];
};

/* ==================== 抽象产品：文本框 ==================== */

typedef struct TextBox TextBox;

typedef struct {
    void (*render)(const TextBox *self);
    void (*set_text)(TextBox *self, const char *text);
    const char* (*get_text)(const TextBox *self);
    void (*destroy)(TextBox *self);
} TextBoxVTable;

struct TextBox {
    const TextBoxVTable *vtable;
    char text[256];
};

/* ==================== Windows风格产品 ==================== */

static void win_button_render(const Button *self) {
    printf("[Windows按钮] ┌─────────────────┐\n");
    printf("               │  %s  │\n", self->text);
    printf("               └─────────────────┘\n");
}

static void win_button_set_text(Button *self, const char *text) {
    strncpy(self->text, text, sizeof(self->text) - 1);
}

static void button_destroy(Button *self) { free(self); }

static const ButtonVTable win_button_vtable = {
    .render = win_button_render,
    .set_text = win_button_set_text,
    .destroy = button_destroy
};

static void win_textbox_render(const TextBox *self) {
    printf("[Windows文本框] ┌─────────────────────┐\n");
    printf("                │ %s│\n", self->text);
    printf("                └─────────────────────┘\n");
}

static void win_textbox_set_text(TextBox *self, const char *text) {
    strncpy(self->text, text, sizeof(self->text) - 1);
}

static const char* win_textbox_get_text(const TextBox *self) {
    return self->text;
}

static void textbox_destroy(TextBox *self) { free(self); }

static const TextBoxVTable win_textbox_vtable = {
    .render = win_textbox_render,
    .set_text = win_textbox_set_text,
    .get_text = win_textbox_get_text,
    .destroy = textbox_destroy
};

/* ==================== Linux风格产品 ==================== */

static void linux_button_render(const Button *self) {
    printf("[Linux按钮] ╔═══════════════════╗\n");
    printf("            ║  %s  ║\n", self->text);
    printf("            ╚═══════════════════╝\n");
}

static void linux_button_set_text(Button *self, const char *text) {
    strncpy(self->text, text, sizeof(self->text) - 1);
}

static const ButtonVTable linux_button_vtable = {
    .render = linux_button_render,
    .set_text = linux_button_set_text,
    .destroy = button_destroy
};

static void linux_textbox_render(const TextBox *self) {
    printf("[Linux文本框] ╔═════════════════════╗\n");
    printf("             ║ %s║\n", self->text);
    printf("             ╚═════════════════════╝\n");
}

static void linux_textbox_set_text(TextBox *self, const char *text) {
    strncpy(self->text, text, sizeof(self->text) - 1);
}

static const char* linux_textbox_get_text(const TextBox *self) {
    return self->text;
}

static const TextBoxVTable linux_textbox_vtable = {
    .render = linux_textbox_render,
    .set_text = linux_textbox_set_text,
    .get_text = linux_textbox_get_text,
    .destroy = textbox_destroy
};

/* ==================== 抽象工厂 ==================== */

typedef struct UIFactory UIFactory;

typedef struct {
    Button*  (*create_button)(void);
    TextBox* (*create_textbox)(void);
} UIFactoryVTable;

struct UIFactory {
    const UIFactoryVTable *vtable;
};

/* Windows工厂 */
static Button* win_create_button(void) {
    Button *btn = calloc(1, sizeof(Button));
    if (btn) btn->vtable = &win_button_vtable;
    return btn;
}

static TextBox* win_create_textbox(void) {
    TextBox *tb = calloc(1, sizeof(TextBox));
    if (tb) tb->vtable = &win_textbox_vtable;
    return tb;
}

static const UIFactoryVTable win_factory_vtable = {
    .create_button = win_create_button,
    .create_textbox = win_create_textbox
};

/* Linux工厂 */
static Button* linux_create_button(void) {
    Button *btn = calloc(1, sizeof(Button));
    if (btn) btn->vtable = &linux_button_vtable;
    return btn;
}

static TextBox* linux_create_textbox(void) {
    TextBox *tb = calloc(1, sizeof(TextBox));
    if (tb) tb->vtable = &linux_textbox_vtable;
    return tb;
}

static const UIFactoryVTable linux_factory_vtable = {
    .create_button = linux_create_button,
    .create_textbox = linux_create_textbox
};

/* ==================== 使用示例 ==================== */

void build_dialog(UIFactory *factory) {
    Button *btn = factory->vtable->create_button();
    TextBox *tb = factory->vtable->create_textbox();

    btn->vtable->set_text(btn, "确定");
    tb->vtable->set_text(tb, "请输入用户名...");

    btn->vtable->render(btn);
    tb->vtable->render(tb);

    btn->vtable->destroy(btn);
    tb->vtable->destroy(tb);
}

int main(void) {
    printf("========== 抽象工厂模式演示 ==========\n\n");

    printf("--- Windows 风格 ---\n");
    UIFactory win_factory = { .vtable = &win_factory_vtable };
    build_dialog(&win_factory);

    printf("\n--- Linux 风格 ---\n");
    UIFactory linux_factory = { .vtable = &linux_factory_vtable };
    build_dialog(&linux_factory);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 产品族风格一致 | 新增产品类型需修改所有工厂 |
| 符合开闭原则（新增产品族） | 代码量大 |
| 调用者与具体产品解耦 | 产品族扩展困难 |

## 4. 建造者模式

### 问题场景

> 💡 构建一个HTTP请求对象需要设置URL、方法、头部、请求体等多个参数，有些参数可选。如果用构造函数，参数太多容易搞混。

### C语言实现

```c
/*
 * ============================================================
 * 建造者模式：分步骤构建复杂对象
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 产品：HTTP请求 ==================== */

#define MAX_HEADERS 16
#define MAX_BODY_SIZE 4096

typedef struct {
    char method[16];           /* GET, POST, PUT, DELETE */
    char url[256];             /* 请求URL */
    char headers[MAX_HEADERS][128]; /* 请求头 */
    int  header_count;         /* 头部数量 */
    char body[MAX_BODY_SIZE];  /* 请求体 */
    int  timeout;              /* 超时时间(ms) */
    int  follow_redirects;     /* 是否跟随重定向 */
} HttpRequest;

void http_request_print(const HttpRequest *req) {
    printf("HTTP请求:\n");
    printf("  方法: %s\n", req->method);
    printf("  URL: %s\n", req->url);
    printf("  超时: %d ms\n", req->timeout);
    printf("  重定向: %s\n", req->follow_redirects ? "是" : "否");
    for (int i = 0; i < req->header_count; i++) {
        printf("  头部: %s\n", req->headers[i]);
    }
    if (req->body[0]) {
        printf("  请求体: %s\n", req->body);
    }
}

/* ==================== 建造者 ==================== */

typedef struct {
    HttpRequest request;  /* 正在构建的请求 */
} HttpRequestBuilder;

/* 创建建造者 */
HttpRequestBuilder* builder_create(void) {
    HttpRequestBuilder *b = calloc(1, sizeof(HttpRequestBuilder));
    if (b) {
        /* 默认值 */
        strcpy(b->request.method, "GET");
        b->request.timeout = 5000;
        b->request.follow_redirects = 1;
    }
    return b;
}

/* 设置方法 */
HttpRequestBuilder* builder_method(HttpRequestBuilder *b, const char *method) {
    if (b) strncpy(b->request.method, method, sizeof(b->request.method) - 1);
    return b;
}

/* 设置URL */
HttpRequestBuilder* builder_url(HttpRequestBuilder *b, const char *url) {
    if (b) strncpy(b->request.url, url, sizeof(b->request.url) - 1);
    return b;
}

/* 添加请求头 */
HttpRequestBuilder* builder_header(HttpRequestBuilder *b, const char *key,
                                    const char *value) {
    if (b && b->request.header_count < MAX_HEADERS) {
        snprintf(b->request.headers[b->request.header_count],
                 sizeof(b->request.headers[0]),
                 "%s: %s", key, value);
        b->request.header_count++;
    }
    return b;
}

/* 设置请求体 */
HttpRequestBuilder* builder_body(HttpRequestBuilder *b, const char *body) {
    if (b) strncpy(b->request.body, body, sizeof(b->request.body) - 1);
    return b;
}

/* 设置超时 */
HttpRequestBuilder* builder_timeout(HttpRequestBuilder *b, int timeout_ms) {
    if (b) b->request.timeout = timeout_ms;
    return b;
}

/* 设置重定向 */
HttpRequestBuilder* builder_redirects(HttpRequestBuilder *b, int follow) {
    if (b) b->request.follow_redirects = follow;
    return b;
}

/* 构建最终对象 */
HttpRequest builder_build(HttpRequestBuilder *b) {
    HttpRequest req = {0};
    if (b) {
        req = b->request;
    }
    return req;
}

/* 释放建造者 */
void builder_destroy(HttpRequestBuilder *b) {
    free(b);
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 建造者模式演示 ==========\n\n");

    /* 链式调用构建GET请求 */
    HttpRequestBuilder *b1 = builder_create();
    HttpRequest req1 = builder_build(
        builder_url(b1, "https://api.example.com/users")
    );
    printf("--- GET 请求 ---\n");
    http_request_print(&req1);
    builder_destroy(b1);

    printf("\n");

    /* 链式调用构建POST请求 */
    HttpRequestBuilder *b2 = builder_create();
    HttpRequest req2 = builder_build(
        builder_method(b2, "POST")
        ->url(b2, "https://api.example.com/users")
        ->header(b2, "Content-Type", "application/json")
        ->header(b2, "Authorization", "Bearer token123")
        ->body(b2, "{\"name\": \"张三\", \"age\": 25}")
        ->timeout(b2, 10000)
    );
    printf("--- POST 请求 ---\n");
    http_request_print(&req2);
    builder_destroy(b2);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 分步构建，参数可选 | 需要额外建造者类 |
| 链式调用，代码可读 | 建造者与产品耦合 |
| 构建逻辑集中 | 简单对象过度设计 |

## 5. 适配器模式

### 问题场景

> 💡 项目中引入了一个第三方日志库，它的接口和现有日志系统不兼容。不想修改现有代码，也不想改第三方库。需要一个"适配器"让两者协同工作。

### C语言实现

```c
/*
 * ============================================================
 * 适配器模式：让不兼容的接口协同工作
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 目标接口（现有系统） ==================== */

typedef struct AppLogger AppLogger;

typedef struct {
    void (*log_info)(AppLogger *self, const char *msg);
    void (*log_error)(AppLogger *self, const char *msg);
    void (*destroy)(AppLogger *self);
} AppLoggerVTable;

struct AppLogger {
    const AppLoggerVTable *vtable;
};

/* ==================== 被适配者（第三方库，不可修改） ==================== */

typedef struct {
    void (*write)(int severity, const char *module, const char *message);
} ThirdPartyLogger;

/* 第三方库的日志级别 */
#define TP_LOG_INFO  0
#define TP_LOG_ERROR 3

static void third_party_write(int severity, const char *module, const char *message) {
    const char *level_str = (severity == TP_LOG_INFO) ? "INFO" :
                            (severity == TP_LOG_ERROR) ? "ERROR" : "UNKNOWN";
    printf("[第三方日志][%s][%s] %s\n", level_str, module, message);
}

static ThirdPartyLogger g_third_party = {
    .write = third_party_write
};

/* ==================== 适配器 ==================== */

typedef struct {
    AppLogger base;              /* 继承目标接口 */
    ThirdPartyLogger *adaptee;   /* 持有被适配者 */
    char module[64];             /* 模块名 */
} LoggerAdapter;

static void adapter_log_info(AppLogger *self, const char *msg) {
    LoggerAdapter *adapter = (LoggerAdapter *)self;
    adapter->adaptee->write(TP_LOG_INFO, adapter->module, msg);
}

static void adapter_log_error(AppLogger *self, const char *msg) {
    LoggerAdapter *adapter = (LoggerAdapter *)self;
    adapter->adaptee->write(TP_LOG_ERROR, adapter->module, msg);
}

static void adapter_destroy(AppLogger *self) {
    free(self);
}

static const AppLoggerVTable adapter_vtable = {
    .log_info = adapter_log_info,
    .log_error = adapter_log_error,
    .destroy = adapter_destroy
};

/* 创建适配器 */
AppLogger* logger_adapter_create(ThirdPartyLogger *third_party,
                                  const char *module) {
    LoggerAdapter *adapter = calloc(1, sizeof(LoggerAdapter));
    if (adapter) {
        adapter->base.vtable = &adapter_vtable;
        adapter->adaptee = third_party;
        strncpy(adapter->module, module, sizeof(adapter->module) - 1);
    }
    return (AppLogger *)adapter;
}

/* ==================== 现有系统代码（不修改） ==================== */

void do_something(AppLogger *logger) {
    logger->vtable->log_info(logger, "开始执行任务");
    logger->vtable->log_error(logger, "任务执行失败");
    logger->vtable->log_info(logger, "重试成功");
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 适配器模式演示 ==========\n\n");

    /* 通过适配器让第三方日志库适配现有系统 */
    AppLogger *logger = logger_adapter_create(&g_third_party, "UserService");

    /* 现有系统代码无需修改 */
    do_something(logger);

    logger->vtable->destroy(logger);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 不修改现有代码 | 增加一层间接调用 |
| 复用已有组件 | 过多适配器使系统复杂 |
| 符合开闭原则 | 可能引入性能开销 |

## 6. 装饰器模式

### 问题场景

> 💡 咖啡店卖咖啡，顾客可以加牛奶、加糖、加奶油等。每种组合如果用一个类，类的数量会爆炸。装饰器模式让你动态地给对象添加功能。

```c
/*
 * ============================================================

 */

#include <stdio.h>

#include <string.h>

    void (*destroy)(Coffee *self);
} CoffeeVTable;

struct Coffee {
    const CoffeeVTable *vtable;
};

/* ==================== 具体组件：浓缩咖啡 ==================== */

    Coffee base;

    (void)self;

}

Coffee* espresso_create(void) {

}

/* ==================== 具体组件：美式咖啡 ==================== */
```

## 7. 代理模式

### 问题场景

> 💡 访问远程数据库很慢，希望加一层缓存代理：先查缓存，缓存没有再访问数据库。调用者不需要知道代理的存在。

### C语言实现

```c
/*
 * ============================================================
 * 代理模式：控制对对象的访问
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 主题接口 ==================== */

typedef struct DataSource DataSource;

typedef struct {
    char* (*query)(DataSource *self, const char *key);
    void (*update)(DataSource *self, const char *key, const char *value);
    void (*destroy)(DataSource *self);
} DataSourceVTable;

struct DataSource {
    const DataSourceVTable *vtable;
};

/* ==================== 真实对象：数据库 ==================== */

typedef struct {
    DataSource base;
    int query_count;  /* 查询次数（统计用） */
} Database;

static char* database_query(DataSource *self, const char *key) {
    Database *db = (Database *)self;
    db->query_count++;
    printf("  [数据库] 查询: %s (第%d次查询)\n", key, db->query_count);

    /* 模拟数据库查询延迟 */
    char *result = malloc(128);
    if (result) {
        snprintf(result, 128, "数据[%s]的值", key);
    }
    return result;
}

static void database_update(DataSource *self, const char *key, const char *value) {
    printf("  [数据库] 更新: %s = %s\n", key, value);
}

static void database_destroy(DataSource *self) { free(self); }

static const DataSourceVTable database_vtable = {
    .query = database_query,
    .update = database_update,
    .destroy = database_destroy
};

DataSource* database_create(void) {
    Database *db = calloc(1, sizeof(Database));
    if (db) db->base.vtable = &database_vtable;
    return (DataSource *)db;
}

/* ==================== 代理：缓存代理 ==================== */

#define CACHE_SIZE 16

typedef struct {
    char key[64];
    char value[128];
    int valid;
} CacheEntry;

typedef struct {
    DataSource base;
    DataSource *real_db;       /* 真实对象 */
    CacheEntry cache[CACHE_SIZE]; /* 缓存 */
    int cache_hits;            /* 缓存命中次数 */
} CacheProxy;

static int find_cache_entry(CacheProxy *proxy, const char *key) {
    for (int i = 0; i < CACHE_SIZE; i++) {
        if (proxy->cache[i].valid &&
            strcmp(proxy->cache[i].key, key) == 0) {
            return i;
        }
    }
    return -1;
}

static char* cache_proxy_query(DataSource *self, const char *key) {
    CacheProxy *proxy = (CacheProxy *)self;

    /* 先查缓存 */
    int idx = find_cache_entry(proxy, key);
    if (idx >= 0) {
        proxy->cache_hits++;
        printf("  [缓存] 命中: %s (累计命中%d次)\n", key, proxy->cache_hits);
        return strdup(proxy->cache[idx].value);
    }

    /* 缓存未命中，查询真实数据库 */
    printf("  [缓存] 未命中: %s，查询数据库\n", key);
    char *result = proxy->real_db->vtable->query(proxy->real_db, key);

    /* 写入缓存 */
    if (result) {
        int slot = -1;
        for (int i = 0; i < CACHE_SIZE; i++) {
            if (!proxy->cache[i].valid) { slot = i; break; }
        }
        if (slot < 0) slot = 0;  /* 简单策略：覆盖第一个 */

        strncpy(proxy->cache[slot].key, key, sizeof(proxy->cache[slot].key) - 1);
        strncpy(proxy->cache[slot].value, result, sizeof(proxy->cache[slot].value) - 1);
        proxy->cache[slot].valid = 1;
    }

    return result;
}

static void cache_proxy_update(DataSource *self, const char *key, const char *value) {
    CacheProxy *proxy = (CacheProxy *)self;

    /* 更新真实数据库 */
    proxy->real_db->vtable->update(proxy->real_db, key, value);

    /* 使缓存失效 */
    int idx = find_cache_entry(proxy, key);
    if (idx >= 0) {
        proxy->cache[idx].valid = 0;
        printf("  [缓存] 失效: %s\n", key);
    }
}

static void cache_proxy_destroy(DataSource *self) {
    CacheProxy *proxy = (CacheProxy *)self;
    proxy->real_db->vtable->destroy(proxy->real_db);
    free(proxy);
}

static const DataSourceVTable cache_proxy_vtable = {
    .query = cache_proxy_query,
    .update = cache_proxy_update,
    .destroy = cache_proxy_destroy
};

DataSource* cache_proxy_create(DataSource *real_db) {
    CacheProxy *proxy = calloc(1, sizeof(CacheProxy));
    if (proxy) {
        proxy->base.vtable = &cache_proxy_vtable;
        proxy->real_db = real_db;
    }
    return (DataSource *)proxy;
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 代理模式演示 ==========\n\n");

    /* 创建真实数据库和缓存代理 */
    DataSource *db = database_create();
    DataSource *proxy = cache_proxy_create(db);

    /* 第一次查询 - 缓存未命中 */
    printf("第1次查询 'user:1':\n");
    char *r1 = proxy->vtable->query(proxy, "user:1");
    printf("  结果: %s\n\n", r1);
    free(r1);

    /* 第二次查询相同key - 缓存命中 */
    printf("第2次查询 'user:1':\n");
    char *r2 = proxy->vtable->query(proxy, "user:1");
    printf("  结果: %s\n\n", r2);
    free(r2);

    /* 查询不同key - 缓存未命中 */
    printf("查询 'user:2':\n");
    char *r3 = proxy->vtable->query(proxy, "user:2");
    printf("  结果: %s\n\n", r3);
    free(r3);

    /* 更新数据 - 缓存失效 */
    printf("更新 'user:1':\n");
    proxy->vtable->update(proxy, "user:1", "新数据");

    /* 再次查询 - 缓存已失效，重新查数据库 */
    printf("\n更新后查询 'user:1':\n");
    char *r4 = proxy->vtable->query(proxy, "user:1");
    printf("  结果: %s\n", r4);
    free(r4);

    proxy->vtable->destroy(proxy);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 控制对象访问 | 增加间接层 |
| 延迟加载（虚拟代理） | 响应可能延迟 |
| 缓存优化（智能代理） | 代理和真实对象需保持接口一致 |
| 访问控制（保护代理） | 增加代码复杂度 |

## 8. 外观模式

### 问题场景

> 💡 启动一个应用需要初始化日志系统、配置管理器、数据库连接、网络模块等多个子系统。调用者不想和每个子系统打交道，只需要一个"一键启动"按钮。

```c
/*

    printf("========== 应用关闭 ==========\n");
    net_system_shutdown(&facade->net);
    db_system_shutdown(&facade->db);
    config_system_shutdown(&facade->config);
    log_system_shutdown(&facade->log);
    printf("========== 关闭完成 ==========\n");
}

/* ==================== 使用示例 ==================== */

int main(void) {
    ApplicationFacade app;

    /* 一键启动 - 不需要和每个子系统打交道 */
    app_facade_init(&app, "app.conf", "mysql://localhost:3306/mydb", 8080);

    /* 应用运行中... */
    printf("应用正在运行...\n\n");

    /* 一键关闭 */
    app_facade_shutdown(&app);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 简化子系统接口 | 外观类可能变成"上帝对象" |
| 解耦调用者与子系统 | 不符合开闭原则（新增子系统需修改外观） |
| 分层设计清晰 | 子系统的精细控制被隐藏 |

## 9. 观察者模式

### 问题场景

> 💡 股票行情系统：当股票价格变化时，多个显示器、手机App、交易机器人都需要收到通知。价格源不应该知道有多少个观察者。

### C语言实现

```c
/*
 * ============================================================
 * 观察者模式：定义对象间一对多的依赖关系
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 观察者接口 ==================== */

typedef struct StockObserver StockObserver;

typedef struct {
    void (*on_price_change)(StockObserver *self, const char *symbol,
                            double old_price, double new_price);
} StockObserverVTable;

struct StockObserver {
    const StockObserverVTable *vtable;
    char name[32];
};

/* ==================== 主题：股票价格 ==================== */

#define MAX_OBSERVERS 16

typedef struct {
    StockObserver *observers[MAX_OBSERVERS];
    int count;
    char symbol[16];
    double price;
} Stock;

Stock* stock_create(const char *symbol, double price) {
    Stock *s = calloc(1, sizeof(Stock));
    if (s) {
        strncpy(s->symbol, symbol, sizeof(s->symbol) - 1);
        s->price = price;
    }
    return s;
}

void stock_subscribe(Stock *s, StockObserver *obs) {
    if (s && s->count < MAX_OBSERVERS) {
        s->observers[s->count++] = obs;
        printf("  [%s] 订阅了 %s\n", obs->name, s->symbol);
    }
}

void stock_unsubscribe(Stock *s, StockObserver *obs) {
    if (!s) return;
    for (int i = 0; i < s->count; i++) {
        if (s->observers[i] == obs) {
            for (int j = i; j < s->count - 1; j++) {
                s->observers[j] = s->observers[j + 1];
            }
            s->count--;
            printf("  [%s] 取消订阅 %s\n", obs->name, s->symbol);
            return;
        }
    }
}

void stock_set_price(Stock *s, double new_price) {
    if (!s) return;
    double old_price = s->price;
    s->price = new_price;

    printf("\n%s 价格变化: %.2f → %.2f\n", s->symbol, old_price, new_price);

    for (int i = 0; i < s->count; i++) {
        s->observers[i]->vtable->on_price_change(
            s->observers[i], s->symbol, old_price, new_price);
    }
}

/* ==================== 具体观察者1：显示器 ==================== */

typedef struct {
    StockObserver base;
} DisplayObserver;

static void display_on_price_change(StockObserver *self, const char *symbol,
                                     double old_price, double new_price) {
    (void)self;
    double change = ((new_price - old_price) / old_price) * 100;
    printf("  [显示器] %s: %.2f (变化 %.2f%%)\n", symbol, new_price, change);
}

static const StockObserverVTable display_vtable = {
    .on_price_change = display_on_price_change
};

/* ==================== 具体观察者2：交易机器人 ==================== */

typedef struct {
    StockObserver base;
    double buy_threshold;   /* 低于此价格买入 */
    double sell_threshold;  /* 高于此价格卖出 */
} TradingBot;

static void bot_on_price_change(StockObserver *self, const char *symbol,
                                 double old_price, double new_price) {
    TradingBot *bot = (TradingBot *)self;
    if (new_price <= bot->buy_threshold) {
        printf("  [交易机器人] %s 价格 %.2f ≤ %.2f，买入！\n",
               symbol, new_price, bot->buy_threshold);
    } else if (new_price >= bot->sell_threshold) {
        printf("  [交易机器人] %s 价格 %.2f ≥ %.2f，卖出！\n",
               symbol, new_price, bot->sell_threshold);
    } else {
        printf("  [交易机器人] %s 价格 %.2f，持有观望\n", symbol, new_price);
    }
    (void)old_price;
}

static const StockObserverVTable bot_vtable = {
    .on_price_change = bot_on_price_change
};

/* ==================== 具体观察者3：报警器 ==================== */

typedef struct {
    StockObserver base;
    double alert_threshold; /* 跌幅超过此比例报警 */
} AlertObserver;

static void alert_on_price_change(StockObserver *self, const char *symbol,
                                   double old_price, double new_price) {
    AlertObserver *ao = (AlertObserver *)self;
    double drop = ((old_price - new_price) / old_price) * 100;
    if (drop >= ao->alert_threshold) {
        printf("  [⚠️报警] %s 暴跌 %.2f%%！\n", symbol, drop);
    }
}

static const StockObserverVTable alert_vtable = {
    .on_price_change = alert_on_price_change
};

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 观察者模式演示 ==========\n\n");

    /* 创建股票 */
    Stock *aapl = stock_create("AAPL", 150.0);

    /* 创建观察者 */
    DisplayObserver display = { .base.vtable = &display_vtable, .base.name = "显示器" };
    TradingBot bot = {
        .base.vtable = &bot_vtable, .base.name = "交易机器人",
        .buy_threshold = 140.0, .sell_threshold = 160.0
    };
    AlertObserver alert = {
        .base.vtable = &alert_vtable, .base.name = "报警器",
        .alert_threshold = 5.0
    };

    /* 订阅 */
    stock_subscribe(aapl, (StockObserver *)&display);
    stock_subscribe(aapl, (StockObserver *)&bot);
    stock_subscribe(aapl, (StockObserver *)&alert);

    /* 价格变化 */
    stock_set_price(aapl, 155.0);   /* 上涨 */
    stock_set_price(aapl, 162.0);   /* 触发卖出 */
    stock_set_price(aapl, 140.0);   /* 触发买入 */
    stock_set_price(aapl, 130.0);   /* 暴跌报警 */

    /* 取消订阅 */
    stock_unsubscribe(aapl, (StockObserver *)&bot);
    stock_set_price(aapl, 145.0);   /* 机器人不再收到通知 */

    free(aapl);
    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 松耦合：主题不知道观察者细节 | 通知顺序不确定 |
| 动态订阅/取消 | 可能引起内存泄漏（忘记取消） |
| 符合开闭原则 | 大量观察者时性能问题 |

## 10. 策略模式

### 问题场景

> 💡 电商系统的折扣计算：双十一打5折、会员打8折、满减优惠。不同时期使用不同折扣策略，策略可以随时切换。

### C语言实现

```c
/*
 * ============================================================
 * 策略模式：将算法封装为可互换的策略
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 策略接口 ==================== */

typedef struct DiscountStrategy DiscountStrategy;

typedef struct {
    double (*calculate)(const DiscountStrategy *self, double original_price);
    const char* (*name)(void);
} DiscountStrategyVTable;

struct DiscountStrategy {
    const DiscountStrategyVTable *vtable;
};

/* ==================== 具体策略1：无折扣 ==================== */

static double no_discount_calc(const DiscountStrategy *self, double price) {
    (void)self;
    return price;
}

static const char* no_discount_name(void) { return "无折扣"; }

static const DiscountStrategyVTable no_discount_vtable = {
    .calculate = no_discount_calc,
    .name = no_discount_name
};

static DiscountStrategy no_discount = { .vtable = &no_discount_vtable };

/* ==================== 具体策略2：百分比折扣 ==================== */

typedef struct {
    DiscountStrategy base;
    double percentage;  /* 折扣百分比，如0.8表示8折 */
} PercentageDiscount;

static double percentage_calc(const DiscountStrategy *self, double price) {
    const PercentageDiscount *pd = (const PercentageDiscount *)self;
    return price * pd->percentage;
}

static const char* percentage_name(void) { return "百分比折扣"; }

static const DiscountStrategyVTable percentage_vtable = {
    .calculate = percentage_calc,
    .name = percentage_name
};

/* ==================== 具体策略3：满减折扣 ==================== */

typedef struct {
    DiscountStrategy base;
    double threshold;  /* 满多少 */
    double reduction;  /* 减多少 */
} FullReductionDiscount;

static double full_reduction_calc(const DiscountStrategy *self, double price) {
    const FullReductionDiscount *fr = (const FullReductionDiscount *)self;
    if (price >= fr->threshold) {
        return price - fr->reduction;
    }
    return price;
}

static const char* full_reduction_name(void) { return "满减折扣"; }

static const DiscountStrategyVTable full_reduction_vtable = {
    .calculate = full_reduction_calc,
    .name = full_reduction_name
};

/* ==================== 上下文：价格计算器 ==================== */

typedef struct {
    DiscountStrategy *strategy;
} PriceCalculator;

void calculator_set_strategy(PriceCalculator *calc, DiscountStrategy *strategy) {
    calc->strategy = strategy;
}

double calculator_calc_price(PriceCalculator *calc, double original_price) {
    if (calc->strategy) {
        double final_price = calc->strategy->vtable->calculate(
            calc->strategy, original_price);
        printf("  原价: ¥%.2f, 策略: %s, 实付: ¥%.2f\n",
               original_price, calc->strategy->vtable->name(), final_price);
        return final_price;
    }
    return original_price;
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 策略模式演示 ==========\n\n");

    PriceCalculator calc = {0};

    /* 无折扣 */
    printf("--- 无折扣 ---\n");
    calculator_set_strategy(&calc, &no_discount);
    calculator_calc_price(&calc, 100.0);

    /* 8折 */
    printf("\n--- 会员8折 ---\n");
    PercentageDiscount member_discount = {
        .base.vtable = &percentage_vtable,
        .percentage = 0.8
    };
    calculator_set_strategy(&calc, (DiscountStrategy *)&member_discount);
    calculator_calc_price(&calc, 100.0);

    /* 双十一5折 */
    printf("\n--- 双十一5折 ---\n");
    PercentageDiscount double11 = {
        .base.vtable = &percentage_vtable,
        .percentage = 0.5
    };
    calculator_set_strategy(&calc, (DiscountStrategy *)&double11);
    calculator_calc_price(&calc, 100.0);

    /* 满200减50 */
    printf("\n--- 满200减50 ---\n");
    FullReductionDiscount full_reduce = {
        .base.vtable = &full_reduction_vtable,
        .threshold = 200.0,
        .reduction = 50.0
    };
    calculator_set_strategy(&calc, (DiscountStrategy *)&full_reduce);
    calculator_calc_price(&calc, 250.0);
    calculator_calc_price(&calc, 150.0);  /* 不满200，不减免 */

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 算法可自由切换 | 客户端必须了解所有策略 |
| 避免多重条件判断 | 策略类数量增多 |
| 符合开闭原则 | 策略对象通常无状态，可能浪费 |

## 11. 命令模式

### 问题场景

> 💡 智能家居系统：通过遥控器控制灯光、空调、窗帘。每个按钮对应一个命令，支持撤销操作。命令可以排队执行、定时执行。

### C语言实现

```c
/*
 * ============================================================
 * 命令模式：将请求封装为对象
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 命令接口 ==================== */

typedef struct Command Command;

typedef struct {
    void (*execute)(Command *self);
    void (*undo)(Command *self);
    const char* (*name)(const Command *self);
    void (*destroy)(Command *self);
} CommandVTable;

struct Command {
    const CommandVTable *vtable;
};

/* ==================== 接收者：灯光 ==================== */

typedef struct {
    int brightness;  /* 0-100 */
    int is_on;
} Light;

void light_on(Light *light) {
    light->is_on = 1;
    light->brightness = 100;
    printf("  灯光: 开启 (亮度=%d%%)\n", light->brightness);
}

void light_off(Light *light) {
    light->is_on = 0;
    printf("  灯光: 关闭\n");
}

void light_set_brightness(Light *light, int level) {
    light->brightness = level;
    if (level > 0) light->is_on = 1;
    else light->is_on = 0;
    printf("  灯光: 亮度=%d%%\n", level);
}

/* ==================== 具体命令：开灯 ==================== */

typedef struct {
    Command base;
    Light *light;
    int prev_brightness;
    int prev_on;
} LightOnCommand;

static void light_on_execute(Command *self) {
    LightOnCommand *cmd = (LightOnCommand *)self;
    cmd->prev_brightness = cmd->light->brightness;
    cmd->prev_on = cmd->light->is_on;
    light_on(cmd->light);
}

static void light_on_undo(Command *self) {
    LightOnCommand *cmd = (LightOnCommand *)self;
    if (cmd->prev_on) {
        light_set_brightness(cmd->light, cmd->prev_brightness);
    } else {
        light_off(cmd->light);
    }
    printf("  (撤销开灯)\n");
}

static const char* light_on_name(const Command *self) {
    (void)self;
    return "开灯";
}

static void light_on_destroy(Command *self) { free(self); }

static const CommandVTable light_on_vtable = {
    .execute = light_on_execute,
    .undo = light_on_undo,
    .name = light_on_name,
    .destroy = light_on_destroy
};

Command* light_on_command_create(Light *light) {
    LightOnCommand *cmd = calloc(1, sizeof(LightOnCommand));
    if (cmd) {
        cmd->base.vtable = &light_on_vtable;
        cmd->light = light;
    }
    return (Command *)cmd;
}

/* ==================== 具体命令：关灯 ==================== */

typedef struct {
    Command base;
    Light *light;
    int prev_brightness;
    int prev_on;
} LightOffCommand;

static void light_off_execute(Command *self) {
    LightOffCommand *cmd = (LightOffCommand *)self;
    cmd->prev_brightness = cmd->light->brightness;
    cmd->prev_on = cmd->light->is_on;
    light_off(cmd->light);
}

static void light_off_undo(Command *self) {
    LightOffCommand *cmd = (LightOffCommand *)self;
    if (cmd->prev_on) {
        light_set_brightness(cmd->light, cmd->prev_brightness);
    }
    printf("  (撤销关灯)\n");
}

static const char* light_off_name(const Command *self) {
    (void)self;
    return "关灯";
}

static void light_off_destroy(Command *self) { free(self); }

static const CommandVTable light_off_vtable = {
    .execute = light_off_execute,
    .undo = light_off_undo,
    .name = light_off_name,
    .destroy = light_off_destroy
};

Command* light_off_command_create(Light *light) {
    LightOffCommand *cmd = calloc(1, sizeof(LightOffCommand));
    if (cmd) {
        cmd->base.vtable = &light_off_vtable;
        cmd->light = light;
    }
    return (Command *)cmd;
}

/* ==================== 调用者：遥控器 ==================== */

#define MAX_BUTTONS 8
#define UNDO_STACK_SIZE 32

typedef struct {
    Command *buttons[MAX_BUTTONS];    /* 按钮对应的命令 */
    Command *undo_stack[UNDO_STACK_SIZE]; /* 撤销栈 */
    int undo_top;
} RemoteControl;

void remote_set_button(RemoteControl *rc, int slot, Command *cmd) {
    if (slot >= 0 && slot < MAX_BUTTONS) {
        rc->buttons[slot] = cmd;
    }
}

void remote_press_button(RemoteControl *rc, int slot) {
    if (slot >= 0 && slot < MAX_BUTTONS && rc->buttons[slot]) {
        printf("按下按钮%d [%s]:\n", slot, rc->buttons[slot]->vtable->name(rc->buttons[slot]));
        rc->buttons[slot]->vtable->execute(rc->buttons[slot]);

        /* 压入撤销栈 */
        if (rc->undo_top < UNDO_STACK_SIZE) {
            rc->undo_stack[rc->undo_top++] = rc->buttons[slot];
        }
    }
}

void remote_undo(RemoteControl *rc) {
    if (rc->undo_top > 0) {
        Command *cmd = rc->undo_stack[--rc->undo_top];
        printf("撤销 [%s]:\n", cmd->vtable->name(cmd));
        cmd->vtable->undo(cmd);
    } else {
        printf("没有可撤销的操作\n");
    }
}

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 命令模式演示 ==========\n\n");

    Light living_room_light = {0};
    RemoteControl remote = {0};

    /* 设置按钮 */
    Command *on_cmd = light_on_command_create(&living_room_light);
    Command *off_cmd = light_off_command_create(&living_room_light);
    remote_set_button(&remote, 0, on_cmd);
    remote_set_button(&remote, 1, off_cmd);

    /* 操作 */
    remote_press_button(&remote, 0);  /* 开灯 */
    remote_press_button(&remote, 1);  /* 关灯 */
    remote_undo(&remote);             /* 撤销关灯 → 恢复开灯 */
    remote_undo(&remote);             /* 撤销开灯 → 恢复关闭 */

    /* 清理 */
    on_cmd->vtable->destroy(on_cmd);
    off_cmd->vtable->destroy(off_cmd);

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 请求发送者与接收者解耦 | 命令类数量增多 |
| 支持撤销/重做 | 简单操作过度设计 |
| 支持队列和定时执行 | 撤销栈可能占用大量内存 |
| 命令可组合（宏命令） | — |

## 12. 状态机模式

### 问题场景

> 💡 自动售货机有多种状态：待投币、已投币、出货中、售罄。不同状态下同一操作的行为不同。用if-else判断状态会非常混乱，状态机让每个状态独立处理。

### C语言实现

```c
/*
 * ============================================================
 * 状态机模式：对象行为随状态改变而改变
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ==================== 前向声明 ==================== */

typedef struct VendingMachine VendingMachine;

/* ==================== 状态接口 ==================== */

typedef struct VendingState VendingState;

typedef struct {
    void (*insert_coin)(VendingMachine *vm, int amount);
    void (*select_product)(VendingMachine *vm, int product_id);
    void (*cancel)(VendingMachine *vm);
    const char* (*name)(void);
} VendingStateVTable;

struct VendingState {
    const VendingStateVTable *vtable;
};

/* ==================== 售货机 ==================== */

#define MAX_PRODUCTS 5

typedef struct {
    const char *name;
    int price;
    int stock;
} Product;

struct VendingMachine {
    const VendingState *current_state;
    int balance;                       /* 当前余额 */
    Product products[MAX_PRODUCTS];    /* 商品列表 */
    int product_count;                 /* 商品数量 */
    int selected_product;              /* 选中的商品 */
};

void vm_init(VendingMachine *vm) {
    vm->current_state = NULL;
    vm->balance = 0;
    vm->selected_product = -1;
    vm->product_count = 0;

    /* 添加商品 */
    vm->products[0] = (Product){"可乐", 3, 10};
    vm->products[1] = (Product){"雪碧", 3, 10};
    vm->products[2] = (Product){"矿泉水", 2, 5};
    vm->products[3] = (Product){"薯片", 5, 3};
    vm->products[4] = (Product){"巧克力", 4, 0};  /* 售罄 */
    vm->product_count = 5;
}

void vm_set_state(VendingMachine *vm, const VendingState *state) {
    if (vm->current_state) {
        printf("  状态: %s → %s\n",
               vm->current_state->vtable->name(),
               state->vtable->name());
    } else {
        printf("  初始状态: %s\n", state->vtable->name());
    }
    vm->current_state = state;
}

void vm_insert_coin(VendingMachine *vm, int amount) {
    printf("投入 %d 元\n", amount);
    vm->current_state->vtable->insert_coin(vm, amount);
}

void vm_select_product(VendingMachine *vm, int id) {
    printf("选择商品 #%d\n", id);
    vm->current_state->vtable->select_product(vm, id);
}

void vm_cancel(VendingMachine *vm) {
    printf("取消操作\n");
    vm->current_state->vtable->cancel(vm);
}

/* ==================== 状态1：待投币 ==================== */

static const VendingState idle_state;

static void idle_insert_coin(VendingMachine *vm, int amount) {
    vm->balance += amount;
    printf("  余额: %d 元\n", vm->balance);
    vm_set_state(vm, &has_coin_state);
}

static void idle_select_product(VendingMachine *vm, int id) {
    (void)id;
    printf("  请先投币！\n");
}

static void idle_cancel(VendingMachine *vm) {
    printf("  没有投币，无需退币\n");
}

static const char* idle_name(void) { return "待投币"; }

static const VendingStateVTable idle_vtable = {
    .insert_coin = idle_insert_coin,
    .select_product = idle_select_product,
    .cancel = idle_cancel,
    .name = idle_name
};

static const VendingState idle_state = { .vtable = &idle_vtable };

/* ==================== 状态2：已投币 ==================== */

static const VendingState has_coin_state;
static const VendingState dispensing_state;
static const VendingState sold_out_state;

static void has_coin_insert_coin(VendingMachine *vm, int amount) {
    vm->balance += amount;
    printf("  余额: %d 元\n", vm->balance);
}

static void has_coin_select_product(VendingMachine *vm, int id) {
    if (id < 0 || id >= vm->product_count) {
        printf("  无效的商品编号\n");
        return;
    }

    Product *p = &vm->products[id];
    if (p->stock <= 0) {
        printf("  %s 已售罄\n", p->name);
        return;
    }

    if (vm->balance < p->price) {
        printf("  余额不足！%s 需要 %d 元，当前 %d 元\n",
               p->name, p->price, vm->balance);
        return;
    }

    vm->balance -= p->price;
    p->stock--;
    vm->selected_product = id;
    printf("  购买 %s，扣除 %d 元，剩余 %d 元\n",
           p->name, p->price, vm->balance);
    vm_set_state(vm, &dispensing_state);
}

static void has_coin_cancel(VendingMachine *vm) {
    printf("  退币 %d 元\n", vm->balance);
    vm->balance = 0;
    vm_set_state(vm, &idle_state);
}

static const char* has_coin_name(void) { return "已投币"; }

static const VendingStateVTable has_coin_vtable = {
    .insert_coin = has_coin_insert_coin,
    .select_product = has_coin_select_product,
    .cancel = has_coin_cancel,
    .name = has_coin_name
};

static const VendingState has_coin_state = { .vtable = &has_coin_vtable };

/* ==================== 状态3：出货中 ==================== */

static void dispensing_insert_coin(VendingMachine *vm, int amount) {
    printf("  正在出货，请稍候...\n");
    vm->balance += amount;  /* 仍然接收投币 */
}

static void dispensing_select_product(VendingMachine *vm, int id) {
    (void)vm; (void)id;
    printf("  正在出货，请稍候...\n");
}

static void dispensing_cancel(VendingMachine *vm) {
    printf("  正在出货，无法取消\n");
}

static const char* dispensing_name(void) { return "出货中"; }

/* 出货完成后自动回到待投币状态 */
static void dispensing_complete(VendingMachine *vm) {
    int id = vm->selected_product;
    if (id >= 0 && id < vm->product_count) {
        printf("  🎉 %s 已出货！\n", vm->products[id].name);
    }
    vm->selected_product = -1;

    /* 如果还有余额，回到已投币状态 */
    if (vm->balance > 0) {
        vm_set_state(vm, &has_coin_state);
    } else {
        /* 检查是否所有商品售罄 */
        int all_sold = 1;
        for (int i = 0; i < vm->product_count; i++) {
            if (vm->products[i].stock > 0) { all_sold = 0; break; }
        }
        vm_set_state(vm, all_sold ? &sold_out_state : &idle_state);
    }
}

static const VendingStateVTable dispensing_vtable = {
    .insert_coin = dispensing_insert_coin,
    .select_product = dispensing_select_product,
    .cancel = dispensing_cancel,
    .name = dispensing_name
};

static const VendingState dispensing_state = { .vtable = &dispensing_vtable };

/* ==================== 状态4：售罄 ==================== */

static void sold_out_insert_coin(VendingMachine *vm, int amount) {
    (void)amount;
    printf("  所有商品已售罄，请勿投币\n");
}

static void sold_out_select_product(VendingMachine *vm, int id) {
    (void)vm; (void)id;
    printf("  所有商品已售罄\n");
}

static void sold_out_cancel(VendingMachine *vm) {
    printf("  没有投币，无需退币\n");
}

static const char* sold_out_name(void) { return "售罄"; }

static const VendingStateVTable sold_out_vtable = {
    .insert_coin = sold_out_insert_coin,
    .select_product = sold_out_select_product,
    .cancel = sold_out_cancel,
    .name = sold_out_name
};

static const VendingState sold_out_state = { .vtable = &sold_out_vtable };

/* ==================== 使用示例 ==================== */

int main(void) {
    printf("========== 状态机模式演示 ==========\n\n");

    VendingMachine vm;
    vm_init(&vm);
    vm_set_state(&vm, &idle_state);

    printf("\n--- 正常购买流程 ---\n");
    vm_insert_coin(&vm, 5);           /* 投币5元 */
    vm_select_product(&vm, 0);        /* 买可乐3元 */
    dispensing_complete(&vm);         /* 出货完成 */

    printf("\n--- 余额不足 ---\n");
    vm_insert_coin(&vm, 1);           /* 投币1元 */
    vm_select_product(&vm, 3);        /* 买薯片5元 → 余额不足 */
    vm_cancel(&vm);                   /* 取消，退币 */

    printf("\n--- 商品售罄 ---\n");
    vm_insert_coin(&vm, 10);
    vm_select_product(&vm, 4);        /* 巧克力售罄 */
    vm_select_product(&vm, 0);        /* 买可乐 */
    dispensing_complete(&vm);

    printf("\n--- 退币 ---\n");
    vm_cancel(&vm);                   /* 退回剩余余额 */

    return 0;
}
```

### 优缺点

| 优点 | 缺点 |
| --- | --- |
| 状态逻辑清晰分离 | 状态类数量增多 |
| 消除大量if-else | 状态转换逻辑分散 |
| 新增状态不影响已有状态 | 状态间通信需要通过上下文 |
| 状态转换显式化 | 状态机可能变得复杂 |

## 总结

### 模式速查表

| 模式 | 类型 | 一句话描述 | 适用场景 |
| --- | --- | --- | --- |
| **单例** | 创建型 | 全局唯一实例 | 配置、日志、连接池 |
| **工厂方法** | 创建型 | 子类决定创建哪个产品 | 日志系统、跨平台创建 |
| **抽象工厂** | 创建型 | 创建一系列相关产品 | UI控件族、跨平台组件 |
| **建造者** | 创建型 | 分步构建复杂对象 | HTTP请求、配置对象 |
| **适配器** | 结构型 | 让不兼容接口协同工作 | 第三方库集成 |
| **装饰器** | 结构型 | 动态添加职责 | 咖啡配料、IO流 |
| **代理** | 结构型 | 控制对象访问 | 缓存、延迟加载 |
| **外观** | 结构型 | 简化子系统接口 | 系统初始化、一键操作 |
| **观察者** | 行为型 | 一对多通知 | 事件系统、数据绑定 |
| **策略** | 行为型 | 算法可互换 | 折扣计算、排序选择 |
| **命令** | 行为型 | 请求封装为对象 | 撤销/重做、任务队列 |
| **状态机** | 行为型 | 行为随状态改变 | 售货机、协议解析 |

### C语言实现设计模式的注意事项

| 要点 | 说明 |
| --- | --- |
| **函数指针是核心** | C语言实现设计模式几乎都依赖函数指针实现多态 |
| **内存管理要小心** | 没有自动析构，每个对象必须手动释放 |
| **类型安全靠自律** | 强制类型转换没有编译期检查，需要开发者保证正确 |
| **不要过度设计** | 简单场景用简单方案，不是每个地方都需要设计模式 |
| **模式可以组合** | 观察者+策略、工厂+单例等组合使用很常见 |
