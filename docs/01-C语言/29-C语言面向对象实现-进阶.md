## 1. 用setjmp/longjmp模拟异常处理

### 1. 为什么需要异常模拟？

C语言没有原生的异常处理机制（如C++的`try/catch`），错误处理通常依赖返回值检查。但在复杂逻辑中，深层嵌套的错误逐层传递非常繁琐。`setjmp/longjmp`提供了一种非局部跳转的能力，可以用来模拟异常处理。

```
┌─────────────────────────────────────────────────────────────────────────┐
│                  setjmp/longjmp 异常模拟原理                            │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   正常流程：                                                            │
│   ┌──────┐    ┌──────┐    ┌──────┐    ┌──────┐                       │
│   │ 调用1 │ -> │ 调用2 │ -> │ 调用3 │ -> │ 返回  │                       │
│   └──────┘    └──────┘    └──────┘    └──────┘                       │
│                                                                         │
│   异常流程：                                                            │
│   ┌──────┐    ┌──────┐    ┌──────┐                                    │
│   │ TRY  │    │ 调用2 │    │ 调用3 │                                    │
│   │setjmp│    │      │    │THROW │                                    │
│   │ 保存 │ <- │      │ <- │longjmp│  直接跳回TRY点！                   │
│   └──────┘    └──────┘    └──────┘                                    │
│                                                                         │
│   setjmp：保存当前执行环境（寄存器、栈指针等）                          │
│   longjmp：恢复之前保存的环境，直接跳转回去                             │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

### 2. TRY/CATCH/FINALLY/THROW宏定义

```c
/*
 * ============================================================
 * 文件: exception.h
 * 描述: 用setjmp/longjmp模拟C++异常处理
 * 功能: 定义TRY/CATCH/FINALLY/THROW宏
 * ============================================================
 */

#ifndef EXCEPTION_H
#define EXCEPTION_H

#include <setjmp.h>
#include <stdio.h>
#include <stdlib.h>

/* ==================== 错误码定义 ==================== */
typedef enum {
    EXC_NONE = 0,           /* 无异常 */
    EXC_MEMORY = 1,         /* 内存分配失败 */
    EXC_NULL_POINTER = 2,   /* 空指针 */
    EXC_OUT_OF_RANGE = 3,   /* 越界访问 */
    EXC_IO_ERROR = 4,       /* 输入输出错误 */
    EXC_INVALID_ARG = 5,    /* 无效参数 */
    EXC_CUSTOM = 100        /* 自定义异常起始值 */
} ExceptionCode;

/* ==================== 异常上下文 ==================== */
typedef struct ExceptionContext {
    jmp_buf env;                        /* 跳转点 */
    int code;                           /* 异常码 */
    const char *message;                /* 异常信息 */
    const char *file;                   /* 文件名 */
    int line;                           /* 行号 */
    struct ExceptionContext *prev;      /* 前一个上下文（支持嵌套） */
} ExceptionContext;

/* ==================== 全局异常栈 ==================== */
static ExceptionContext *_exc_stack = NULL;

/* ==================== 宏定义 ==================== */

/**
 * TRY - 设置异常捕获点
 * 原理：setjmp保存当前环境，首次返回0（正常执行）
 *       longjmp跳回时返回非0值（异常发生）
 */
#define TRY \
    do { \
        ExceptionContext _exc_ctx; \
        _exc_ctx.prev = _exc_stack; \
        _exc_ctx.code = EXC_NONE; \
        _exc_ctx.message = NULL; \
        _exc_ctx.file = NULL; \
        _exc_ctx.line = 0; \
        _exc_stack = &_exc_ctx; \
        \
        if (setjmp(_exc_ctx.env) == 0) {

/**
 * CATCH - 捕获异常
 * 原理：检查异常码是否匹配
 */
#define CATCH(exc_code) \
        } else if (_exc_ctx.code == (exc_code)) {

/**
 * CATCH_ALL - 捕获所有异常
 */
#define CATCH_ALL \
        } else {

/**
 * FINALLY - 无论如何都执行的清理代码
 */
#define FINALLY \
        } {

/**
 * END_TRY - 结束异常处理块
 */
#define END_TRY \
        } \
        _exc_stack = _exc_ctx.prev; \
    } while(0)

/**
 * THROW - 抛出异常
 * 原理：设置异常信息，然后longjmp跳回最近的TRY点
 */
#define THROW(exc_code, msg) \
    do { \
        if (_exc_stack) { \
            _exc_stack->code = (exc_code); \
            _exc_stack->message = (msg); \
            _exc_stack->file = __FILE__; \
            _exc_stack->line = __LINE__; \
            longjmp(_exc_stack->env, (exc_code)); \
        } else { \
            fprintf(stderr, "未捕获的异常: [%d] %s (%s:%d)\n", \
                    (exc_code), (msg), __FILE__, __LINE__); \
            abort(); \
        } \
    } while(0)

/**
 * RETHROW - 重新抛出当前异常
 */
#define RETHROW() \
    do { \
        if (_exc_stack && _exc_stack->prev) { \
            _exc_stack->prev->code = _exc_stack->code; \
            _exc_stack->prev->message = _exc_stack->message; \
            _exc_stack->prev->file = _exc_stack->file; \
            _exc_stack->prev->line = _exc_stack->line; \
            longjmp(_exc_stack->prev->env, _exc_stack->code); \
        } \
    } while(0)

/**
 * 获取当前异常信息
 */
#define EXCEPTION_CODE()    (_exc_ctx.code)
#define EXCEPTION_MSG()     (_exc_ctx.message)
#define EXCEPTION_FILE()    (_exc_ctx.file)
#define EXCEPTION_LINE()    (_exc_ctx.line)

#endif /* EXCEPTION_H */
```

### 3. 完整使用示例

```c
/*
 * ============================================================
 * 文件: exception_demo.c
 * 描述: setjmp/longjmp异常处理完整示例
 * ============================================================
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "exception.h"

/* ==================== 模拟可能出错的函数 ==================== */

void* safe_malloc(size_t size) {
    void *ptr = malloc(size);
    if (!ptr) {
        THROW(EXC_MEMORY, "内存分配失败");
    }
    return ptr;
}

void process_array(int *arr, int index, int len) {
    if (!arr) {
        THROW(EXC_NULL_POINTER, "数组指针为空");
    }
    if (index < 0 || index >= len) {
        THROW(EXC_OUT_OF_RANGE, "数组索引越界");
    }
    printf("arr[%d] = %d\n", index, arr[index]);
}

int read_file(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (!fp) {
        THROW(EXC_IO_ERROR, "无法打开文件");
    }
    /* 读取操作... */
    fclose(fp);
    return 0;
}

/* ==================== 基本TRY/CATCH使用 ==================== */

void demo_basic_try_catch(void) {
    printf("=== 基本TRY/CATCH演示 ===\n\n");

    TRY
        printf("尝试分配内存...\n");
        int *data = safe_malloc(100 * sizeof(int));
        printf("内存分配成功！\n");

        printf("尝试越界访问...\n");
        process_array(data, 111, 100);  /* 将抛出异常 */
        printf("这行不会执行\n");        /* 被跳过 */

        free(data);
    CATCH(EXC_MEMORY)
        printf("捕获内存异常: %s\n", EXCEPTION_MSG());
    CATCH(EXC_OUT_OF_RANGE)
        printf("捕获越界异常: %s\n", EXCEPTION_MSG());
    CATCH_ALL
        printf("捕获其他异常: [%d] %s\n", EXCEPTION_CODE(), EXCEPTION_MSG());
    END_TRY

    printf("\n");
}

/* ==================== FINALLY使用 ==================== */

void demo_finally(void) {
    printf("=== FINALLY演示 ===\n\n");

    int *data = NULL;

    TRY
        data = safe_malloc(50 * sizeof(int));
        printf("内存分配成功\n");

        /* 模拟一个异常 */
        THROW(EXC_NULL_POINTER, "模拟异常测试FINALLY");
    CATCH(EXC_NULL_POINTER)
        printf("捕获异常: %s\n", EXCEPTION_MSG());
    FINALLY
        /* 无论是否异常，都会执行清理 */
        printf("FINALLY: 执行清理工作\n");
        if (data) {
            free(data);
            data = NULL;
            printf("FINALLY: 内存已释放\n");
        }
    END_TRY

    printf("\n");
}

/* ==================== 嵌套TRY/CATCH ==================== */

void inner_operation(void) {
    TRY
        printf("  内层: 尝试打开不存在的文件...\n");
        read_file("nonexistent.txt");
    CATCH(EXC_IO_ERROR)
        printf("  内层: 捕获IO异常: %s\n", EXCEPTION_MSG());
        printf("  内层: 重新抛出异常...\n");
        RETHROW();
    END_TRY
}

void demo_nested_try_catch(void) {
    printf("=== 嵌套TRY/CATCH演示 ===\n\n");

    TRY
        printf("外层: 开始操作\n");
        inner_operation();
        printf("外层: 这行不会执行\n");
    CATCH(EXC_IO_ERROR)
        printf("外层: 捕获从内层传播的异常: %s\n", EXCEPTION_MSG());
    END_TRY

    printf("\n");
}

/* ==================== 主函数 ==================== */

int main(void) {
    demo_basic_try_catch();
    demo_finally();
    demo_nested_try_catch();

    printf("=== 所有异常处理演示完成 ===\n");
    return 0;
}
```

### 4. 限制与风险

```
┌─────────────────────────────────────────────────────────────────────────┐
│                  setjmp/longjmp 的限制与风险                            │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  ⚠️ 1. 不调用析构函数                                                   │
│     longjmp直接跳转，不会执行中间函数的清理代码                         │
│     → 局部变量指向的堆内存可能泄漏                                     │
│     → 打开的文件句柄不会被关闭                                         │
│                                                                         │
│  ⚠️ 2. volatile要求                                                     │
│     setjmp返回后，非volatile局部变量可能被优化器覆盖                    │
│     → 在TRY块中修改的局部变量必须声明为volatile                        │
│                                                                         │
│  ⚠️ 3. 栈帧限制                                                        │
│     longjmp只能跳回仍然有效的栈帧                                      │
│     → 不能从已返回的函数中longjmp                                      │
│     → 不能跨线程使用                                                   │
│                                                                         │
│  ⚠️ 4. 资源安全                                                        │
│     跳转路径上的所有资源需要手动管理                                   │
│     → 建议配合FINALLY使用，确保资源释放                                │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

**volatile示例：**

```c
void demo_volatile(void) {
    /*
     * 非volatile变量在setjmp/longjmp后
     * 可能恢复为setjmp时的旧值
     */
    volatile int important_value = 0;  /* 必须加volatile！ */

    TRY
        important_value = 42;          /* 在TRY块中修改 */
        THROW(EXC_CUSTOM, "测试");
    CATCH(EXC_CUSTOM)
        /* 不加volatile，important_value可能是0而非42 */
        printf("important_value = %d\n", important_value);
    END_TRY
}
```

### 5. 与C++异常的对比

| 特性         | C setjmp/longjmp              | C++ try/catch/throw        |
| ---------- | ----------------------------- | -------------------------- |
| **语法**     | 宏定义，不直观                    | 语言原生，语法清晰             |
| **类型安全**   | 无，仅传递int错误码              | 有，可抛出任意类型异常对象       |
| **栈展开**   | 无，直接跳转                     | 有，自动调用析构函数           |
| **资源清理**   | 需手动FINALLY                  | RAII自动清理                |
| **嵌套支持**   | 需手动维护异常栈                 | 编译器自动管理               |
| **性能**     | setjmp保存寄存器有开销          | 零开销异常（无异常时无开销）     |
| **线程安全**   | 全局异常栈不安全                 | 线程局部存储，安全            |
| **调试**     | 难以跟踪跳转路径                 | 调试器可捕获异常              |
| **volatile** | 必须使用                        | 不需要                      |
| **适用场景**   | 简单C项目、嵌入式               | C++项目、复杂异常逻辑         |

### 2. RefCounted基类与Shape继承

```c
/*
 * ============================================================
 * 文件: refcounted.h
 * 描述: 引用计数基类 - 与虚函数表配合
 * ============================================================
 */

#ifndef REFCOUNTED_H
#define REFCOUNTED_H

#include <stdio.h>
#include <stdlib.h>

/* ==================== 引用计数虚函数表 ==================== */
typedef struct RefCounted RefCounted;

typedef struct {
    void (*destroy)(RefCounted *self);   /* 析构函数 */
    const char* (*get_type_name)(void);  /* 获取类型名 */
} RefCountedVTable;

/* ==================== RefCounted基类 ==================== */
struct RefCounted {
    const RefCountedVTable *vtable;  /* 虚函数表 */
    int ref_count;                   /* 引用计数 */
};

/* ==================== 引用计数操作 ==================== */

static inline void ref_counted_init(RefCounted *self, const RefCountedVTable *vtable) {
    self->vtable = vtable;
    self->ref_count = 1;
}

static inline void ref_counted_add_ref(RefCounted *self) {
    if (self) {
        self->ref_count++;
    }
}

static inline void ref_counted_release(RefCounted *self) {
    if (self && --self->ref_count == 0) {
        if (self->vtable && self->vtable->destroy) {
            self->vtable->destroy(self);
        } else {
            free(self);
        }
    }
}

static inline int ref_counted_get_count(const RefCounted *self) {
    return self ? self->ref_count : 0;
}

#endif /* REFCOUNTED_H */
```

```c
/*
 * ============================================================
 * 文件: ref_shape.h
 * 描述: 带引用计数的Shape基类
 * ============================================================
 */

#ifndef REF_SHAPE_H
#define REF_SHAPE_H

#include "refcounted.h"

/* ==================== Shape虚函数表 ==================== */
typedef struct Shape Shape;

typedef struct {
    RefCountedVTable base;                  /* 继承RefCounted的虚函数表 */
    double (*area)(const Shape *self);      /* 计算面积 */
    double (*perimeter)(const Shape *self); /* 计算周长 */
    void (*draw)(const Shape *self);        /* 绘制形状 */
} ShapeVTable;

/* ==================== Shape结构体 ==================== */
struct Shape {
    RefCounted base;     /* 继承引用计数基类 */
    int id;              /* 形状ID */
    char name[32];       /* 形状名称 */
};

/* ==================== Shape公共接口 ==================== */

static inline double shape_area(const Shape *shape) {
    if (shape) {
        const ShapeVTable *vtable = (const ShapeVTable *)shape->base.vtable;
        return vtable->area(shape);
    }
    return 0.0;
}

static inline double shape_perimeter(const Shape *shape) {
    if (shape) {
        const ShapeVTable *vtable = (const ShapeVTable *)shape->base.vtable;
        return vtable->perimeter(shape);
    }
    return 0.0;
}

static inline void shape_draw(const Shape *shape) {
    if (shape) {
        const ShapeVTable *vtable = (const ShapeVTable *)shape->base.vtable;
        vtable->draw(shape);
    }
}

static inline void shape_add_ref(Shape *shape) {
    ref_counted_add_ref(&shape->base);
}

static inline void shape_release(Shape *shape) {
    ref_counted_release(&shape->base);
}

#endif /* REF_SHAPE_H */
```

```c
/*
 * ============================================================
 * 文件: ref_circle.c
 * 描述: 带引用计数的圆形类实现
 * ============================================================
 */

#include "ref_circle.h"
#include <stdio.h>
#include <math.h>

/* ==================== 虚函数实现 ==================== */

static void circle_destroy_impl(RefCounted *self) {
    Shape *shape = (Shape *)self;
    printf("  [析构] 销毁圆形 ID=%d, 名称=%s\n", shape->id, shape->name);
    free(self);
}

static const char* circle_type_name_impl(void) {
    return "Circle";
}

static double circle_area_impl(const Shape *self) {
    const Circle *c = (const Circle *)self;
    return 3.14159265 * c->radius * c->radius;
}

static double circle_perimeter_impl(const Shape *self) {
    const Circle *c = (const Circle *)self;
    return 2 * 3.14159265 * c->radius;
}

static void circle_draw_impl(const Shape *self) {
    const Circle *c = (const Circle *)self;
    printf("绘制圆形: 半径=%.2f\n", c->radius);
}

/* ==================== 虚函数表 ==================== */
static const ShapeVTable circle_vtable = {
    .base = {
        .destroy = circle_destroy_impl,
        .get_type_name = circle_type_name_impl
    },
    .area = circle_area_impl,
    .perimeter = circle_perimeter_impl,
    .draw = circle_draw_impl
};

/* ==================== 构造函数 ==================== */
Circle* circle_create(double radius) {
    Circle *c = malloc(sizeof(Circle));
    if (c) {
        /* 初始化RefCounted部分 */
        ref_counted_init(&c->base.base, (const RefCountedVTable *)&circle_vtable);
        /* 初始化Shape部分 */
        c->base.id = 1;
        snprintf(c->base.name, sizeof(c->base.name), "Circle(r=%.2f)", radius);
        /* 初始化Circle部分 */
        c->radius = radius;
        printf("  [构造] 创建圆形 半径=%.2f, 引用计数=%d\n",
               radius, ref_counted_get_count(&c->base.base));
    }
    return c;
}
```

### 4. 智能指针包装 AutoRef\<T\>

```c
/*
 * ============================================================
 * 文件: autoref.h
 * 描述: 智能指针包装 - 类似C++ shared_ptr
 *       利用GCC cleanup属性实现自动引用计数管理
 * ============================================================
 */

#ifndef AUTOREF_H
#define AUTOREF_H

#include "refcounted.h"

/* ==================== AutoRef结构体 ==================== */
typedef struct {
    RefCounted *ptr;    /* 指向被管理对象的RefCounted部分 */
} AutoRef;

/* ==================== AutoRef操作 ==================== */

static inline AutoRef autoref_create(RefCounted *obj) {
    AutoRef ref = { obj };
    if (obj) {
        ref_counted_add_ref(obj);
    }
    return ref;
}

static inline void autoref_reset(AutoRef *ref, RefCounted *new_obj) {
    if (ref->ptr) {
        ref_counted_release(ref->ptr);
    }
    ref->ptr = new_obj;
    if (new_obj) {
        ref_counted_add_ref(new_obj);
    }
}

static inline void autoref_release(AutoRef *ref) {
    if (ref && ref->ptr) {
        ref_counted_release(ref->ptr);
        ref->ptr = NULL;
    }
}

static inline RefCounted* autoref_get(AutoRef *ref) {
    return ref ? ref->ptr : NULL;
}

static inline int autoref_use_count(AutoRef *ref) {
    return (ref && ref->ptr) ? ref_counted_get_count(ref->ptr) : 0;
}

/* ==================== 基于cleanup的自动管理 ==================== */

static inline void autoref_cleanup(AutoRef *ref) {
    autoref_release(ref);
}

/* AUTO_REF宏：声明一个自动管理的智能指针 */
#define AUTO_REF(var) \
    __attribute__((cleanup(autoref_cleanup))) AutoRef var = { .ptr = NULL }

#endif /* AUTOREF_H */
```

### 6. 循环引用问题与弱引用

```
┌─────────────────────────────────────────────────────────────────────────┐
│                    循环引用问题                                          │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│   正常引用：                                                            │
│   ┌────────┐         ┌────────┐                                       │
│   │  对象A  │ ──────> │  对象B  │    A引用B，B不引用A                  │
│   │ ref=1  │         │ ref=1  │    两者都可正常释放                    │
│   └────────┘         └────────┘                                       │
│                                                                         │
│   循环引用（内存泄漏）：                                                │
│   ┌────────┐         ┌────────┐                                       │
│   │  对象A  │ ──────> │  对象B  │                                      │
│   │ ref=2  │ <────── │ ref=2  │    A引用B，B也引用A                   │
│   └────────┘         └────────┘    两者都无法释放！                    │
│                                                                         │
│   弱引用解决：                                                          │
│   ┌────────┐         ┌────────┐                                       │
│   │  对象A  │ ══════> │  对象B  │    强引用（计数+1）                  │
│   │ ref=1  │ - - - > │ ref=1  │    弱引用（计数不变）                 │
│   └────────┘ < - - - └────────┘    B可以正常释放，A检测到后清除弱引用  │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

```c
/*
 * ============================================================
 * 弱引用实现 - 解决循环引用
 * ============================================================
 */

typedef struct WeakRef WeakRef;

struct WeakRef {
    RefCounted **target;     /* 指向目标指针的指针 */
};

static inline WeakRef weak_ref_create(RefCounted **target) {
    WeakRef ref = { target };
    return ref;
}

static inline RefCounted* weak_ref_lock(WeakRef *ref) {
    if (ref && ref->target && *ref->target) {
        ref_counted_add_ref(*ref->target);
        return *ref->target;
    }
    return NULL;  /* 对象已被销毁 */
}

static inline void weak_ref_unlock(RefCounted *obj) {
    if (obj) {
        ref_counted_release(obj);
    }
}

static inline int weak_ref_expired(WeakRef *ref) {
    return (!ref || !ref->target || *ref->target == NULL);
}

/* 使用示例：双向关联 */
typedef struct {
    RefCounted base;
    char name[32];
    WeakRef partner;        /* 用弱引用避免循环 */
} Person;

void demo_weak_ref(void) {
    /*
     * 创建两个人，互相引用
     * 如果都用强引用，则内存泄漏
     * 一方用弱引用，可以正常释放
     */
    Person *alice = person_create("Alice");  /* ref=1 */
    Person *bob = person_create("Bob");      /* ref=1 */

    /* Alice强引用Bob */
    ref_counted_add_ref(&bob->base);         /* bob ref=2 */

    /* Bob弱引用Alice（不增加计数） */
    bob->partner = weak_ref_create((RefCounted **)&alice);

    /* 释放Alice的原始引用 */
    ref_counted_release(&alice->base);       /* alice ref=0, 销毁 */

    /* Bob的弱引用检测到Alice已销毁 */
    if (weak_ref_expired(&bob->partner)) {
        printf("Alice已被销毁，弱引用失效\n");
    }

    /* 释放Bob */
    ref_counted_release(&bob->base);         /* bob ref=1 */
    ref_counted_release(&bob->base);         /* bob ref=0, 销毁 */
}
```

### 7. 与C++ shared_ptr/weak_ptr的对比

| 特性         | C AutoRef                   | C++ shared_ptr              |
| ---------- | --------------------------- | --------------------------- |
| **引用计数**   | 手动嵌入RefCounted基类        | 控制块自动管理               |
| **线程安全**   | 非原子操作，不安全            | 原子操作，线程安全           |
| **弱引用**   | 手动WeakRef                  | weak_ptr语言支持             |
| **自动析构**   | 需cleanup属性或手动release   | RAII自动析构                |
| **类型安全**   | 需手动类型转换               | 模板参数保证类型安全         |
| **自定义删除器** | 通过虚函数表destroy          | 构造时传入deleter           |
| **循环引用**   | 需手动使用WeakRef            | 使用weak_ptr                |
| **运行时开销** | 极小（仅int计数）            | 控制块分配+原子操作          |
| **使用难度**   | 较高，容易出错               | 低，编译器保障               |

### 2. 核心概念

#### 1. GType类型系统

GType是GObject的类型基础设施，为每个类型分配唯一的运行时类型ID。

```c
/*
 * GType是GObject类型系统的核心
 * 每个类型在运行时都有一个唯一的GType标识
 */

/* 基本类型 */
G_TYPE_INT       /* 整型 */
G_TYPE_STRING    /* 字符串 */
G_TYPE_OBJECT    /* GObject基类 */

/* 自定义类型通过注册获得GType */
GType my_type = G_TYPE_INVALID;  /* 初始值，首次使用时注册 */
```

#### 2. 类结构体与实例结构体

GObject将"类"和"实例"分离为两个结构体：

```c
/*
 * GObject的类/实例分离设计：
 *
 * 类结构体（Class Struct）：每个类型只有一份，存储虚函数表和类数据
 *   → 类似于C++的vtable + static成员
 *
 * 实例结构体（Instance Struct）：每个对象一份，存储实例数据
 *   → 类似于C++的非static成员
 */

/* 示例：自定义MyButton类型 */

/* 实例结构体 - 每个对象独有 */
typedef struct _MyButton MyButton;
struct _MyButton {
    GObject parent_instance;  /* 继承GObject，必须作为第一个成员 */

    /* 实例数据 */
    gchar *label;
    gint  pressed_count;
};

/* 类结构体 - 所有对象共享 */
typedef struct _MyButtonClass MyButtonClass;
struct _MyButtonClass {
    GObjectClass parent_class;  /* 继承GObjectClass */

    /* 虚函数 */
    void (*clicked)(MyButton *self);
    void (*released)(MyButton *self);
};
```

#### 3. 属性系统

GObject提供了运行时可查询、可修改的属性系统：

```c
/*
 * GObject属性系统：
 * - 运行时可列举对象的所有属性
 * - 通过统一接口get/set属性
 * - 属性变化时可发出通知信号

/* 安装属性 */
GParamSpec *pspec = g_param_spec_string(

    "default",            /* 默认值 */

g_object_class_install_property(object_class, PROP_LABEL, pspec);

/* 使用属性 */
g_object_set(button, "label", "Click Me", NULL);
gchar *label;
g_object_get(button, "label", &label, NULL);
```

#### 4. 信号系统

GObject的信号系统实现了观察者模式：

```c
/*
 * GObject信号系统：
 * - 对象发出信号，多个回调函数可连接到同一信号
 * - 支持信号累积、阻塞、断开等高级操作
 */

/* 定义信号ID */
enum {
    CLICKED_SIGNAL,
    LAST_SIGNAL
};

static guint my_button_signals[LAST_SIGNAL] = { 0 };

/* 注册信号 */
my_button_signals[CLICKED_SIGNAL] =
    g_signal_new("clicked",
                 G_TYPE_FROM_CLASS(klass),
                 G_SIGNAL_RUN_LAST,
                 G_STRUCT_OFFSET(MyButtonClass, clicked),
                 NULL, NULL,
                 g_cclosure_marshal_VOID__VOID,
                 G_TYPE_NONE, 0);

/* 发出信号 */
g_signal_emit(button, my_button_signals[CLICKED_SIGNAL], 0);

/* 连接信号处理函数 */
g_signal_connect(button, "clicked",
                 G_CALLBACK(on_button_clicked), user_data);

/* 信号处理回调 */
static void on_button_clicked(MyButton *button, gpointer user_data) {
    g_print("按钮被点击了！\n");
}
```

#### 5. 引用计数内存管理

GObject内置引用计数，与本章前面手写的RefCounted原理相同：

```c
/*
 * GObject引用计数：

GObject *obj = g_object_new(MY_TYPE_BUTTON, NULL);
```

```c
                     G_STRUCT_OFFSET(MyButtonClass, clicked),
                     NULL, NULL,
                     g_cclosure_marshal_VOID__VOID,
                     G_TYPE_NONE, 0);
}

/* ==================== 实例初始化 ==================== */

static void my_button_init(MyButton *self) {
    self->label = NULL;
    self->click_count = 0;
}

/* ==================== 公共接口实现 ==================== */

MyButton* my_button_new(const gchar *label) {
    return g_object_new(MY_TYPE_BUTTON, "label", label, NULL);
}

void my_button_click(MyButton *self) {
    g_return_if_fail(MY_IS_BUTTON(self));
    g_signal_emit(self, obj_signals[CLICKED_SIGNAL], 0);
}

const gchar* my_button_get_label(MyButton *self) {
    g_return_val_if_fail(MY_IS_BUTTON(self), NULL);
    return self->label;
}

void my_button_set_label(MyButton *self, const gchar *label) {
    g_return_if_fail(MY_IS_BUTTON(self));
    g_free(self->label);
    self->label = g_strdup(label);
}
```

```c
 * ============================================================

#include <glib-object.h>
#include "my_button.h"

/* 信号回调函数 */
static void on_button_clicked(MyButton *button, gpointer user_data) {
    const gchar *name = (const gchar *)user_data;
    g_print("  [回调] %s 收到点击信号\n", name);
}

int main(int argc, char *argv[]) {
    g_print("=== GObject最小类示例 ===\n\n");

    /* 创建对象（引用计数=1） */
    MyButton *btn = my_button_new("确定");
    g_print("创建按钮，标签: %s\n\n", my_button_get_label(btn));

    /* 使用属性系统 */
    g_object_set(btn, "label", "取消", NULL);
    gchar *label = NULL;
    g_object_get(btn, "label", &label, NULL);
    g_print("修改标签: %s\n\n", label);
    g_free(label);

    /* 连接信号 */
    g_signal_connect(btn, "clicked",
                     G_CALLBACK(on_button_clicked),
                     (gpointer)"观察者A");
    g_signal_connect(btn, "clicked",
                     G_CALLBACK(on_button_clicked),
                     (gpointer)"观察者B");

    /* 触发信号 */
    g_print("点击按钮:\n");
    my_button_click(btn);
    g_print("\n");

    g_print("再次点击:\n");
    my_button_click(btn);
    g_print("\n");

    /* 释放对象 */
    g_print("释放按钮:\n");
    g_object_unref(btn);

    g_print("\n=== 示例结束 ===\n");
    return 0;
}
```

### 4. GObject的优缺点

| 优点                     | 缺点                          |
| ---------------------- | --------------------------- |
| **工业级成熟**：被大量开源项目验证     | **学习曲线陡峭**：概念多，宏复杂       |
| **类型系统**：运行时类型查询和检查     | **样板代码多**：每个类需要大量宏和模板     |
| **信号系统**：解耦的观察者模式实现     | **编译依赖**：需要GLib库           |
| **属性系统**：运行时可查询的对象属性    | **调试困难**：宏展开后代码难以阅读       |
| **引用计数**：内置内存管理         | **性能开销**：类型注册、信号分发有运行时开销  |
| **接口系统**：类似Java接口的多继承   | **文档复杂**：官方文档庞大，入门门槛高     |
| **工具链支持**：glib-mkenums等代码生成 | **C语言局限**：无法完全避免手动错误      |

### 6. 与本章手写OOP的对比

| 特性         | 手写OOP（本章）               | GObject                      |
| ---------- | ----------------------------- | ---------------------------- |
| **封装**     | 不透明指针+函数               | 不透明指针+函数（相同原理）   |
| **继承**     | 结构体组合                    | 结构体组合+类型注册           |
| **多态**     | 手写虚函数表                  | 类结构体中的函数指针          |
| **引用计数** | 手写RefCounted               | 内置g_object_ref/unref       |
| **信号系统** | 无（需手写回调）              | 完整的信号/回调系统           |
| **属性系统** | 手写getter/setter             | 运行时可查询的属性系统        |
| **类型检查** | 手写宏或无                    | G_TYPE_CHECK_INSTANCE等宏    |
| **接口**    | 无（需手写多个vtable）        | GInterface接口系统           |
| **代码量**   | 少，直观                      | 多，样板代码                  |
| **依赖**    | 无（纯C标准库）               | 需要GLib库                   |
| **可维护性** | 小项目好，大项目差            | 大项目好，工具链支持          |
| **学习成本** | 低                            | 高                           |

## 总结

### 7. 核心要点

1. **封装**：使用不透明指针和函数接口隐藏实现细节
2. **继承**：通过结构体组合实现代码复用
3. **多态**：使用虚函数表实现运行时方法绑定
4. **生命周期**：create/destroy函数管理资源
5. **异常模拟**：用setjmp/longjmp实现TRY/CATCH/FINALLY/THROW，注意volatile要求和资源清理
6. **引用计数深化**：RefCounted与虚函数表结合，AutoRef智能指针，弱引用解决循环引用
7. **GObject系统**：工业级C语言OOP框架，提供类型系统、信号系统、属性系统和引用计数
8. **设计模式**：详见 [第30章：C语言设计模式实现](30-C语言设计模式实现.md)

### 8. 学习建议

1. 理解虚函数表机制，这是C++多态的底层实现
2. 掌握资源管理思想，为C++ RAII打下基础
3. 学习设计模式的C实现，加深对模式本质的理解
4. 在实际项目中权衡抽象层次和性能需求
