> 💡 **通俗理解 - 什么是智能指针？**

回想一下C语言内存管理章节讲的"酒店"比喻：
- 普通指针就像你自己管理房间
- 经常忘记退房（内存泄漏）
- 有时候退错房（重复释放）

**智能指针就像"酒店服务员"！**
- 你只需要告诉服务员"我要住这间房"
- 服务员自动帮你办理入住
- 你退房的时候，服务员自动帮你处理
- 不用担心忘记退房，也不用担心退错房

**三种智能指针就像不同的"服务员"：**

1. **unique_ptr** - 就像"专属服务员"
   - 这间房只为你服务
   - 只能有一个服务员负责这间房
   - 你走了，服务员自动帮你退房

2. **shared_ptr** - 就像"共享服务员"
   - 多个服务员可以同时服务同一间房
   - 大家轮班照顾这间房
   - 所有服务员都下班了（引用计数为0），自动退房

3. **weak_ptr** - 就像"临时工"
   - 只能看，不能住
   - 不会影响房间的退房时间
   - 用来解决"互相等待"的问题（循环引用）

> 🔬 **抽象理解 - RAII与资源管理**：
> - **RAII**（Resource Acquisition Is Initialization）：资源获取即初始化，是一种"自动化管理"资源的思想
> - **智能指针**：本质是 RAII 资源管理。unique_ptr 独占所有权（无引用计数），shared_ptr 共享所有权（引用计数），weak_ptr 解决循环引用
> - **所有权语义**：unique_ptr表示"独占所有权"，shared_ptr表示"共享所有权"
> - **weak_ptr**：是一种"非拥有性引用"，用于打破循环引用
> - **内存管理的本质**：是"生命周期管理"，谁创建、谁负责释放

## 1. 智能指针概述

### 1. 概念与定义

**智能指针（smart pointer）**：C++中用于自动管理内存的机制。智能指针可以自动释放内存，避免内存泄漏、重复释放、悬空指针等问题。

**内存泄漏（memory leak）**：程序运行时，分配的内存没有被释放，导致内存占用越来越大。内存泄漏会导致程序性能下降，甚至崩溃。

**重复释放（double free）**：程序运行时，同一内存被释放多次。重复释放会导致程序崩溃。

**悬空指针（dangling pointer）**：程序运行时，指针指向已释放的内存。悬空指针会导致程序崩溃或出现未定义行为。

**异常安全（exception safety）**：程序在抛出异常时，不会出现资源泄漏、数据损坏等问题。异常安全可以分为基本异常安全、强异常安全、不抛异常安全。

**unique_ptr**：C++11引入的新特性，用于独占所有权的智能指针。`unique_ptr`可以确保同一时刻只有一个指针指向对象，避免重复释放。例如`std::unique_ptr<int> p = std::make_unique<int>(42);`。

**shared_ptr**：C++11引入的新特性，用于共享所有权的智能指针。`shared_ptr`可以确保多个指针指向同一对象，当所有指针都被释放时，对象才会被释放。例如`std::shared_ptr<int> p = std::make_shared<int>(42);`。

**weak_ptr**：C++11引入的新特性，用于弱引用的智能指针。`weak_ptr`可以打破循环引用，避免内存泄漏。例如`std::weak_ptr<int> wp = p;`。

**引用计数（reference counting）**：用于跟踪对象被引用的次数。当引用计数为0时，对象会被释放。`shared_ptr`使用引用计数来管理对象的生命周期。

**循环引用（circular reference）**：两个或多个对象相互引用，导致引用计数永远不会为0，从而导致内存泄漏。`weak_ptr`可以打破循环引用。

```cpp
// ✅ 智能指针：自动管理内存
void safeFunction() {
    // 自动释放，无需手动 delete
    std::unique_ptr<int> p = std::make_unique<int>(42);
    
    // 即使抛出异常，内存也会被正确释放
```

## 2. unique_ptr独占所有权

### 1. 基本使用

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    Resource() { std::cout << "Resource acquired\n"; }
    ~Resource() { std::cout << "Resource released\n"; }
    
    void doSomething() { std::cout << "Using resource\n"; }
};

int main() {
    // 创建 unique_ptr（推荐使用 make_unique）
    std::unique_ptr<Resource> ptr1 = std::make_unique<Resource>();
    
    // 使用 -> 访问成员
    ptr1->doSomething();
    
    // 使用 * 解引用
    Resource& ref = *ptr1;
    ref.doSomething();
    
    // 检查是否为空
    if (ptr1) {
        std::cout << "ptr1 is not null\n";
    }
    
    // 离开作用域时自动释放
    return 0;
}
```

### 2. 创建方式

```cpp
#include <iostream>
#include <memory>

class Widget {
public:
    int id;
    explicit Widget(int i) : id(i) {
        std::cout << "Widget " << id << " created\n";
    }
    ~Widget() {
        std::cout << "Widget " << id << " destroyed\n";
    }
};

int main() {
    // 方式1：make_unique（推荐，C++14）
    std::unique_ptr<Widget> w1 = std::make_unique<Widget>(1);
    
    // 方式2：直接构造
    std::unique_ptr<Widget> w2(new Widget(2));
    
    // 方式3：重置
    std::unique_ptr<Widget> w3;
    w3 = std::make_unique<Widget>(3);
    
    // 方式4：自动类型推导
    auto w4 = std::make_unique<Widget>(4);
    
    return 0;
}
```

### 3. 移动语义

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    int id;
    explicit Resource(int i) : id(i) {
        std::cout << "Resource " << id << " created\n";
    }
    ~Resource() {
        std::cout << "Resource " << id << " destroyed\n";
    }
};

// unique_ptr 不能拷贝，只能移动
std::unique_ptr<Resource> createResource(int id) {
    return std::make_unique<Resource>(id);  // 返回时自动移动
}

void transferOwnership() {
    std::unique_ptr<Resource> ptr1 = std::make_unique<Resource>(1);
    
    // 移动所有权
    std::unique_ptr<Resource> ptr2 = std::move(ptr1);
    
    // ptr1 现在为空
    if (!ptr1) {
        std::cout << "ptr1 is null after move\n";
    }
    
    // ptr2 拥有资源
    std::cout << "ptr2 owns resource " << ptr2->id << "\n";
    
    // 移动到函数
    auto ptr3 = createResource(3);
    std::cout << "ptr3 owns resource " << ptr3->id << "\n";
}

int main() {
    transferOwnership();
    return 0;
}
```

### 4. release与reset

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    int id;
    explicit Resource(int i) : id(i) {
        std::cout << "Resource " << id << " created\n";
    }
    ~Resource() {
        std::cout << "Resource " << id << " destroyed\n";
    }
};

int main() {
    // reset：释放当前对象，接管新对象
    std::unique_ptr<Resource> ptr = std::make_unique<Resource>(1);
    
    // 重置为新对象（旧对象被释放）
    ptr.reset(new Resource(2));
    
    // 重置为空（对象被释放）
    ptr.reset();
    
    if (!ptr) {
        std::cout << "ptr is null\n";
    }
    
    // release：放弃所有权，返回原始指针
    ptr = std::make_unique<Resource>(3);
    Resource* raw = ptr.release();  // ptr 变为空
    
    std::cout << "Raw pointer: " << raw->id << "\n";
    std::cout << "ptr is null: " << (!ptr ? "yes" : "no") << "\n";
    
    // 必须手动释放原始指针
    delete raw;
    
    return 0;
}
```

### 5. 在容器中使用

```cpp
#include <iostream>
#include <memory>
#include <vector>
#include <algorithm>

class Item {
public:
    int id;
    explicit Item(int i) : id(i) {
        std::cout << "Item " << id << " created\n";
    }
    ~Item() {
        std::cout << "Item " << id << " destroyed\n";
    }
    
    void print() const {
        std::cout << "Item id: " << id << "\n";
    }
};

int main() {
    // 在容器中存储 unique_ptr
    std::vector<std::unique_ptr<Item>> items;
    
    // 添加元素（使用移动语义）
    items.push_back(std::make_unique<Item>(1));
    items.push_back(std::make_unique<Item>(2));
    items.push_back(std::make_unique<Item>(3));
    
    // 遍历
    for (const auto& item : items) {
        item->print();
    }
    
    // 使用算法
    auto it = std::find_if(items.begin(), items.end(),
        [](const std::unique_ptr<Item>& item) {
            return item->id == 2;
        });
    
    if (it != items.end()) {
        std::cout << "Found item with id 2\n";
    }
    
    // 移除元素
    items.erase(items.begin());
    
    std::cout << "Vector size: " << items.size() << "\n";
    
    return 0;
}
```

### 6. 工厂模式

```cpp

// 产品接口
class Product {

    virtual ~Product() = default;

    else if (type == "B") {
        return std::make_unique<ProductB>();
    }

    std::vector<std::unique_ptr<Product>> products;
    products.push_back(createProduct("A"));
    products.push_back(createProduct("B"));
    
    for (const auto& p : products) {
        p->use();
    }
    
    return 0;
```

## 3. shared_ptr共享所有权

### 1. 基本使用

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    int id;
    explicit Resource(int i) : id(i) {
        std::cout << "Resource " << id << " created\n";
    }
    ~Resource() {
        std::cout << "Resource " << id << " destroyed\n";
    }
    
    void use() {
        std::cout << "Using resource " << id << "\n";
    }
};

int main() {
    // 创建 shared_ptr（推荐使用 make_shared）
    std::shared_ptr<Resource> ptr1 = std::make_shared<Resource>(1);
    
    std::cout << "ptr1 use_count: " << ptr1.use_count() << "\n";  // 1
    
    // 共享所有权
    {
        std::shared_ptr<Resource> ptr2 = ptr1;  // 拷贝
        std::cout << "ptr1 use_count: " << ptr1.use_count() << "\n";  // 2
        std::cout << "ptr2 use_count: " << ptr2.use_count() << "\n";  // 2
        
        ptr2->use();
    }  // ptr2 离开作用域
    
    std::cout << "ptr1 use_count: " << ptr1.use_count() << "\n";  // 1
    
    return 0;
}
```

### 2. 引用计数机制

```cpp
#include <iostream>

class Widget {
public:
    int id;

    ~Widget() {
        std::cout << "Widget " << id << " destroyed\n";
    }
};

void demonstrateRefCount() {
    auto ptr1 = std::make_shared<Widget>(1);
    std::cout << "After creation: " << ptr1.use_count() << "\n";
    
    {
        auto ptr2 = ptr1;
        auto ptr3 = ptr1;
        std::cout << "Three pointers: " << ptr1.use_count() << "\n";
        
        // 移动构造
        std::shared_ptr<Widget> ptr4 = std::move(ptr3);
        std::cout << "After move: " << ptr1.use_count() << "\n";
        std::cout << "ptr3 is null: " << (!ptr3 ? "yes" : "no") << "\n";
    }
    
    std::cout << "After block: " << ptr1.use_count() << "\n";
    
    // 重置
    ptr1.reset();
    std::cout << "After reset: ptr1 is null\n";
}

int main() {
    demonstrateRefCount();
    std::cout << "End of main\n";
    return 0;
}
```

### 3. make_shared的优势

```cpp
#include <iostream>
#include <memory>

class Widget {
public:
    int a, b;
    Widget(int x, int y) : a(x), b(y) {
        std::cout << "Widget created\n";
    }
    ~Widget() {
        std::cout << "Widget destroyed\n";
    }
};

int main() {
    // ✅ 推荐：make_shared
    // 优点1：一次内存分配（对象 + 控制块）
    // 优点2：异常安全
    // 优点3：代码简洁
    auto ptr1 = std::make_shared<Widget>(1, 2);
    
    // ❌ 不推荐：new + shared_ptr
    // 缺点1：两次内存分配
    // 缺点2：可能的内存泄漏（异常安全）
    std::shared_ptr<Widget> ptr2(new Widget(3, 4));
    
    // make_shared 内存布局
    /*
    ┌────────────────────────────────────┐
    │  Widget 对象                       │
    │  ├─ a = 1                          │
    │  └─ b = 2                          │
    ├────────────────────────────────────┤
    │  控制块                            │
    │  ├─ strong_count = 1               │
    │  ├─ weak_count = 0                 │
    │  └─ 其他管理信息                   │
    └────────────────────────────────────┘
    */
    
    return 0;
}
```

### 4. 共享所有权示例

```cpp
#include <iostream>
#include <memory>
#include <vector>
#include <string>

// 文档类：被多个用户共享
class Document {
public:
    std::string title;
    std::string content;
    
    Document(const std::string& t) : title(t) {
        std::cout << "Document '" << title << "' created\n";
    }
    
    ~Document() {
        std::cout << "Document '" << title << "' destroyed\n";
    }
    
    void append(const std::string& text) {
        content += text;
    }
};

// 用户类：持有文档的引用
class User {
public:
    std::string name;
    std::shared_ptr<Document> doc;
    
    User(const std::string& n) : name(n) {
        std::cout << "User '" << name << "' created\n";
    }
    
    ~User() {
        std::cout << "User '" << name << "' destroyed\n";
    }
    
    void setDocument(std::shared_ptr<Document> d) {
        doc = d;
        std::cout << name << " now has document '" << doc->title << "'\n";
    }
    
    void editDocument(const std::string& text) {
        if (doc) {
            doc->append(text);
            std::cout << name << " edited document\n";
        }
    }
};

int main() {
    // 创建文档
    auto doc = std::make_shared<Document>("Project Plan");
    
    // 创建用户
    auto user1 = std::make_shared<User>("Alice");
    auto user2 = std::make_shared<User>("Bob");
    auto user3 = std::make_shared<User>("Charlie");
    
    // 共享文档
    user1->setDocument(doc);
    std::cout << "Document use_count: " << doc.use_count() << "\n";
    
    user2->setDocument(doc);
    std::cout << "Document use_count: " << doc.use_count() << "\n";
    
    user3->setDocument(doc);
    std::cout << "Document use_count: " << doc.use_count() << "\n";
    
    // 任何用户都可以编辑文档
    user1->editDocument("Introduction\n");
    user2->editDocument("Main content\n");
    user3->editDocument("Conclusion\n");
    
    std::cout << "\nDocument content:\n" << doc->content << "\n";
    
    // 用户离开，文档仍然存在
    user1.reset();
    std::cout << "\nAfter user1 left, use_count: " << doc.use_count() << "\n";
    
    user2.reset();
    std::cout << "After user2 left, use_count: " << doc.use_count() << "\n";
    
    // 最后一个用户离开，文档被销毁
    user3.reset();
    std::cout << "After user3 left\n";
    
    return 0;
}
```

### 5. this指针与shared_ptr

```cpp
#include <iostream>
#include <memory>
#include <vector>

// 使用 enable_shared_from_this 获取 this 的 shared_ptr
class Node : public std::enable_shared_from_this<Node> {
public:
    int id;
    std::vector<std::shared_ptr<Node>> neighbors;
    std::vector<std::weak_ptr<Node>> weak_neighbors;
    
    explicit Node(int i) : id(i) {
        std::cout << "Node " << id << " created\n";
    }
    
    ~Node() {
        std::cout << "Node " << id << " destroyed\n";
    }
    
    void addNeighbor(std::shared_ptr<Node> other) {
        neighbors.push_back(other);
        other->weak_neighbors.push_back(weak_from_this());
    }
    
    void printNeighbors() const {
        std::cout << "Node " << id << " neighbors: ";
        for (const auto& n : neighbors) {
            std::cout << n->id << " ";
        }
        for (const auto& wn : weak_neighbors) {
            if (auto n = wn.lock()) {
                std::cout << n->id << " ";
            }
        }
        std::cout << "\n";
    }
};

int main() {
    // 创建节点
    auto node1 = std::make_shared<Node>(1);
    auto node2 = std::make_shared<Node>(2);
    auto node3 = std::make_shared<Node>(3);
    
    // 建立连接
    node1->addNeighbor(node2);
    node1->addNeighbor(node3);
    node2->addNeighbor(node3);
    
    // 打印邻居
    node1->printNeighbors();
    node2->printNeighbors();
    node3->printNeighbors();
    
    return 0;
}
```

## 4. weak_ptr弱引用

### 1. 基本概念

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    int id;
    explicit Resource(int i) : id(i) {
        std::cout << "Resource " << id << " created\n";
    }
    ~Resource() {
        std::cout << "Resource " << id << " destroyed\n";
    }
};

int main() {
    std::weak_ptr<Resource> weak;
    
    {
        auto shared = std::make_shared<Resource>(1);
        weak = shared;  // 不增加引用计数
        
        std::cout << "shared use_count: " << shared.use_count() << "\n";  // 1
        std::cout << "weak use_count: " << weak.use_count() << "\n";      // 1
        std::cout << "weak expired: " << weak.expired() << "\n";          // false
        
        // 使用 lock() 获取 shared_ptr
        if (auto locked = weak.lock()) {
            std::cout << "Resource id: " << locked->id << "\n";
        }
        
    }  // shared 离开作用域，资源被释放
    
    std::cout << "After scope:\n";
    std::cout << "weak expired: " << weak.expired() << "\n";  // true
    
    // 尝试获取资源
    if (auto locked = weak.lock()) {
        std::cout << "Resource still exists\n";
    } else {
        std::cout << "Resource has been destroyed\n";
    }
    
    return 0;
}
```

### 2. 解决循环引用

```cpp
#include <iostream>
#include <memory>

// ❌ 问题：循环引用导致内存泄漏
class NodeBad {
public:
    int id;
    std::shared_ptr<NodeBad> next;
    std::shared_ptr<NodeBad> prev;
    
    explicit NodeBad(int i) : id(i) {
        std::cout << "NodeBad " << id << " created\n";
    }
    ~NodeBad() {
        std::cout << "NodeBad " << id << " destroyed\n";
    }
};

void demonstrateLeak() {
    auto node1 = std::make_shared<NodeBad>(1);
    auto node2 = std::make_shared<NodeBad>(2);
    
    // 循环引用
    node1->next = node2;
    node2->prev = node1;
    
    std::cout << "node1 use_count: " << node1.use_count() << "\n";  // 2
    std::cout << "node2 use_count: " << node2.use_count() << "\n";  // 2
    
    // 离开作用域时，引用计数不会降为 0
    // 内存泄漏！
}

// ✅ 解决方案：使用 weak_ptr 打破循环
class NodeGood {
public:
    int id;
    std::shared_ptr<NodeGood> next;
    std::weak_ptr<NodeGood> prev;  // 使用 weak_ptr
    
    explicit NodeGood(int i) : id(i) {
        std::cout << "NodeGood " << id << " created\n";
    }
    ~NodeGood() {
        std::cout << "NodeGood " << id << " destroyed\n";
    }
};

void demonstrateNoLeak() {
    auto node1 = std::make_shared<NodeGood>(1);
    auto node2 = std::make_shared<NodeGood>(2);
    
    // 建立连接
    node1->next = node2;
    node2->prev = node1;  // weak_ptr 不增加引用计数
    
    std::cout << "node1 use_count: " << node1.use_count() << "\n";  // 1
    std::cout << "node2 use_count: " << node2.use_count() << "\n";  // 2
    
    // 离开作用域时，正确释放
}

int main() {
    std::cout << "=== Demonstrating memory leak ===\n";
    demonstrateLeak();
    
    std::cout << "\n=== Demonstrating no leak ===\n";
    demonstrateNoLeak();
    
    return 0;
}
```

### 3. 观察者模式

```cpp
#include <iostream>

#include <vector>
#include <string>

// 被观察的主题
class Subject {
public:
    std::string name;
    
    explicit Subject(const std::string& n) : name(n) {
        std::cout << "Subject '" << name << "' created\n";
    }
    
    ~Subject() {
        std::cout << "Subject '" << name << "' destroyed\n";

        std::cout << name << ": " << message << "\n";
    }

// 观察者
class Observer {
public:
    std::string name;
    std::weak_ptr<Subject> subject;  // 使用 weak_ptr 观察主题
    
    explicit Observer(const std::string& n) : name(n) {
        std::cout << "Observer '" << name << "' created\n";
    }
    
    ~Observer() {
        std::cout << "Observer '" << name << "' destroyed\n";
    }
    
    void update(const std::string& message) {
        // 检查主题是否还存在
        if (auto subj = subject.lock()) {
            std::cout << name << " received: ";
            subj->notify(message);
        } else {
            std::cout << name << ": Subject no longer exists\n";
        }
    }
};

int main() {
    auto observer1 = std::make_shared<Observer>("Observer1");
    auto observer2 = std::make_shared<Observer>("Observer2");
    
    {
        // 创建主题
        auto subject = std::make_shared<Subject>("News");
        
        // 观察者订阅主题
        observer1->subject = subject;
        observer2->subject = subject;
        
        // 更新
        observer1->update("Breaking news 1");
        observer2->update("Breaking news 2");
        
        std::cout << "\nSubject use_count: " << subject.use_count() << "\n\n";
        
    }  // 主题被销毁
    
    // 观察者尝试更新
    std::cout << "\nAfter subject destroyed:\n";
    observer1->update("Breaking news 3");
    observer2->update("Breaking news 4");
    
    return 0;
}
```

### 4. 缓存实现

```cpp
#include <iostream>
#include <memory>
#include <map>
#include <string>

class ExpensiveResource {
public:
    std::string key;
    
    explicit ExpensiveResource(const std::string& k) : key(k) {
        std::cout << "Creating expensive resource: " << key << "\n";
    }
    
    ~ExpensiveResource() {
        std::cout << "Destroying expensive resource: " << key << "\n";
    }
    
    void use() {
        std::cout << "Using resource: " << key << "\n";
    }
};

class ResourceCache {
private:
    std::map<std::string, std::weak_ptr<ExpensiveResource>> cache;
    
public:
    std::shared_ptr<ExpensiveResource> get(const std::string& key) {
        // 检查缓存中是否存在
        auto it = cache.find(key);
        
        if (it != cache.end()) {
            // 尝试获取资源
            if (auto resource = it->second.lock()) {
                std::cout << "Cache hit: " << key << "\n";
                return resource;
            }
        }
        
        // 缓存未命中，创建新资源
        std::cout << "Cache miss: " << key << "\n";
        auto resource = std::make_shared<ExpensiveResource>(key);
        cache[key] = resource;
        return resource;
    }
    
    void cleanup() {
        // 清理已过期的缓存项
        for (auto it = cache.begin(); it != cache.end(); ) {
            if (it->second.expired()) {
                std::cout << "Removing expired cache entry: " << it->first << "\n";
                it = cache.erase(it);
            } else {
                ++it;
            }
        }
    }
};

int main() {
    ResourceCache cache;
    
    // 第一次获取
    {
        auto r1 = cache.get("resource1");
        r1->use();
        
        auto r2 = cache.get("resource1");  // 缓存命中
        r2->use();
        
        auto r3 = cache.get("resource2");
        r3->use();
        
    }  // 资源被释放
    
    std::cout << "\nAfter resources released:\n";
    cache.cleanup();
    
    // 再次获取
    auto r4 = cache.get("resource1");  // 需要重新创建
    r4->use();
    
    return 0;
}
```

## 5. 自定义删除器

### 1. 基本用法

```cpp
#include <iostream>
#include <memory>
#include <fstream>
#include <cstdio>

// 自定义删除器：函数指针
void fileDeleter(FILE* fp) {
    if (fp) {
        std::cout << "Closing file\n";
        fclose(fp);
    }
}

// 自定义删除器：Lambda
auto arrayDeleter = [](int* p) {
    std::cout << "Deleting array\n";
    delete[] p;
};

int main() {
    // 使用函数指针作为删除器
    std::unique_ptr<FILE, decltype(&fileDeleter)> 
        file1(fopen("test.txt", "w"), fileDeleter);
    
    // 使用 Lambda 作为删除器
    std::unique_ptr<int[], decltype(arrayDeleter)>
        arr1(new int[10], arrayDeleter);
    
    // 更简洁的方式（C++20）
    auto file2 = std::unique_ptr<FILE, decltype(&fileDeleter)>{
        fopen("test2.txt", "w"), fileDeleter
    };
    
    return 0;
}
```

### 2. 管理非内存资源

```cpp
#include <iostream>
#include <memory>
#include <mutex>

// 管理互斥锁
std::mutex mtx;

auto lockDeleter = [](std::mutex* m) {
    std::cout << "Unlocking mutex\n";
    m->unlock();
};

std::unique_ptr<std::mutex, decltype(lockDeleter)> lockMutex() {
    mtx.lock();
    std::cout << "Mutex locked\n";
    return {&mtx, lockDeleter};
}

// 管理自定义资源
class NetworkConnection {
public:
    int socket;
    
    NetworkConnection(int s) : socket(s) {
        std::cout << "Connection opened: socket " << socket << "\n";
    }
    
    void close() {
        std::cout << "Connection closed: socket " << socket << "\n";
    }
};

auto connectionDeleter = [](NetworkConnection* conn) {
    if (conn) {
        conn->close();
        delete conn;
    }
};

int main() {
    // 自动管理互斥锁
    {
        auto lock = lockMutex();
        std::cout << "Critical section\n";
    }  // 自动解锁
    
    // 管理网络连接
    {
        std::unique_ptr<NetworkConnection, decltype(connectionDeleter)>
            conn(new NetworkConnection(42), connectionDeleter);
        
        std::cout << "Using connection\n";
    }  // 自动关闭连接
    
    return 0;
}
```

### 3. shared_ptr自定义删除器

```cpp
#include <iostream>
#include <memory>

        std::cout << "Resource " << id << " created\n";
    }

        std::cout << "Resource " << id << " destroyed\n";

};

        delete p;
    };
    
    std::shared_ptr<Resource> ptr2(new Resource(2), deleter2);
    
    // 可以放在同一个容器中
    std::vector<std::shared_ptr<Resource>> resources;
    resources.push_back(ptr);
    resources.push_back(ptr2);
    
    return 0;
}
```

### 2. 最佳实践

```cpp
#include <iostream>
#include <memory>
#include <vector>

// ✅ 最佳实践总结

// 1. 默认使用 unique_ptr
class Example1 {
    std::unique_ptr<int> data;
public:
    Example1() : data(std::make_unique<int>(42)) {}
};

// 2. 需要共享所有权时使用 shared_ptr
class Example2 {
    std::shared_ptr<std::vector<int>> data;
public:
    std::shared_ptr<std::vector<int>> getData() {
        return data;  // 共享数据
    }
};

// 3. 观察者或打破循环使用 weak_ptr
class Example3 {
    std::weak_ptr<int> observer;
public:
    void observe(std::shared_ptr<int> target) {
        observer = target;
    }
    
    bool isAlive() {
        return !observer.expired();
    }
};

// 4. 使用 make_unique 和 make_shared
void example4() {
    // ✅ 推荐
    auto p1 = std::make_unique<int>(42);
    auto p2 = std::make_shared<int>(42);
    
    // ❌ 不推荐
    std::unique_ptr<int> p3(new int(42));
    std::shared_ptr<int> p4(new int(42));
}

// 5. 不要从原始指针创建多个智能指针
void example5() {
    int* raw = new int(42);
    
    // ❌ 错误：两个智能指针管理同一个对象
    // std::unique_ptr<int> p1(raw);
    // std::unique_ptr<int> p2(raw);  // 双重删除！
    
    // ✅ 正确：立即转移所有权
    std::unique_ptr<int> p1(raw);
    // raw 不再使用
}

// 6. 函数参数传递
class Widget {
public:
    int id;
    explicit Widget(int i) : id(i) {}
};

// 独占所有权：按值传递
void takeOwnership(std::unique_ptr<Widget> w) {
    std::cout << "Took ownership of widget " << w->id << "\n";
}

// 共享所有权：按值传递
void shareOwnership(std::shared_ptr<Widget> w) {
    std::cout << "Shared widget " << w->id << "\n";
}

// 只读访问：传递引用
void useWidget(const Widget& w) {
    std::cout << "Using widget " << w.id << "\n";
}

// 可选所有权：传递原始指针
void maybeModify(Widget* w) {
    if (w) {
        w->id *= 2;
    }
}

int main() {
    // 函数调用示例
    auto widget1 = std::make_unique<Widget>(1);
    takeOwnership(std::move(widget1));
    
    auto widget2 = std::make_shared<Widget>(2);
    shareOwnership(widget2);
    
    auto widget3 = std::make_unique<Widget>(3);
    useWidget(*widget3);
    
    auto widget4 = std::make_unique<Widget>(4);
    maybeModify(widget4.get());
    
    return 0;
}
```

## 总结

### 4. 关键要点

| 智能指针 | 所有权 | 性能开销 | 使用场景 |
|----------|--------|----------|----------|
| `unique_ptr` | 独占 | 几乎无 | 默认选择、工厂函数 |
| `shared_ptr` | 共享 | 引用计数 | 共享资源、观察者模式 |
| `weak_ptr` | 无 | 几乎无 | 打破循环、缓存 |

### 5. 最佳实践清单

```
□ 默认使用 unique_ptr
□ 需要共享时使用 shared_ptr
□ 打破循环引用使用 weak_ptr
□ 优先使用 make_unique 和 make_shared
□ 不要从原始指针创建多个智能指针
□ 函数参数按需传递
□ 使用 vector 代替智能指针数组
□ 自定义删除器管理特殊资源
```
