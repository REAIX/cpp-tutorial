## 1. auto深入

### 1. auto推导规则

```cpp
// 规则1：auto 忽略顶层const和引用
const int ci = 10;
auto a = ci;          // int（忽略const）
const auto b = ci;    // const int（显式加const）

// 规则2：auto& 保留const
auto& c = ci;         // const int&（保留const）

// 规则3：auto&& 万能引用
int x = 10;
auto&& r1 = x;        // int&（左值 → int& && → int&）
auto&& r2 = 10;       // int&&（右值 → int&&）
```

### 2. auto推导的更多细节

```cpp
// auto的推导实际上是模板参数推导的等价形式
// template<typename U> void f(U u);  f(expr);  ≈  auto u = expr;
// template<typename U> void f(U& u); f(expr);  ≈  auto& u = expr;

// 推导规则详解
const volatile int cvi = 100;
auto v1 = cvi;           // int（去掉顶层const和volatile）
auto& v2 = cvi;          // const volatile int&（引用保留修饰符）

// 指针的auto推导
const int* pci = &ci;
auto p1 = pci;           // const int*（指针本身的const被忽略，指向的const保留）
auto* p2 = pci;          // const int*（等价于p1）
const auto p3 = pci;     // const int* const（指针本身为const）

// auto与std::initializer_list
auto list = {1, 2, 3};   // C++11: std::initializer_list<int>
// auto list2{1, 2, 3};  // C++11: 错误！C++17: 错误，需要直接初始化

// 结构化绑定中的auto
std::map<std::string, int> scores;
for (const auto& [name, score] : scores) {
    // name: const std::string&
    // score: const int&
}
```

### 3. auto与初始化列表

```cpp
auto x1 = {1, 2, 3};     // std::initializer_list<int>
auto x2 = {1};            // std::initializer_list<int>
auto x3{1};               // int（C++17）
auto x4 = 1;              // int

// 函数返回auto不能推导initializer_list
auto f() { return {1, 2, 3}; }  // 编译错误！
```

### 4. 数组与函数

```cpp
int arr[5] = {1, 2, 3, 4, 5};
auto a = arr;          // int*（数组退化为指针）
auto& b = arr;         // int(&)[5]（引用保留数组类型）

void func(int);
auto f = func;         // void(*)(int)（函数退化为函数指针）
auto& g = func;        // void(&)(int)（引用保留函数类型）
```

### 5. auto在范围for循环中的推导

```cpp
#include <vector>
#include <map>

void range_for_deduction() {
    std::vector<int> v = {1, 2, 3, 4, 5};
    
    // 按值拷贝
    for (auto x : v) {        // int x
        x *= 2;               // 不影响原容器
    }
    
    // 按引用修改
    for (auto& x : v) {       // int& x
        x *= 2;               // 修改原容器
    }
    
    // const引用（只读）
    for (const auto& x : v) { // const int& x
        // x *= 2;            // 编译错误
    }
    
    // 通用引用（用于代理迭代器）
    for (auto&& x : v) {      // int& x（左值折叠）
        // 对于vector<bool>这类有代理迭代器的容器尤其有用
    }
    
    // 结构化绑定
    std::map<int, std::string> m = {{1, "one"}, {2, "two"}};
    for (const auto& [key, val] : m) {
        // key: const int&
        // val: const std::string&
    }
}
```

### 6. C++20的auto参数（缩写函数模板）

```cpp

auto add(auto a, auto b) {
    return a + b;
}
// 等价于：
// template<typename T, typename U>
// auto add(T a, U b) -> decltype(a + b)

// 模板的可约束auto参数
```

## 2. decltype

### 1. 基本规则

```cpp
int x = 10;
const int& cx = x;

decltype(x) a;         // int
decltype(cx) b = x;    // const int&
decltype(10) c;        // int（右值→int）

int&& rr = 5;
decltype(rr) d = 5;    // int&&

// decltype 不忽略const和引用！
```

### 2. decltype与表达式

```cpp
int x = 10;
decltype(x) a;           // int（变量名 → 变量类型）
decltype((x)) b = x;     // int&（加括号 → 左值引用）
decltype(x + 1) c;       // int（表达式 → 右值）
decltype(x = 5) d = x;   // int&（赋值表达式 → 左值引用）
```

### 3. decltype的更多表达式示例

```cpp
#include <vector>
#include <iostream>

struct Widget {
    int value;
    int& get_ref() { return value; }
    int get_val() const { return value; }
};

void decltype_expressions() {
    Widget w{42};
    const Widget& cw = w;
    
    // 成员访问
    decltype(w.value) d1;           // int
    decltype((w.value)) d2 = w.value;  // int&（括号导致左值）
    decltype(cw.value) d3;          // int（const通过顶层const被忽略?）
    // 实际上是const int，因为cw是const引用，value被视为const int
    
    // 成员函数调用
    decltype(w.get_ref()) d4 = w.value;  // int&
    decltype(w.get_val()) d5;            // int
    decltype(cw.get_ref()) d6 = w.value; // int&（即使cw是const）
    
    // 指针与解引用
    Widget* pw = &w;
    decltype(pw) d7;                // Widget*
    decltype(*pw) d8 = w;           // Widget&（解引用产生左值）
    
    // 条件表达式
    decltype(true ? 1 : 2) d9;      // int（编译期确定）
    int i = 0;
    decltype(true ? i : i) d10 = i; // int&（条件表达式结果是左值）
}

// decltype在SFINAE中的应用
template<typename T>
auto get_size(const T& container) -> decltype(container.size()) {
    return container.size();
}
// 如果T没有size()成员，SFINAE会忽略这个重载
```

### 4. decltype与函数返回值

```cpp
int f();
decltype(f()) a;         // int（函数调用返回类型）

int& g();
decltype(g()) b = a;     // int&（返回引用）
```

## 3. decltype(auto)

```cpp
// decltype(auto) = 用decltype规则推导auto
int x = 10;
const int& ref = x;

auto a = ref;              // int（auto忽略const和引用）
decltype(auto) b = ref;    // const int&（decltype保留）

// 函数返回中使用
template<typename Container>
decltype(auto) get_element(Container& c, size_t i) {
    return c[i];  // 保留operator[]的返回类型（可能是引用）
}

std::vector<int> v{1, 2, 3};
get_element(v, 0) = 10;  // OK，返回int&
```

### 1. decltype(auto)的进阶用法

```cpp
#include <utility>
#include <iostream>

// 完美转发包装器
template<typename F, typename... Args>
decltype(auto) invoke(F&& f, Args&&... args) {
    return std::forward<F>(f)(std::forward<Args>(args)...);
}

// 使用decltype(auto)实现属性访问器
template<typename T, typename Prop>
decltype(auto) access_prop(T& obj, Prop prop) {
    return (obj.*prop);  // 返回成员指针引用的类型
}

class Data {
public:
    int x = 10;
    std::string s = "hello";
};

// 返回函数对象
decltype(auto) get_lambda() {
    static int counter = 0;
    return [&counter]() { return ++counter; };
}

// decltype(auto)的注意事项
struct Container {
    int data[10];
    
    // 正确：返回int&
    decltype(auto) operator[](size_t i) {
        return data[i];
    }
    
    // 危险：返回int&，但data是成员数组
    // decltype(auto) get_data() { return data; }  // 返回int(&)[10]，没问题
    
    // 特别危险的情况
    decltype(auto) bad_get() {
        int local = 42;
        return (local);  // 返回int& → 悬空引用！
    }
};
```

### 2. decltype(auto)在递归模板中的应用

```cpp
#include <iostream>

// 递归计算嵌套类型的值
template<typename T>
decltype(auto) get_nested_value(T& t) {
    if constexpr (requires { t.value; }) {
        return get_nested_value(t.value);
    } else {
        return t;
    }
}

struct Level1 { int value; };
struct Level2 { Level1 value; };
struct Level3 { Level2 value; };

void test_nested() {
    Level3 l3{Level2{Level1{42}}};
    decltype(auto) result = get_nested_value(l3);
    std::cout << result << std::endl;  // 42
}
```

## 4. 返回类型推导

### 1. auto返回类型

```cpp
auto add(int a, int b) {
    return a + b;  // 推导为int
}

// C++14支持
auto get_vector() {
    std::vector<int> v{1, 2, 3};
    return v;  // 推导为std::vector<int>
}
```

### 2. decltype(auto)返回类型

```cpp
// 需要返回引用时
template<typename T>
decltype(auto) forward_value(T&& x) {
    return std::forward<T>(x);
}

// 注意：返回局部变量的引用是危险的
decltype(auto) dangerous() {
    int x = 10;
    return (x);  // 返回int& → 悬空引用！
}

decltype(auto) safe() {
    int x = 10;
    return x;  // 返回int → OK
}
```

### 3. 尾置返回类型

```cpp
template<typename T, typename U>
auto add(T t, U u) -> decltype(t + u) {
    return t + u;
}
```

### 4. 多返回语句的推导一致性

```cpp
#include <string>
#include <stdexcept>

// 多个return语句必须推导出相同类型
auto conditional_return(bool flag) {
    if (flag) {
        return 42;     // int
    } else {
        return 0;      // int，与上面一致
    }
}

// 编译错误：推导不一致
// auto bad_return(bool flag) {
//     if (flag) return 42;          // int
//     else return 3.14;             // double → 冲突！
// }

// 解决方案：显式转换或使用decltype
auto good_return(bool flag) -> double {
    if (flag) return 42;        // 隐式转换为double
    else return 3.14;
}

// 递归函数的返回类型推导
auto factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);  // 需要先看到递归调用
    // C++14要求递归调用前的return语句能够推导出类型
}
```

### 5. 返回类型推导与模板

```cpp
#include <vector>
#include <algorithm>

// 模板函数返回类型推导
template<typename Container>
auto find_middle(Container& c) {
    auto it = c.begin();
    std::advance(it, c.size() / 2);
    return it;  // 推导为Container::iterator
}

// 结合decltype(auto)的完美转发返回
template<typename T>
decltype(auto) get_forward(T&& t) {
    return std::forward<T>(t);
}

// 工厂函数
template<typename T, typename... Args>
auto make_vector(Args&&... args) {
    return std::vector<T>{std::forward<Args>(args)...};
}
```

## 5. CTAD类模板参数推导

### 1. C++17 CTAD

```cpp
// 之前
std::pair<int, double> p1(1, 2.5);
std::vector<int> v1{1, 2, 3};
std::tuple<int, double, std::string> t1(1, 2.5, "hello");

// C++17 CTAD
std::pair p2(1, 2.5);              // 推导为 pair<int, double>
std::vector v2{1, 2, 3};           // 推导为 vector<int>
std::tuple t2(1, 2.5, "hello");    // 推导为 tuple<int, double, const char*>
std::optional opt{42};              // 推导为 optional<int>
std::lock_guard lock(mutex);        // 推导为 lock_guard<mutex>
```

### 2. 自定义推导指引

```cpp
template<typename T>
struct Wrapper {
    T value;
    Wrapper(T v) : value(v) {}
};

// 隐式推导：Wrapper w(42) → Wrapper<int>

// 自定义推导指引
Wrapper(const char*) -> Wrapper<std::string>;

Wrapper w1(42);           // Wrapper<int>
Wrapper w2("hello");      // Wrapper<std::string>（自定义指引）
```

### 3. CTAD的高级案例

```cpp
#include <vector>
#include <string>
#include <memory>
#include <tuple>
#include <array>

// 1. 聚合类型的CTAD
template<typename T, size_t N>
struct ArrayWrapper {
    T data[N];
};

// C++20起支持聚合类型的CTAD
// ArrayWrapper aw{1, 2, 3};  // C++20: ArrayWrapper<int, 3>

// 2. 复杂推导指引
template<typename T>
struct JsonValue {
    T value;
    JsonValue(T v) : value(v) {}
};

// 字符串字面量推导指引
JsonValue(const char*) -> JsonValue<std::string>;
// 整数推导指引（选择int64_t）
JsonValue(int) -> JsonValue<int64_t>;

// 3. 非类型模板参数的CTAD
template<typename T, auto Default>
struct Config {
    T value = Default;
};

// Config c{42};  // C++17: Config<int, 42>

// 4. pair/tuple的异质推导
void ctad_examples() {
    auto p = std::pair(1, 3.14);         // pair<int, double>
    auto t = std::tuple(1, "hello", 3.14f);  // tuple<int, const char*, float>
    
    // 推导指引的解析规则：优先选择最匹配的
    auto v = std::vector{1, 2, 3};       // vector<int>
    // std::vector v2{v.begin(), v.end()}; // C++17: vector<int>（用迭代器对）
}

// 5. CTAD与工厂模式的取舍
template<typename T>
struct SharedWrapper {
    std::shared_ptr<T> ptr;
};

// 对比：
auto w1 = SharedWrapper<int>{std::make_shared<int>(42)};  // 显式
// auto w2 = SharedWrapper{std::make_shared<int>(42)};       // CTAD: SharedWrapper<int>
// 实际上，make_shared<int>返回shared_ptr<int>，所以推导为SharedWrapper<int>
```

### 4. CTAD的限制与注意事项

```cpp
#include <vector>
#include <array>

// CTAD的限制：
```

## 6. 类型推导在模板元编程中的应用

### 1. 类型萃取与推导

```cpp
#include <type_traits>
#include <iostream>

// 用decltype实现类型函数
template<typename T>
struct AddPointer {
    using type = std::add_pointer_t<T>;
};

// 用decltype实现编译期类型计算
template<typename T, typename U>
struct CommonType {
    using type = decltype(true ? std::declval<T>() : std::declval<U>());
};

// 推导函数返回类型
template<typename F, typename... Args>
using InvokeResult = decltype(std::declval<F>()(std::declval<Args>()...));

// 使用示例
auto lambda = [](int x, double y) { return x + y; };
using ResultType = InvokeResult<decltype(lambda), int, double>;
// ResultType 是 double
```

### 2. 用auto推导实现编译期函数

```cpp
// C++17 constexpr lambda实现编译期计算
constexpr auto compile_time_factorial = [](int n) {
    int result = 1;
    for (int i = 2; i <= n; ++i) result *= i;
    return result;
};

// 编译期计算的两种方式
template<int N>
struct Factorial {
    static constexpr int value = N * Factorial<N - 1>::value;
};

template<>
struct Factorial<0> {
    static constexpr int value = 1;
};

// auto结合constexpr
constexpr auto factorial_v = Factorial<5>::value;  // 120
constexpr auto lambda_fact = compile_time_factorial(5);  // 120
```

### 3. 类型列表与推导

```cpp
#include <tuple>
#include <type_traits>

// 类型列表
template<typename... Types>
struct TypeList {};

// 获取类型列表中的第N个类型
template<size_t N, typename List>
struct TypeAt;

template<size_t N, typename First, typename... Rest>
struct TypeAt<N, TypeList<First, Rest...>> {
    using type = typename TypeAt<N - 1, TypeList<Rest...>>::type;
};

template<typename First, typename... Rest>
struct TypeAt<0, TypeList<First, Rest...>> {
    using type = First;
};

// 用auto推导实现类型过滤
template<typename T, typename List>
struct FilterIntegral;

template<typename T, typename... Rest>
struct FilterIntegral<T, TypeList<T, Rest...>> {
    using type = std::conditional_t<
        std::is_integral_v<T>,
        decltype(TypeList<T>{}),
        TypeList<>
    >;
};

// decltype在元编程中的核心作用
template<typename... Args>
auto make_tuple_from_types(Args&&... args) {
    return std::make_tuple(std::forward<Args>(args)...);
    // 返回类型由编译器自动推导为 std::tuple<Args...>
}
```

### 4. 表达式模板与类型推导

```cpp
#include <vector>
#include <iostream>

template<typename L, typename Op, typename R>
struct Expression {
    const L& lhs;
    const R& rhs;
    
    decltype(auto) operator[](size_t i) const {
        return Op::apply(lhs[i], rhs[i]);
    }
};

struct Add {
    template<typename T, typename U>
    static auto apply(const T& a, const U& b) {
        return a + b;
    }
};

// 延迟求值的向量加法
template<typename L, typename R>
auto operator+(const L& lhs, const R& rhs) {
    return Expression<L, Add, R>{lhs, rhs};
}

// 使用表达式模板
void expression_template_demo() {
    std::vector<double> a = {1.0, 2.0, 3.0};
    std::vector<double> b = {4.0, 5.0, 6.0};
    std::vector<double> c = {7.0, 8.0, 9.0};
    
    // 类型推导自动构建表达式树
    auto expr = a + b + c;  // Expression<Expression<...>, Add, vector<double>>
    // 不会立即计算，只有在访问元素时才计算
    
    double result = expr[0];  // 1.0 + 4.0 + 7.0 = 12.0
    std::cout << result << std::endl;
}
```

### 5. SFINAE与类型推导

```cpp
#include <type_traits>

// SFINAE通过decltype实现
template<typename T>
auto increment_if_possible(T& t) -> decltype(++t, void()) {
    ++t;

// C++20的requires更简洁，但decltype的SFINAE仍广泛使用

template<typename T, typename U, typename = void>
struct HasPushBack : std::false_type {};

template<typename T, typename U>
struct HasPushBack<T, U, std::void_t<push_back_result<T, U>>> 
    : std::true_type {};
```

## 7. 类型推导陷阱

### 1. auto与引用

```cpp
std::vector<bool> v{true, false, true};
auto x = v[0];  // std::vector<bool>::reference，不是bool！
// x 是代理对象，不是真正的bool
```

### 2. auto与字符串字面量

```cpp
auto s1 = "hello";       // const char*
auto s2 = std::string("hello");  // std::string
auto s3 = "hello"s;      // std::string（using namespace std::literals）
```

### 3. 花括号初始化

```cpp
auto x1 = 42;       // int
auto x2(42);        // int
auto x3{42};        // int（C++17）
auto x4 = {42};     // std::initializer_list<int>
```

### 4. decltype(auto)的引用陷阱

```cpp
#include <vector>

// 意外返回引用
template<typename T>
decltype(auto) get_value_bad(T& container) {
    return container[0];
    // 如果T是std::vector<int>，返回int&
    // 如果T是std::vector<bool>，返回代理对象的引用（悬空！）
}

// 修正版本
template<typename T>
auto get_value_good(T& container) -> typename T::value_type {
    return container[0];
}

// 括号导致的引用
decltype(auto) confusing() {
    int x = 42;
    return x;   // 返回int（拷贝）
}

decltype(auto) also_confusing() {
    int x = 42;
    return (x); // 返回int&（悬空引用！）
}
```

### 5. 完美转发中的推导问题

```cpp
#include <utility>
#include <vector>

// 错误：按值传递丢失引用信息
template<typename T>
void wrapper_bad(T arg) {
    some_function(std::forward<T>(arg));  // T始终是值类型
}

// 正确：转发引用
template<typename T>
void wrapper_good(T&& arg) {
    some_function(std::forward<T>(arg));
}

// 初始化列表不能完美转发
template<typename T>
void forwarding_func(T&& arg) {}

void test_initializer_list() {
    // forwarding_func({1, 2, 3});  // 编译错误！
    auto il = {1, 2, 3};
    forwarding_func(il);  // OK
}
```

### 6. 泛型Lambda的类型推导

```cpp
// 每个auto参数产生独立的模板参数
auto lambda = [](auto a, auto b) { return a + b; };
// 相当于：
// template<typename T1, typename T2>
// auto operator()(T1 a, T2 b) const { return a + b; }

// C++14的泛型lambda可以捕获
auto counter = [count = 0]() mutable { return ++count; };

// C++20的模板lambda（显式模板参数）
auto explicit_lambda = []<typename T>(T x) {
    return std::to_string(x);
};

// 显式指定模板参数
auto result = explicit_lambda.template operator()<int>(42);
```

### 7. CTAD推导指引的歧义

```cpp
#include <string>
#include <vector>

// 推导指引冲突
template<typename T>
struct Container {
    Container(T val) {}
    Container(T* begin, T* end) {}
};

// 自定义指引可能导致歧义
Container(const char*) -> Container<std::string>;

// 使用：
Container c1("hello");       // Container<std::string>（自定义指引）
Container c2(42);            // Container<int>（隐式推导）

// 两个迭代器参数的歧义
std::vector<int> v = {1, 2, 3};
// Container c3(v.begin(), v.end());
// 可能推导为Container<std::vector<int>::iterator>
// 或Container<int>（如果begin/end返回int*）
```

## 8. 本章小结

| 主题 | 核心内容 |
|------|---------|
| **auto** | 忽略顶层const/引用，数组/函数退化 |
| **decltype** | 保留const/引用，(x)是左值引用 |
| **decltype(auto)** | 用decltype规则推导auto |

| **陷阱** | vector\<bool\>、花括号初始化、字符串字面量 |

### 1. 类型推导规则速查

| 语法 | 值类别 | 推导规则 |
|------|--------|---------|
| `auto` | 值 | 忽略顶层const/引用/volatile |
| `auto&` | 引用 | 保留const |
| `auto&&` | 万能引用 | 左值→左值引用，右值→右值引用 |
| `decltype(var)` | 值 | 返回声明类型 |
| `decltype((var))` | 引用 | 加括号→左值引用 |
| `decltype(expr)` | 取决于表达式 | 左值→左值引用，右值→值 |
| `decltype(auto)` | 取决于初始化 | 用decltype规则推导 |
| `CTAD` | 值 | 从构造函数参数推导 |
