## 2. 自动类型推导

### 1. auto关键字

```cpp
#include <iostream>
#include <vector>
#include <map>

int main() {
    // 基本类型推导
    auto i = 42;              // int
    auto d = 3.14;            // double
    auto s = "hello";         // const char*
    auto str = std::string("hello");  // std::string
    
    // 复杂类型推导（最常用场景）
    std::map<std::string, std::vector<int>> data;
    
    // C++98写法：冗长
    std::map<std::string, std::vector<int>>::iterator it1 = data.begin();
    
    // C++11写法：简洁
    auto it2 = data.begin();  // 自动推导迭代器类型
    
    // 遍历时使用auto
    for (auto& pair : data) {
        std::cout << pair.first << std::endl;
    }
    
    return 0;
}
```

### 2. auto使用规则

```cpp
#include <iostream>

int main() {
    int x = 10;
    
    // 1. auto会忽略顶层const
    const int cx = 20;
    auto a = cx;        // int（const被忽略）
    
    // 2. auto会保留底层const（指针/引用）
    const int* ptr = &x;
    auto b = ptr;       // const int*（底层const保留）
    
    // 3. 引用必须显式指定
    auto& ref = x;      // int&
    const auto& cref = x;  // const int&
    
    // 4. auto与指针
    auto* p1 = &x;      // int*
    auto p2 = &x;       // int*（等价）
    
    // 5. 初始化列表
    auto list = {1, 2, 3};  // std::initializer_list<int>
    
    return 0;
}
```

### 3. decltype关键字

```cpp
#include <iostream>
#include <type_traits>

// decltype：获取表达式的类型

int main() {
    int x = 10;
    const int& ref = x;
    
    // 基本用法
    decltype(x) a = x;      // int
    decltype(ref) b = x;    // const int&
    decltype((x)) c = x;    // int&（注意：双层括号返回引用）
    
    // 用于返回类型推导
    auto add = [](int a, int b) -> decltype(a + b) {
        return a + b;
    };
    
    return 0;
}

// 用于泛型编程（模板函数必须在全局作用域或命名空间作用域定义）
template<typename T, typename U>
auto add(T t, U u) -> decltype(t + u) {
    return t + u;
}
```

### 4. auto vs decltype

```cpp
#include <iostream>
#include <type_traits>

int x = 10;

int& getRef() { return x; }
int getValue() { return x; }

int main() {
    // auto：推导初始化表达式的类型
    auto a = getRef();      // int（引用被忽略）
    auto b = getValue();    // int
    
    // decltype：保留表达式的精确类型
    decltype(getRef()) c = x;  // int&（引用保留）
    decltype(getValue()) d = x; // int
    
    // 实际应用场景
    std::vector<int> vec = {1, 2, 3};
    
    // auto用于简化
    for (auto it = vec.begin(); it != vec.end(); ++it) {
        std::cout << *it << " ";
    }
    
    // decltype用于精确类型
    decltype(vec)::value_type val = 42;  // int
    
    return 0;
}
```

## 3. 右值引用与移动语义

### 1. 概念与定义

**左值（lvalue）**：有名字、有地址，可以取地址的表达式。例如变量名、数组元素、函数返回的左值引用等。

**右值（rvalue）**：无名、临时对象，不能取地址的表达式。例如字面量、表达式结果、函数返回的临时对象等。

**左值引用（lvalue reference）**：绑定到左值的引用，使用`&`符号。例如`int& ref = x;`。

**右值引用（rvalue reference）**：C++11引入的新特性，绑定到右值的引用，使用`&&`符号。例如`int&& ref = 10;`。

**移动语义（move semantics）**：C++11引入的新特性，用于支持高效的资源转移。移动语义可以将资源从一个对象转移到另一个对象，避免拷贝，提高性能。

**std::move**：C++11引入的函数模板，用于将左值转换为右值引用，用于移动构造和移动赋值。例如`std::move(x)`。

### 2. 左值与右值

```cpp
#include <iostream>
#include <string>

int main() {
    // 左值：有名字、有地址，可以取地址
    int x = 10;           // x是左值
    int* p = &x;          // 可以取地址
    std::string str = "hello";  // str是左值
    
    // 右值：无名、临时对象，不能取地址
    int y = 10 + 20;      // 10 + 20 是右值
    // int* p2 = &(10 + 20);  // 错误：不能取右值地址
    std::string s = std::string("world");  // 临时对象是右值
    
    // 左值引用
    int& lr = x;          // 正确：左值引用绑定左值
    // int& lr2 = 10;     // 错误：左值引用不能绑定右值
    const int& clr = 10;  // 正确：const左值引用可以绑定右值
    
    // 右值引用（C++11）
    int&& rr = 10;        // 正确：右值引用绑定右值
    int&& rr2 = x + y;    // 正确：表达式结果是右值
    // int&& rr3 = x;     // 错误：右值引用不能绑定左值
    
    return 0;
}
```

### 3. 移动语义

```cpp
#include <iostream>
#include <string>
#include <utility>

class String {
private:
    char* data;
    size_t len;
    
public:
    // 构造函数
    String(const char* s = "") {
        len = strlen(s);
        data = new char[len + 1];
        strcpy(data, s);
        std::cout << "构造: " << data << "\n";
    }
    
    // 析构函数
    ~String() {
        std::cout << "析构: " << (data ? data : "null") << "\n";
        delete[] data;
    }
    
    // 拷贝构造（深拷贝）
    String(const String& other) {
        len = other.len;
        data = new char[len + 1];
        strcpy(data, other.data);
        std::cout << "拷贝构造: " << data << "\n";
    }
    
    // 移动构造（浅拷贝+置空）
    String(String&& other) noexcept {
        // 直接"偷"走资源
        data = other.data;
        len = other.len;
        other.data = nullptr;  // 置空源对象
        other.len = 0;
        std::cout << "移动构造: " << data << "\n";
    }
    
    // 拷贝赋值
    String& operator=(const String& other) {
        if (this != &other) {
            delete[] data;
            len = other.len;
            data = new char[len + 1];
            strcpy(data, other.data);
            std::cout << "拷贝赋值: " << data << "\n";
        }
        return *this;
    }
    
    // 移动赋值
    String& operator=(String&& other) noexcept {
        if (this != &other) {
            delete[] data;
            data = other.data;
            len = other.len;
            other.data = nullptr;
            other.len = 0;
            std::cout << "移动赋值: " << data << "\n";
        }
        return *this;
    }
    
    const char* c_str() const { return data; }
};

int main() {
    std::cout << "=== 创建对象 ===\n";
    String s1("Hello");
    
    std::cout << "\n=== 拷贝构造 ===\n";
    String s2 = s1;  // 调用拷贝构造
    
    std::cout << "\n=== 移动构造 ===\n";
    String s3 = std::move(s1);  // 调用移动构造
    
    std::cout << "\n=== 返回临时对象 ===\n";
    auto createString = []() { return String("World"); };
    String s4 = createString();  // 可能触发移动构造或RVO
    
    std::cout << "\n=== 程序结束 ===\n";
    return 0;
}
```

### 4. std::move的使用

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <utility>

int main() {
    // std::move：将左值转换为右值引用
    std::string str = "Hello";
    std::string str2 = std::move(str);  // 移动后str为空
    
    std::cout << "str: " << str << "\n";    // 空（或未定义）
    std::cout << "str2: " << str2 << "\n";  // Hello
    
    // 在容器中使用
    std::vector<std::string> vec;
    std::string s = "World";
    
    vec.push_back(s);           // 拷贝
    vec.push_back(std::move(s)); // 移动，s变为空
    
    std::cout << "s after move: " << s << "\n";
    
    // 移动语义的性能优势
    std::vector<std::string> vec2;
    vec2.reserve(3);
    
    std::string a = "a", b = "b", c = "c";
    vec2.push_back(std::move(a));  // 移动，无需分配新内存
    vec2.push_back(std::move(b));
    vec2.push_back(std::move(c));
    
    return 0;
}
```

## 4. 智能指针

### 1. unique_ptr

```cpp
#include <iostream>
#include <memory>

class Resource {
public:
    Resource(int id) : id_(id) {
        std::cout << "Resource " << id_ << " created\n";
    }
    ~Resource() {
        std::cout << "Resource " << id_ << " destroyed\n";
    }
    void use() { std::cout << "Using resource " << id_ << "\n"; }
private:
    int id_;
};

int main() {
    // 创建unique_ptr
    std::unique_ptr<Resource> p1(new Resource(1));
    p1->use();
    
    // C++14推荐：make_unique
    auto p2 = std::make_unique<Resource>(2);
    p2->use();
    
    // 不能拷贝
    // auto p3 = p1;  // 错误！
    
    // 可以移动
    auto p3 = std::move(p1);
    // p1现在是nullptr
    
    if (!p1) {
        std::cout << "p1 is null\n";
    }
    
    // 自定义删除器
    auto deleter = [](Resource* p) {
        std::cout << "Custom delete\n";
        delete p;
    };
    std::unique_ptr<Resource, decltype(deleter)> p4(new Resource(3), deleter);
    
    return 0;
}
```

### 2. shared_ptr

```cpp
#include <iostream>
#include <memory>

class SharedResource {
public:
    SharedResource(const std::string& name) : name_(name) {
        std::cout << name_ << " created\n";
    }
    ~SharedResource() {
        std::cout << name_ << " destroyed\n";
    }
private:
    std::string name_;
};

int main() {
    // 创建shared_ptr
    auto sp1 = std::make_shared<SharedResource>("Resource1");
    std::cout << "count: " << sp1.use_count() << "\n";  // 1
    
    {
        auto sp2 = sp1;  // 共享所有权
        std::cout << "count: " << sp1.use_count() << "\n";  // 2
        
        auto sp3 = sp1;
        std::cout << "count: " << sp1.use_count() << "\n";  // 3
    }
    // sp2, sp3离开作用域
    std::cout << "count: " << sp1.use_count() << "\n";  // 1
    
    return 0;
}
```

### 3. weak_ptr

```cpp
#include <iostream>
#include <memory>

class Node {
public:
    std::string name;
    std::shared_ptr<Node> next;
    std::weak_ptr<Node> prev;  // 使用weak_ptr避免循环引用
    
    Node(const std::string& n) : name(n) {
        std::cout << name << " created\n";
    }
    ~Node() {
        std::cout << name << " destroyed\n";
    }
};

int main() {
    auto node1 = std::make_shared<Node>("Node1");
    auto node2 = std::make_shared<Node>("Node2");
    
    // 建立双向链表
    node1->next = node2;
    node2->prev = node1;  // weak_ptr不增加引用计数
    
    std::cout << "node1 count: " << node1.use_count() << "\n";  // 1
    std::cout << "node2 count: " << node2.use_count() << "\n";  // 2
    
    // 使用weak_ptr
    if (auto locked = node2->prev.lock()) {
        std::cout << "prev node: " << locked->name << "\n";
    }
    
    return 0;
}
```

## 5. Lambda表达式

### 1. 概念与定义

**Lambda表达式**：C++11引入的新特性，用于创建匿名函数。Lambda表达式可以简化代码，提高可读性。Lambda表达式可以捕获变量，用于函数式编程。

**Lambda语法**：`[capture](params) -> return_type { body }`

- **capture**：捕获列表，用于捕获外部变量。例如`[a]`表示值捕获`a`，`[&a]`表示引用捕获`a`，`[=]`表示全部值捕获，`[&]`表示全部引用捕获。
- **params**：参数列表，用于定义函数参数。例如`(int a, int b)`表示两个int类型的参数。
- **return_type**：返回类型，用于定义函数返回类型。例如`-> int`表示返回int类型。
- **body**：函数体，用于定义函数实现。例如`{ return a + b; }`表示返回`a + b`。

**捕获方式**：
- **值捕获**：拷贝外部变量，在Lambda内部使用。例如`[a]`表示值捕获`a`。
- **引用捕获**：引用外部变量，在Lambda内部可以修改。例如`[&a]`表示引用捕获`a`。
- **全部值捕获**：拷贝所有外部变量，在Lambda内部使用。例如`[=]`表示全部值捕获。
- **全部引用捕获**：引用所有外部变量，在Lambda内部可以修改。例如`[&]`表示全部引用捕获。
- **混合捕获**：混合使用值捕获和引用捕获。例如`[=, &a]`表示值捕获所有变量，引用捕获`a`。
- **mutable**：允许修改值捕获的变量。例如`[a] mutable { a = 10; }`表示可以修改值捕获的`a`。

### 2. 基本语法

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    // Lambda语法：[capture](params) -> return_type { body }
    
    // 1. 最简单的lambda
    auto sayHello = []() {
        std::cout << "Hello, Lambda!\n";
    };
    sayHello();
    
    // 2. 带参数
    auto add = [](int a, int b) {
        return a + b;
    };
    std::cout << "3 + 4 = " << add(3, 4) << "\n";
    
    // 3. 显式返回类型
    auto divide = [](double a, double b) -> double {
        if (b == 0) return 0;
        return a / b;
    };
    
    // 4. 在算法中使用
    std::vector<int> nums = {3, 1, 4, 1, 5, 9};
    
    // 排序（降序）
    std::sort(nums.begin(), nums.end(), [](int a, int b) {
        return a > b;
    });
    
    // 查找
    auto it = std::find_if(nums.begin(), nums.end(), [](int n) {
        return n > 5;
    });
    
    // 遍历
    std::for_each(nums.begin(), nums.end(), [](int n) {
        std::cout << n << " ";
    });
    std::cout << "\n";
    
    return 0;
}
```

### 3. 捕获方式

```cpp
#include <iostream>
#include <functional>

int main() {
    int a = 10;
    int b = 20;
    
    // 1. 不捕获
    auto f1 = []() { return 42; };
    
    // 2. 值捕获
    auto f2 = [a]() { return a; };  // 拷贝a
    
    // 3. 引用捕获
    auto f3 = [&a]() { a = 100; };  // 可修改外部a
    f3();
    std::cout << "a after f3: " << a << "\n";  // 100
    
    // 4. 全部值捕获
    auto f4 = [=]() { return a + b; };
    
    // 5. 全部引用捕获
    auto f5 = [&]() { a = b = 0; };
    
    // 6. 混合捕获
    auto f6 = [=, &a]() { a = b; };  // b值捕获，a引用捕获
    
    // 7. mutable：允许修改值捕获的变量
    int x = 10;
    auto f7 = [x]() mutable {
        x++;  // 修改的是副本
        return x;
    };
    std::cout << "f7(): " << f7() << "\n";  // 11
    std::cout << "x: " << x << "\n";        // 10（原值不变）
    
    return 0;
}
```

## 6. 范围for循环

### 1. 基本用法

```cpp
#include <iostream>
#include <vector>
#include <map>
#include <array>

int main() {
    // 数组
    int arr[] = {1, 2, 3, 4, 5};
    for (int x : arr) {
        std::cout << x << " ";
    }
    std::cout << "\n";
    
    // vector
    std::vector<int> vec = {10, 20, 30};
    for (int x : vec) {
        std::cout << x << " ";
    }
    std::cout << "\n";
    
    // 引用修改
    for (int& x : vec) {
        x *= 2;  // 修改原元素
    }
    
    // const引用（只读）
    for (const int& x : vec) {
        std::cout << x << " ";
    }
    std::cout << "\n";
    
    // map
    std::map<std::string, int> scores = {
        {"Alice", 95},
        {"Bob", 87}
    };
    for (const auto& pair : scores) {
        std::cout << pair.first << ": " << pair.second << "\n";
    }
    
    // 使用auto
    for (auto x : vec) {
        std::cout << x << " ";
    }
```

## 9. constexpr与静态断言

### 1. constexpr

```cpp
#include <iostream>
#include <array>

// constexpr：编译期常量表达式

// 编译期常量
constexpr int MAX_SIZE = 100;
constexpr double PI = 3.14159265358979;

// constexpr函数
constexpr int square(int x) {
    return x * x;
}

constexpr int factorial(int n) {
    return n <= 1 ? 1 : n * factorial(n - 1);
}

// C++14：允许更复杂的constexpr函数
constexpr int fibonacci(int n) {
    if (n <= 1) return n;
    
    int a = 0, b = 1;
    for (int i = 2; i <= n; ++i) {
        int temp = a + b;
        a = b;
        b = temp;
    }
    return b;
}

int main() {
    // 编译期计算
    constexpr int sq = square(5);      // 编译期计算
    int arr[square(3)];                // 数组大小
    std::array<int, square(4)> arr2;   // 标准库容器大小
    
    // 运行时也可以使用
    int x = 5;
    int result = square(x);  // 运行期计算
    
    // 编译期验证
    static_assert(square(5) == 25, "square(5) should be 25");
    static_assert(factorial(5) == 120, "factorial(5) should be 120");
    
    std::cout << "5! = " << factorial(5) << "\n";
    std::cout << "fib(10) = " << fibonacci(10) << "\n";
    
    return 0;
}
```
