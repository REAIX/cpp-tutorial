## 1. 学生成绩管理系统 🟢

> 🟢 **入门级**：巩固类、容器、算法的基础用法

### 1. 概念与定义

**学生成绩管理系统**：C++中用于管理学生成绩的系统。学生成绩管理系统可以用于添加学生信息、查询学生成绩、统计成绩分布、排序输出等。

### 2. 需求

- 添加学生信息（姓名、成绩）
- 查询学生成绩
- 统计成绩分布
- 排序输出

### 3. 实现

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <map>
#include <algorithm>
#include <iomanip>

class Student {
public:
    std::string name;
    double score;
    
    Student(const std::string& n, double s) : name(n), score(s) {}
    
    bool operator<(const Student& other) const {
        return score > other.score;  // 降序
    }
};

class GradeManager {
private:
    std::vector<Student> students;
    
public:
    void addStudent(const std::string& name, double score) {
        students.emplace_back(name, score);
    }
    
    void displayAll() const {
        std::cout << "\n=== 学生列表 ===\n";
        std::cout << std::setw(20) << "姓名" << std::setw(10) << "成绩\n";
        std::cout << std::string(30, '-') << "\n";
        
        for (const auto& s : students) {
            std::cout << std::setw(20) << s.name 
                      << std::setw(10) << std::fixed << std::setprecision(2) << s.score << "\n";
        }
    }
    
    void sortByScore() {
        std::sort(students.begin(), students.end());
    }
    
    void showStatistics() const {
        if (students.empty()) return;
        
        double total = 0;
        double maxScore = students[0].score;
        double minScore = students[0].score;
        
        for (const auto& s : students) {
            total += s.score;
            maxScore = std::max(maxScore, s.score);
            minScore = std::min(minScore, s.score);
        }
        
        std::cout << "\n=== 统计信息 ===\n";
        std::cout << "学生人数: " << students.size() << "\n";
        std::cout << "平均分: " << std::fixed << std::setprecision(2) 
                  << total / students.size() << "\n";
        std::cout << "最高分: " << maxScore << "\n";
        std::cout << "最低分: " << minScore << "\n";
    }
    
    void showDistribution() const {
        std::map<char, int> distribution;
        
        for (const auto& s : students) {
            char grade;
            if (s.score >= 90) grade = 'A';
            else if (s.score >= 80) grade = 'B';
            else if (s.score >= 70) grade = 'C';
            else if (s.score >= 60) grade = 'D';
            else grade = 'F';
            
            distribution[grade]++;
        }
        
        std::cout << "\n=== 成绩分布 ===\n";
        for (const auto& [grade, count] : distribution) {
            std::cout << grade << ": " << std::string(count, '*') << " (" << count << ")\n";
        }
    }
};

int main() {
    GradeManager manager;
    
    manager.addStudent("张三", 85.5);
    manager.addStudent("李四", 92.0);
    manager.addStudent("王五", 78.5);
    manager.addStudent("赵六", 65.0);
    manager.addStudent("钱七", 88.5);
    
    manager.displayAll();
    manager.showStatistics();
    manager.showDistribution();
    
    std::cout << "\n按成绩排序后:\n";
    manager.sortByScore();
    manager.displayAll();
    
    return 0;
}
```

## 2. 简单计算器 🟢

> 🟢 **入门级**：练习栈、字符串处理、运算符优先级

### 1. 需求

- 支持基本四则运算
- 支持括号
- 支持变量存储

### 2. 实现

```cpp
#include <iostream>
#include <string>
#include <stack>
#include <map>
#include <cmath>
#include <cctype>

class Calculator {
private:
    std::map<std::string, double> variables;
    
    int precedence(char op) {
        if (op == '+' || op == '-') return 1;
        if (op == '*' || op == '/') return 2;
        if (op == '^') return 3;
        return 0;
    }
    
    double applyOp(double a, double b, char op) {
        switch (op) {
            case '+': return a + b;
            case '-': return a - b;
            case '*': return a * b;
            case '/': return a / b;
            case '^': return std::pow(a, b);
            default: return 0;
        }
    }
    
public:
    double evaluate(std::string expr) {
        std::stack<double> values;
        std::stack<char> ops;
        
        for (size_t i = 0; i < expr.length(); ++i) {
            if (expr[i] == ' ') continue;
            
            if (expr[i] == '(') {
                ops.push(expr[i]);
            }
            else if (isdigit(expr[i]) || expr[i] == '.') {
                std::string num;
                while (i < expr.length() && (isdigit(expr[i]) || expr[i] == '.')) {
                    num += expr[i++];
                }
                --i;
                values.push(std::stod(num));
            }
            else if (isalpha(expr[i])) {
                std::string var;
                while (i < expr.length() && isalpha(expr[i])) {
                    var += expr[i++];
                }
                --i;
                if (variables.count(var)) {
                    values.push(variables[var]);
                } else {
                    std::cout << "变量 " << var << " 未定义\n";
                    values.push(0);
                }
            }
            else if (expr[i] == ')') {
                while (!ops.empty() && ops.top() != '(') {
                    double b = values.top(); values.pop();
                    double a = values.top(); values.pop();
                    char op = ops.top(); ops.pop();
                    values.push(applyOp(a, b, op));
                }
                ops.pop();
            }
            else {
                while (!ops.empty() && precedence(ops.top()) >= precedence(expr[i])) {
                    double b = values.top(); values.pop();
                    double a = values.top(); values.pop();
                    char op = ops.top(); ops.pop();
                    values.push(applyOp(a, b, op));
                }
                ops.push(expr[i]);
            }
        }
        
        while (!ops.empty()) {
            double b = values.top(); values.pop();
            double a = values.top(); values.pop();
            char op = ops.top(); ops.pop();
            values.push(applyOp(a, b, op));
        }
        
        return values.top();
    }
    
    void setVariable(const std::string& name, double value) {
        variables[name] = value;
    }
};

int main() {
    Calculator calc;
    
    calc.setVariable("x", 10);
    calc.setVariable("y", 5);
    
    std::cout << "2 + 3 * 4 = " << calc.evaluate("2 + 3 * 4") << "\n";
    std::cout << "(2 + 3) * 4 = " << calc.evaluate("(2 + 3) * 4") << "\n";
    std::cout << "x + y = " << calc.evaluate("x + y") << "\n";
    std::cout << "x * y + 10 = " << calc.evaluate("x * y + 10") << "\n";
    
    return 0;
}
```

### 1. 需求

- 文件复制
- 文件搜索
- 行数统计

### 2. 实现

```cpp
#include <iostream>
#include <fstream>
#include <filesystem>
#include <string>
#include <vector>

namespace fs = std::filesystem;

class FileTool {
public:
    static bool copyFile(const std::string& src, const std::string& dest) {
        try {
            fs::copy_file(src, dest, fs::copy_options::overwrite_existing);
            return true;
        } catch (const std::exception& e) {
            std::cerr << "Copy failed: " << e.what() << "\n";
            return false;
        }
    }
    
    static std::vector<fs::path> searchFiles(const std::string& dir, 
                                              const std::string& pattern) {
        std::vector<fs::path> results;
        
        for (const auto& entry : fs::recursive_directory_iterator(dir)) {
            if (entry.path().filename().string().find(pattern) != std::string::npos) {
                results.push_back(entry.path());
            }
        }
        
        return results;
    }
    
    static std::map<std::string, size_t> countLines(const std::string& dir) {
        std::map<std::string, size_t> counts;
        
        for (const auto& entry : fs::recursive_directory_iterator(dir)) {
            if (entry.is_regular_file() && entry.path().extension() == ".cpp") {
                std::ifstream file(entry.path());
                size_t lines = std::count(std::istreambuf_iterator<char>(file),
                                          std::istreambuf_iterator<char>(), '\n');
                counts[entry.path().string()] = lines;
            }
        }
        
        return counts;
    }
    
    static void printDirectoryTree(const fs::path& path, int indent = 0) {
        for (const auto& entry : fs::directory_iterator(path)) {
            std::cout << std::string(indent, ' ');
            
            if (entry.is_directory()) {
                std::cout << "[D] " << entry.path().filename() << "\n";
                printDirectoryTree(entry.path(), indent + 2);
            } else {
                std::cout << "[F] " << entry.path().filename() 
                          << " (" << entry.file_size() << " bytes)\n";
            }
        }
    }
};

int main() {
    std::cout << "=== 文件搜索 ===\n";
    auto files = FileTool::searchFiles(".", ".cpp");
    for (const auto& f : files) {
        std::cout << f << "\n";
    }
    
    std::cout << "\n=== 目录树 ===\n";
    FileTool::printDirectoryTree(".");
    
    return 0;
}
```

## 4. 任务调度器 🟡

> 🟡 **进阶级**：综合运用多线程、互斥量、条件变量、优先队列

### 1. 需求

- 定时任务执行
- 任务队列管理
- 并发执行

### 2. 实现

```cpp
#include <iostream>
#include <queue>
#include <functional>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <chrono>
#include <atomic>

class TaskScheduler {
private:
    struct Task {
        std::function<void()> func;
        std::chrono::steady_clock::time_point executeTime;
        std::string name;
        
        bool operator>(const Task& other) const {
            return executeTime > other.executeTime;
        }
    };
    
    std::priority_queue<Task, std::vector<Task>, std::greater<Task>> tasks;
    std::mutex mtx;
    std::condition_variable cv;
    std::atomic<bool> running{true};
    std::thread worker;
    
public:
    TaskScheduler() {
        worker = std::thread([this]() {
            while (running) {
                std::unique_lock<std::mutex> lock(mtx);
                
                if (tasks.empty()) {
                    cv.wait(lock);
                    continue;
                }
                
                auto now = std::chrono::steady_clock::now();
                auto& top = tasks.top();
                
                if (top.executeTime <= now) {
                    Task task = top;
                    tasks.pop();
                    lock.unlock();
                    
                    std::cout << "执行任务: " << task.name << "\n";
                    task.func();
                } else {
                    cv.wait_until(lock, top.executeTime);
                }
            }
        });
    }
    
    ~TaskScheduler() {
        running = false;
        cv.notify_all();
        if (worker.joinable()) {
            worker.join();
        }
    }
    
    void schedule(std::function<void()> func, int delayMs, const std::string& name = "") {
        auto executeTime = std::chrono::steady_clock::now() + 
                          std::chrono::milliseconds(delayMs);
        
        {
            std::lock_guard<std::mutex> lock(mtx);
            tasks.push({func, executeTime, name});
        }
        cv.notify_one();
    }
};

int main() {
    TaskScheduler scheduler;
    
    scheduler.schedule([]() {
        std::cout << "任务1执行\n";
    }, 1000, "Task 1");
    
    scheduler.schedule([]() {
        std::cout << "任务2执行\n";
    }, 2000, "Task 2");
    
    scheduler.schedule([]() {
        std::cout << "任务3执行\n";
    }, 500, "Task 3");
    
    std::cout << "任务已调度，等待执行...\n";
    
    std::this_thread::sleep_for(std::chrono::seconds(3));
    
    return 0;
}
```

## 5. 高并发HTTP服务器 🔴

> 🔴 **高阶级**：综合运用Reactor模式、线程池、epoll、协议解析

### 1. 架构设计

```
┌─────────────────────────────────────────┐
│           主线程 (Main Thread)           │
│  ┌─────────┐  ┌─────────────────────┐  │
│  │ Acceptor │→│   epoll_wait        │  │
│  └─────────┘  │   (监听新连接)        │  │
│               └─────────────────────┘  │
└─────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────┐
│         工作线程池 (Worker Pool)         │
│  ┌─────────┐ ┌─────────┐ ┌─────────┐  │
│  │ Worker1 │ │ Worker2 │ │ WorkerN │  │
│  │ 处理请求 │ │ 处理请求 │ │ 处理请求 │  │
│  └─────────┘ └─────────┘ └─────────┘  │
└─────────────────────────────────────────┘
```

### 2. 核心实现

```cpp
// ===== http_server.hpp =====
#pragma once

#include <iostream>
#include <string>
#include <vector>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <functional>
#include <sys/epoll.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <fcntl.h>
#include <cstring>

class ThreadPool {
public:
    explicit ThreadPool(size_t numThreads);
    ~ThreadPool();
    void enqueue(std::function<void()> task);

private:
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    std::mutex queueMutex;
    std::condition_variable condition;
    bool stop = false;
};

class HttpServer {
public:
    explicit HttpServer(int port = 8080, size_t threadCount = 4);
    ~HttpServer();
    void start();
    void stop();

private:
    int serverFd;
    int epollFd;
    int port;
    bool running = false;
    ThreadPool threadPool;
    static constexpr int MAX_EVENTS = 1024;
    
    void setNonBlocking(int fd);
    void handleAccept();
    void handleClient(int clientFd);
    std::string parseRequest(const std::string& request);
    void sendResponse(int clientFd, int statusCode, const std::string& body);
};

// ===== thread_pool.cpp =====
ThreadPool::ThreadPool(size_t numThreads) {
    for (size_t i = 0; i < numThreads; ++i) {
        workers.emplace_back([this]() {
            while (true) {
                std::function<void()> task;
                {
                    std::unique_lock<std::mutex> lock(queueMutex);
                    condition.wait(lock, [this]() { return stop || !tasks.empty(); });
                    if (stop && tasks.empty()) return;
                    task = std::move(tasks.front());
                    tasks.pop();
                }
                task();
            }
        });
    }
}

ThreadPool::~ThreadPool() {
    {
        std::unique_lock<std::mutex> lock(queueMutex);
        stop = true;
    }
    condition.notify_all();
    for (auto& worker : workers) {
        if (worker.joinable()) worker.join();
    }
}

void ThreadPool::enqueue(std::function<void()> task) {
    {
        std::unique_lock<std::mutex> lock(queueMutex);
        tasks.push(std::move(task));
    }
    condition.notify_one();
}

// ===== http_server.cpp =====
HttpServer::HttpServer(int port, size_t threadCount) 
    : port(port), threadPool(threadCount) {}

HttpServer::~HttpServer() {
    stop();
}

void HttpServer::setNonBlocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    fcntl(fd, F_SETFL, flags | O_NONBLOCK);
}

void HttpServer::start() {
    serverFd = socket(AF_INET, SOCK_STREAM, 0);
    if (serverFd < 0) {
        perror("socket failed");
        return;
    }
    
    int opt = 1;
    setsockopt(serverFd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(port);
    
    if (bind(serverFd, (sockaddr*)&addr, sizeof(addr)) < 0) {
        perror("bind failed");
        return;
    }
    
    if (listen(serverFd, 128) < 0) {
        perror("listen failed");
        return;
    }
    
    setNonBlocking(serverFd);
    
    epollFd = epoll_create1(0);
    epoll_event ev{};
    ev.events = EPOLLIN;
    ev.data.fd = serverFd;
    epoll_ctl(epollFd, EPOLL_CTL_ADD, serverFd, &ev);
    
    running = true;
    std::cout << "HTTP Server listening on port " << port << "\n";
    
    epoll_event events[MAX_EVENTS];
    while (running) {
        int nfds = epoll_wait(epollFd, events, MAX_EVENTS, -1);
        for (int i = 0; i < nfds; ++i) {
            if (events[i].data.fd == serverFd) {
                handleAccept();
            } else {
                int clientFd = events[i].data.fd;
                threadPool.enqueue([this, clientFd]() {
                    handleClient(clientFd);
                });
            }
        }
    }
}

void HttpServer::handleAccept() {
    sockaddr_in clientAddr{};
    socklen_t len = sizeof(clientAddr);
    int clientFd = accept(serverFd, (sockaddr*)&clientAddr, &len);
    if (clientFd < 0) return;
    
    setNonBlocking(clientFd);
    epoll_event ev{};
    ev.events = EPOLLIN | EPOLLET;
    ev.data.fd = clientFd;
    epoll_ctl(epollFd, EPOLL_CTL_ADD, clientFd, &ev);
}

void HttpServer::handleClient(int clientFd) {
    char buffer[4096];
    ssize_t n = read(clientFd, buffer, sizeof(buffer) - 1);
    if (n <= 0) {
        close(clientFd);
        return;
    }
    buffer[n] = '\0';
    
    std::string request(buffer);
    std::string path = parseRequest(request);
    
    std::string body = "<html><body><h1>Hello from C++ HTTP Server!</h1>";
    body += "<p>Path: " + path + "</p></body></html>";
    
    sendResponse(clientFd, 200, body);
    close(clientFd);
}

std::string HttpServer::parseRequest(const std::string& request) {
    size_t firstSpace = request.find(' ');
    size_t secondSpace = request.find(' ', firstSpace + 1);
    if (firstSpace != std::string::npos && secondSpace != std::string::npos) {
        return request.substr(firstSpace + 1, secondSpace - firstSpace - 1);
    }
    return "/";
}

void HttpServer::sendResponse(int clientFd, int statusCode, const std::string& body) {
    std::string response = "HTTP/1.1 " + std::to_string(statusCode) + " OK\r\n";
    response += "Content-Type: text/html\r\n";
    response += "Content-Length: " + std::to_string(body.length()) + "\r\n";
    response += "Connection: close\r\n\r\n";
    response += body;
    
    write(clientFd, response.c_str(), response.length());
}

void HttpServer::stop() {
    running = false;
    if (serverFd >= 0) close(serverFd);
    if (epollFd >= 0) close(epollFd);
}

// ===== main.cpp =====
int main() {
    HttpServer server(8080, 4);
    server.start();
    return 0;
}
```

### 3. 性能优化要点

| 优化点 | 实现方式 | 效果 |
|--------|---------|------|
| Reactor模式 | epoll + 单线程Accept | 高并发连接处理 |
| 线程池 | 固定工作线程数 | 避免线程创建销毁开销 |
| 非阻塞IO | fcntl O_NONBLOCK | 单线程处理多个连接 |
| ET模式 | EPOLLET | 减少epoll触发次数 |
| 零拷贝 | sendfile（静态文件） | 减少内核态/用户态拷贝 |

## 6. 极简KV存储引擎 🔴

> 🔴 **高阶级**：综合运用数据结构、文件IO、内存管理、并发控制

### 1. 架构设计

```
┌─────────────────────────────────────────┐
│              内存索引 (MemIndex)          │
```

```cpp
// ===== kv_store.hpp =====
#pragma once

#include <string>
#include <unordered_map>
#include <vector>
#include <mutex>
#include <fstream>
#include <cstdint>

struct Record {
    uint32_t keySize;
    uint32_t valueSize;
    std::string key;
    std::string value;
    
    size_t totalSize() const {
        return sizeof(keySize) + sizeof(valueSize) + key.size() + value.size();
    }
};

struct IndexEntry {
    uint64_t offset;
    uint32_t size;
};

class KVStore {
public:
    explicit KVStore(const std::string& filename);
    ~KVStore();
    
    bool put(const std::string& key, const std::string& value);
    bool get(const std::string& key, std::string& value);
    bool del(const std::string& key);
    
private:
    std::string filename;
    std::fstream dataFile;
    std::unordered_map<std::string, IndexEntry> index;
    std::mutex mtx;
    uint64_t currentOffset = 0;
    
    bool writeRecord(const Record& record);
    bool readRecord(uint64_t offset, Record& record);
    void rebuildIndex();
};

// ===== kv_store.cpp =====
KVStore::KVStore(const std::string& filename) : filename(filename) {
    dataFile.open(filename, std::ios::in | std::ios::out | std::ios::app | std::ios::binary);
    if (!dataFile.is_open()) {
        dataFile.open(filename, std::ios::out | std::ios::binary);
        dataFile.close();
        dataFile.open(filename, std::ios::in | std::ios::out | std::ios::app | std::ios::binary);
    }
    rebuildIndex();
}

KVStore::~KVStore() {
    if (dataFile.is_open()) {
        dataFile.close();
    }
}

bool KVStore::put(const std::string& key, const std::string& value) {
    std::lock_guard<std::mutex> lock(mtx);
    
    Record record;
    record.keySize = key.size();
    record.valueSize = value.size();
    record.key = key;
    record.value = value;
    
    uint64_t offset = currentOffset;
    if (!writeRecord(record)) return false;
    
    index[key] = {offset, static_cast<uint32_t>(record.totalSize())};
    return true;
}

bool KVStore::get(const std::string& key, std::string& value) {
    std::lock_guard<std::mutex> lock(mtx);
    
    auto it = index.find(key);
    if (it == index.end()) return false;
    
    Record record;
    if (!readRecord(it->second.offset, record)) return false;
    
    value = record.value;
    return true;
}

bool KVStore::del(const std::string& key) {
    std::lock_guard<std::mutex> lock(mtx);
    
    auto it = index.find(key);
    if (it == index.end()) return false;
    
    // 标记删除（墓碑机制）
    Record record;
    record.keySize = key.size();
    record.valueSize = 0;  // 0表示删除
    record.key = key;
    record.value = "";
    
    writeRecord(record);
    index.erase(it);
    return true;
}

bool KVStore::writeRecord(const Record& record) {
    dataFile.write(reinterpret_cast<const char*>(&record.keySize), sizeof(record.keySize));
    dataFile.write(reinterpret_cast<const char*>(&record.valueSize), sizeof(record.valueSize));
    dataFile.write(record.key.data(), record.key.size());
    dataFile.write(record.value.data(), record.value.size());
    dataFile.flush();
    
    currentOffset += record.totalSize();
    return dataFile.good();
}

bool KVStore::readRecord(uint64_t offset, Record& record) {
    dataFile.seekg(offset);
    
    dataFile.read(reinterpret_cast<char*>(&record.keySize), sizeof(record.keySize));
    dataFile.read(reinterpret_cast<char*>(&record.valueSize), sizeof(record.valueSize));
    
    record.key.resize(record.keySize);
    record.value.resize(record.valueSize);
    
    dataFile.read(record.key.data(), record.keySize);
    dataFile.read(record.value.data(), record.valueSize);
    
    return dataFile.good();
}

void KVStore::rebuildIndex() {
    dataFile.seekg(0, std::ios::end);
    uint64_t fileSize = dataFile.tellg();
    dataFile.seekg(0, std::ios::beg);
    
    currentOffset = 0;
    while (currentOffset < fileSize) {
        Record record;
        if (!readRecord(currentOffset, record)) break;
        
        if (record.valueSize > 0) {
            index[record.key] = {currentOffset, static_cast<uint32_t>(record.totalSize())};
        } else {
            index.erase(record.key);
        }
        
        currentOffset += record.totalSize();
    }
}

// ===== main.cpp =====
int main() {
    KVStore store("test.db");
    
    store.put("name", "Alice");
    store.put("age", "25");
    store.put("city", "Beijing");
    
    std::string value;
    if (store.get("name", value)) {
        std::cout << "name = " << value << "\n";
    }
    
    store.del("age");
    
    if (!store.get("age", value)) {
        std::cout << "age deleted\n";
    }
    
    return 0;
}
```

## 7. C++协程库实现 🔴

> 🔴 **高阶级**：深入理解协程状态机、调度器、无栈协程

### 1. 核心概念

C++20协程不是线程，而是**可暂停和恢复的函数**。编译器将协程转换为状态机。

```cpp
// ===== generator.hpp =====
#pragma once

#include <coroutine>
#include <optional>

template<typename T>
class Generator {
public:
    struct promise_type {
        T current_value;
        
        auto get_return_object() {
            return Generator{std::coroutine_handle<promise_type>::from_promise(*this)};
        }
        
        auto initial_suspend() { return std::suspend_always{}; }
        auto final_suspend() noexcept { return std::suspend_always{}; }
        void unhandled_exception() { std::terminate(); }
        void return_void() {}
        
        auto yield_value(T value) {
            current_value = value;
            return std::suspend_always{};
        }
    };
    
    using handle_type = std::coroutine_handle<promise_type>;
    
    explicit Generator(handle_type h) : handle(h) {}
    ~Generator() { if (handle) handle.destroy(); }
    
    // 禁止拷贝，允许移动
    Generator(const Generator&) = delete;
    Generator& operator=(const Generator&) = delete;
    Generator(Generator&& other) noexcept : handle(other.handle) {
        other.handle = nullptr;
    }
    
    class Iterator {
    public:
        Iterator(handle_type h) : handle(h) {}
        
        Iterator& operator++() {
            handle.resume();
            if (handle.done()) handle = nullptr;
            return *this;
        }
        
        T operator*() const { return handle.promise().current_value; }
        bool operator!=(const Iterator& other) const { return handle != other.handle; }
        
    private:
        handle_type handle;
    };
    
    Iterator begin() {
        if (handle && !handle.done()) {
            handle.resume();
            if (handle.done()) return Iterator{nullptr};
        }
        return Iterator{handle};
    }
    
    Iterator end() { return Iterator{nullptr}; }
    
private:
    handle_type handle;
};

// ===== 使用示例 =====
Generator<int> fibonacci(int n) {
    int a = 0, b = 1;
    for (int i = 0; i < n; ++i) {
        co_yield a;
        int next = a + b;
        a = b;
        b = next;
    }
}

int main() {
    std::cout << "Fibonacci: ";
    for (auto val : fibonacci(10)) {
        std::cout << val << " ";
    }
    std::cout << "\n";
    return 0;
}
```

```cpp
// ===== task.hpp =====
#pragma once

#include <coroutine>
#include <optional>
#include <exception>

template<typename T = void>
class Task {
public:
    struct promise_type {
        std::optional<T> result;
        std::exception_ptr exception;
        std::coroutine_handle<> continuation;
        
        auto get_return_object() {
            return Task{std::coroutine_handle<promise_type>::from_promise(*this)};
        }
        
        auto initial_suspend() { return std::suspend_always{}; }
        
        struct final_awaiter {
            bool await_ready() noexcept { return false; }
            void await_suspend(std::coroutine_handle<promise_type> h) noexcept {
                if (h.promise().continuation) {
                    h.promise().continuation.resume();
                }
            }
            void await_resume() noexcept {}
        };
        
        auto final_suspend() noexcept { return final_awaiter{}; }
        void unhandled_exception() { exception = std::current_exception(); }
        
        template<typename U>
        void return_value(U&& value) {
            result = std::forward<U>(value);
        }
    };
    
    using handle_type = std::coroutine_handle<promise_type>;
    
    explicit Task(handle_type h) : handle(h) {}
    ~Task() { if (handle) handle.destroy(); }
    
    Task(const Task&) = delete;
    Task& operator=(const Task&) = delete;
    Task(Task&& other) noexcept : handle(other.handle) { other.handle = nullptr; }
    
    bool await_ready() { return handle.done(); }
    
    void await_suspend(std::coroutine_handle<> h) {
        handle.promise().continuation = h;
        handle.resume();
    }
    
    T await_resume() {
        if (handle.promise().exception) {
            std::rethrow_exception(handle.promise().exception);
        }
        return *handle.promise().result;
    }
    
    T get() {
        if (!handle.done()) {
            handle.resume();
        }
        if (handle.promise().exception) {
            std::rethrow_exception(handle.promise().exception);
        }
        return *handle.promise().result;
    }
    
private:
    handle_type handle;
};

// 特化void版本
template<>
class Task<void> {
public:
    struct promise_type {
        std::exception_ptr exception;
        std::coroutine_handle<> continuation;

        }
        auto initial_suspend() { return std::suspend_always{}; }
        auto final_suspend() noexcept {
            struct awaiter {
                bool await_ready() noexcept { return false; }
                void await_suspend(std::coroutine_handle<promise_type> h) noexcept {
                    if (h.promise().continuation) h.promise().continuation.resume();
                }
                void await_resume() noexcept {}
            };
            return awaiter{};
        }
        void unhandled_exception() { exception = std::current_exception(); }
        void return_void() {}
    };
    
    using handle_type = std::coroutine_handle<promise_type>;
    explicit Task(handle_type h) : handle(h) {}
    ~Task() { if (handle) handle.destroy(); }
    
    bool await_ready() { return handle.done(); }
    void await_suspend(std::coroutine_handle<> h) {
        handle.promise().continuation = h;
        handle.resume();
    }
    void await_resume() {
        if (handle.promise().exception) std::rethrow_exception(handle.promise().exception);
    }
    
private:

};
```

### 1. 伙伴系统实现

```cpp
// ===== buddy_allocator.hpp =====
#pragma once

#include <cstddef>
#include <cstdint>
#include <vector>
#include <memory>
#include <cstring>

class BuddyAllocator {
public:
    explicit BuddyAllocator(size_t totalSize);
    ~BuddyAllocator();
    
    void* allocate(size_t size);
    void deallocate(void* ptr, size_t size);
    
    // 禁止拷贝
    BuddyAllocator(const BuddyAllocator&) = delete;
    BuddyAllocator& operator=(const BuddyAllocator&) = delete;

private:
    static constexpr size_t MIN_BLOCK_SIZE = 16;
    static constexpr size_t MAX_ORDER = 20;  // 支持到 16 * 2^20 = 16MB
    
    struct Block {
        size_t order;
        bool free;
        Block* next;
        Block* buddy;
    };
    
    uint8_t* memory;
    size_t totalSize;
    std::vector<Block*> freeLists;  // 按order组织的空闲链表
    
    size_t getOrder(size_t size) const;
    Block* splitBlock(Block* block, size_t targetOrder);
    Block* mergeBlock(Block* block);
    size_t ptrToOffset(void* ptr) const;
    void* offsetToPtr(size_t offset) const;
    Block* getBuddy(Block* block);
};

// ===== buddy_allocator.cpp =====
BuddyAllocator::BuddyAllocator(size_t totalSize) : totalSize(totalSize) {
    // 对齐到2的幂
    size_t alignedSize = MIN_BLOCK_SIZE;
    while (alignedSize < totalSize) {
        alignedSize <<= 1;
    }
    this->totalSize = alignedSize;
    
    memory = new uint8_t[this->totalSize];
    freeLists.resize(MAX_ORDER + 1, nullptr);
    
    // 初始化整个内存为一个空闲块
    Block* initial = reinterpret_cast<Block*>(memory);
    initial->order = getOrder(this->totalSize);
    initial->free = true;
    initial->next = nullptr;
    initial->buddy = nullptr;
    
    freeLists[initial->order] = initial;
}

BuddyAllocator::~BuddyAllocator() {
    delete[] memory;
}

size_t BuddyAllocator::getOrder(size_t size) const {
    size_t order = 0;
    size_t blockSize = MIN_BLOCK_SIZE;
    while (blockSize < size && order < MAX_ORDER) {
        blockSize <<= 1;
        ++order;
    }
    return order;
}

void* BuddyAllocator::allocate(size_t size) {
    if (size == 0) return nullptr;
    
    size_t order = getOrder(size + sizeof(Block));
    if (order > MAX_ORDER) return nullptr;
    
    // 找到合适的空闲块
    size_t foundOrder = order;
    while (foundOrder <= MAX_ORDER && freeLists[foundOrder] == nullptr) {
        ++foundOrder;
    }
    
    if (foundOrder > MAX_ORDER) return nullptr;  // 内存不足
    
    // 分裂大块直到达到所需order
    Block* block = freeLists[foundOrder];
    freeLists[foundOrder] = block->next;
    
    while (block->order > order) {
        block = splitBlock(block, order);
    }
    
    block->free = false;
    return reinterpret_cast<void*>(reinterpret_cast<uint8_t*>(block) + sizeof(Block));
}

void BuddyAllocator::deallocate(void* ptr, size_t size) {
    if (!ptr) return;
    
    Block* block = reinterpret_cast<Block*>(
        reinterpret_cast<uint8_t*>(ptr) - sizeof(Block));
    
    block->free = true;
    
    // 尝试合并伙伴块
    block = mergeBlock(block);
    
    // 放回空闲链表
    block->next = freeLists[block->order];
    freeLists[block->order] = block;
}

Block* BuddyAllocator::splitBlock(Block* block, size_t targetOrder) {
    size_t newOrder = block->order - 1;
    size_t blockSize = MIN_BLOCK_SIZE << newOrder;
    
    Block* buddy = reinterpret_cast<Block*>(
        reinterpret_cast<uint8_t*>(block) + blockSize);
    
    buddy->order = newOrder;
    buddy->free = true;
    buddy->next = freeLists[newOrder];
    buddy->buddy = block;
    freeLists[newOrder] = buddy;
    
    block->order = newOrder;
    block->buddy = buddy;
    
    return block;
}

Block* BuddyAllocator::mergeBlock(Block* block) {
    while (block->order < MAX_ORDER) {
        Block* buddy = getBuddy(block);
        if (!buddy || !buddy->free || buddy->order != block->order) {
            break;
        }
        
        // 从空闲链表中移除buddy
        Block** current = &freeLists[buddy->order];
        while (*current && *current != buddy) {
            current = &(*current)->next;
        }
        if (*current) {
            *current = buddy->next;
        }
        
        // 合并（地址小的作为头）
        if (buddy < block) {
            block = buddy;
        }
        block->order++;
    }
    return block;
}

Block* BuddyAllocator::getBuddy(Block* block) {
    size_t blockSize = MIN_BLOCK_SIZE << block->order;
    size_t offset = reinterpret_cast<uint8_t*>(block) - memory;
    size_t buddyOffset = offset ^ blockSize;
    
    if (buddyOffset >= totalSize) return nullptr;
    return reinterpret_cast<Block*>(memory + buddyOffset);
}

// ===== 使用示例 =====
int main() {
    BuddyAllocator allocator(1024 * 1024);  // 1MB
    
    void* p1 = allocator.allocate(100);
    void* p2 = allocator.allocate(200);
    void* p3 = allocator.allocate(500);
    
    allocator.deallocate(p2, 200);
    allocator.deallocate(p1, 100);
    allocator.deallocate(p3, 500);
    
    return 0;
}
```
