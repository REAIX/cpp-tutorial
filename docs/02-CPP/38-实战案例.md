## 1. 学生成绩管理系统

### 1. 概念与定义

**学生成绩管理系统**：C++中用于管理学生成绩的系统。学生成绩管理系统可以用于添加学生信息、查询学生成绩、统计成绩分布、排序输出等。例如`class GradeManager { /* ... */ };`用于实现学生成绩管理系统。

**简单计算器**：C++中用于实现简单计算的系统。简单计算器可以用于实现加法、减法、乘法、除法等运算。例如`class Calculator { /* ... */ };`用于实现简单计算器。

**文件处理工具**：C++中用于处理文件的工具。文件处理工具可以用于创建文件、删除文件、复制文件、移动文件等。例如`class FileHandler { /* ... */ };`用于实现文件处理工具。

**任务调度器**：C++中用于调度任务的系统。任务调度器可以用于调度定时任务、周期性任务等。例如`class TaskScheduler { /* ... */ };`用于实现任务调度器。

**最佳实践（best practice）**：C++中用于编写高质量代码的实践。最佳实践可以包括代码风格、命名规范、注释规范、性能优化、调试技巧等。例如使用`std::unique_ptr`管理内存、使用`const`关键字保证函数的不可变性、使用`noexcept`关键字保证函数的不抛出异常等。

### 2. 需求

- 添加学生信息（姓名、成绩）
- 查询学生成绩
- 统计成绩分布
- 排序输出

### 3. 实现

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <map>
#include <algorithm>
#include <iomanip>

class Student {
public:
    std::string name;
    double score;
    
    Student(const std::string& n, double s) : name(n), score(s) {}
    
    bool operator<(const Student& other) const {
        return score > other.score;  // 降序
    }
};

class GradeManager {
private:
    std::vector<Student> students;
    
public:
    void addStudent(const std::string& name, double score) {
        students.emplace_back(name, score);
    }
    
    void displayAll() const {
        std::cout << "\n=== 学生列表 ===\n";
        std::cout << std::setw(20) << "姓名" << std::setw(10) << "成绩\n";
        std::cout << std::string(30, '-') << "\n";
        
        for (const auto& s : students) {
            std::cout << std::setw(20) << s.name 
                      << std::setw(10) << std::fixed << std::setprecision(2) << s.score << "\n";
        }
    }
    
    void sortByScore() {
        std::sort(students.begin(), students.end());
    }
    
    void showStatistics() const {
        if (students.empty()) return;
        
        double total = 0;
        double maxScore = students[0].score;
        double minScore = students[0].score;
        
        for (const auto& s : students) {
            total += s.score;
            maxScore = std::max(maxScore, s.score);
            minScore = std::min(minScore, s.score);
        }
        
        std::cout << "\n=== 统计信息 ===\n";
        std::cout << "学生人数: " << students.size() << "\n";
        std::cout << "平均分: " << std::fixed << std::setprecision(2) 
                  << total / students.size() << "\n";
        std::cout << "最高分: " << maxScore << "\n";
        std::cout << "最低分: " << minScore << "\n";
    }
    
    void showDistribution() const {
        std::map<char, int> distribution;
        
        for (const auto& s : students) {
            char grade;
            if (s.score >= 90) grade = 'A';
            else if (s.score >= 80) grade = 'B';
            else if (s.score >= 70) grade = 'C';
            else if (s.score >= 60) grade = 'D';
            else grade = 'F';
            
            distribution[grade]++;
        }
        
        std::cout << "\n=== 成绩分布 ===\n";
        for (const auto& [grade, count] : distribution) {
            std::cout << grade << ": " << std::string(count, '*') << " (" << count << ")\n";
        }
    }
};

int main() {
    GradeManager manager;
    
    manager.addStudent("张三", 85.5);
    manager.addStudent("李四", 92.0);
    manager.addStudent("王五", 78.5);
    manager.addStudent("赵六", 65.0);
    manager.addStudent("钱七", 88.5);
    
    manager.displayAll();
    manager.showStatistics();
    manager.showDistribution();
    
    std::cout << "\n按成绩排序后:\n";
    manager.sortByScore();
    manager.displayAll();
    
    return 0;
}
```

## 2. 简单计算器

### 1. 需求

- 支持基本四则运算
- 支持括号
- 支持变量存储

### 2. 实现

```cpp
#include <iostream>
#include <string>
#include <stack>
#include <map>
#include <cmath>
#include <cctype>

class Calculator {
private:
    std::map<std::string, double> variables;
    
    int precedence(char op) {
        if (op == '+' || op == '-') return 1;
        if (op == '*' || op == '/') return 2;
        if (op == '^') return 3;
        return 0;
    }
    
    double applyOp(double a, double b, char op) {
        switch (op) {
            case '+': return a + b;
            case '-': return a - b;
            case '*': return a * b;
            case '/': return a / b;
            case '^': return std::pow(a, b);
            default: return 0;
        }
    }
    
public:
    double evaluate(std::string expr) {
        std::stack<double> values;
        std::stack<char> ops;
        
        for (size_t i = 0; i < expr.length(); ++i) {
            if (expr[i] == ' ') continue;
            
            if (expr[i] == '(') {
                ops.push(expr[i]);
            }
            else if (isdigit(expr[i]) || expr[i] == '.') {
                std::string num;
                while (i < expr.length() && (isdigit(expr[i]) || expr[i] == '.')) {
                    num += expr[i++];
                }
                --i;
                values.push(std::stod(num));
            }
            else if (isalpha(expr[i])) {
                std::string var;
                while (i < expr.length() && isalpha(expr[i])) {
                    var += expr[i++];
                }
                --i;
                if (variables.count(var)) {
                    values.push(variables[var]);
                } else {
                    std::cout << "变量 " << var << " 未定义\n";
                    values.push(0);
                }
            }
            else if (expr[i] == ')') {
                while (!ops.empty() && ops.top() != '(') {
                    double b = values.top(); values.pop();
                    double a = values.top(); values.pop();
                    char op = ops.top(); ops.pop();
                    values.push(applyOp(a, b, op));
                }
                ops.pop();
            }
            else {
                while (!ops.empty() && precedence(ops.top()) >= precedence(expr[i])) {
                    double b = values.top(); values.pop();
                    double a = values.top(); values.pop();
                    char op = ops.top(); ops.pop();
                    values.push(applyOp(a, b, op));
                }
                ops.push(expr[i]);
            }
        }
        
        while (!ops.empty()) {
            double b = values.top(); values.pop();
            double a = values.top(); values.pop();
            char op = ops.top(); ops.pop();
            values.push(applyOp(a, b, op));
        }
        
        return values.top();
    }
    
    void setVariable(const std::string& name, double value) {
        variables[name] = value;
    }
};

int main() {
    Calculator calc;
    
    calc.setVariable("x", 10);
    calc.setVariable("y", 5);
    
    std::cout << "2 + 3 * 4 = " << calc.evaluate("2 + 3 * 4") << "\n";
    std::cout << "(2 + 3) * 4 = " << calc.evaluate("(2 + 3) * 4") << "\n";
    std::cout << "x + y = " << calc.evaluate("x + y") << "\n";
    std::cout << "x * y + 10 = " << calc.evaluate("x * y + 10") << "\n";
    
    return 0;
}
```

## 3. 文件处理工具

### 1. 需求

- 文件复制
- 文件搜索
- 行数统计

### 2. 实现

```cpp
#include <iostream>
#include <fstream>
#include <filesystem>
#include <string>
#include <vector>

namespace fs = std::filesystem;

class FileTool {
public:
    static bool copyFile(const std::string& src, const std::string& dest) {
        try {
            fs::copy_file(src, dest, fs::copy_options::overwrite_existing);
            return true;
        } catch (const std::exception& e) {
            std::cerr << "Copy failed: " << e.what() << "\n";
            return false;
        }
    }
    
    static std::vector<fs::path> searchFiles(const std::string& dir, 
                                              const std::string& pattern) {
        std::vector<fs::path> results;
        
        for (const auto& entry : fs::recursive_directory_iterator(dir)) {
            if (entry.path().filename().string().find(pattern) != std::string::npos) {
                results.push_back(entry.path());
            }
        }
        
        return results;
    }
    
    static std::map<std::string, size_t> countLines(const std::string& dir) {
        std::map<std::string, size_t> counts;
        
        for (const auto& entry : fs::recursive_directory_iterator(dir)) {
            if (entry.is_regular_file() && entry.path().extension() == ".cpp") {
                std::ifstream file(entry.path());
                size_t lines = std::count(std::istreambuf_iterator<char>(file),
                                          std::istreambuf_iterator<char>(), '\n');
                counts[entry.path().string()] = lines;
            }
        }
        
        return counts;
    }
    
    static void printDirectoryTree(const fs::path& path, int indent = 0) {
        for (const auto& entry : fs::directory_iterator(path)) {
            std::cout << std::string(indent, ' ');
            
            if (entry.is_directory()) {
                std::cout << "[D] " << entry.path().filename() << "\n";
                printDirectoryTree(entry.path(), indent + 2);
            } else {
                std::cout << "[F] " << entry.path().filename() 
                          << " (" << entry.file_size() << " bytes)\n";
            }
        }
    }
};

int main() {
    std::cout << "=== 文件搜索 ===\n";
    auto files = FileTool::searchFiles(".", ".cpp");
    for (const auto& f : files) {
        std::cout << f << "\n";
    }
    
    std::cout << "\n=== 目录树 ===\n";
    FileTool::printDirectoryTree(".");
    
    return 0;
}
```

## 4. 任务调度器

### 1. 需求

- 定时任务执行
- 任务队列管理
- 并发执行

### 2. 实现

```cpp
#include <iostream>
#include <queue>
#include <functional>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <chrono>
#include <atomic>

class TaskScheduler {
private:
    struct Task {
        std::function<void()> func;
        std::chrono::steady_clock::time_point executeTime;
        std::string name;
        
        bool operator>(const Task& other) const {
            return executeTime > other.executeTime;
        }
    };
    
    std::priority_queue<Task, std::vector<Task>, std::greater<Task>> tasks;
    std::mutex mtx;
    std::condition_variable cv;
    std::atomic<bool> running{true};
    std::thread worker;
    
public:
    TaskScheduler() {
        worker = std::thread([this]() {
            while (running) {
                std::unique_lock<std::mutex> lock(mtx);
                
                if (tasks.empty()) {
                    cv.wait(lock);
                    continue;
                }
                
                auto now = std::chrono::steady_clock::now();
                auto& top = tasks.top();
                
                if (top.executeTime <= now) {
                    Task task = top;
                    tasks.pop();
                    lock.unlock();
                    
                    std::cout << "执行任务: " << task.name << "\n";
                    task.func();
                } else {
                    cv.wait_until(lock, top.executeTime);
                }
            }
        });
    }
    
    ~TaskScheduler() {
        running = false;
        cv.notify_all();
        if (worker.joinable()) {
            worker.join();
        }
    }
    
    void schedule(std::function<void()> func, int delayMs, const std::string& name = "") {
        auto executeTime = std::chrono::steady_clock::now() + 
                          std::chrono::milliseconds(delayMs);
        
        {
            std::lock_guard<std::mutex> lock(mtx);
            tasks.push({func, executeTime, name});
        }
        cv.notify_one();
    }
};

int main() {
    TaskScheduler scheduler;
    
    scheduler.schedule([]() {
        std::cout << "任务1执行\n";
    }, 1000, "Task 1");
    
    scheduler.schedule([]() {
        std::cout << "任务2执行\n";
    }, 2000, "Task 2");
    
    scheduler.schedule([]() {
        std::cout << "任务3执行\n";
    }, 500, "Task 3");
    
    std::cout << "任务已调度，等待执行...\n";
    
    std::this_thread::sleep_for(std::chrono::seconds(3));
    
    return 0;
}
```

## 5. 注意事项与最佳实践

### 1. 代码质量

```cpp
#include <memory>  // 智能指针所需头文件
// 1. 使用 const 和引用
void process(const std::string& data);  // 好
void process(std::string data);         // 避免

// 2. 使用 RAII
{
    std::lock_guard<std::mutex> lock(mtx);
    // 自动解锁
}

// 3. 使用智能指针
auto ptr = std::make_unique<Resource>();
auto shared = std::make_shared<Resource>();

// 4. 使用范围 for
for (const auto& item : container) { }

// 5. 使用 auto 简化
auto it = map.find(key);
```

### 2. 性能优化

```cpp
// 1. 避免不必要的拷贝
std::string& ref = get_string();  // 引用

// 2. 使用移动语义
std::vector<int> v = std::move(other);

// 3. 预分配容量
std::vector<int> v;
v.reserve(1000);

// 4. 使用 emplace
vec.emplace_back(arg1, arg2);

// 5. 选择合适的容器
// 随机访问: vector
// 中间插入: list
// 查找: unordered_map
```
