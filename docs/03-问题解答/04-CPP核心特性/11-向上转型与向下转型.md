# 向上转型与向下转型
> 📖 相关章节：[类与对象](../../02-CPP/03-类与对象.md)、[继承与多态](../../02-CPP/04-继承与多态.md)、[核心机制](../../02-CPP/05-核心机制.md)、[异常处理](../../02-CPP/07-异常处理.md)、[移动语义](../../02-CPP/09-移动语义与完美转发.md)

### 1. 本质洞察

**向上转型**（Upcasting）= 派生类→基类，永远安全，隐式转换；**向下转型**（Downcasting）= 基类→派生类，不安全，必须用 dynamic_cast 检查。频繁向下转型通常意味着设计有问题。

***

### 2. 向上转型详解

向上转型是将派生类指针/引用转换为基类指针/引用，这是安全的隐式转换。

```cpp
class Animal {
public:
    virtual ~Animal() = default;
    virtual void speak() const { cout << "Animal sound" << endl; }
    int age_;
};

class Dog : public Animal {
public:
    void speak() const override { cout << "Woof!" << endl; }
    void fetch() const { cout << "Fetch!" << endl; }
    string breed_;
};

Dog dog;
// 以下都是向上转型，全部安全、隐式
Animal* a = &dog;                    // 指针向上转型
Animal& ref = dog;                   // 引用向上转型
Animal animal = dog;                 // 对象切片！注意！
```

**为什么安全**：派生类一定包含基类的所有成员，转成基类只是"缩小视野"，不会访问到不存在的数据。

**对象切片陷阱**：

```cpp
Dog dog;
dog.breed_ = "Golden";
dog.age_ = 3;

Animal animal = dog;                 // 对象切片！
// animal 只有 Animal 的部分，breed_ 被切掉了
// animal.speak() 调用 Animal::speak，不是 Dog::speak
// 因为 animal 是对象，不是指针/引用，没有多态

// 正确做法：用指针或引用
Animal& ref = dog;
ref.speak();                         // 输出 "Woof!"，多态生效
```

**对象切片 vs 多态对比**：

| 方式 | 类型 | 多态 | 数据 |
|------|------|:---:|------|
| `Animal a = dog;` | 值（对象） | 不生效 | 派生类部分被切掉 |
| `Animal* p = &dog;` | 指针 | 生效 | 完整 |
| `Animal& r = dog;` | 引用 | 生效 | 完整 |

### 3. 向下转型详解

向下转型是将基类指针/引用转换为派生类指针/引用，这是不安全的操作。

```cpp
Animal* a = new Dog();

// 不安全的向下转型：static_cast
Dog* d1 = static_cast<Dog*>(a);      // 不检查，如果 a 不是 Dog 则 UB
d1->fetch();                          // 看似正常，但 a 可能不是 Dog

// 安全的向下转型：dynamic_cast
Dog* d2 = dynamic_cast<Dog*>(a);     // 运行时检查
if (d2 != nullptr) {
    d2->fetch();                      // 安全
}

// 引用版本的 dynamic_cast
try {
    Dog& d3 = dynamic_cast<Dog&>(*a);
    d3.fetch();
} catch (const std::bad_cast& e) {
    cout << "转型失败: " << e.what() << endl;
}
```

**为什么不安全**：基类指针可能指向任何派生类，不一定是目标类型。

```cpp
Animal* animals[] = {
    new Dog(),
    new Cat(),
    new Bird()
};

// 危险：盲目向下转型
Dog* d = static_cast<Dog*>(animals[1]);  // animals[1] 是 Cat！
d->fetch();                               // UB！访问了不存在的 Dog 成员

// 安全：dynamic_cast 检查
for (auto a : animals) {
    if (Dog* d = dynamic_cast<Dog*>(a)) {
        d->fetch();                       // 只有真正的 Dog 才会执行
    }
}
```

### 4. static_cast vs dynamic_cast 向下转型对比

| 特性 | static_cast | dynamic_cast |
|------|:---:|:---:|
| 运行时检查 | 无 | 有（RTTI） |
| 失败时行为 | UB（未定义行为） | 指针返回 nullptr，引用抛 bad_cast |
| 性能 | 零开销 | 有开销（查 vtable） |
| 前提条件 | 无 | 基类必须有虚函数 |
| 安全性 | 低 | 高 |
| 适用场景 | 确定类型时 | 不确定类型时 |

**何时可以用 static_cast 向下转型**：

```cpp
// 场景：你100%确定对象的实际类型
void processDog(Animal* a) {
    // 如果调用方保证传入的一定是 Dog，可以用 static_cast
    // 但这种设计本身就有问题——为什么不用 Dog* 作为参数？
    Dog* d = static_cast<Dog*>(a);
    d->fetch();
}

// 更好的设计：直接用派生类类型
void processDog(Dog* d) {
    d->fetch();  // 不需要转型
}
```

### 5. 为什么要转型

| 场景 | 转型方向 | 原因 | 示例 |
|------|----------|------|------|
| 传参给接受基类的函数 | 向上 | 多态 | `void draw(Shape*)` |
| 存入基类容器 | 向上 | 统一管理 | `vector<Shape*>` |
| 访问派生类特有方法 | 向下 | 需要具体功能 | `dog->fetch()` |
| 访问派生类特有数据 | 向下 | 需要具体数据 | `circle->getRadius()` |
| 实现双重分派 | 向下 | 访问者模式 | `visitor.visit(this)` |

**向上转型的典型用法**：

```cpp
// 统一管理不同派生类对象
vector<unique_ptr<Shape>> shapes;
shapes.push_back(make_unique<Circle>(5.0));      // 向上转型
shapes.push_back(make_unique<Rectangle>(3, 4));  // 向上转型

for (const auto& s : shapes) {
    s->draw();       // 多态调用，各自画自己
}
```

### 6. 向下转型的替代方案：多态设计

频繁使用向下转型通常意味着设计有问题。应该优先使用多态来替代。

**不好的设计：频繁向下转型**：

```cpp
enum class ShapeType { Circle, Rectangle, Triangle };

class Shape {
public:
    ShapeType type;
    virtual ~Shape() = default;
};

class Circle : public Shape {
public:
    double radius;
    Circle(double r) : radius(r) { type = ShapeType::Circle; }
};

class Rectangle : public Shape {
public:
    double width, height;
    Rectangle(double w, double h) : width(w), height(h) { type = ShapeType::Rectangle; }
};

// 不好：根据类型做不同处理
double computeArea(Shape* s) {
    switch (s->type) {
        case ShapeType::Circle:
            return 3.14159 * static_cast<Circle*>(s)->radius
                             * static_cast<Circle*>(s)->radius;
        case ShapeType::Rectangle:
            return static_cast<Rectangle*>(s)->width
                 * static_cast<Rectangle*>(s)->height;
        default:
            return 0;
    }
}
```

**好的设计：利用多态**：

```cpp
class Shape {
public:
    virtual ~Shape() = default;
    virtual double area() const = 0;       // 每个派生类自己实现
    virtual void draw() const = 0;
};

class Circle : public Shape {
    double radius_;
public:
    Circle(double r) : radius_(r) {}
    double area() const override { return 3.14159 * radius_ * radius_; }
    void draw() const override { cout << "Circle"; }
};

class Rectangle : public Shape {
    double w_, h_;
public:
    Rectangle(double w, double h) : w_(w), h_(h) {}
    double area() const override { return w_ * h_; }
    void draw() const override { cout << "Rectangle"; }
};

// 简洁：不需要转型
double computeArea(const Shape& s) {
    return s.area();  // 多态自动调用正确的实现
}
```

**访问者模式：需要根据类型做不同操作时**：

```cpp
class Circle;
class Rectangle;

class ShapeVisitor {
public:
    virtual void visit(Circle& c) = 0;
    virtual void visit(Rectangle& r) = 0;
    virtual ~ShapeVisitor() = default;
};

class Shape {
public:
    virtual void accept(ShapeVisitor& v) = 0;
    virtual ~Shape() = default;
};

class Circle : public Shape {
    double radius_;
public:
    Circle(double r) : radius_(r) {}
    double getRadius() const { return radius_; }
    void accept(ShapeVisitor& v) override { v.visit(*this); }
};

class Rectangle : public Shape {
    double w_, h_;
public:
    Rectangle(double w, double h) : w_(w), h_(h) {}
    double getWidth() const { return w_; }
    void accept(ShapeVisitor& v) override { v.visit(*this); }
};

// 具体操作在 Visitor 中实现，不需要 dynamic_cast
class AreaCalculator : public ShapeVisitor {
    double result_;
public:
    void visit(Circle& c) override {
        result_ = 3.14159 * c.getRadius() * c.getRadius();
    }
    void visit(Rectangle& r) override {
        result_ = r.getWidth();
    }
    double result() const { return result_; }
};
```

### 7. dynamic_cast 的性能开销

`dynamic_cast` 需要遍历继承层次进行类型匹配，开销不可忽略。

```cpp
// dynamic_cast 的大致实现过程：
// 1. 获取对象的 vptr
// 2. 从 vtable 中获取 type_info
// 3. 沿继承链逐层比较 type_info
// 4. 对于多重继承，还需要调整 this 指针偏移
```

**性能对比**：

| 操作 | 相对开销 |
|------|----------|
| static_cast 向下转型 | ~1x（零开销） |
| dynamic_cast 单继承 | ~2-5x |
| dynamic_cast 多继承 | ~5-20x |
| 虚函数调用 | ~1.5x |

**优化建议**：

```cpp
// 不好：在热循环中反复 dynamic_cast
for (auto& shape : shapes) {
    if (Circle* c = dynamic_cast<Circle*>(shape)) {
        c->drawCircle();
    }
}

// 好：用多态替代
for (auto& shape : shapes) {
    shape->draw();  // 虚函数调用，比 dynamic_cast 快
}

// 好：如果必须转型，转型一次缓存结果
Circle* c = dynamic_cast<Circle*>(shape);
if (c) {
    c->drawCircle();
    c->setRadius(5);
    // 后续操作不再转型
}
```

### 8. 交叉转型（Cross-Cast）

交叉转型是在多重继承的兄弟类之间转型，只能用 `dynamic_cast`。

```cpp
class Animal {
public:
    virtual ~Animal() = default;
    virtual void speak() = 0;
};

class Pet {
public:
    virtual ~Pet() = default;
    virtual string getName() const = 0;
};

class Dog : public Animal, public Pet {
public:
    void speak() override { cout << "Woof!"; }
    string getName() const override { return "Buddy"; }
    void fetch() { cout << "Fetch!"; }
};

// 交叉转型：Animal* → Pet*
Animal* a = new Dog();
Pet* p = dynamic_cast<Pet*>(a);      // 交叉转型
if (p) {
    cout << p->getName();             // OK
}

// static_cast 不能做交叉转型
// Pet* p2 = static_cast<Pet*>(a);   // 编译错误！
```

**交叉转型的内存布局**：

```
Dog 对象内存布局：
+------------------+
| Animal 子对象     |  ← a 指向这里
|   vptr_Animal    |
+------------------+
| Pet 子对象       |  ← p 指向这里（偏移调整）
|   vptr_Pet       |
+------------------+
| Dog 成员         |
+------------------+

dynamic_cast 自动处理 this 指针偏移
```

### 9. 完整示例：图形编辑器中的转型

```cpp
#include <iostream>
#include <vector>
#include <memory>
using namespace std;

class Shape {
public:
    virtual ~Shape() = default;
    virtual void draw() const = 0;
    virtual double area() const = 0;
    virtual void scale(double factor) = 0;
};

class Circle : public Shape {
    double radius_;
public:
    Circle(double r) : radius_(r) {}
    void draw() const override { cout << "Circle(r=" << radius_ << ")\n"; }
    double area() const override { return 3.14159 * radius_ * radius_; }
    void scale(double f) override { radius_ *= f; }
    double getRadius() const { return radius_; }
};

class Rectangle : public Shape {
    double w_, h_;
public:
    Rectangle(double w, double h) : w_(w), h_(h) {}
    void draw() const override { cout << "Rect(" << w_ << "x" << h_ << ")\n"; }
    double area() const override { return w_ * h_; }
    void scale(double f) override { w_ *= f; h_ *= f; }
    double getWidth() const { return w_; }
};

class Triangle : public Shape {
    double base_, height_;
public:
    Triangle(double b, double h) : base_(b), height_(h) {}
    void draw() const override { cout << "Triangle(b=" << base_ << ",h=" << height_ << ")\n"; }
    double area() const override { return 0.5 * base_ * height_; }
    void scale(double f) override { base_ *= f; height_ *= f; }
};

int main() {
    // 向上转型：统一管理
    vector<unique_ptr<Shape>> shapes;
    shapes.push_back(make_unique<Circle>(5.0));
    shapes.push_back(make_unique<Rectangle>(3.0, 4.0));
    shapes.push_back(make_unique<Triangle>(6.0, 8.0));

    // 多态调用，不需要转型
    for (const auto& s : shapes) {
        s->draw();
        cout << "  area = " << s->area() << "\n";
    }

    // 需要访问派生类特有接口时，用 dynamic_cast
    for (const auto& s : shapes) {
        if (Circle* c = dynamic_cast<Circle*>(s.get())) {
            cout << "  Circle radius = " << c->getRadius() << "\n";
        }
    }

    // 安全缩放：用多态，不需要转型
    for (auto& s : shapes) {
        s->scale(2.0);
    }

    return 0;
}
```

### 10. 常见陷阱

**陷阱1：忘记检查 dynamic_cast 返回值**

```cpp
// 危险：不检查返回值
Dog* d = dynamic_cast<Dog*>(animal);
d->fetch();  // 如果 d 是 nullptr，崩溃！

// 正确：必须检查
if (Dog* d = dynamic_cast<Dog*>(animal)) {
    d->fetch();  // 安全
}
```

**陷阱2：对非多态类型使用 dynamic_cast**

```cpp
class Base {};              // 没有虚函数
class Derived : public Base {};

Base* b = new Derived();
// Derived* d = dynamic_cast<Derived*>(b);  // 编译错误！基类不是多态类型
Derived* d = static_cast<Derived*>(b);       // 只能用 static_cast，但不安全
```

**陷阱3：对象切片导致多态失效**

```cpp
Dog dog;
Animal a = dog;            // 切片！丢失 Dog 部分
// dynamic_cast<Dog*>(&a)  // 返回 nullptr，a 不是 Dog 对象
```

### 11. 极简总结

**向上转型 = 派生→基类 = 安全隐式 = 多态基础 | 向下转型 = 基类→派生类 = 不安全用 dynamic_cast | 频繁向下转型 = 设计有问题 = 优先用多态替代**
