# 什么是裸机编程
> 📖 相关章节：[嵌入式概述](../../12-嵌入式与实时系统/00-嵌入式系统概述.md)

> **裸机编程就是没有操作系统兜底——你写的代码直接跑在硬件上，没有文件系统、没有标准库、没有内存保护，一切都要自己来。** 它让你真正理解"程序是怎么跑起来的"。

***

### 1. 精髓速览

**裸机编程（Bare-metal Programming）** 是指在没有操作系统的硬件上直接编写和运行程序。你的代码直接与CPU寄存器、内存映射外设、中断控制器等硬件交互，没有OS提供的抽象和服务——这意味着你拥有完全的控制权，但也承担了全部的责任。

***

### 2. 生活类比

| 类比 | 有OS | 裸机 |
|------|------|------|
| 住房 | 精装公寓（水电暖齐全） | 毛坯房（自己接水管、拉电线） |
| 出行 | 公交车（固定路线、固定时间） | 自己修路、自己开车 |
| 吃饭 | 餐厅（点菜就上） | 自己种地、做饭 |
| 安全 | 保安巡逻 | 自己锁门 |

**具体场景**：在OS上调用`printf("hello")`，OS帮你把字符送到显示器；裸机上你需要自己配置UART控制器、设置波特率、逐字节发送——每一步都要和硬件打交道。

***

### 3. 裸机 vs 有OS的区别

| 维度 | 有操作系统 | 裸机 |
|------|-----------|------|
| 启动 | OS引导→加载程序 | 自己写启动代码 |
| 内存管理 | OS提供malloc/free | 自己管理每一字节 |
| 文件系统 | OS提供文件API | 自己实现或不用 |
| 输入输出 | OS驱动程序 | 直接操作硬件寄存器 |
| 多任务 | OS调度器 | 自己实现或单任务 |
| 中断 | OS处理并转发 | 自己写ISR |
| 错误处理 | OS异常处理 | 一个错误就死机 |
| 调试 | GDB/IDE | JTAG/串口打印 |
| 标准库 | 完整可用 | 不可用或极度精简 |

### 4. 裸机启动流程

#### 4.1 从上电到main函数

```
上电
 │
 ↓
CPU从复位向量地址取指令（通常是0x00000000或0xFFFF0000）
 │
 ↓
执行启动代码（Startup/Boot Code）：
  1. 初始化栈指针（SP）
  2. 初始化.data段（从Flash拷贝到RAM）
  3. 清零.bss段
  4. 调用main函数
 │
 ↓
main函数执行你的应用逻辑
```

#### 4.2 启动代码示例（ARM Cortex-M）

```assembly
; ARM Cortex-M 启动文件（简化版）
; 文件：startup.s

; 中断向量表
    .section .isr_vector, "a"
    .word   _estack           ; 栈顶地址（MSP初始值）
    .word   Reset_Handler     ; 复位向量
    .word   NMI_Handler       ; NMI中断
    .word   HardFault_Handler ; 硬件错误
    ; ... 更多中断向量

; 复位处理函数
    .section .text
    .thumb_func
Reset_Handler:
    ; 1. 拷贝.data段从Flash到RAM
    ldr     r0, =_sdata       ; .data段在RAM中的起始地址
    ldr     r1, =_edata       ; .data段在RAM中的结束地址
    ldr     r2, =_sidata      ; .data段在Flash中的起始地址
copy_data:
    cmp     r0, r1
    bge     zero_bss
    ldr     r3, [r2]
    str     r3, [r0]
    adds    r0, r0, #4
    adds    r2, r2, #4
    b       copy_data

    ; 2. 清零.bss段
zero_bss:
    ldr     r0, =_sbss
    ldr     r1, =_ebss
    movs    r3, #0
clear_bss:
    cmp     r0, r1
    bge     call_main
    str     r3, [r0]
    adds    r0, r0, #4
    b       clear_bss

    ; 3. 调用main函数
call_main:
    bl      main

    ; 如果main返回，进入死循环
    b       .
```

#### 4.3 链接脚本

```
/* ARM Cortex-M 链接脚本（简化版） */
/* 文件：linker.ld */

MEMORY {
    FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 512K
    RAM (rwx)   : ORIGIN = 0x20000000, LENGTH = 128K
}

SECTIONS {
    /* 中断向量表放在Flash最前面 */
    .isr_vector : {
        . = ALIGN(4);
        KEEP(*(.isr_vector))
        . = ALIGN(4);
    } > FLASH

    /* 代码段 */
    .text : {
        . = ALIGN(4);
        *(.text)
        *(.text*)
        . = ALIGN(4);
    } > FLASH

    /* 只读数据 */
    .rodata : {
        . = ALIGN(4);
        *(.rodata)
        *(.rodata*)
        . = ALIGN(4);
    } > FLASH

    /* .data段：存储在Flash，运行时拷贝到RAM */
    _sidata = LOADADDR(.data);
    .data : {
        . = ALIGN(4);
        _sdata = .;
        *(.data)
        *(.data*)
        . = ALIGN(4);
        _edata = .;
    } > RAM AT > FLASH

    /* .bss段：运行时清零 */
    .bss : {
        . = ALIGN(4);
        _sbss = .;
        *(.bss)
        *(.bss*)
        *(COMMON)
        . = ALIGN(4);
        _ebss = .;
    } > RAM

    /* 栈 */
    _estack = ORIGIN(RAM) + LENGTH(RAM);
}
```

### 5. 为什么需要理解硬件

#### 5.1 直接操作硬件寄存器

```cpp
// ARM Cortex-M4：点亮LED（GPIO操作）
// STM32F407的GPIOA基地址
#define PERIPH_BASE       0x40000000UL
#define AHB1PERIPH_BASE   (PERIPH_BASE + 0x00020000UL)
#define GPIOA_BASE        (AHB1PERIPH_BASE + 0x0000UL)

// GPIO寄存器结构体
typedef struct {
    volatile uint32_t MODER;    // 模式寄存器
    volatile uint32_t OTYPER;   // 输出类型寄存器
    volatile uint32_t OSPEEDR;  // 输出速度寄存器
    volatile uint32_t PUPDR;    // 上拉下拉寄存器
    volatile uint32_t IDR;      // 输入数据寄存器
    volatile uint32_t ODR;      // 输出数据寄存器
    volatile uint32_t BSRR;     // 置位/复位寄存器
    volatile uint32_t LCKR;     // 锁定寄存器
    volatile uint32_t AFR[2];   // 复用功能寄存器
} GPIO_TypeDef;

#define GPIOA ((GPIO_TypeDef*)GPIOA_BASE)

// RCC时钟控制寄存器
#define RCC_BASE          (AHB1PERIPH_BASE + 0x3800UL)
typedef struct {
    volatile uint32_t CR;
    volatile uint32_t PLLCFGR;
    volatile uint32_t CFGR;
    volatile uint32_t CIR;
    volatile uint32_t AHB1ENR;  // AHB1外设时钟使能
    // ...
} RCC_TypeDef;

#define RCC ((RCC_TypeDef*)RCC_BASE)

// 初始化GPIOA第5引脚（LED）
void led_init(void) {
    // 1. 使能GPIOA时钟
    RCC->AHB1ENR |= (1 << 0);  // GPIOAEN = bit 0

    // 2. 配置PA5为输出模式
    GPIOA->MODER &= ~(3 << (5 * 2));   // 清除bit[11:10]
    GPIOA->MODER |=  (1 << (5 * 2));   // 设置为01（通用输出）

    // 3. 配置为推挽输出
    GPIOA->OTYPER &= ~(1 << 5);        // 0=推挽

    // 4. 配置输出速度
    GPIOA->OSPEEDR |= (3 << (5 * 2));  // 高速

    // 5. 无上拉下拉
    GPIOA->PUPDR &= ~(3 << (5 * 2));   // 无上拉下拉
}

// 点亮LED
void led_on(void) {
    GPIOA->BSRR = (1 << 5);  // 置位PA5
}

// 熄灭LED
void led_off(void) {
    GPIOA->BSRR = (1 << (5 + 16));  // 复位PA5
}
```

#### 5.2 UART串口通信

```cpp
// STM32 UART发送（无OS，直接操作寄存器）
#define USART1_BASE  (APB2PERIPH_BASE + 0x1000UL)
typedef struct {
    volatile uint32_t SR;    // 状态寄存器
    volatile uint32_t DR;    // 数据寄存器
    volatile uint32_t BRR;   // 波特率寄存器
    volatile uint32_t CR1;   // 控制寄存器1
    volatile uint32_t CR2;   // 控制寄存器2
    volatile uint32_t CR3;   // 控制寄存器3
} USART_TypeDef;

#define USART1 ((USART_TypeDef*)USART1_BASE)

// 初始化UART1（115200波特率，8N1）
void uart_init(void) {
    // 使能USART1和GPIOA时钟
    RCC->APB2ENR |= (1 << 4);   // USART1EN
    RCC->AHB1ENR |= (1 << 0);   // GPIOAEN

    // 配置PA9(TX)为复用功能
    GPIOA->MODER &= ~(3 << (9 * 2));
    GPIOA->MODER |=  (2 << (9 * 2));   // 复用模式
    GPIOA->AFR[1] &= ~(0xF << ((9-8) * 4));
    GPIOA->AFR[1] |=  (7 << ((9-8) * 4));  // AF7 = USART1_TX

    // 配置波特率（假设APB2时钟84MHz）
    // BRR = 84MHz / 115200 ≈ 729
    USART1->BRR = 729;

    // 使能发送和UART
    USART1->CR1 |= (1 << 3);   // TE = 发送使能
    USART1->CR1 |= (1 << 13);  // UE = USART使能
}

// 发送一个字节
void uart_send_byte(uint8_t data) {
    // 等待发送数据寄存器为空
    while (!(USART1->SR & (1 << 7)));  // TXE位
    USART1->DR = data;
}

// 发送字符串
void uart_send_string(const char* str) {
    while (*str) {
        uart_send_byte(*str++);
    }
}
```

### 6. 裸机编程的挑战

#### 6.1 没有标准库

```cpp
// 不能使用标准库函数，需要自己实现

// 自定义malloc（极简版，仅用于演示）
#define HEAP_SIZE  (16 * 1024)  // 16KB堆空间
static uint8_t heap[HEAP_SIZE];
static size_t heap_offset = 0;

void* my_malloc(size_t size) {
    // 对齐到4字节
    size = (size + 3) & ~3;

    if (heap_offset + size > HEAP_SIZE) {
        return NULL;  // 内存不足
    }

    void* ptr = &heap[heap_offset];
    heap_offset += size;
    return ptr;
}

// 注意：没有free()！这是最简单的线性分配器
// 实际项目中需要更复杂的内存管理策略

// 自定义printf（极简版）
void my_printf(const char* fmt, ...) {
    va_list args;
    va_start(args, fmt);

    while (*fmt) {
        if (*fmt == '%') {
            fmt++;
            switch (*fmt) {
                case 'd': {
                    int val = va_arg(args, int);
                    // 整数转字符串
                    char buf[16];
                    int i = 0;
                    if (val < 0) {
                        uart_send_byte('-');
                        val = -val;
                    }
                    do {
                        buf[i++] = '0' + (val % 10);
                        val /= 10;
                    } while (val > 0);
                    while (--i >= 0) uart_send_byte(buf[i]);
                    break;
                }
                case 's': {
                    const char* s = va_arg(args, const char*);
                    uart_send_string(s);
                    break;
                }
                case 'x': {
                    uint32_t val = va_arg(args, uint32_t);
                    for (int i = 7; i >= 0; i--) {
                        uint8_t nibble = (val >> (i * 4)) & 0xF;
                        uart_send_byte(nibble < 10 ? '0' + nibble : 'A' + nibble - 10);
                    }
                    break;
                }
                default:
                    uart_send_byte(*fmt);
            }
        } else {
            uart_send_byte(*fmt);
        }
        fmt++;
    }
    va_end(args);
}
```

#### 6.2 没有内存保护

```cpp
// 裸机上没有MMU，任何指针错误都可能导致灾难

// 在OS上：段错误，程序崩溃，OS回收资源
// 在裸机上：可能覆盖关键数据，系统行为不可预测

// 危险示例：数组越界
uint8_t buffer[16];
void dangerous_write(void) {
    for (int i = 0; i < 32; i++) {  // 越界！
        buffer[i] = 0xFF;  // 可能覆盖其他变量或硬件寄存器
    }
}

// 防御性编程：使用断言
#define ASSERT(cond) do { \
    if (!(cond)) { \
        uart_send_string("ASSERT FAIL: " #cond "\n"); \
        while(1); \
    } \
} while(0)

void safe_write(void) {
    ASSERT(sizeof(buffer) >= 16);
    for (int i = 0; i < 16; i++) {
        buffer[i] = 0xFF;
    }
}
```

### 7. 裸机项目结构

```
my_baremetal_project/
├── startup.s          # 启动文件（汇编）
├── linker.ld          # 链接脚本
├── Makefile           # 构建脚本
├── src/
│   ├── main.c         # 主程序
│   ├── gpio.c         # GPIO驱动
│   ├── uart.c         # UART驱动
│   ├── timer.c        # 定时器驱动
│   └── interrupts.c   # 中断处理
├── inc/
│   ├── gpio.h
│   ├── uart.h
│   ├── timer.h
│   └── interrupts.h
└── tools/
    └── openocd.cfg    # 调试器配置
```

```makefile
# Makefile示例（ARM Cortex-M）
CC = arm-none-eabi-gcc
OBJCOPY = arm-none-eabi-objcopy

CFLAGS = -mcpu=cortex-m4 -mthumb -Wall -O2
CFLAGS += -ffunction-sections -fdata-sections
CFLAGS += -Iinc

LDFLAGS = -Tlinker.ld -Wl,--gc-sections -nostdlib -nostartfiles

SOURCES = startup.s src/main.c src/gpio.c src/uart.c src/timer.c
OBJECTS = $(SOURCES:.c=.o)
OBJECTS := $(OBJECTS:.s=.o)

all: firmware.bin

firmware.elf: $(OBJECTS)
	$(CC) $(LDFLAGS) -o $@ $^

firmware.bin: firmware.elf
	$(OBJCOPY) -O binary $< $@

flash: firmware.bin
	openocd -f tools/openocd.cfg -c "program firmware.bin verify reset exit 0x08000000"

clean:
	rm -f $(OBJECTS) firmware.elf firmware.bin
```

### 8. 常见误区

| 误区 | 事实 |
|------|------|
| 裸机编程必须用汇编 | 大部分代码可以用C写，只有启动代码需要汇编 |
| 裸机编程已经过时 | IoT、安全关键系统、超低功耗场景仍在广泛使用 |
| 裸机编程比有OS简单 | 恰恰相反，裸机编程需要理解更多底层细节 |
| 裸机不能跑C++ | 可以，但需要适配运行时支持（new/delete等） |
| 裸机程序不需要链接脚本 | 必须需要，否则不知道代码和数据放在哪里 |

***

### 9. 总结

| 维度 | 核心要点 |
|------|---------|
| 本质 | 无OS，代码直接运行在硬件上 |
| 启动流程 | 复位向量→初始化栈/数据段→main |
| 硬件交互 | 直接读写寄存器、内存映射I/O |
| 主要挑战 | 无标准库、无内存保护、调试困难 |
| 适用场景 | 超低功耗、安全关键、资源极度受限 |

**核心记忆**：裸机编程 = 没有OS兜底 + 直接操作硬件 + 自己管理一切。它让你从"用电脑"变成"造电脑"，真正理解程序是怎么跑起来的。
