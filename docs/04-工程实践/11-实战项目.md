```c
/* buffer.c */
#include "buffer.h"
#include <stdlib.h>
#include <string.h>

Line *line_create(const char *text) {
    Line *line = malloc(sizeof(Line));
    if (!line) return NULL;
    
    line->length = text ? strlen(text) : 0;
    line->text = malloc(line->length + 1);
    
    if (text) {
        strcpy(line->text, text);
    } else {
        line->text[0] = '\0';
    }
    
    line->next = NULL;
    line->prev = NULL;
    
    return line;
}

void line_free(Line *line) {
    if (line) {
        free(line->text);
        free(line);
    }
}

TextBuffer *buffer_create(void) {
    TextBuffer *buf = malloc(sizeof(TextBuffer));
    if (!buf) return NULL;
    
    /* 创建一个空行 */
    Line *line = line_create("");
    buf->head = line;
    buf->tail = line;
    buf->current = line;
    buf->total_lines = 1;
    buf->cursor_row = 0;
    buf->cursor_col = 0;
    buf->filename[0] = '\0';
    buf->modified = 0;
    
    return buf;
}

int buffer_insert_line(TextBuffer *buf, int row, const char *text) {
    if (!buf || row < 0 || row > buf->total_lines) {
        return -1;
    }
    
    Line *new_line = line_create(text);
    if (!new_line) return -1;
    
    /* 插入到指定位置 */
    if (row == 0) {
        /* 插入到头部 */
        new_line->next = buf->head;
        buf->head->prev = new_line;
        buf->head = new_line;
    } else if (row == buf->total_lines) {
        /* 插入到尾部 */
        new_line->prev = buf->tail;
        buf->tail->next = new_line;
        buf->tail = new_line;
    } else {
        /* 插入到中间 */
        Line *current = buf->head;
        for (int i = 0; i < row; i++) {
            current = current->next;
        }
        
        new_line->prev = current->prev;
        new_line->next = current;
        current->prev->next = new_line;
        current->prev = new_line;
    }
    
    buf->total_lines++;
    buf->modified = 1;
    
    return 0;
}

int buffer_delete_line(TextBuffer *buf, int row) {
    if (!buf || row < 0 || row >= buf->total_lines) {
        return -1;
    }
    
    /* 找到要删除的行 */
    Line *line = buf->head;
    for (int i = 0; i < row; i++) {
        line = line->next;
    }
    
    /* 更新指针 */
    if (line->prev) {
        line->prev->next = line->next;
    } else {
        buf->head = line->next;
    }
    
    if (line->next) {
        line->next->prev = line->prev;
    } else {
        buf->tail = line->prev;
    }
    
    line_free(line);
    buf->total_lines--;
    buf->modified = 1;
    
    /* 如果删除后没有行了，创建一个空行 */
    if (buf->total_lines == 0) {
        Line *empty = line_create("");
        buf->head = empty;
        buf->tail = empty;
        buf->current = empty;
        buf->total_lines = 1;
    }
    
    return 0;
}
```

#### 2. 文件读写

```c
/* fileio.c */
#include "buffer.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int buffer_load_file(TextBuffer *buf, const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (!fp) return -1;
    
    /* 清空当前缓冲区 */
    while (buf->total_lines > 0) {
        buffer_delete_line(buf, 0);
    }
    
    /* 读取文件 */
    char line[4096];
    int row = 0;
    
    while (fgets(line, sizeof(line), fp)) {
        /* 去除换行符 */
        int len = strlen(line);
        if (len > 0 && line[len - 1] == '\n') {
            line[len - 1] = '\0';
        }
        
        buffer_insert_line(buf, row++, line);
    }
    
    fclose(fp);
    
    /* 保存文件名 */
    strncpy(buf->filename, filename, sizeof(buf->filename) - 1);
    buf->modified = 0;
    
    return 0;
}

int buffer_save_file(TextBuffer *buf, const char *filename) {
    FILE *fp = fopen(filename, "w");
    if (!fp) return -1;
    
    /* 写入所有行 */
    Line *line = buf->head;
    while (line) {
        fprintf(fp, "%s\n", line->text);
        line = line->next;
    }
    
    fclose(fp);
    
    /* 更新文件名和修改标志 */
    if (filename != buf->filename) {
        strncpy(buf->filename, filename, sizeof(buf->filename) - 1);
    }
    buf->modified = 0;
    
    return 0;
}
```

### 5. 遇到的陷阱

#### 1. 陷阱1：链表指针操作错误

```c
/* 错误示例：删除节点时指针操作顺序错误 */
int buffer_delete_line_wrong(TextBuffer *buf, int row) {
    Line *line = /* ... 找到要删除的行 ... */;
    
    /* 错误：先释放了内存，再访问指针 */
    line_free(line);
    
    if (line->prev) {  /* 错误！line已经被释放 */
        line->prev->next = line->next;
    }
    
    return 0;
}

/* 正确做法：先更新指针，再释放内存 */
int buffer_delete_line_correct(TextBuffer *buf, int row) {
    Line *line = /* ... 找到要删除的行 ... */;
    
    /* 先更新指针 */
    if (line->prev) {
        line->prev->next = line->next;
    }
    if (line->next) {
        line->next->prev = line->prev;
    }
    
    /* 最后释放内存 */
    line_free(line);
    
    return 0;
}
```

#### 2. 陷阱2：缓冲区溢出

```c
/* 错误示例：没有检查字符串长度 */
int buffer_insert_char_wrong(TextBuffer *buf, char ch) {
    Line *line = buf->current;
    
    /* 错误：没有检查是否超出缓冲区 */
    line->text[line->length] = ch;  /* 可能越界！ */
    line->text[line->length + 1] = '\0';
    line->length++;
    
    return 0;
}

/* 正确做法：先检查并扩容 */
int buffer_insert_char_correct(TextBuffer *buf, char ch) {
    Line *line = buf->current;
    
    /* 检查是否需要扩容 */
    if (line->length >= buf->cursor_col) {
        /* 分配更多空间 */
        int new_size = line->length * 2 + 1;
        char *new_text = realloc(line->text, new_size);
        if (!new_text) return -1;
        
        line->text = new_text;
    }
    
    /* 插入字符 */
    memmove(&line->text[buf->cursor_col + 1], 
            &line->text[buf->cursor_col],
            line->length - buf->cursor_col + 1);
    
    line->text[buf->cursor_col] = ch;
    line->length++;
    buf->cursor_col++;
    buf->modified = 1;
    
    return 0;
}
```

### 6. 优化技巧

#### 1. 技巧1：使用缓冲区减少磁盘I/O

```c
/* 优化前：逐字符写入 */
int save_slow(TextBuffer *buf, FILE *fp) {
    Line *line = buf->head;
    while (line) {
        for (int i = 0; i < line->length; i++) {
            fputc(line->text[i], fp);  /* 每个字符一次I/O */
        }
        fputc('\n', fp);
        line = line->next;
    }
    return 0;
}

/* 优化后：使用缓冲区 */
int save_fast(TextBuffer *buf, FILE *fp) {
    char buffer[8192];  /* 8KB缓冲区 */
    int pos = 0;
    
    Line *line = buf->head;
    while (line) {
        /* 写入到缓冲区 */
        int len = line->length;
        if (pos + len + 1 >= sizeof(buffer)) {
            /* 缓冲区满，写入文件 */
            fwrite(buffer, 1, pos, fp);
            pos = 0;
        }
        
        memcpy(buffer + pos, line->text, len);
        pos += len;
        buffer[pos++] = '\n';
        
        line = line->next;
    }
    
    /* 写入剩余数据 */
    if (pos > 0) {
        fwrite(buffer, 1, pos, fp);
    }
    
    return 0;
}
```

```c
    }
}
```

### 3. 核心实现

#### 1. 服务器端

```c
/* server.c */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pthread.h>
#include <sys/socket.h>
#include <netinet/in.h>

#define PORT 8888
#define MAX_CLIENTS 100

/* 客户端结构 */
typedef struct Client {
    int fd;                 /* socket描述符 */
    int user_id;            /* 用户ID */
    char username[50];      /* 用户名 */
    int room_id;            /* 当前房间ID */
    struct Client *next;
} Client;

/* 房间结构 */
typedef struct Room {
    int id;
    char name[50];
    Client *clients;        /* 房间内的客户端 */
    struct Room *next;
} Room;

/* 全局变量 */
Client *client_list = NULL;
Room *room_list = NULL;
pthread_mutex_t clients_mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t rooms_mutex = PTHREAD_MUTEX_INITIALIZER;

/* 添加客户端 */
void add_client(Client *client) {
    pthread_mutex_lock(&clients_mutex);
    client->next = client_list;
    client_list = client;
    pthread_mutex_unlock(&clients_mutex);
}

/* 移除客户端 */
void remove_client(Client *client) {
    pthread_mutex_lock(&clients_mutex);
    
    if (client_list == client) {
        client_list = client->next;
    } else {
        Client *prev = client_list;
        while (prev && prev->next != client) {
            prev = prev->next;
        }
        if (prev) {
            prev->next = client->next;
        }
    }
    
    pthread_mutex_unlock(&clients_mutex);
}

/* 广播消息到房间 */
void broadcast_to_room(int room_id, Message *msg, Client *sender) {
    pthread_mutex_lock(&rooms_mutex);
    
    /* 找到房间 */
    Room *room = room_list;
    while (room && room->id != room_id) {
        room = room->next;
    }
    
    if (!room) {
        pthread_mutex_unlock(&rooms_mutex);
        return;
    }
    
    /* 广播给房间内的所有客户端 */
    Client *client = room->clients;
    while (client) {
        if (client != sender) {  /* 不发给自己 */
            send(client->fd, msg, sizeof(MessageHeader) + msg->header.length, 0);
        }
        client = client->next;
    }
    
    pthread_mutex_unlock(&rooms_mutex);
}

/* 处理客户端消息 */
void *handle_client(void *arg) {
    Client *client = (Client *)arg;
    Message msg;
    
    while (1) {
        /* 接收消息头 */
        int n = recv(client->fd, &msg.header, sizeof(MessageHeader), 0);
        if (n <= 0) break;  /* 连接断开 */
        
        /* 接收消息体 */
        if (msg.header.length > 0) {
            n = recv(client->fd, msg.body, msg.header.length, 0);
            if (n <= 0) break;
        }
        
        /* 处理不同类型的消息 */
        switch (msg.header.type) {
            case MSG_CHAT:
                /* 设置发送者信息 */
                msg.header.user_id = client->user_id;
                /* 广播到房间 */
                broadcast_to_room(client->room_id, &msg, client);
                break;
                
            case MSG_PRIVATE:
                /* 私聊：找到目标用户并发送 */
                /* ... */
                break;
                
            case MSG_JOIN_ROOM:
                /* 加入房间 */
                /* ... */
                break;
                
            default:
                break;
        }
    }
    
    /* 清理 */
    close(client->fd);
    remove_client(client);
    free(client);
    
    return NULL;
}

int main() {
    int server_fd, client_fd;
    struct sockaddr_in address;
    
    /* 创建socket */
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    
    /* 设置地址重用 */
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    /* 绑定地址 */
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);
    
    bind(server_fd, (struct sockaddr *)&address, sizeof(address));
    
    /* 监听 */
    listen(server_fd, 10);
    printf("Server started on port %d\n", PORT);
    
    /* 接受连接 */
    while (1) {
        socklen_t addrlen = sizeof(address);
        client_fd = accept(server_fd, (struct sockaddr *)&address, &addrlen);
        
        printf("New client connected\n");
        
        /* 创建客户端结构 */
        Client *client = malloc(sizeof(Client));
        client->fd = client_fd;
        client->user_id = 0;
        client->room_id = 0;
        
        add_client(client);
        
        /* 创建线程处理客户端 */
        pthread_t thread;
        pthread_create(&thread, NULL, handle_client, client);
        pthread_detach(thread);
    }
    
    return 0;
}
```

### 4. 遇到的陷阱

#### 1. 陷阱1：竞态条件

```c
/* 错误示例：没有加锁 */
void broadcast_wrong(Message *msg) {
    Client *client = client_list;  /* 可能在遍历时被修改 */
    while (client) {
        send(client->fd, msg, sizeof(Message), 0);
        client = client->next;  /* 可能访问已释放的内存 */
    }
}

/* 正确做法：使用互斥锁 */
void broadcast_correct(Message *msg) {
    pthread_mutex_lock(&clients_mutex);
    
    Client *client = client_list;
    while (client) {
        send(client->fd, msg, sizeof(Message), 0);
        client = client->next;
    }
    
    pthread_mutex_unlock(&clients_mutex);
}
```

#### 2. 陷阱2：消息分片

```c
/* 错误示例：假设recv一次接收完整消息 */
Message msg;
recv(fd, &msg, sizeof(Message), 0);  /* 可能只收到部分数据 */

/* 正确做法：循环接收直到完整 */
int recv_full(int fd, void *buf, size_t len) {
    size_t total = 0;
    while (total < len) {
        int n = recv(fd, (char *)buf + total, len - total, 0);
        if (n <= 0) return n;  /* 错误或连接关闭 */
        total += n;
    }
    return total;
}

Message msg;
recv_full(fd, &msg.header, sizeof(MessageHeader));
recv_full(fd, msg.body, msg.header.length);
```

```c
Table *table_create(const char *name, Field *fields, int field_count) {

        fwrite(fields, sizeof(Field), field_count, fp);
        
        /* 写入行数（初始为0） */

        fclose(fp);
```

```c
int table_insert(Table *table, Row *row) {
    FILE *fp = fopen(table->data_file, "ab");
    if (!fp) return -1;
    
    /* 写入每列的值 */
    for (int i = 0; i < table->field_count; i++) {
        Field *field = &table->fields[i];
        
        switch (field->type) {
            case TYPE_INT:
                fwrite(row->values[i], sizeof(int), 1, fp);
                break;
            case TYPE_FLOAT:
                fwrite(row->values[i], sizeof(float), 1, fp);
                break;
            case TYPE_STRING:
                fwrite(row->values[i], sizeof(char), field->size, fp);
                break;
            case TYPE_BOOL:
                fwrite(row->values[i], sizeof(char), 1, fp);
                break;
        }
    }
    
    fclose(fp);
    
    /* 更新行数 */
    table->row_count++;
    
    /* 更新文件头中的行数 */
    fp = fopen(table->data_file, "r+b");
    if (fp) {
        fseek(fp, 50 + sizeof(int) + table->field_count * sizeof(Field), SEEK_SET);
        fwrite(&table->row_count, sizeof(int), 1, fp);
        fclose(fp);
    }
    
    return 0;
}
```

```c
/* 查询条件 */
typedef struct {
    int field_index;       /* 字段索引 */
    CompareOp op;          /* 比较操作 */
    void *value;           /* 比较值 */
} Condition;

typedef enum {
    OP_EQ,    /* = */
    OP_NE,    /* != */
    OP_GT,    /* > */
    OP_LT,    /* < */
    OP_GE,    /* >= */
    OP_LE     /* <= */
} CompareOp;

/* 查询 */
Row **table_select(Table *table, Condition *conditions, int cond_count, int *result_count) {
    FILE *fp = fopen(table->data_file, "rb");
    if (!fp) return NULL;
    
    /* 跳过表头 */
    fseek(fp, 50 + sizeof(int) + table->field_count * sizeof(Field) + sizeof(int), SEEK_SET);
    
    /* 分配结果数组 */
    Row **results = malloc(table->row_count * sizeof(Row *));
    *result_count = 0;
    
    /* 读取每一行 */
    for (int i = 0; i < table->row_count; i++) {
        Row *row = malloc(sizeof(Row));
        row->values = malloc(table->field_count * sizeof(void *));
        
        /* 读取每列的值 */
        for (int j = 0; j < table->field_count; j++) {
            Field *field = &table->fields[j];
            
            switch (field->type) {
                case TYPE_INT:
                    row->values[j] = malloc(sizeof(int));
                    fread(row->values[j], sizeof(int), 1, fp);
                    break;
                case TYPE_FLOAT:
                    row->values[j] = malloc(sizeof(float));
                    fread(row->values[j], sizeof(float), 1, fp);
                    break;
                case TYPE_STRING:
                    row->values[j] = malloc(field->size);
                    fread(row->values[j], sizeof(char), field->size, fp);
                    break;
                case TYPE_BOOL:
                    row->values[j] = malloc(sizeof(char));
                    fread(row->values[j], sizeof(char), 1, fp);
                    break;
            }
        }
        
        /* 检查条件 */
        int match = 1;
        for (int k = 0; k < cond_count && match; k++) {
            Condition *cond = &conditions[k];
            void *value = row->values[cond->field_index];
            
            /* 比较值 */
            switch (table->fields[cond->field_index].type) {
                case TYPE_INT:
                    match = compare_int(*(int *)value, *(int *)cond->value, cond->op);
                    break;
                case TYPE_STRING:
                    match = compare_string((char *)value, (char *)cond->value, cond->op);
                    break;
                /* ... 其他类型 */
            }
        }
        
        if (match) {
            results[*result_count] = row;
            (*result_count)++;
        } else {
            /* 释放不匹配的行 */
            for (int j = 0; j < table->field_count; j++) {
                free(row->values[j]);
            }
            free(row->values);
            free(row);
        }
    }
    
    fclose(fp);
    return results;
}
```

## 4. 项目4：贪吃蛇游戏

### 1. 需求分析

```
- 定时器
```

### 2. 核心实现

```c
#include <ncurses.h>
#include <unistd.h>
#include <stdlib.h>
#include <time.h>

#define WIDTH 40
#define HEIGHT 20
#define DELAY 100000  /* 微秒 */

/* 方向 */
typedef enum {
    UP, DOWN, LEFT, RIGHT
} Direction;

/* 蛇身节点 */
typedef struct SnakeNode {
    int x, y;
    struct SnakeNode *next;
} SnakeNode;

/* 游戏状态 */
typedef struct {
    SnakeNode *head;        /* 蛇头 */
    SnakeNode *tail;        /* 蛇尾 */
    Direction direction;    /* 方向 */
    int food_x, food_y;     /* 食物位置 */
    int score;              /* 分数 */
    int game_over;          /* 游戏结束标志 */
} Game;

/* 初始化蛇 */
void init_snake(Game *game) {
    /* 创建初始蛇身（3节） */
    SnakeNode *node1 = malloc(sizeof(SnakeNode));
    node1->x = WIDTH / 2;
    node1->y = HEIGHT / 2;
    node1->next = NULL;
    
    SnakeNode *node2 = malloc(sizeof(SnakeNode));
    node2->x = WIDTH / 2 - 1;
    node2->y = HEIGHT / 2;
    node2->next = node1;
    
    SnakeNode *node3 = malloc(sizeof(SnakeNode));
    node3->x = WIDTH / 2 - 2;
    node3->y = HEIGHT / 2;
    node3->next = node2;
    
    game->head = node1;
    game->tail = node3;
    game->direction = RIGHT;
    game->score = 0;
    game->game_over = 0;
}

/* 生成食物 */
void spawn_food(Game *game) {
    game->food_x = rand() % WIDTH;
    game->food_y = rand() % HEIGHT;
}

/* 移动蛇 */
void move_snake(Game *game) {
    /* 计算新头部位置 */
    int new_x = game->head->x;
    int new_y = game->head->y;
    
    switch (game->direction) {
        case UP:    new_y--; break;
        case DOWN:  new_y++; break;
        case LEFT:  new_x--; break;
        case RIGHT: new_x++; break;
    }
    
    /* 检查碰撞 */
    if (new_x < 0 || new_x >= WIDTH || new_y < 0 || new_y >= HEIGHT) {
        game->game_over = 1;
        return;
    }
    
    /* 检查是否撞到自己 */
    SnakeNode *node = game->tail;
    while (node) {
        if (node->x == new_x && node->y == new_y) {
            game->game_over = 1;
            return;
        }
        node = node->next;
    }
    
    /* 创建新头部 */
    SnakeNode *new_head = malloc(sizeof(SnakeNode));
    new_head->x = new_x;
    new_head->y = new_y;
    new_head->next = NULL;
    
    game->head->next = new_head;
    game->head = new_head;
    
    /* 检查是否吃到食物 */
    if (new_x == game->food_x && new_y == game->food_y) {
        game->score += 10;
        spawn_food(game);
    } else {
        /* 删除尾部 */
        SnakeNode *old_tail = game->tail;
        game->tail = game->tail->next;
        free(old_tail);
    }
}

/* 绘制游戏 */
void draw(Game *game) {
    clear();
    
    /* 绘制边框 */
    for (int x = 0; x < WIDTH + 2; x++) {
        mvprintw(0, x, "#");
        mvprintw(HEIGHT + 1, x, "#");
    }
    for (int y = 0; y < HEIGHT + 2; y++) {
        mvprintw(y, 0, "#");
        mvprintw(y, WIDTH + 1, "#");
    }
    
    /* 绘制蛇 */
    SnakeNode *node = game->tail;
    while (node) {
        mvprintw(node->y + 1, node->x + 1, "O");
        node = node->next;
    }
    
    /* 绘制食物 */
    mvprintw(game->food_y + 1, game->food_x + 1, "*");
    
    /* 显示分数 */
    mvprintw(0, WIDTH + 5, "Score: %d", game->score);
    
    refresh();
}

int main() {
    /* 初始化ncurses */
    initscr();
    noecho();
    curs_set(FALSE);
    keypad(stdscr, TRUE);
    nodelay(stdscr, TRUE);
    
    /* 初始化随机数 */
    srand(time(NULL));
    
    /* 初始化游戏 */
    Game game;
    init_snake(&game);
    spawn_food(&game);
    
    /* 游戏循环 */
    while (!game.game_over) {
        /* 处理输入 */
        int ch = getch();
        switch (ch) {
            case KEY_UP:
                if (game.direction != DOWN) game.direction = UP;
                break;
            case KEY_DOWN:
                if (game.direction != UP) game.direction = DOWN;
                break;
            case KEY_LEFT:
                if (game.direction != RIGHT) game.direction = LEFT;
                break;
            case KEY_RIGHT:
                if (game.direction != LEFT) game.direction = RIGHT;
                break;
            case 'q':
                game.game_over = 1;
                break;
        }
        
        /* 移动蛇 */
        move_snake(&game);
        
        /* 绘制 */
        draw(&game);
        
        /* 延迟 */
        usleep(DELAY);
    }
    
    /* 游戏结束 */
    mvprintw(HEIGHT / 2, WIDTH / 2 - 5, "Game Over!");
    mvprintw(HEIGHT / 2 + 1, WIDTH / 2 - 10, "Press any key to exit");
    refresh();
    
    nodelay(stdscr, FALSE);
    getch();
    
    /* 清理 */
    endwin();
    
    return 0;
}
```
