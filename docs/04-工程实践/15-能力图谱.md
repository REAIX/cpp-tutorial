**案例1：实现动态数组**

```c
/* 动态数组：自动扩容 */
typedef struct {
    int *data;        /* 数据指针 */
    size_t size;      /* 当前元素数量 */
    size_t capacity;  /* 容量 */
} DynamicArray;

/* 初始化 */
void array_init(DynamicArray *arr, size_t initial_capacity) {
    arr->data = malloc(initial_capacity * sizeof(int));
    arr->size = 0;
    arr->capacity = initial_capacity;
}

/* 添加元素（自动扩容） */
void array_push(DynamicArray *arr, int value) {
    if (arr->size >= arr->capacity) {
        /* 扩容：容量翻倍 */
        arr->capacity *= 2;
        arr->data = realloc(arr->data, arr->capacity * sizeof(int));
    }
    arr->data[arr->size++] = value;
}

/* 使用场景：读取未知数量的数据 */
DynamicArray arr;
array_init(&arr, 10);

int num;
while (scanf("%d", &num) == 1) {
    array_push(&arr, num);  /* 自动扩容，无需预先知道数量 */
}
```

**案例2：内存池**

```c
/* 内存池：预分配大块内存，减少malloc调用 */
typedef struct {
    char *pool;           /* 内存池 */
    size_t pool_size;     /* 总大小 */
    size_t used;          /* 已使用 */
} MemoryPool;

/* 初始化内存池 */
void pool_init(MemoryPool *mp, size_t size) {
    mp->pool = malloc(size);
    mp->pool_size = size;
    mp->used = 0;
}

/* 从池中分配 */
void *pool_alloc(MemoryPool *mp, size_t size) {
    if (mp->used + size > mp->pool_size) {
        return NULL;  /* 空间不足 */
    }
    void *ptr = mp->pool + mp->used;
    mp->used += size;
    return ptr;
}

/* 重置池（不释放内存，可重用） */
void pool_reset(MemoryPool *mp) {
    mp->used = 0;
}

/* 使用场景：游戏中的对象分配 */
MemoryPool game_pool;
pool_init(&game_pool, 1024 * 1024);  /* 1MB池 */

/* 游戏循环 */
while (game_running) {
    pool_reset(&game_pool);  /* 每帧重置，快速回收 */
    
    /* 分配临时对象 */
    Enemy *enemy = pool_alloc(&game_pool, sizeof(Enemy));
    Bullet *bullet = pool_alloc(&game_pool, sizeof(Bullet));
    
    /* ... 游戏逻辑 ... */
}
```

**案例1：实现链表**

```c
/* 链表节点 */
typedef struct Node {
    int data;
    struct Node *next;  /* 指针：指向下一个节点 */
} Node;

/* 在头部插入 */
Node *list_insert_head(Node *head, int value) {
    Node *new_node = malloc(sizeof(Node));
    new_node->data = value;
    new_node->next = head;  /* 新节点指向原头节点 */
    return new_node;        /* 返回新头节点 */
}

/* 遍历链表 */
void list_print(Node *head) {
    Node *current = head;  /* 指针遍历 */
    while (current != NULL) {
        printf("%d -> ", current->data);
        current = current->next;  /* 移动到下一个节点 */
    }
    printf("NULL\n");
}

/* 使用场景：动态数据集合 */
Node *list = NULL;
list = list_insert_head(list, 3);
list = list_insert_head(list, 2);
list = list_insert_head(list, 1);
/* 结果：1 -> 2 -> 3 -> NULL */
```

**案例2：函数指针实现回调**

```c
/* 比较函数类型 */
typedef int (*CompareFunc)(const void *, const void *);

/* 通用排序函数 */
void my_sort(void *arr, size_t count, size_t size, CompareFunc cmp) {
    char *base = (char *)arr;  /* 转换为字节指针 */
    char *temp = malloc(size);
    
    for (size_t i = 0; i < count - 1; i++) {
        for (size_t j = 0; j < count - 1 - i; j++) {
            void *a = base + j * size;      /* 指针运算 */
            void *b = base + (j + 1) * size;
            
            if (cmp(a, b) > 0) {  /* 使用回调函数比较 */
                memcpy(temp, a, size);
                memcpy(a, b, size);
                memcpy(b, temp, size);
            }
        }
    }
    free(temp);
}

/* 使用场景：不同类型的排序 */
int compare_int(const void *a, const void *b) {
    return *(int *)a - *(int *)b;
}

int compare_string(const void *a, const void *b) {
    return strcmp(*(const char **)a, *(const char **)b);
}

int numbers[] = {5, 2, 8, 1, 9};
my_sort(numbers, 5, sizeof(int), compare_int);

char *names[] = {"Charlie", "Alice", "Bob"};
my_sort(names, 3, sizeof(char *), compare_string);
```

**案例1：配置文件解析**

```c
/* 配置文件格式：
   key=value
   # 注释
*/

typedef struct {
    char key[50];
    char value[100];
} ConfigEntry;

int parse_config(const char *filename, ConfigEntry *entries, int max_entries) {
    FILE *fp = fopen(filename, "r");
    if (!fp) return -1;
    
    char line[256];
    int count = 0;
    
    while (fgets(line, sizeof(line), fp) && count < max_entries) {
        /* 跳过注释和空行 */
        if (line[0] == '#' || line[0] == '\n') continue;
        
        /* 解析 key=value */
        char *eq = strchr(line, '=');
        if (eq) {
            *eq = '\0';  /* 分割 */
            strcpy(entries[count].key, line);
            strcpy(entries[count].value, eq + 1);
            
            /* 去除换行符 */
            char *nl = strchr(entries[count].value, '\n');
            if (nl) *nl = '\0';
            
            count++;
        }
    }
    
    fclose(fp);
    return count;
}

/* 使用场景：读取程序配置 */
ConfigEntry config[100];
int count = parse_config("app.conf", config, 100);

for (int i = 0; i < count; i++) {
    if (strcmp(config[i].key, "port") == 0) {
        server_port = atoi(config[i].value);
    }
}
```

**案例2：二进制文件格式**

```c
/* 文件头 */
typedef struct {
    char magic[4];       /* 魔数：标识文件类型 */
    uint32_t version;    /* 版本号 */
    uint32_t count;      /* 数据条数 */
} FileHeader;

/* 数据条目 */
typedef struct {
    uint32_t id;
    char name[32];
    double value;
} DataEntry;

/* 写入二进制文件 */
int write_data(const char *filename, DataEntry *entries, int count) {
    FILE *fp = fopen(filename, "wb");  /* 二进制写模式 */
    if (!fp) return -1;
    
    /* 写入文件头 */
    FileHeader header = {"DATA", 1, count};
    fwrite(&header, sizeof(FileHeader), 1, fp);
    
    /* 写入数据 */
    fwrite(entries, sizeof(DataEntry), count, fp);
    
    fclose(fp);
    return 0;
}

/* 读取二进制文件 */
int read_data(const char *filename, DataEntry **entries, int *count) {
    FILE *fp = fopen(filename, "rb");  /* 二进制读模式 */
    if (!fp) return -1;
    
    /* 读取文件头 */
    FileHeader header;
    fread(&header, sizeof(FileHeader), 1, fp);
    
    /* 验证文件类型 */
    if (memcmp(header.magic, "DATA", 4) != 0) {
        fclose(fp);
        return -2;  /* 文件类型错误 */
    }
    
    /* 分配内存并读取数据 */
    *entries = malloc(header.count * sizeof(DataEntry));
    fread(*entries, sizeof(DataEntry), header.count, fp);
    *count = header.count;
    
    fclose(fp);
    return 0;
}

/* 使用场景：游戏存档、数据备份 */
DataEntry data[] = {
    {1, "Item1", 100.0},
    {2, "Item2", 200.0}
};
write_data("save.dat", data, 2);

DataEntry *loaded;
int count;
read_data("save.dat", &loaded, &count);
```

**案例1：线程池**

```c
/* 任务结构 */
typedef struct Task {
    void (*function)(void *arg);
    void *arg;
    struct Task *next;
} Task;

/* 线程池 */
typedef struct {
    Task *task_queue;          /* 任务队列 */
    pthread_t *threads;        /* 线程数组 */
    int thread_count;          /* 线程数量 */
    pthread_mutex_t lock;      /* 互斥锁 */
    pthread_cond_t notify;     /* 条件变量 */
    int shutdown;              /* 关闭标志 */
} ThreadPool;

/* 工作线程函数 */
void *thread_worker(void *arg) {
    ThreadPool *pool = (ThreadPool *)arg;
    
    while (1) {
        pthread_mutex_lock(&pool->lock);
        
        /* 等待任务 */
        while (pool->task_queue == NULL && !pool->shutdown) {
            pthread_cond_wait(&pool->notify, &pool->lock);
        }
        
        if (pool->shutdown) {
            pthread_mutex_unlock(&pool->lock);
            break;
        }
        
        /* 取出任务 */
        Task *task = pool->task_queue;
        pool->task_queue = task->next;
        
        pthread_mutex_unlock(&pool->lock);
        
        /* 执行任务 */
        task->function(task->arg);
        free(task);
    }
    
    return NULL;
}

/* 使用场景：Web服务器、并行计算 */
ThreadPool pool;
/* 初始化线程池... */

/* 提交任务 */
void process_request(void *arg) {
    /* 处理网络请求 */
}

Task *task = malloc(sizeof(Task));
task->function = process_request;
task->arg = client_data;

pthread_mutex_lock(&pool.lock);
task->next = pool.task_queue;
pool.task_queue = task;
pthread_cond_signal(&pool.notify);
pthread_mutex_unlock(&pool.lock);
```

```c
/* 键值对节点 */
typedef struct KVNode {
    char *key;
    char *value;
    struct KVNode *next;
} KVNode;

/* 哈希表 */
typedef struct {
    KVNode **buckets;  /* 桶数组 */
    int size;          /* 桶数量 */
} HashTable;

/* 创建哈希表 */
HashTable *hashtable_create(int size) {
    HashTable *ht = malloc(sizeof(HashTable));
    ht->buckets = calloc(size, sizeof(KVNode *));
    ht->size = size;
    return ht;
}

/* 哈希函数 */
unsigned int hash(const char *key, int size) {
    unsigned int hash = 0;
    while (*key) {
        hash = (hash << 5) + *key++;
    }
    return hash % size;
}

/* 插入键值对 */
void hashtable_set(HashTable *ht, const char *key, const char *value) {
    unsigned int index = hash(key, ht->size);
    
    /* 创建新节点 */
    KVNode *node = malloc(sizeof(KVNode));
    node->key = strdup(key);
    node->value = strdup(value);
    
    /* 插入到链表头部 */
    node->next = ht->buckets[index];
    ht->buckets[index] = node;
}

/* 查找值 */
char *hashtable_get(HashTable *ht, const char *key) {
    unsigned int index = hash(key, ht->size);
    KVNode *node = ht->buckets[index];
    
    while (node) {
        if (strcmp(node->key, key) == 0) {
            return node->value;
        }
        node = node->next;
    }
    
    return NULL;  /* 未找到 */
}

/* 使用场景：缓存系统、配置管理 */
HashTable *cache = hashtable_create(100);
hashtable_set(cache, "user:1001", "Alice");
hashtable_set(cache, "user:1002", "Bob");

printf("User: %s\n", hashtable_get(cache, "user:1001"));  /* Alice */
```

```c
/* CSV文件读写 */

typedef struct {
    int id;
    char name[50];
    int age;
    double score;
} Student;

/* 写入CSV */
int write_students_csv(const char *filename, Student *students, int count) {
    FILE *fp = fopen(filename, "w");
    if (!fp) return -1;
    
    /* 写入表头 */
    fprintf(fp, "id,name,age,score\n");
    
    /* 写入数据 */
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%d,%s,%d,%.2f\n",
                students[i].id,
                students[i].name,
                students[i].age,
                students[i].score);
    }
    
    fclose(fp);
    return 0;
}

/* 读取CSV */
int read_students_csv(const char *filename, Student **students, int *count) {
    FILE *fp = fopen(filename, "r");
    if (!fp) return -1;
    
    /* 跳过表头 */
    char line[256];
    fgets(line, sizeof(line), fp);
    
    /* 分配内存 */
    *students = malloc(100 * sizeof(Student));
    *count = 0;
    
    /* 读取数据 */
    while (fgets(line, sizeof(line), fp)) {
        Student *s = &(*students)[*count];
        
        /* 解析CSV行 */
        char *token = strtok(line, ",");
        s->id = atoi(token);
        
        token = strtok(NULL, ",");
        strcpy(s->name, token);
        
        token = strtok(NULL, ",");
        s->age = atoi(token);
        
        token = strtok(NULL, ",");
        s->score = atof(token);
        
        (*count)++;
    }
    
    fclose(fp);
    return 0;
}

/* 使用场景：数据导入导出 */
Student students[] = {
    {1, "Alice", 20, 85.5},
    {2, "Bob", 21, 90.0}
};
write_students_csv("students.csv", students, 2);

Student *loaded;
int count;
read_students_csv("students.csv", &loaded, &count);
```

### 3. 组合3：并发 + 网络 + I/O = 网络服务器

```
应用场景：
- Web服务器
- 聊天服务器
- 游戏服务器
```

```c
/* 简单的Echo服务器 */

#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>

#define PORT 8080
#define BUFFER_SIZE 1024

/* 客户端处理线程 */
void *handle_client(void *arg) {
    int client_fd = *(int *)arg;
    free(arg);
    
    char buffer[BUFFER_SIZE];
    ssize_t bytes_read;
    
    /* 读取并回显 */
    while ((bytes_read = read(client_fd, buffer, BUFFER_SIZE)) > 0) {
        write(client_fd, buffer, bytes_read);  /* Echo回客户端 */
    }
    
    close(client_fd);
    return NULL;
}

int main() {
    int server_fd, client_fd;
    struct sockaddr_in address;
    
    /* 创建socket */
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    
    /* 绑定地址 */
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);
    
    bind(server_fd, (struct sockaddr *)&address, sizeof(address));
    
    /* 监听 */
    listen(server_fd, 5);
    printf("Server listening on port %d\n", PORT);
    
    /* 接受连接 */
    while (1) {
        socklen_t addrlen = sizeof(address);
        client_fd = accept(server_fd, (struct sockaddr *)&address, &addrlen);
        
        printf("Client connected\n");
        
        /* 创建线程处理客户端 */
        pthread_t thread;
        int *fd = malloc(sizeof(int));
        *fd = client_fd;
        pthread_create(&thread, NULL, handle_client, fd);
        pthread_detach(thread);  /* 自动回收线程资源 */
    }
    
    return 0;
}
```
