## 🟢 1. 系统级编程的定义与范围

### 1.1 什么是系统级编程

系统级编程（Systems Programming）是指编写与计算机硬件和操作系统紧密交互的软件。它关注的是底层资源的管理与优化，而非业务逻辑的实现。

**系统级编程的核心范围：**

| 层次 | 内容 | 示例 |
|------|------|------|
| **硬件接口层** | CPU指令集、寄存器、缓存 | SIMD、缓存行对齐 |
| **操作系统层** | 系统调用、进程/线程、文件系统 | epoll、mmap、fork |
| **内存管理层** | 虚拟内存、页面管理、TLB | 大页内存、NUMA感知 |
| **I/O层** | 网络栈、磁盘I/O、设备驱动 | io_uring、DPDK、SPDK |
| **并发层** | 原子操作、锁、无锁数据结构 | CAS、RCU、Hazard Pointer |

### 1.2 系统级编程 vs 应用级编程

```cpp
// ============================================
// 应用级编程：关注业务逻辑，依赖框架和库
// ============================================

#include <vector>
#include <algorithm>
#include <iostream>

// 应用级：使用STL，不关心内存分配细节
void application_level() {
    std::vector<int> data;
    for (int i = 0; i < 1000000; ++i) {
        data.push_back(i);  // 自动管理内存，可能多次重新分配
    }
    std::sort(data.begin(), data.end());  // 使用标准库排序
    std::cout << "数据量: " << data.size() << std::endl;
}

// ============================================
// 系统级编程：关注底层细节，主动控制资源
// ============================================

#include <cstdlib>
#include <cstring>
#include <iostream>

// 系统级：手动管理内存，避免重新分配开销
void system_level() {
    // 一次性分配足够内存，避免多次realloc
    const size_t capacity = 1000000;
    int* data = static_cast<int*>(malloc(capacity * sizeof(int)));
    if (!data) {
        std::cerr << "内存分配失败" << std::endl;
        return;
    }

    size_t size = 0;
    for (int i = 0; i < static_cast<int>(capacity); ++i) {
        data[size++] = i;  // 直接写入，无额外开销
    }

    // 使用更高效的排序（可能利用CPU缓存特性）
    // qsort相比std::sort在某些场景下缓存更友好
    qsort(data, size, sizeof(int), [](const void* a, const void* b) -> int {
        return *static_cast<const int*>(a) - *static_cast<const int*>(b);
    });

    std::cout << "数据量: " << size << std::endl;
    free(data);  // 手动释放
}
```

## 🟢 2. 用户态与内核态

### 2.1 特权级与运行模式

现代CPU提供不同的特权级别（Protection Ring），操作系统利用这一机制隔离用户程序与内核代码：

```
┌─────────────────────────────────────────┐
│  Ring 3 - 用户态 (User Mode)            │
│  - 应用程序运行在此                      │
│  - 受限的指令集                          │
│  - 无法直接访问硬件                      │
│  - 虚拟地址空间隔离                      │
├─────────────────────────────────────────┤
│  Ring 0 - 内核态 (Kernel Mode)          │
│  - 操作系统内核运行在此                   │
│  - 完整的指令集                          │
│  - 可直接访问硬件                        │
│  - 可访问所有内存                        │
└─────────────────────────────────────────┘
```

### 2.2 用户态与内核态的切换

用户态到内核态的切换（上下文切换）是有开销的，理解这一点对性能优化至关重要：

```cpp
#include <iostream>
#include <chrono>
#include <unistd.h>
#include <sys/syscall.h>

// ============================================
// 测量系统调用的开销
// ============================================

// 纯用户态操作：无系统调用
void user_mode_operation(int iterations) {
    volatile int sum = 0;  // volatile防止优化
    for (int i = 0; i < iterations; ++i) {
        sum += i;
    }
}

// 系统调用操作：每次循环都进入内核态
void kernel_mode_operation(int iterations) {
    volatile int sum = 0;
    for (int i = 0; i < iterations; ++i) {
        // getpid() 是最简单的系统调用之一
        // 但仍然需要用户态→内核态→用户态的切换
        pid_t pid = getpid();
        sum += pid;
    }
}

void benchmark_syscall_overhead() {
    const int iterations = 1000000;

    // 测量纯用户态操作
    auto start = std::chrono::high_resolution_clock::now();
    user_mode_operation(iterations);
    auto end = std::chrono::high_resolution_clock::now();
    auto user_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();

    // 测量系统调用操作
    start = std::chrono::high_resolution_clock::now();
    kernel_mode_operation(iterations);
    end = std::chrono::high_resolution_clock::now();
    auto kernel_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();

    std::cout << "纯用户态: " << user_ns / iterations << " ns/次" << std::endl;
    std::cout << "系统调用:  " << kernel_ns / iterations << " ns/次" << std::endl;
    std::cout << "系统调用额外开销: " << (kernel_ns - user_ns) / iterations << " ns/次" << std::endl;
}
```

### 2.3 减少内核态切换的策略

```cpp
#include <iostream>
#include <vector>
#include <unistd.h>

// ============================================
// 策略1：批量操作减少系统调用次数
// ============================================

// 低效：每次写一个字节，触发一次系统调用
void write_one_by_one(int fd, const char* data, size_t len) {
    for (size_t i = 0; i < len; ++i) {
        write(fd, &data[i], 1);  // 每次一个字节，len次系统调用
    }
}

// 高效：一次性写入所有数据，只触发一次系统调用
void write_batch(int fd, const char* data, size_t len) {
    write(fd, data, len);  // 一次系统调用
}

// ============================================
// 策略2：使用缓冲区减少系统调用
// ============================================

class BufferedWriter {
private:
    int fd_;
    std::vector<char> buffer_;
    static constexpr size_t BUFFER_SIZE = 64 * 1024;  // 64KB缓冲区

public:
    explicit BufferedWriter(int fd) : fd_(fd), buffer_(BUFFER_SIZE) {}

    // 写入数据到缓冲区，缓冲区满时才触发系统调用
    void write(const char* data, size_t len) {
        size_t offset = 0;
        while (offset < len) {
            size_t available = BUFFER_SIZE - buffer_.size();
            size_t to_copy = std::min(available, len - offset);
            buffer_.insert(buffer_.end(), data + offset, data + offset + to_copy);
            offset += to_copy;

            if (buffer_.size() >= BUFFER_SIZE) {
                flush();  // 缓冲区满，触发系统调用
            }
        }
    }

    void flush() {
        if (!buffer_.empty()) {
            ::write(fd_, buffer_.data(), buffer_.size());
            buffer_.clear();
        }
    }

    ~BufferedWriter() {
        flush();  // 析构时确保数据写入
    }
};

// ============================================
// 策略3：使用vDSO避免系统调用
// ============================================

// vDSO (Virtual Dynamic Shared Object) 是内核映射到
// 每个进程地址空间的特殊页面，某些系统调用可以不切换到内核态

void vdso_example() {
    // clock_gettime 在现代Linux上通过vDSO实现
    // 不需要真正的系统调用！
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);

    // gettimeofday 也通过vDSO实现
    struct timeval tv;
    gettimeofday(&tv, nullptr);

    std::cout << "这些时间获取函数通过vDSO在用户态完成" << std::endl;
    std::cout << "无需切换到内核态，开销极低" << std::endl;
}
```

## 🟡 3. 系统调用机制

### 3.1 系统调用的执行流程

系统调用是用户程序请求内核服务的唯一合法途径。以下是完整的执行流程：

```
┌──────────┐    syscall指令     ┌──────────┐
│  用户态   │ ───────────────→  │  内核态   │
│          │                    │          │
│ 1.准备参数 │                   │ 4.保存上下文│
│ (寄存器)  │                    │          │
│          │                    │ 5.查找系统│
│ 2.执行    │                    │ 调用表    │
│ syscall  │                    │          │
│          │    sysret指令      │ 6.执行内核│
│ 9.返回   │ ←───────────────  │ 函数      │
│ 用户空间  │                    │          │
│          │                    │ 7.恢复上下文│
│ 10.检查  │                    │          │
│ 返回值   │                    │ 8.设置返回值│
└──────────┘                    └──────────┘
```

### 3.2 系统调用表与调用号

```cpp
#include <iostream>
#include <sys/syscall.h>
#include <unistd.h>

// ============================================
// 直接使用syscall()进行系统调用
// ============================================

void direct_syscall_example() {
    // 方法1：使用glibc封装的函数（推荐）
    pid_t pid1 = getpid();

    // 方法2：直接使用syscall()，通过系统调用号调用

    // 常见系统调用号（x86_64 Linux）
    std::cout << "\n常见系统调用号:" << std::endl;
    std::cout << "read:    " << SYS_read << std::endl;
    std::cout << "write:   " << SYS_write << std::endl;
    std::cout << "open:    " << SYS_open << std::endl;
    std::cout << "close:   " << SYS_close << std::endl;
    std::cout << "mmap:    " << SYS_mmap << std::endl;
    std::cout << "fork:    " << SYS_fork << std::endl;
```

### 3.3 系统调用的开销分析

```cpp
#include <iostream>
#include <chrono>
#include <unistd.h>
#include <fcntl.h>

// ============================================
// 不同系统调用的开销对比
// ============================================

void measure_syscall_costs() {
    const int iterations = 100000;

    // 1. getpid - 最轻量的系统调用
    {
        auto start = std::chrono::high_resolution_clock::now();
        for (int i = 0; i < iterations; ++i) {
            getpid();
        }
        auto end = std::chrono::high_resolution_clock::now();
        auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
        std::cout << "getpid:    " << ns / iterations << " ns/次" << std::endl;
    }

    // 2. clock_gettime - 通过vDSO可能更快
    {
        struct timespec ts;
        auto start = std::chrono::high_resolution_clock::now();
        for (int i = 0; i < iterations; ++i) {
            clock_gettime(CLOCK_MONOTONIC, &ts);
        }
        auto end = std::chrono::high_resolution_clock::now();
        auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
        std::cout << "clock_gettime: " << ns / iterations << " ns/次 (可能走vDSO)" << std::endl;
    }

    // 3. write - 涉及文件系统的系统调用
    {
        int fd = open("/dev/null", O_WRONLY);
        char buf[1] = {'x'};
        auto start = std::chrono::high_resolution_clock::now();
        for (int i = 0; i < iterations; ++i) {
            write(fd, buf, 1);
        }
        auto end = std::chrono::high_resolution_clock::now();
        auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
        std::cout << "write:     " << ns / iterations << " ns/次" << std::endl;
        close(fd);
    }

    // 4. stat - 涉及路径解析的系统调用
    {
        struct stat st;
        auto start = std::chrono::high_resolution_clock::now();
        for (int i = 0; i < iterations; ++i) {
            stat("/dev/null", &st);
        }
        auto end = std::chrono::high_resolution_clock::now();
        auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
        std::cout << "stat:      " << ns / iterations << " ns/次" << std::endl;
    }
}
```

### 3.4 避免系统调用的替代方案

```cpp
#include <iostream>
#include <sys/mman.h>
#include <unistd.h>

// ============================================
// 用mmap替代read/write系统调用
// ============================================

class MemoryMappedFile {
private:
    void* mapped_data_;
    size_t size_;

public:
    // 使用mmap映射文件，后续读写不需要系统调用
    bool open(const char* filename) {
        int fd = ::open(filename, O_RDWR);
        if (fd < 0) return false;

        // 获取文件大小
        struct stat st;
        if (fstat(fd, &st) < 0) {
            ::close(fd);
            return false;
        }
        size_ = st.st_size;

        // 映射到内存，之后读写就像访问内存一样
        mapped_data_ = mmap(nullptr, size_, PROT_READ | PROT_WRITE,
                           MAP_SHARED, fd, 0);
        ::close(fd);  // 映射后可以关闭fd

        if (mapped_data_ == MAP_FAILED) {
            return false;
        }
        return true;
    }

    // 直接通过指针访问，无需系统调用
    char* data() { return static_cast<char*>(mapped_data_); }
    size_t size() const { return size_; }

    ~MemoryMappedFile() {
        if (mapped_data_ != MAP_FAILED) {
            munmap(mapped_data_, size_);
        }
    }
};

// ============================================
// 用futex替代传统信号量（减少内核态切换）
// ============================================

class FastMutex {
private:
    std::atomic<int> state_{0};  // 0=未锁定, 1=锁定

public:
    void lock() {
        // 快速路径：尝试一次CAS，无系统调用
        int expected = 0;
        if (state_.compare_exchange_strong(expected, 1,
                                           std::memory_order_acquire)) {
            return;  // 成功获取锁，无需进入内核
        }

        // 慢速路径：竞争激烈时才调用futex
        while (state_.exchange(2, std::memory_order_acquire) != 0) {
            // futex系统调用，让内核挂起当前线程
            syscall(SYS_futex, &state_, FUTEX_WAIT_PRIVATE, 2, nullptr, nullptr, 0);
        }
    }

    void unlock() {
        if (state_.exchange(0, std::memory_order_release) == 2) {
            // 唤醒一个等待的线程
            syscall(SYS_futex, &state_, FUTEX_WAKE_PRIVATE, 1, nullptr, nullptr, 0);
        }
    }
};
```

## 🟡 4. POSIX标准

### 4.1 POSIX简介

POSIX（Portable Operating System Interface）是IEEE定义的一套操作系统接口标准，确保软件在不同Unix-like系统间的可移植性。

**POSIX的核心组成部分：**

| 标准 | 内容 | 示例API |
|------|------|---------|
| **POSIX.1** | 核心服务 | `open`, `read`, `write`, `fork`, `exec` |
| **POSIX.1b** | 实时扩展 | `mq_open`, `sem_init`, `clock_gettime` |
| **POSIX.1c** | 线程扩展 | `pthread_create`, `pthread_mutex_lock` |
| **POSIX.1d** | 实时信号 | `sigqueue`, `sigwaitinfo` |

### 4.2 POSIX API示例

```cpp
#include <iostream>
#include <pthread.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/mman.h>

// ============================================
// POSIX线程编程
// ============================================

struct ThreadData {
    int id;
    int result;
};

// 线程函数
void* thread_func(void* arg) {
    ThreadData* data = static_cast<ThreadData*>(arg);
    data->result = data->id * 100;
    std::cout << "线程 " << data->id << " 正在运行" << std::endl;
    return nullptr;
}

void posix_thread_example() {
    const int NUM_THREADS = 4;
    pthread_t threads[NUM_THREADS];
    ThreadData data[NUM_THREADS];

    // 创建线程
    for (int i = 0; i < NUM_THREADS; ++i) {
        data[i].id = i;
        data[i].result = 0;
        int ret = pthread_create(&threads[i], nullptr, thread_func, &data[i]);
        if (ret != 0) {
            std::cerr << "创建线程失败: " << ret << std::endl;
        }
    }

    // 等待线程完成
    for (int i = 0; i < NUM_THREADS; ++i) {
        pthread_join(threads[i], nullptr);
        std::cout << "线程 " << i << " 结果: " << data[i].result << std::endl;
    }
}

void posix_shared_memory_example() {
    const char* shm_name = "/my_shm";
    const size_t shm_size = 4096;

    // 创建共享内存对象
    int fd = shm_open(shm_name, O_CREAT | O_RDWR, 0666);
    if (fd < 0) {
        std::cerr << "创建共享内存失败" << std::endl;
        return;
    }

    // 设置大小
    ftruncate(fd, shm_size);

    // 映射共享内存
    void* ptr = mmap(nullptr, shm_size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) {
        std::cerr << "映射共享内存失败" << std::endl;
        close(fd);
        shm_unlink(shm_name);
        return;
    }

    // 写入数据
    int* shared_int = static_cast<int*>(ptr);
    *shared_int = 42;
    std::cout << "写入共享内存: " << *shared_int << std::endl;

    // 清理
    munmap(ptr, shm_size);
    close(fd);
    shm_unlink(shm_name);
}
```

```cpp
    ssize_t write(const void* buffer, size_t size) override {

        if (fd_ >= 0) {

            fd_ = -1;

                creation = OPEN_EXISTING;
                break;
            case Mode::WRITE:
                access = GENERIC_WRITE;
                creation = CREATE_ALWAYS;
                break;
            case Mode::READ_WRITE:
                access = GENERIC_READ | GENERIC_WRITE;
                creation = OPEN_ALWAYS;
                break;

            return static_cast<ssize_t>(bytesRead);
        }
        return -1;
    }

    ssize_t write(const void* buffer, size_t size) override {
        DWORD bytesWritten;
        if (WriteFile(handle_, buffer, static_cast<DWORD>(size), &bytesWritten, nullptr)) {
            return static_cast<ssize_t>(bytesWritten);
        }
        return -1;
    }

    void close() override {
        if (handle_ != INVALID_HANDLE_VALUE) {
            CloseHandle(handle_);
            handle_ = INVALID_HANDLE_VALUE;
        }
    }
```

### 5.2 跨平台内存映射

```cpp
#include <iostream>
#include <string>

// ============================================
// 跨平台内存映射抽象
// ============================================

class MemoryMap {
private:
    void* data_ = nullptr;
    size_t size_ = 0;

#if PLATFORM_LINUX || PLATFORM_MACOS
    int fd_ = -1;
#elif PLATFORM_WINDOWS
    HANDLE file_handle_ = INVALID_HANDLE_VALUE;
    HANDLE map_handle_ = nullptr;
#endif

public:
    bool map(const std::string& path, bool read_only = true) {
#if PLATFORM_LINUX || PLATFORM_MACOS
        // POSIX实现
        int flags = read_only ? O_RDONLY : O_RDWR;
        fd_ = ::open(path.c_str(), flags);
        if (fd_ < 0) return false;

        struct stat st;
        if (fstat(fd_, &st) < 0) { ::close(fd_); return false; }
        size_ = st.st_size;

        int prot = read_only ? PROT_READ : (PROT_READ | PROT_WRITE);
        data_ = mmap(nullptr, size_, prot, MAP_SHARED, fd_, 0);
        if (data_ == MAP_FAILED) {
            data_ = nullptr;
            ::close(fd_);
            return false;
        }
        return true;

#elif PLATFORM_WINDOWS
        // Windows实现
        DWORD access = read_only ? GENERIC_READ : (GENERIC_READ | GENERIC_WRITE);
        DWORD share = FILE_SHARE_READ;
        DWORD creation = OPEN_EXISTING;

        file_handle_ = CreateFileA(path.c_str(), access, share, nullptr,
                                   creation, FILE_ATTRIBUTE_NORMAL, nullptr);
        if (file_handle_ == INVALID_HANDLE_VALUE) return false;

        LARGE_INTEGER file_size;
        GetFileSizeEx(file_handle_, &file_size);
        size_ = static_cast<size_t>(file_size.QuadPart);

        DWORD protect = read_only ? PAGE_READONLY : PAGE_READWRITE;
        map_handle_ = CreateFileMappingA(file_handle_, nullptr, protect, 0, 0, nullptr);
        if (!map_handle_) {
            CloseHandle(file_handle_);
            return false;
        }

        DWORD map_access = read_only ? FILE_MAP_READ : FILE_MAP_ALL_ACCESS;
        data_ = MapViewOfFile(map_handle_, map_access, 0, 0, size_);
        if (!data_) {
            CloseHandle(map_handle_);
            CloseHandle(file_handle_);
            return false;
        }
        return true;
#endif
    }

    void* data() { return data_; }
    size_t size() const { return size_; }

    ~MemoryMap() {
#if PLATFORM_LINUX || PLATFORM_MACOS
        if (data_) munmap(data_, size_);
        if (fd_ >= 0) ::close(fd_);
#elif PLATFORM_WINDOWS
        if (data_) UnmapViewOfFile(data_);
        if (map_handle_) CloseHandle(map_handle_);
        if (file_handle_ != INVALID_HANDLE_VALUE) CloseHandle(file_handle_);
#endif
    }
};
```

```cpp
    #define ALIGNAS(N) __attribute__((aligned(N)))
    #define LIKELY(x)   __builtin_expect(!!(x), 1)
    #define UNLIKELY(x) __builtin_expect(!!(x), 0)
    #define FORCEINLINE __attribute__((always_inline)) inline
    #define NOINLINE    __attribute__((noinline))
#elif defined(_MSC_VER)
    #define ALIGNAS(N) __declspec(align(N))
    #define LIKELY(x)   (x)
    #define UNLIKELY(x) (x)
    #define FORCEINLINE __forceinline
    #define NOINLINE    __declspec(noinline)
#endif

// 跨平台原子操作
class SpinLock {
    std::atomic<bool> locked_{false};

public:
    void lock() {
        while (UNLIKELY(locked_.exchange(true, std::memory_order_acquire))) {
            // 自旋等待
#if PLATFORM_LINUX || PLATFORM_MACOS
            // 使用pause指令减少功耗
            __asm__ __volatile__("pause" ::: "memory");
#elif PLATFORM_WINDOWS
            _mm_pause();
#endif
        }
    }

    void unlock() {
        locked_.store(false, std::memory_order_release);
    }
};
```
