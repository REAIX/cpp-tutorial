**不剖析就优化 = 蒙着眼睛打靶**

- 精确定位哪行代码最耗时

- 用数据证明优化效果

```bash
# ============================================

    ./my_program

perf record -g ./my_program
# -g: 记录调用栈

# 4. 分析采样数据
perf report
# 交互式界面，显示热点函数

# 5. 实时热点监控
perf top

# 6. 源码级分析
perf annotate main
# 显示每行源码的采样比例

# 7. 追踪特定事件
perf record -e cache-misses ./my_program
perf report

# 8. 分析缓存未命中的具体位置
perf record -e mem-loads,mem-stores ./my_program

# 9. 统计IPC (Instructions Per Cycle)
perf stat -e cycles,instructions ./my_program
# IPC > 1.0 = 好 (指令级并行好)
# IPC < 0.5 = 差 (可能有缓存/分支问题)
```

```
├──────────────────────────────────────────────────────────┤
│                                                          │

│ │          main              │                          │
│ └────────────────────────────┘                          │
│                                                          │
│  解读:                                                   │
│  - funcA最宽 = 占用CPU时间最多 = 优化目标               │
│  - funcC和funcD是funcA的热点子函数                      │
│  - "平顶" = CPU密集型函数（优化重点）                    │
```

### 2.3 火焰图分析技巧

```cpp
#include <iostream>

// ============================================
// 火焰图分析实战
// ============================================

// 模拟一个有性能问题的程序
class DataProcessor {
public:
    void process(const std::vector<int>& data) {
        // 热点1：排序（占40%时间）
        std::vector<int> sorted = data;
        slow_sort(sorted);

        // 热点2：过滤（占30%时间）
        std::vector<int> filtered = slow_filter(sorted);

        // 热点3：聚合（占20%时间）
        int64_t sum = slow_aggregate(filtered);

        // 非热点：输出（占10%时间）
        output_result(sum);
    }

private:
    // 故意写慢的排序（冒泡排序）
    void slow_sort(std::vector<int>& data) {
        for (size_t i = 0; i < data.size(); ++i) {
            for (size_t j = i + 1; j < data.size(); ++j) {
                if (data[j] < data[i]) {
                    std::swap(data[i], data[j]);
                }
            }
        }
    }

    // 故意写慢的过滤（逐元素拷贝）
    std::vector<int> slow_filter(const std::vector<int>& data) {
        std::vector<int> result;
        for (size_t i = 0; i < data.size(); ++i) {
            if (data[i] > 0) {
                result.push_back(data[i]);  // 可能多次重新分配
            }
        }
        return result;
    }

    // 故意写慢的聚合（逐元素累加，无SIMD）
    int64_t slow_aggregate(const std::vector<int>& data) {
        int64_t sum = 0;
        for (int val : data) {
            sum += val * val;  // 逐元素，无向量化
        }
        return sum;
    }

    void output_result(int64_t sum) {
        std::cout << "结果: " << sum << std::endl;
    }
};

// ============================================
// 火焰图分析步骤
// ============================================

void flamegraph_analysis_guide() {
    std::cout << "=== 火焰图分析步骤 ===" << std::endl;

    std::cout << "\n1. 找最宽的'平顶'" << std::endl;
    std::cout << "   → 这是CPU时间最多的函数" << std::endl;
    std::cout << "   → 在上例中是slow_sort（40%时间）" << std::endl;

    std::cout << "\n2. 分析热点函数的调用路径" << std::endl;
    std::cout << "   → 从底部到顶部的调用链" << std::endl;
    std::cout << "   → 确认是哪个调用路径贡献了最多时间" << std::endl;

    std::cout << "\n3. 确定优化策略" << std::endl;
    std::cout << "   → slow_sort: 改用std::sort（O(n log n)）" << std::endl;
    std::cout << "   → slow_filter: 预分配内存 + 移除条件分支" << std::endl;
    std::cout << "   → slow_aggregate: 使用SIMD向量化" << std::endl;

    std::cout << "\n4. 优化后重新生成火焰图验证" << std::endl;
    std::cout << "   → 对比优化前后的火焰图" << std::endl;
    std::cout << "   → 使用差分火焰图可视化变化" << std::endl;

    std::cout << "\n5. 常见火焰图模式" << std::endl;
    std::cout << "   → 宽平顶: CPU密集型 → 算法/向量化优化" << std::endl;
    std::cout << "   → 多小塔: 函数调用开销大 → 内联/减少调用" << std::endl;
    std::cout << "   → 左右不对称: 数据依赖 → 数据布局优化" << std::endl;
    std::cout << "   → 大量[unknown]: 编译选项问题 → 加-g -fno-omit-frame-pointer" << std::endl;
}
```

```
│          ┌────────┤ 前端  ├──→ 前端瓶颈                   │
│          │        │       │    (指令解码/取指慢)           │
│          │        └───────┘                               │
│          │                                                │
│          │        ┌───────┐                               │
│   已退役  ├────────┤ 后端  ├──→ 后端瓶颈                   │
│   指令    │        │       │    (执行单元/缓存慢)          │
│          │        └───────┘                               │
│          │                                                │
│          │        ┌───────┐                               │
│          └────────┤ 猜测  ├──→ 分支预测瓶颈               │
│                   │       │    (误预测惩罚)               │
│                   └───────┘                               │
│                                                          │
│  后端瓶颈细分:                                           │
│  ┌──────────────────────────────────────────────┐       │
│  │  内存绑定 → L1/L2/L3 miss, TLB miss          │       │
│  │  核心绑定 → 执行端口饱和                      │       │
│  └──────────────────────────────────────────────┘       │
│                                                          │
│  优化方向:                                               │
│  前端瓶颈 → 减少指令数、代码布局优化                      │
│  后端-内存 → 缓存优化、数据布局、预取                     │
│  后端-核心 → 向量化、减少数据依赖                         │
│  猜测瓶颈 → 消除分支、likely/unlikely                     │
└──────────────────────────────────────────────────────────┘
```

```cpp
static void BM_VectorSum(benchmark::State& state) {
    std::vector<int> data(state.range(0));
    std::iota(data.begin(), data.end(), 0);

}

// 冒泡排序
static void BM_BubbleSort(benchmark::State& state) {
    std::vector<int> data(state.range(0));
    for (auto _ : state) {
        state.PauseTiming();
        // 每次迭代重新初始化数据
        std::iota(data.begin(), data.end(), 0);
        std::reverse(data.begin(), data.end());  // 最坏情况
        state.ResumeTiming();

        // 冒泡排序
        for (size_t i = 0; i < data.size(); ++i) {
            for (size_t j = i + 1; j < data.size(); ++j) {
                if (data[j] < data[i]) {
                    std::swap(data[i], data[j]);
                }
            }
        }
        benchmark::DoNotOptimize(data.data());
    }
}
BENCHMARK(BM_BubbleSort)->Arg(100)->Arg(500);

static void BM_ParallelSum(benchmark::State& state) {
    const size_t size = 10000000;
    std::vector<int> data(size, 1);
    const int threads = state.range(0);

    for (auto _ : state) {
        std::vector<int64_t> partial_sums(threads, 0);
        std::vector<std::thread> thread_list;

        size_t chunk = size / threads;
        for (int t = 0; t < threads; ++t) {
            thread_list.emplace_back([&data, &partial_sums, t, chunk, size]() {
                size_t start = t * chunk;
                size_t end = std::min(start + chunk, size);
                int64_t sum = 0;
                for (size_t i = start; i < end; ++i) {
                    sum += data[i];
                }
                partial_sums[t] = sum;
            });
        }

        for (auto& th : thread_list) th.join();

        int64_t total = 0;
        for (auto s : partial_sums) total += s;
        benchmark::DoNotOptimize(total);
    }
}
BENCHMARK(BM_ParallelSum)->Arg(1)->Arg(2)->Arg(4)->Arg(8);
```

```python
#!/usr/bin/env python3
"""
性能回归检测脚本
对比两次基准测试结果，检测性能退化
"""

import json
import sys

def compare_benchmarks(baseline_file, current_file, threshold_pct=5):
    """对比两次基准测试结果"""

    with open(baseline_file) as f:
        baseline = json.load(f)
    with open(current_file) as f:
        current = json.load(f)

    # 构建基准映射
    baseline_map = {}
    for bench in baseline['benchmarks']:
        name = bench['name']
        baseline_map[name] = bench['real_time']

    regressions = []
    improvements = []

    for bench in current['benchmarks']:
        name = bench['name']
        current_time = bench['real_time']

        if name not in baseline_map:
            print(f"新增基准: {name}")
            continue

        baseline_time = baseline_map[name]
        change_pct = (current_time - baseline_time) / baseline_time * 100

        if change_pct > threshold_pct:
            regressions.append((name, baseline_time, current_time, change_pct))
        elif change_pct < -threshold_pct:
            improvements.append((name, baseline_time, current_time, change_pct))

    # 输出结果
    if regressions:
        print("\n⚠️  性能退化:")
        for name, base, curr, pct in regressions:
            print(f"  {name}: {base:.1f} → {curr:.1f} ns (+{pct:.1f}%)")

    if improvements:
        print("\n✅ 性能改善:")
        for name, base, curr, pct in improvements:
            print(f"  {name}: {base:.1f} → {curr:.1f} ns ({pct:.1f}%)")

    if not regressions and not improvements:
        print("\n✅ 无显著性能变化")

    # 如果有退化，返回非零退出码
    if regressions:
        return 1
    return 0

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("用法: compare_benchmarks.py baseline.json current.json [threshold%]")
        sys.exit(1)

    threshold = float(sys.argv[3]) if len(sys.argv) > 3 else 5.0
    sys.exit(compare_benchmarks(sys.argv[1], sys.argv[2], threshold))
```

## 🔴 6. Amdahl定律与优化优先级

### 6.1 Amdahl定律

```
┌──────────────────────────────────────────────────────────┐
│              Amdahl定律                                   │
├──────────────────────────────────────────────────────────┤
│                                                          │
│  加速比 = 1 / ((1 - P) + P/S)                           │
│                                                          │
│  P = 可优化部分占总时间的比例                            │
│  S = 可优化部分的加速倍数                                │
│                                                          │
│  示例:                                                   │
│  - 程序80%时间在函数A（P=0.8）                          │
│  - 优化函数A使其快2倍（S=2）                             │
│  - 总加速比 = 1 / (0.2 + 0.8/2) = 1/0.6 = 1.67x       │
│                                                          │
│  关键洞察:                                               │
│  - 优化非热点代码几乎无用                                │
│  - 即使无限优化热点，加速比也有上限: 1/(1-P)            │
│  - P=0.8时，最大加速比 = 1/0.2 = 5x                     │
│  - P=0.5时，最大加速比 = 1/0.5 = 2x                     │
│                                                          │
│  结论: 先优化最耗时的部分！                              │
└──────────────────────────────────────────────────────────┘
```

### 6.2 优化优先级计算

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <iomanip>

// ============================================
// Amdahl定律计算器
// ============================================

struct Hotspot {
    std::string name;
    double time_fraction;  // 占总时间的比例
    double speedup;        // 可达到的加速倍数
};

double calculate_speedup(const std::vector<Hotspot>& hotspots) {
    // Amdahl定律的推广形式
    // Speedup = 1 / Σ(time_fraction_i / speedup_i)
    double denominator = 0.0;
    for (const auto& h : hotspots) {
        denominator += h.time_fraction / h.speedup;
    }
    return 1.0 / denominator;
}

void amdahl_analysis() {
    std::cout << "=== Amdahl定律分析 ===" << std::endl;

    // 示例：一个Web服务器的性能分析
    std::vector<Hotspot> hotspots = {
        {"数据库查询",  0.40, 1.0},  // 40%时间，尚未优化
        {"JSON序列化",  0.20, 1.0},  // 20%时间
        {"网络I/O",    0.15, 1.0},  // 15%时间
        {"业务逻辑",   0.15, 1.0},  // 15%时间
        {"其他",       0.10, 1.0},  // 10%时间
    };

    std::cout << "\n当前状态（无优化）:" << std::endl;
    std::cout << "总加速比: " << calculate_speedup(hotspots) << "x" << std::endl;

    // 场景1：只优化数据库查询（2倍加速）
    {
        auto hs = hotspots;
        hs[0].speedup = 2.0;
        double speedup = calculate_speedup(hs);
        std::cout << "\n场景1: 数据库查询优化2倍" << std::endl;
        std::cout << "总加速比: " << std::fixed << std::setprecision(2) << speedup << "x" << std::endl;
        // 结果: 1/(0.4/2 + 0.6) = 1/0.8 = 1.25x
    }

    // 场景2：只优化JSON序列化（5倍加速）
    {
        auto hs = hotspots;
        hs[1].speedup = 5.0;
        double speedup = calculate_speedup(hs);
        std::cout << "\n场景2: JSON序列化优化5倍" << std::endl;
        std::cout << "总加速比: " << speedup << "x" << std::endl;
        // 结果: 1/(0.4 + 0.2/5 + 0.4) = 1/0.84 = 1.19x
    }

    // 场景3：同时优化数据库和JSON
    {
        auto hs = hotspots;
        hs[0].speedup = 2.0;
        hs[1].speedup = 5.0;
        double speedup = calculate_speedup(hs);
        std::cout << "\n场景3: 数据库2倍 + JSON5倍" << std::endl;
        std::cout << "总加速比: " << speedup << "x" << std::endl;
    }

    // 场景4：优化非热点（业务逻辑10倍加速）
    {
        auto hs = hotspots;
        hs[3].speedup = 10.0;
        double speedup = calculate_speedup(hs);
        std::cout << "\n场景4: 业务逻辑优化10倍（非热点）" << std::endl;
        std::cout << "总加速比: " << speedup << "x" << std::endl;
        // 结果: 1/(0.4 + 0.2 + 0.15 + 0.15/10 + 0.1) = 1/0.865 = 1.16x
        // 优化非热点效果有限！
    }

    // 最大加速比
    std::cout << "\n=== 最大加速比分析 ===" << std::endl;
    for (const auto& h : hotspots) {
        double max_speedup = 1.0 / (1.0 - h.time_fraction);
        std::cout << h.name << " (占" << (h.time_fraction * 100) << "%): "
                  << "最大加速比 = " << max_speedup << "x" << std::endl;
    }
}

// ============================================
// 优化投资回报率分析
// ============================================

void optimization_roi() {
    std::cout << "\n=== 优化投资回报率 ===" << std::endl;

    struct Optimization {
        std::string name;
        double time_fraction;   // 占总时间比例
        double speedup;         // 优化后加速倍数
        double effort;          // 开发工作量（人天）
    };

    std::vector<Optimization> opts = {
        {"算法优化",     0.40, 3.0, 5},   // 高收益，中等工作量
        {"缓存优化",     0.30, 2.0, 3},   // 高收益，低工作量
        {"向量化",       0.15, 4.0, 10},  // 中收益，高工作量
        {"无分支优化",   0.10, 1.5, 2},   // 低收益，低工作量
        {"内存池",       0.05, 2.0, 3},   // 极低收益，低工作量
    };

    std::cout << "\n优化方案排序（按ROI从高到低）:" << std::endl;

    // 计算每个优化的总加速比和ROI
    std::vector<std::tuple<double, double, std::string>> results;
    for (const auto& opt : opts) {
        double total_speedup = 1.0 / ((1.0 - opt.time_fraction) + opt.time_fraction / opt.speedup);
        double improvement = total_speedup - 1.0;  // 提升百分比
        double roi = improvement / opt.effort;
        results.push_back({roi, improvement, opt.name});
    }

    std::sort(results.begin(), results.end(), std::greater<>());

    for (const auto& [roi, improvement, name] : results) {
        std::cout << "  " << name << ": 提升" << (improvement * 100) << "%, "
                  << "ROI=" << std::fixed << std::setprecision(3) << roi << "/人天" << std::endl;
    }
}
```

## 🟢 7. 小结

### 核心要点回顾

| 工具/方法 | 用途 | 适用场景 |
|----------|------|---------|
| **perf stat** | 统计性能事件 | 快速了解IPC、缓存命中率 |
| **perf record/report** | 采样分析热点 | 定位热点函数 |
| **VTune** | 深度微架构分析 | 分析缓存/分支/流水线瓶颈 |
| **火焰图** | 可视化调用栈 | 直观找到最宽的"平顶" |
| **google benchmark** | 微基准测试 | 精确测量优化效果 |
| **Amdahl定律** | 计算优化收益 | 确定优化优先级 |

### 性能优化黄金法则

```
1. 测量先行，不要猜测
   → 先profile，再优化

2. 优化热点，忽略冷点
   → Amdahl定律：优化占比最大的部分

3. 一次只改一个变量
   → 控制变量法，确保知道哪个改动有效

4. 验证优化效果
   → 每次优化后重新测量

5. 关注端到端性能
   → 微基准优化不等于整体优化

6. 防止性能回归
   → CI中集成性能测试

7. 记录优化过程
   → 方便回溯和知识积累
```

### 性能分析检查清单

```
□ 是否定义了明确的性能目标？
□ 是否建立了性能基线？
□ 是否使用了perf/VTune定位热点？
□ 是否生成了火焰图？
□ 是否根据Amdahl定律确定优化优先级？
□ 是否用微基准测试验证优化效果？
□ 是否测量了端到端性能？
□ 是否设置了性能回归检测？
```
