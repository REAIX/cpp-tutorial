### 1.2 高性能日志系统实现

```cpp
#include <iostream>
#include <string>
#include <string_view>
#include <atomic>
#include <thread>
#include <chrono>
#include <fstream>
#include <cstring>
#include <ctime>
#include <array>

// ============================================
// 日志级别
// ============================================

enum class LogLevel : uint8_t {
    TRACE = 0,
    DEBUG = 1,
    INFO = 2,
    WARN = 3,
    ERROR = 4,
    FATAL = 5
};

constexpr const char* LEVEL_NAMES[] = {
    "TRACE", "DEBUG", "INFO", "WARN", "ERROR", "FATAL"
};

// ============================================
// 日志条目（固定大小，避免动态分配）
// ============================================

struct LogEntry {
    static constexpr size_t MAX_MESSAGE_SIZE = 256;

    uint64_t timestamp_ns;  // 纳秒时间戳
    LogLevel level;
    uint32_t thread_id;
    char message[MAX_MESSAGE_SIZE];
    uint16_t message_len;

    void set(const char* msg, size_t len, LogLevel lvl, uint32_t tid) {
        auto now = std::chrono::high_resolution_clock::now();
        timestamp_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(
            now.time_since_epoch()).count();
        level = lvl;
        thread_id = tid;
        message_len = static_cast<uint16_t>(std::min(len, MAX_MESSAGE_SIZE - 1));
        std::memcpy(message, msg, message_len);
        message[message_len] = '\0';
    }
};

// ============================================
// 无锁SPSC环形缓冲区
// ============================================

template<typename T, size_t CAPACITY>
class SPSCRingBuffer {
    static_assert((CAPACITY & (CAPACITY - 1)) == 0, "CAPACITY必须是2的幂");

public:
    bool push(const T& item) {
        const size_t head = head_.load(std::memory_order_relaxed);
        const size_t next_head = (head + 1) & (CAPACITY - 1);

        if (next_head == tail_.load(std::memory_order_acquire)) {
            return false;  // 缓冲区满
        }

        buffer_[head] = item;
        head_.store(next_head, std::memory_order_release);
        return true;
    }

    bool pop(T& item) {
        const size_t tail = tail_.load(std::memory_order_relaxed);

        if (tail == head_.load(std::memory_order_acquire)) {
            return false;  // 缓冲区空
        }

        item = buffer_[tail];
        tail_.store((tail + 1) & (CAPACITY - 1), std::memory_order_release);
        return true;
    }

    // 批量弹出
    size_t popBatch(T* items, size_t max_count) {
        size_t count = 0;
        while (count < max_count && pop(items[count])) {
            count++;
        }
        return count;
    }

    bool empty() const {
        return head_.load(std::memory_order_acquire) ==
               tail_.load(std::memory_order_acquire);
    }

private:
    alignas(64) std::atomic<size_t> head_{0};
    alignas(64) std::atomic<size_t> tail_{0};
    T buffer_[CAPACITY];
};

// ============================================
// 多生产者无锁日志缓冲
// ============================================

template<size_t CAPACITY = 65536>
class MPSCLogBuffer {
public:
    bool push(const LogEntry& entry) {
        // 使用CAS实现多生产者安全
        size_t pos = write_pos_.fetch_add(1, std::memory_order_acq_rel);
        size_t idx = pos & (CAPACITY - 1);

        // 等待该位置可写（简单的自旋等待）
        size_t expected_seq = pos;
        while (sequence_[idx].load(std::memory_order_acquire) != expected_seq) {
            std::this_thread::yield();
        }

        buffer_[idx] = entry;
        sequence_[idx].store(pos + 1, std::memory_order_release);
        return true;
    }

    size_t popBatch(LogEntry* entries, size_t max_count) {
        size_t count = 0;
        size_t read_pos = read_pos_.load(std::memory_order_relaxed);

        while (count < max_count) {
            size_t idx = read_pos & (CAPACITY - 1);
            size_t seq = sequence_[idx].load(std::memory_order_acquire);

            if (seq != read_pos + 1) {
                break;  // 没有数据或数据未就绪
            }

            entries[count] = buffer_[idx];
            sequence_[idx].store(read_pos + CAPACITY, std::memory_order_release);
            count++;
            read_pos++;
        }

        if (count > 0) {
            read_pos_.store(read_pos, std::memory_order_release);
        }
        return count;
    }

private:
    LogEntry buffer_[CAPACITY];
    std::atomic<size_t> sequence_[CAPACITY];
    alignas(64) std::atomic<size_t> write_pos_{0};
    alignas(64) std::atomic<size_t> read_pos_{0};
};

// ============================================
// 高性能日志器
// ============================================

class HighPerfLogger {
public:
    static HighPerfLogger& instance() {
        static HighPerfLogger logger;
        return logger;
    }

    void init(const std::string& filename, LogLevel min_level = LogLevel::INFO) {
        min_level_ = min_level;
        output_file_.open(filename, std::ios::app | std::ios::out);
        running_.store(true);

        // 启动写入线程
        writer_thread_ = std::thread(&HighPerfLogger::writerLoop, this);
    }

    void shutdown() {
        running_.store(false);
        if (writer_thread_.joinable()) {
            writer_thread_.join();
        }
        // 刷新剩余日志
        flushBuffer();
        output_file_.close();
    }

    // 写入日志（业务线程调用，极低延迟）
    void log(LogLevel level, const char* fmt, ...) {
        if (level < min_level_) return;

        // 格式化消息
        char buf[LogEntry::MAX_MESSAGE_SIZE];
        va_list args;
        va_start(args, fmt);
        int len = vsnprintf(buf, sizeof(buf), fmt, args);
        va_end(args);

        if (len < 0) len = 0;

        // 创建日志条目（栈上分配，零堆分配）
        LogEntry entry;
        entry.set(buf, static_cast<size_t>(len), level,
                 static_cast<uint32_t>(std::hash<std::thread::id>{}(
                     std::this_thread::get_id())));

        // 推入无锁缓冲区
        buffer_.push(entry);
    }

    // 便捷宏
    #define LOG_TRACE(fmt, ...) HighPerfLogger::instance().log(LogLevel::TRACE, fmt, ##__VA_ARGS__)
    #define LOG_DEBUG(fmt, ...) HighPerfLogger::instance().log(LogLevel::DEBUG, fmt, ##__VA_ARGS__)
    #define LOG_INFO(fmt, ...)  HighPerfLogger::instance().log(LogLevel::INFO, fmt, ##__VA_ARGS__)
    #define LOG_WARN(fmt, ...)  HighPerfLogger::instance().log(LogLevel::WARN, fmt, ##__VA_ARGS__)
    #define LOG_ERROR(fmt, ...) HighPerfLogger::instance().log(LogLevel::ERROR, fmt, ##__VA_ARGS__)
    #define LOG_FATAL(fmt, ...) HighPerfLogger::instance().log(LogLevel::FATAL, fmt, ##__VA_ARGS__)

private:
    HighPerfLogger() = default;

    void writerLoop() {
        constexpr size_t BATCH_SIZE = 256;
        LogEntry batch[BATCH_SIZE];

        while (running_.load(std::memory_order_relaxed)) {
            size_t count = buffer_.popBatch(batch, BATCH_SIZE);

            if (count > 0) {
                // 批量写入文件
                for (size_t i = 0; i < count; ++i) {
                    formatEntry(batch[i]);
                }
                output_file_.flush();
            } else {
                // 无数据，短暂休眠
                std::this_thread::sleep_for(std::chrono::microseconds(100));
            }
        }
    }

    void formatEntry(const LogEntry& entry) {
        // 格式化时间
        auto seconds = entry.timestamp_ns / 1000000000;
        auto nanos = entry.timestamp_ns % 1000000000;
        std::time_t t = static_cast<std::time_t>(seconds);
        char time_buf[32];
        std::strftime(time_buf, sizeof(time_buf), "%Y-%m-%d %H:%M:%S",
                     std::localtime(&t));

        output_file_ << time_buf << "." << std::setfill('0') << std::setw(9) << nanos
                     << " [" << LEVEL_NAMES[static_cast<int>(entry.level)] << "]"
                     << " [T" << entry.thread_id << "] "
                     << entry.message << "\n";
    }

    void flushBuffer() {
        constexpr size_t BATCH_SIZE = 256;
        LogEntry batch[BATCH_SIZE];
        while (true) {
            size_t count = buffer_.popBatch(batch, BATCH_SIZE);
            if (count == 0) break;
            for (size_t i = 0; i < count; ++i) {
                formatEntry(batch[i]);
            }
        }
    }

    MPSCLogBuffer<65536> buffer_;
    std::ofstream output_file_;
    std::thread writer_thread_;
    std::atomic<bool> running_{false};
    LogLevel min_level_{LogLevel::INFO};
};
```

## 🟡 2. 零分配消息队列

### 2.1 零分配设计原则

```
┌──────────────────────────────────────────────────────────┐
│              零分配消息队列设计                            │
├──────────────────────────────────────────────────────────┤
│                                                          │
│  传统消息队列:                                           │
│  发送: new Message() → queue.push() → 接收: delete      │
│  每条消息2次堆分配/释放 → GC压力/碎片                    │
│                                                          │
│  零分配消息队列:                                         │
│  预分配消息池 → 环形缓冲区 → 消息回收到池                │
│  运行时零堆分配 → 无GC压力/无碎片                        │
│                                                          │
│  关键技术:                                               │
│  1. 预分配固定大小消息池                                 │
│  2. 环形缓冲区传递消息引用                               │
│  3. 引用计数或所有权模型管理消息生命周期                  │
│  4. 缓存行对齐避免false sharing                          │
└──────────────────────────────────────────────────────────┘
```

### 2.2 零分配消息队列实现

```cpp
#include <iostream>
#include <atomic>
#include <array>
#include <functional>
#include <chrono>

// ============================================
// 零分配消息队列
// ============================================

template<typename T, size_t CAPACITY = 1048576>  // 默认100万消息
class ZeroAllocMessageQueue {
    static_assert((CAPACITY & (CAPACITY - 1)) == 0, "CAPACITY必须是2的幂");

private:
    // 消息槽（缓存行对齐避免false sharing）
    struct alignas(64) Slot {
        std::atomic<uint64_t> sequence{0};
        T data{};

        Slot() {
            sequence.store(0, std::memory_order_relaxed);
        }
    };

    std::array<Slot, CAPACITY> buffer_;
    alignas(64) std::atomic<uint64_t> enqueue_pos_{0};
    alignas(64) std::atomic<uint64_t> dequeue_pos_{0};

public:
    // 入队（多生产者安全）
    bool enqueue(const T& item) {
        uint64_t pos = enqueue_pos_.load(std::memory_order_relaxed);

        while (true) {
            Slot& slot = buffer_[pos & (CAPACITY - 1)];
            uint64_t seq = slot.sequence.load(std::memory_order_acquire);
            intptr_t diff = static_cast<intptr_t>(seq) - static_cast<intptr_t>(pos);

            if (diff == 0) {
                // 槽位可用，尝试占用
                if (enqueue_pos_.compare_exchange_weak(
                        pos, pos + 1,
                        std::memory_order_relaxed)) {
                    slot.data = item;
                    slot.sequence.store(pos + 1, std::memory_order_release);
                    return true;
                }
            } else if (diff < 0) {
                // 队列满
                return false;
            } else {
                // 位置已被其他生产者占用，重新获取
                pos = enqueue_pos_.load(std::memory_order_relaxed);
            }
        }
    }

    // 出队（单消费者）
    bool dequeue(T& item) {
        uint64_t pos = dequeue_pos_.load(std::memory_order_relaxed);
        Slot& slot = buffer_[pos & (CAPACITY - 1)];
        uint64_t seq = slot.sequence.load(std::memory_order_acquire);

        intptr_t diff = static_cast<intptr_t>(seq) - static_cast<intptr_t>(pos + 1);

        if (diff == 0) {
            // 数据就绪
            dequeue_pos_.store(pos + 1, std::memory_order_relaxed);
            item = slot.data;
            slot.sequence.store(pos + CAPACITY, std::memory_order_release);
            return true;
        }

        return false;  // 队列空
    }

    // 批量出队
    size_t dequeueBatch(T* items, size_t max_count) {
        size_t count = 0;
        while (count < max_count && dequeue(items[count])) {
            count++;
        }
        return count;
    }

    size_t capacity() const { return CAPACITY; }
};

// ============================================
// 消息类型定义
// ============================================

struct MarketData {
    uint32_t symbol_id;
    int64_t price;
    int32_t volume;
    uint64_t timestamp_ns;
    uint8_t side;  // 0=买, 1=卖
};

// ============================================
// 使用示例：金融行情分发
// ============================================

void market_data_demo() {
    ZeroAllocMessageQueue<MarketData, 1048576> queue;

    // 生产者线程：接收行情数据
    std::thread producer([&queue]() {
        for (uint32_t i = 0; i < 1000000; ++i) {
            MarketData data;
            data.symbol_id = i % 1000;
            data.price = 10000 + i;
            data.volume = 100 + (i % 50);
            data.timestamp_ns = std::chrono::high_resolution_clock::now()
                .time_since_epoch().count();
            data.side = i % 2;

            while (!queue.enqueue(data)) {
                std::this_thread::yield();
            }
        }
    });

    // 消费者线程：处理行情数据
    std::thread consumer([&queue]() {
        MarketData data;
        size_t count = 0;
        auto start = std::chrono::high_resolution_clock::now();

        while (count < 1000000) {
            if (queue.dequeue(data)) {
                count++;
                // 处理行情数据...
            } else {
                std::this_thread::yield();
            }
        }

        auto end = std::chrono::high_resolution_clock::now();
        auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(end - start).count();
        std::cout << "处理100万条行情: " << ms << " ms ("
                  << 1000000.0 / ms * 1000 << " 条/秒)" << std::endl;
    });

    producer.join();
    consumer.join();
}
```

## 🔴 3. 实时系统编程

### 3.1 实时系统基础

```
┌──────────────────────────────────────────────────────────┐
│              实时系统分类                                  │
├──────────────────────────────────────────────────────────┤
│                                                          │
│  硬实时 (Hard Real-time)                                 │
│  └→ 错过截止时间 = 系统失败（航空航天、医疗设备）        │
│                                                          │
│  固实时 (Firm Real-time)                                 │
│  └→ 错过截止时间 = 结果无用但系统不崩溃                  │
│                                                          │
│  软实时 (Soft Real-time)                                 │
│  └→ 错过截止时间 = 性能下降但系统继续运行               │
│                                                          │
│  实时编程关键:                                           │
│  1. 确定性：最坏情况执行时间(WCET)可预测                 │
│  2. 低延迟：中断响应时间极短                              │
│  3. 无优先级反转：优先级继承或优先级天花板协议            │
│  4. 无内存分配：避免GC和malloc的不确定性                  │
│  5. 无阻塞I/O：所有I/O操作异步                           │
└──────────────────────────────────────────────────────────┘
```

### 3.2 实时编程实践

```cpp
#include <iostream>
#include <atomic>
#include <thread>
#include <chrono>
#include <vector>

// ============================================
// 实时线程配置
// ============================================

class RealtimeThread {
public:
    // 设置线程为实时优先级
    static bool setRealtimePriority(std::thread& t, int priority = 80) {
#ifdef __linux__
        sched_param param;
        param.sched_priority = priority;
        int policy = SCHED_FIFO;  // 先进先出实时调度

        if (pthread_setschedparam(t.native_handle(), policy, &param) != 0) {
            std::cerr << "设置实时优先级失败（需要root权限）" << std::endl;
            return false;
        }
        return true;
#else
        std::cerr << "非Linux系统，实时调度不可用" << std::endl;
        return false;
#endif
    }

    // 锁定内存（避免页面错误导致的延迟）
    static bool lockMemory() {
#ifdef __linux__
        if (mlockall(MCL_CURRENT | MCL_FUTURE) != 0) {
            std::cerr << "锁定内存失败（需要root权限）" << std::endl;
            return false;
        }
        return true;
#else
        return false;
#endif
    }

    // 设置CPU亲和性（绑定到特定CPU核心）
    static bool setCpuAffinity(std::thread& t, int cpu) {
#ifdef __linux__
        cpu_set_t cpuset;
        CPU_ZERO(&cpuset);
        CPU_SET(cpu, &cpuset);

        if (pthread_setaffinity_np(t.native_handle(), sizeof(cpuset), &cpuset) != 0) {
            std::cerr << "设置CPU亲和性失败" << std::endl;
            return false;
        }
        return true;
#else
        return false;
#endif
    }
};

// ============================================
// 实时安全的数据结构
// ============================================

// 实时安全的环形缓冲区（无锁、无分配、固定大小）
template<typename T, size_t N>
class RealtimeRingBuffer {
    static_assert((N & (N - 1)) == 0, "N必须是2的幂");

public:
    bool push(const T& item) {
        const size_t head = head_.load(std::memory_order_relaxed);
        const size_t next = (head + 1) & (N - 1);

        if (next == tail_.load(std::memory_order_acquire)) {
            return false;  // 满
        }

        buffer_[head] = item;
        head_.store(next, std::memory_order_release);
        return true;
    }

    bool pop(T& item) {
        const size_t tail = tail_.load(std::memory_order_relaxed);

        if (tail == head_.load(std::memory_order_acquire)) {
            return false;  // 空
        }

        item = buffer_[tail];
        tail_.store((tail + 1) & (N - 1), std::memory_order_release);
        return true;
    }

private:
    alignas(64) std::atomic<size_t> head_{0};
    alignas(64) std::atomic<size_t> tail_{0};
    T buffer_[N];
};

// ============================================
// 实时任务调度器
// ============================================

class RealtimeTaskScheduler {
public:
    using TaskFunc = std::function<void()>;

    struct Task {
        TaskFunc func;
        uint64_t period_ns;     // 周期（纳秒）
        uint64_t next_run_ns;   // 下次运行时间
        int priority;
    };

    void addTask(TaskFunc func, uint64_t period_ns, int priority = 0) {
        Task task;
        task.func = func;
        task.period_ns = period_ns;
        task.next_run_ns = getCurrentTimeNs();
        task.priority = priority;
        tasks_.push_back(task);
    }

    void run() {
        while (running_.load()) {
            uint64_t now = getCurrentTimeNs();

            for (auto& task : tasks_) {
                if (now >= task.next_run_ns) {
                    task.func();
                    task.next_run_ns += task.period_ns;

                    // 如果严重超时，重置下次运行时间
                    if (task.next_run_ns < now) {
                        task.next_run_ns = now + task.period_ns;
                    }
                }
            }

            // 计算最近的任务时间
            uint64_t next_time = UINT64_MAX;
            for (const auto& task : tasks_) {
                if (task.next_run_ns < next_time) {
                    next_time = task.next_run_ns;
                }
            }

            // 休眠到下次任务时间
            int64_t sleep_ns = static_cast<int64_t>(next_time - getCurrentTimeNs());
            if (sleep_ns > 0) {
                std::this_thread::sleep_for(std::chrono::nanoseconds(sleep_ns));
            }
        }
    }

    void stop() { running_.store(false); }

private:
    static uint64_t getCurrentTimeNs() {
        return std::chrono::duration_cast<std::chrono::nanoseconds>(
            std::chrono::high_resolution_clock::now().time_since_epoch()).count();
    }

    std::vector<Task> tasks_;
    std::atomic<bool> running_{true};
};

// ============================================
// 实时系统示例
// ============================================

void realtime_demo() {
    RealtimeTaskScheduler scheduler;

    // 任务1：每1ms执行一次（1000Hz）
    std::atomic<int> fast_count{0};
    scheduler.addTask([&fast_count]() {
        fast_count.fetch_add(1, std::memory_order_relaxed);
    }, 1000000);  // 1ms = 1000000ns

    // 任务2：每10ms执行一次（100Hz）
    std::atomic<int> slow_count{0};
    scheduler.addTask([&slow_count]() {
        slow_count.fetch_add(1, std::memory_order_relaxed);
    }, 10000000);  // 10ms = 10000000ns

    // 运行1秒
    std::thread scheduler_thread([&scheduler]() {
        scheduler.run();
    });

    std::this_thread::sleep_for(std::chrono::seconds(1));
    scheduler.stop();
    scheduler_thread.join();

    std::cout << "快速任务执行: " << fast_count.load() << " 次 (预期~1000)" << std::endl;
    std::cout << "慢速任务执行: " << slow_count.load() << " 次 (预期~100)" << std::endl;
}
```

### 4.1 DPDK原理

```
┌──────────────────────────────────────────────────────────┐
│              传统网络栈 vs DPDK                            │
├──────────────────────────────────────────────────────────┤
│                                                          │
│  传统网络栈:                                             │
│  NIC → 内核协议栈 → 系统调用 → 用户空间                  │
│  └→ 中断开销 + 上下文切换 + 数据拷贝                     │
│                                                          │
│  DPDK用户态网络栈:                                       │
│  NIC → 用户态驱动 → 用户态协议栈 → 应用                  │
│  └→ 轮询模式 + 零拷贝 + 无内核参与                      │
│                                                          │
│  DPDK关键技术:                                           │
│  1. UIO: 用户态I/O，绕过内核                             │
│  2. PMD: 轮询模式驱动，无中断开销                        │
│  3. HugePages: 减少TLB miss                              │
│  4. NUMA: 本地内存分配                                   │
│  5. RSS: 网卡多队列 + CPU亲和性                          │
│  6. 零拷贝: 数据不经过内核                               │
```

```cpp
// 数据包结构
struct Packet {
    static constexpr size_t MAX_SIZE = 2048;

    uint8_t data[MAX_SIZE];
    uint16_t length;
    uint32_t port_id;
    uint64_t timestamp_ns;

    void clear() {
        length = 0;
        port_id = 0;
        timestamp_ns = 0;
    }
};

// 数据包缓冲区（模拟rte_mbuf）
class PacketBufferPool {
public:
    explicit PacketBufferPool(size_t pool_size = 65536)
        : pool_(pool_size), free_stack_(pool_size) {
        for (size_t i = 0; i < pool_size; ++i) {
            free_stack_[i] = &pool_[i];
        }
        free_top_.store(pool_size, std::memory_order_relaxed);
    }

    Packet* allocate() {
        size_t top = free_top_.fetch_sub(1, std::memory_order_acq_rel);
        if (top == 0 || top > pool_.size()) {
            free_top_.fetch_add(1, std::memory_order_relaxed);
            return nullptr;  // 池耗尽
        }
        return free_stack_[top - 1];
    }

    void deallocate(Packet* pkt) {
        pkt->clear();
        size_t top = free_top_.fetch_add(1, std::memory_order_acq_rel);
        free_stack_[top] = pkt;
    }

private:
    std::vector<Packet> pool_;
    std::vector<Packet*> free_stack_;
    std::atomic<size_t> free_top_;
};

// 数据包批处理（模拟rte_mbuf数组）
struct PacketBatch {
    static constexpr size_t MAX_BURST = 32;
    Packet* packets[MAX_BURST];
    size_t count = 0;

    void clear() { count = 0; }
    void add(Packet* pkt) {
        if (count < MAX_BURST) {
            packets[count++] = pkt;
        }
    }
};

class PacketProcessor {
public:
    // 处理一批数据包
    void processBatch(PacketBatch& batch, PacketBufferPool& pool) {
        for (size_t i = 0; i < batch.count; ++i) {
            Packet* pkt = batch.packets[i];
            processPacket(pkt, pool);
        }
    }

private:
    void processPacket(Packet* pkt, PacketBufferPool& pool) {
        // 简化的以太网头部检查
        uint8_t* data = pkt->data;

        // 检查以太网类型（偏移12-13字节）
        uint16_t ether_type = (data[12] << 8) | data[13];

        if (ether_type == 0x0800) {
            // IPv4包
            processIPv4(pkt, pool);
        } else {
            // 不支持的协议，丢弃
            pool.deallocate(pkt);
        }
    }

    void processIPv4(Packet* pkt, PacketBufferPool& pool) {
        IPv4Header* ip = reinterpret_cast<IPv4Header*>(pkt->data + 14);

        if (ip->protocol == 6) {
            // TCP包
            processTCP(pkt, pool);
        } else if (ip->protocol == 17) {
            // UDP包
            processUDP(pkt, pool);
        } else {
            pool.deallocate(pkt);
        }
    }

    void processTCP(Packet* pkt, PacketBufferPool& pool) {
        // TCP处理逻辑...
        // 简化：直接统计
        tcp_count_.fetch_add(1, std::memory_order_relaxed);
        pool.deallocate(pkt);
    }

    void processUDP(Packet* pkt, PacketBufferPool& pool) {
        // UDP处理逻辑...
        udp_count_.fetch_add(1, std::memory_order_relaxed);
        pool.deallocate(pkt);
    }

    std::atomic<uint64_t> tcp_count_{0};
    std::atomic<uint64_t> udp_count_{0};
};

// ============================================
// DPDK风格的主循环
// ============================================

void dpdk_style_main_loop() {
    PacketBufferPool pool;
    PacketProcessor processor;

    // 模拟收包
    auto receivePackets = [&pool](PacketBatch& batch, size_t port) -> size_t {
        size_t count = 0;
        for (size_t i = 0; i < PacketBatch::MAX_BURST; ++i) {
            Packet* pkt = pool.allocate();
            if (!pkt) break;

            // 模拟填充数据
            pkt->length = 64 + (rand() % 1400);
            pkt->port_id = port;
            pkt->timestamp_ns = std::chrono::high_resolution_clock::now()
                .time_since_epoch().count();

            // 模拟以太网头部
            memset(pkt->data, 0, 14);
            pkt->data[12] = 0x08;
            pkt->data[13] = 0x00;  // IPv4

            batch.add(pkt);
            count++;
        }
        return count;
    };

    // 主循环（轮询模式）
    std::cout << "DPDK风格主循环启动..." << std::endl;

    uint64_t total_packets = 0;
    auto start = std::chrono::high_resolution_clock::now();

    for (int iter = 0; iter < 100000; ++iter) {
        PacketBatch rx_batch;
        rx_batch.clear();

        // 从网卡接收一批包
        size_t rx_count = receivePackets(rx_batch, 0);

        if (rx_count > 0) {
            // 处理这批包
            processor.processBatch(rx_batch, pool);
            total_packets += rx_count;
        }
    }

    auto end = std::chrono::high_resolution_clock::now();
    auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(end - start).count();

    std::cout << "处理 " << total_packets << " 个包, 耗时 " << ms << " ms" << std::endl;
    if (ms > 0) {
        std::cout << "吞吐量: " << total_packets / ms * 1000 << " pps" << std::endl;
    }
}
```

```cpp
// ============================================
// 共享内存IPC框架
// ============================================

class SharedMemoryIPC {
private:
    // IPC消息头
    struct alignas(64) MessageHeader {
        uint32_t msg_type;
        uint32_t msg_size;
        uint64_t timestamp_ns;
        uint64_t sequence;
    };

    // IPC通道
    struct alignas(4096) Channel {
        static constexpr size_t DATA_SIZE = 1024 * 1024;  // 1MB数据区

        alignas(64) std::atomic<uint64_t> write_seq{0};
        alignas(64) std::atomic<uint64_t> read_seq{0};
        alignas(64) std::atomic<uint64_t> total_sent{0};
        alignas(64) std::atomic<uint64_t> total_recv{0};

        char data[DATA_SIZE];
    };

    void* shm_ptr_;
    Channel* channel_;
    size_t shm_size_;
    int shm_fd_;
    bool is_owner_;

public:
    // 创建共享内存IPC
    static SharedMemoryIPC create(const char* name) {
        SharedMemoryIPC ipc;
        ipc.is_owner_ = true;

#ifdef __linux__
        shm_unlink(name);  // 清理旧的
        ipc.shm_fd_ = shm_open(name, O_CREAT | O_RDWR, 0666);
        ipc.shm_size_ = sizeof(Channel);

        ftruncate(ipc.shm_fd_, ipc.shm_size_);
        ipc.shm_ptr_ = mmap(nullptr, ipc.shm_size_,
                            PROT_READ | PROT_WRITE, MAP_SHARED,
                            ipc.shm_fd_, 0);

        ipc.channel_ = new (ipc.shm_ptr_) Channel();  // placement new
#endif
        return ipc;
    }

    // 连接到已有共享内存
    static SharedMemoryIPC connect(const char* name) {
        SharedMemoryIPC ipc;
        ipc.is_owner_ = false;

#ifdef __linux__
        ipc.shm_fd_ = shm_open(name, O_RDWR, 0666);
        ipc.shm_size_ = sizeof(Channel);
        ipc.shm_ptr_ = mmap(nullptr, ipc.shm_size_,
                            PROT_READ | PROT_WRITE, MAP_SHARED,
                            ipc.shm_fd_, 0);
        ipc.channel_ = static_cast<Channel*>(ipc.shm_ptr_);
#endif
        return ipc;
    }

    // 发送消息
    bool send(const void* data, size_t size, uint32_t msg_type = 0) {
        size_t total_size = sizeof(MessageHeader) + size;
        if (total_size > Channel::DATA_SIZE) return false;

        uint64_t seq = channel_->write_seq.fetch_add(1, std::memory_order_acq_rel);
        size_t offset = (seq * total_size) % Channel::DATA_SIZE;

        // 检查是否会覆盖未读数据
        if (offset + total_size > Channel::DATA_SIZE) {
            offset = 0;  // 简化：回到开头
        }

        // 写入消息头
        MessageHeader header;
        header.msg_type = msg_type;
        header.msg_size = static_cast<uint32_t>(size);
        header.timestamp_ns = std::chrono::high_resolution_clock::now()
            .time_since_epoch().count();
        header.sequence = seq;

        std::memcpy(channel_->data + offset, &header, sizeof(header));
        std::memcpy(channel_->data + offset + sizeof(header), data, size);

        channel_->total_sent.fetch_add(1, std::memory_order_relaxed);
        return true;
    }

    // 接收消息
    bool receive(void* buffer, size_t buffer_size,
                 uint32_t* msg_type = nullptr, size_t* msg_size = nullptr) {
        uint64_t read_seq = channel_->read_seq.load(std::memory_order_relaxed);
        uint64_t write_seq = channel_->write_seq.load(std::memory_order_acquire);

        if (read_seq >= write_seq) {
            return false;  // 无新消息
        }

        size_t total_size = sizeof(MessageHeader) + buffer_size;
        size_t offset = (read_seq * total_size) % Channel::DATA_SIZE;

        MessageHeader header;
        std::memcpy(&header, channel_->data + offset, sizeof(header));

        if (header.msg_size > buffer_size) {
            return false;  // 缓冲区太小
        }

        std::memcpy(buffer, channel_->data + offset + sizeof(header),
                    header.msg_size);

        if (msg_type) *msg_type = header.msg_type;
        if (msg_size) *msg_size = header.msg_size;

        channel_->read_seq.fetch_add(1, std::memory_order_release);
        channel_->total_recv.fetch_add(1, std::memory_order_relaxed);
        return true;
    }

    ~SharedMemoryIPC() {
        if (shm_ptr_) {
            munmap(shm_ptr_, shm_size_);
        }
        if (shm_fd_ >= 0) {
            close(shm_fd_);
        }
    }
};
```

```cpp
    }

    void releaseBuffer(char* buffer) {
        buffer_pool_.push_back(buffer);
    }

private:
    static constexpr size_t MAX_RESPONSE_SIZE = 4096;

    char* allocateBuffer() {
        std::lock_guard<std::mutex> lock(pool_mutex_);
        if (buffer_pool_.empty()) return nullptr;
        char* buf = buffer_pool_.back();
        buffer_pool_.pop_back();
        return buf;
    }

    // SIMD加速的大写转换
    size_t toUpperSIMD(const char* src, size_t len, char* dst) {
        size_t i = 0;
#if defined(__AVX2__)
        const __m256i lower_a = _mm256_set1_epi8('a');
        const __m256i lower_z = _mm256_set1_epi8('z');
        const __m256i diff = _mm256_set1_epi8('a' - 'A');

        for (; i + 32 <= len; i += 32) {
            __m256i v = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(src + i));
            __m256i mask = _mm256_and_si256(
                _mm256_cmpgt_epi8(v, lower_a - 1),
                _mm256_cmpgt_epi8(lower_z + 1, v));
            __m256i upper = _mm256_sub_epi8(v, _mm256_and_si256(diff, mask));
            _mm256_storeu_si256(reinterpret_cast<__m256i*>(dst + i), upper);
        }
#endif
        // 标量处理剩余
        for (; i < len; ++i) {
            char c = src[i];
            dst[i] = (c >= 'a' && c <= 'z') ? (c - 32) : c;
        }
        return len;
    }

    std::vector<char*> buffer_pool_;
    std::mutex pool_mutex_;
};

class OptimizedServerV4 {
public:
    // 零拷贝处理：直接在接收缓冲区上操作
    void handleRequestZeroCopy(char* buffer, size_t len) {
        // 直接在原缓冲区上转换（零拷贝）
        toUpperInPlace(buffer, len);
    }

private:
    void toUpperInPlace(char* data, size_t len) {
        size_t i = 0;
#if defined(__AVX2__)
        const __m256i lower_a = _mm256_set1_epi8('a');
        const __m256i lower_z = _mm256_set1_epi8('z');
        const __m256i diff = _mm256_set1_epi8('a' - 'A');

        for (; i + 32 <= len; i += 32) {
            __m256i v = _mm256_loadu_si256(reinterpret_cast<__m256i*>(data + i));
            __m256i mask = _mm256_and_si256(
                _mm256_cmpgt_epi8(v, lower_a - 1),
                _mm256_cmpgt_epi8(lower_z + 1, v));
            __m256i upper = _mm256_sub_epi8(v, _mm256_and_si256(diff, mask));
            _mm256_storeu_si256(reinterpret_cast<__m256i*>(data + i), upper);
        }
#endif
        for (; i < len; ++i) {
            if (data[i] >= 'a' && data[i] <= 'z') {
                data[i] -= 32;
            }
        }
    }
};
```

## 🟢 7. 小结

### 核心要点回顾

| 实战项目 | 关键技术 | 性能提升 |
|---------|---------|---------|
| **高性能日志** | 异步写入 + 无锁缓冲 + 零分配 | 10-100x吞吐量 |
| **零分配消息队列** | 预分配 + 环形缓冲 + 无锁 | 消除GC压力 |
| **实时系统** | 实时调度 + 锁内存 + CPU绑定 | 确定性延迟 |
| **DPDK网络栈** | 轮询模式 + 零拷贝 + 大页 | 10M+ pps |
| **共享内存IPC** | mmap + 原子操作 + 无锁 | 1M+ msg/s |
| **性能调优** | 逐步优化 + 测量驱动 | 100x QPS |

### 系统级编程实战原则

```
1. 架构先行
   → 异步优于同步、批量优于逐个、零拷贝优于拷贝

2. 测量驱动
   → 先profile再优化、每步验证效果

3. 逐步优化
   → 从最简单的版本开始，逐步应用高级技术

4. 保持简单
   → 复杂度是性能的敌人、过早优化是万恶之源

5. 可观测性
   → 埋点监控、性能计数器、日志

6. 权衡取舍
   → 延迟 vs 吞吐量、内存 vs CPU、复杂度 vs 性能
```

### 从1000到100K QPS的优化清单

```
□ 全局锁 → 细粒度锁 → 无锁
□ 同步I/O → 异步I/O → 零拷贝
□ 动态分配 → 内存池 → 栈分配
□ 逐个处理 → 批量处理 → SIMD
□ 通用算法 → 特化算法 → 查表法
□ 内核网络栈 → 用户态网络栈
□ 默认内存 → 大页内存 → NUMA感知
□ 单线程 → 多线程 → 无锁并发
```
