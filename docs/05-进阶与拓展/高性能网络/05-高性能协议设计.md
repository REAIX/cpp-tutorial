> 掌握网络协议设计原则与序列化技术

---

> **A good protocol is like a good language: it makes communication efficient and unambiguous.**
> （好的协议就像好的语言：让通信高效且无歧义。）

> **协议：网络世界的通用语言。**

---

> 🎯 **约定优于配置，协议优于约定。**

> （精心设计的协议是高性能网络应用的基石。）

---

> 💡 **通俗理解 - 协议设计**

想象两个人通电话：
- **没有协议**：各说各的，听不懂对方 → 通信失败
- **有协议**：先说"你好"，再说正事，最后说"再见" → 通信成功

**协议就是通信双方约定的"说话规则"——怎么开始、怎么结束、数据怎么组织！**

> 🔬 **抽象理解 - 协议设计**：
> - **二进制协议**：紧凑高效，解析快，适合高性能场景
> - **文本协议**：可读性好，调试方便，适合开发调试
> - **帧格式**：定义消息的边界和结构，是协议的核心
> - **序列化**：将数据结构转换为可传输的字节序列
> - **协议版本化**：保证前后兼容，平滑升级

### 2.2 二进制协议代码示例

```cpp
#include <iostream>
#include <cstring>
#include <cstdint>
#include <vector>
#include <string>

// 二进制协议工具类
class BinaryWriter {
public:
    // 写入1字节
    void write_uint8(uint8_t value) {
        buffer_.push_back(value);
    }

    // 写入2字节（大端序）
    void write_uint16(uint16_t value) {
        buffer_.push_back((value >> 8) & 0xFF);
        buffer_.push_back(value & 0xFF);
    }

    // 写入4字节（大端序）
    void write_uint32(uint32_t value) {
        buffer_.push_back((value >> 24) & 0xFF);
        buffer_.push_back((value >> 16) & 0xFF);
        buffer_.push_back((value >> 8) & 0xFF);
        buffer_.push_back(value & 0xFF);
    }

    // 写入字符串（长度前缀）
    void write_string(const std::string& str) {
        write_uint16(static_cast<uint16_t>(str.size()));
        buffer_.insert(buffer_.end(), str.begin(), str.end());
    }

    // 写入原始字节
    void write_bytes(const void* data, size_t len) {
        const auto* ptr = static_cast<const uint8_t*>(data);
        buffer_.insert(buffer_.end(), ptr, ptr + len);
    }

    const std::vector<uint8_t>& data() const { return buffer_; }
    size_t size() const { return buffer_.size(); }

private:
    std::vector<uint8_t> buffer_;
};

// 二进制读取器
class BinaryReader {
public:
    BinaryReader(const uint8_t* data, size_t size)
        : data_(data), size_(size), pos_(0) {}

    uint8_t read_uint8() {
        return data_[pos_++];
    }

    uint16_t read_uint16() {
        uint16_t value = (static_cast<uint16_t>(data_[pos_]) << 8) |
                          data_[pos_ + 1];
        pos_ += 2;
        return value;
    }

    uint32_t read_uint32() {
        uint32_t value = (static_cast<uint32_t>(data_[pos_]) << 24) |
                          (static_cast<uint32_t>(data_[pos_ + 1]) << 16) |
                          (static_cast<uint32_t>(data_[pos_ + 2]) << 8) |
                          data_[pos_ + 3];
        pos_ += 4;
        return value;
    }

    std::string read_string() {
        uint16_t len = read_uint16();
        std::string str(reinterpret_cast<const char*>(data_ + pos_), len);
        pos_ += len;
        return str;
    }

    bool has_more() const { return pos_ < size_; }
    size_t remaining() const { return size_ - pos_; }

private:
    const uint8_t* data_;
    size_t size_;
    size_t pos_;
};

// 使用示例
void binary_protocol_example() {
    BinaryWriter writer;

    // 写入协议头
    writer.write_uint8(0x01);       // 版本号
    writer.write_uint8(0x01);       // 消息类型：登录请求
    writer.write_uint32(0);         // 请求ID（占位）
    writer.write_uint16(0);         // 载荷长度（占位）

    // 写入载荷
    size_t payload_start = writer.size();
    writer.write_string("alice");   // 用户名
    writer.write_string("pass123"); // 密码

    // 计算并回填载荷长度
    // （实际实现中需要支持回填）

    std::cout << "二进制协议数据大小: " << writer.size() << " 字节" << std::endl;

    // 解析
    BinaryReader reader(writer.data().data(), writer.size());
    uint8_t version = reader.read_uint8();
    uint8_t msg_type = reader.read_uint8();
    uint32_t request_id = reader.read_uint32();
    uint16_t payload_len = reader.read_uint16();

    std::cout << "版本: " << (int)version
              << ", 类型: " << (int)msg_type
              << ", 请求ID: " << request_id
              << ", 载荷长度: " << payload_len << std::endl;

    std::string username = reader.read_string();
    std::string password = reader.read_string();
    std::cout << "用户名: " << username
              << ", 密码: " << password << std::endl;
}
```

### 2.3 文本协议代码示例

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <sstream>
#include <unordered_map>

// 简单的文本协议解析器（类似Redis协议）
class TextProtocolParser {
public:
    // RESP协议（Redis序列化协议）
    // 简单字符串: +OK\r\n
    // 错误: -ERR message\r\n
    // 整数: :1000\r\n
    // 批量字符串: $6\r\nfoobar\r\n
    // 数组: *2\r\n$3\r\nfoo\r\n$3\r\nbar\r\n

    enum class Type { SIMPLE_STRING, ERROR, INTEGER, BULK_STRING, ARRAY };

    struct Value {
        Type type;
        std::string str;
        int64_t integer;
        std::vector<Value> array;
    };

    static Value parse(const std::string& data) {
        size_t pos = 0;
        return parse_one(data, pos);
    }

private:
    static Value parse_one(const std::string& data, size_t& pos) {
        char prefix = data[pos++];

        switch (prefix) {
            case '+': return parse_simple_string(data, pos);
            case '-': return parse_error(data, pos);
            case ':': return parse_integer(data, pos);
            case '$': return parse_bulk_string(data, pos);
            case '*': return parse_array(data, pos);
            default:
                return {Type::ERROR, "Unknown prefix: " + std::string(1, prefix), 0, {}};
        }
    }

    static Value parse_simple_string(const std::string& data, size_t& pos) {
        size_t end = data.find("\r\n", pos);
        std::string str = data.substr(pos, end - pos);
        pos = end + 2;
        return {Type::SIMPLE_STRING, str, 0, {}};
    }

    static Value parse_error(const std::string& data, size_t& pos) {
        size_t end = data.find("\r\n", pos);
        std::string str = data.substr(pos, end - pos);
        pos = end + 2;
        return {Type::ERROR, str, 0, {}};
    }

    static Value parse_integer(const std::string& data, size_t& pos) {
        size_t end = data.find("\r\n", pos);
        int64_t val = std::stoll(data.substr(pos, end - pos));
        pos = end + 2;
        return {Type::INTEGER, "", val, {}};
    }

    static Value parse_bulk_string(const std::string& data, size_t& pos) {
        size_t end = data.find("\r\n", pos);
        int64_t len = std::stoll(data.substr(pos, end - pos));
        pos = end + 2;

        if (len < 0) {
            return {Type::BULK_STRING, "", 0, {}};  // null
        }

        std::string str = data.substr(pos, len);
        pos += len + 2;  // 跳过\r\n
        return {Type::BULK_STRING, str, 0, {}};
    }

    static Value parse_array(const std::string& data, size_t& pos) {
        size_t end = data.find("\r\n", pos);
        int64_t count = std::stoll(data.substr(pos, end - pos));
        pos = end + 2;

        Value result{Type::ARRAY, "", 0, {}};

        if (count < 0) {
            return result;  // null array
        }

        for (int64_t i = 0; i < count; ++i) {
            result.array.push_back(parse_one(data, pos));
        }

        return result;
    }
};

// HTTP请求解析器（简化版）
class HttpRequestParser {
public:
    struct Request {
        std::string method;
        std::string path;
        std::string version;
        std::unordered_map<std::string, std::string> headers;
        std::string body;
    };

    static Request parse(const std::string& raw) {
        Request req;
        std::istringstream stream(raw);
        std::string line;

        // 解析请求行
        std::getline(stream, line);
        line.erase(line.find('\r'));
        std::istringstream request_line(line);
        request_line >> req.method >> req.path >> req.version;

        // 解析头部
        while (std::getline(stream, line)) {
            if (line == "\r" || line.empty()) break;
            line.erase(line.find('\r'));
            size_t colon = line.find(':');
            if (colon != std::string::npos) {
                std::string key = line.substr(0, colon);
                std::string value = line.substr(colon + 2);
                req.headers[key] = value;
            }
        }

        // 解析body
        std::string body_line;
        while (std::getline(stream, body_line)) {
            req.body += body_line + "\n";
        }

        return req;
    }
};
```

### 3.2 完整帧格式设计

```cpp
#include <iostream>
#include <cstring>
#include <cstdint>
#include <vector>
#include <algorithm>

// 高性能二进制帧格式
// ┌──────┬──────┬──────┬──────┬──────┬──────────┬──────┐
// │ Magic│ Ver  │ Type │ ReqId│ Len  │ Payload  │ CRC  │
// │ 2B   │ 1B   │ 1B   │ 4B   │ 4B   │ N B      │ 2B   │
// └──────┴──────┴──────┴──────┴──────┴──────────┴──────┘

#pragma pack(push, 1)  // 确保无填充字节
struct FrameHeader {
    uint16_t magic;      // 魔数，用于识别协议
    uint8_t  version;    // 协议版本
    uint8_t  msg_type;   // 消息类型
    uint32_t request_id; // 请求ID（用于匹配请求和响应）
    uint32_t length;     // 载荷长度
};
#pragma pack(pop)

constexpr uint16_t FRAME_MAGIC = 0xABCD;
constexpr size_t FRAME_HEADER_SIZE = sizeof(FrameHeader);
constexpr size_t FRAME_CRC_SIZE = 2;
constexpr size_t FRAME_MAX_PAYLOAD = 1024 * 1024;  // 1MB最大载荷

// CRC16校验
uint16_t crc16(const uint8_t* data, size_t len) {
    uint16_t crc = 0xFFFF;
    for (size_t i = 0; i < len; ++i) {
        crc ^= data[i];
        for (int j = 0; j < 8; ++j) {
            if (crc & 1) {
                crc = (crc >> 1) ^ 0xA001;
            } else {
                crc >>= 1;
            }
        }
    }
    return crc;
}

// 帧编码器
class FrameEncoder {
public:
    // 编码一帧
    static std::vector<uint8_t> encode(uint8_t msg_type,
                                        uint32_t request_id,
                                        const void* payload,
                                        size_t payload_len) {
        std::vector<uint8_t> frame;

        // 填充头部
        FrameHeader header{};
        header.magic = htons(FRAME_MAGIC);
        header.version = 1;
        header.msg_type = msg_type;
        header.request_id = htonl(request_id);
        header.length = htonl(payload_len);

        // 写入头部
        const auto* header_bytes = reinterpret_cast<const uint8_t*>(&header);
        frame.insert(frame.end(), header_bytes,
                     header_bytes + FRAME_HEADER_SIZE);

        // 写入载荷
        const auto* payload_bytes = static_cast<const uint8_t*>(payload);
        frame.insert(frame.end(), payload_bytes,
                     payload_bytes + payload_len);

        // 计算并写入CRC（覆盖头部+载荷）
        uint16_t crc = crc16(frame.data(), frame.size());
        uint8_t crc_bytes[2] = {
            static_cast<uint8_t>(crc >> 8),
            static_cast<uint8_t>(crc & 0xFF)
        };
        frame.insert(frame.end(), crc_bytes, crc_bytes + 2);

        return frame;
    }
};

// 帧解码器
class FrameDecoder {
public:
    struct DecodeResult {
        bool success;
        std::string error;
        uint8_t msg_type;
        uint32_t request_id;
        std::vector<uint8_t> payload;
        size_t bytes_consumed;
    };

    // 尝试从缓冲区解码一帧
    static DecodeResult decode(const uint8_t* data, size_t available) {
        DecodeResult result{};
        result.success = false;

        // 检查是否有足够的头部数据
        if (available < FRAME_HEADER_SIZE) {
            result.error = "数据不足，无法读取头部";
            return result;
        }

        // 读取头部
        FrameHeader header;
        memcpy(&header, data, FRAME_HEADER_SIZE);

        // 转换字节序
        header.magic = ntohs(header.magic);
        header.length = ntohl(header.length);
        header.request_id = ntohl(header.request_id);

        // 验证魔数
        if (header.magic != FRAME_MAGIC) {
            result.error = "魔数不匹配";
            return result;
        }

        // 验证载荷长度
        if (header.length > FRAME_MAX_PAYLOAD) {
            result.error = "载荷超过最大限制";
            return result;
        }

        // 检查是否有完整的帧
        size_t frame_size = FRAME_HEADER_SIZE + header.length + FRAME_CRC_SIZE;
        if (available < frame_size) {
            result.error = "数据不足，等待更多数据";
            return result;
        }

        // 验证CRC
        uint16_t expected_crc = (data[FRAME_HEADER_SIZE + header.length] << 8) |
                                 data[FRAME_HEADER_SIZE + header.length + 1];
        uint16_t actual_crc = crc16(data, FRAME_HEADER_SIZE + header.length);

        if (expected_crc != actual_crc) {
            result.error = "CRC校验失败";
            return result;
        }

        // 解码成功
        result.success = true;
        result.msg_type = header.msg_type;
        result.request_id = header.request_id;
        result.bytes_consumed = frame_size;

        const auto* payload_start = data + FRAME_HEADER_SIZE;
        result.payload.assign(payload_start, payload_start + header.length);

        return result;
    }
};

// 使用示例
void frame_protocol_example() {
    // 编码
    std::string message = "Hello, Protocol!";
    auto frame = FrameEncoder::encode(
        0x01,           // 消息类型
        12345,          // 请求ID
        message.data(),
        message.size()
    );

    std::cout << "编码帧大小: " << frame.size() << " 字节" << std::endl;
    std::cout << "  头部: " << FRAME_HEADER_SIZE << " 字节" << std::endl;
    std::cout << "  载荷: " << message.size() << " 字节" << std::endl;
    std::cout << "  CRC: " << FRAME_CRC_SIZE << " 字节" << std::endl;

    // 解码
    auto result = FrameDecoder::decode(frame.data(), frame.size());

    if (result.success) {
        std::string payload(result.payload.begin(), result.payload.end());
        std::cout << "解码成功: 类型=" << (int)result.msg_type
                  << ", 请求ID=" << result.request_id
                  << ", 载荷=" << payload << std::endl;
    } else {
        std::cout << "解码失败: " << result.error << std::endl;
```

```cpp
#include <iostream>

// 协议版本管理器
class ProtocolVersionManager {
public:
    // 注册协议版本处理器
    using Handler = std::function<void(const uint8_t*, size_t)>;

    void register_handler(uint8_t version, Handler handler) {
        handlers_[version] = std::move(handler);
    }

    // 根据版本分发处理
    void dispatch(uint8_t version, const uint8_t* data, size_t len) {
        auto it = handlers_.find(version);
        if (it != handlers_.end()) {
            it->second(data, len);
        } else {
            // 尝试使用最低兼容版本处理
            if (!handlers_.empty()) {
                auto lowest = handlers_.begin();
                std::cout << "未知版本 " << (int)version
                          << "，使用版本 " << (int)lowest->first
                          << " 处理" << std::endl;
                lowest->second(data, len);
            }
        }
    }

private:
    std::unordered_map<uint8_t, Handler> handlers_;
};

// V1协议消息
struct LoginV1 {
    std::string username;
    std::string password;
};

// V2协议消息（新增token字段）
struct LoginV2 {
    std::string username;
    std::string password;
    std::string token;      // V2新增
    std::string device_id;  // V2新增
};

// 版本化协议编解码
class VersionedProtocol {
public:
    // V1编码
    static std::vector<uint8_t> encode_v1(const LoginV1& login) {
        std::vector<uint8_t> data;
        // 版本号
        data.push_back(1);
        // 用户名长度+数据
        uint16_t len = login.username.size();
        data.push_back(len >> 8);
        data.push_back(len & 0xFF);
        data.insert(data.end(), login.username.begin(), login.username.end());
        // 密码长度+数据
        len = login.password.size();
        data.push_back(len >> 8);
        data.push_back(len & 0xFF);
        data.insert(data.end(), login.password.begin(), login.password.end());
        return data;
    }

    // V2编码
    static std::vector<uint8_t> encode_v2(const LoginV2& login) {
        std::vector<uint8_t> data;
        // 版本号
        data.push_back(2);
        // 用户名
        append_string(data, login.username);
        // 密码
        append_string(data, login.password);
        // V2新增字段
        append_string(data, login.token);
        append_string(data, login.device_id);
        return data;
    }

    // V2解码器可以兼容V1数据
    static LoginV2 decode(const uint8_t* data, size_t len) {
        LoginV2 result{};
        size_t pos = 0;

        uint8_t version = data[pos++];

        result.username = read_string(data, pos);
        result.password = read_string(data, pos);

        if (version >= 2 && pos < len) {
            result.token = read_string(data, pos);
        }
        if (version >= 2 && pos < len) {
            result.device_id = read_string(data, pos);
        }

        return result;
    }

private:
    static void append_string(std::vector<uint8_t>& data,
                               const std::string& str) {
        uint16_t len = str.size();
        data.push_back(len >> 8);
        data.push_back(len & 0xFF);
        data.insert(data.end(), str.begin(), str.end());
    }

    static std::string read_string(const uint8_t* data, size_t& pos) {
        uint16_t len = (data[pos] << 8) | data[pos + 1];
        pos += 2;
        std::string str(reinterpret_cast<const char*>(data + pos), len);
        pos += len;
        return str;
    }
};
```

```cpp
// 手动实现简化版Protobuf编码（理解原理）
class SimpleProtobuf {
public:
    // Varint编码（Protobuf的核心编码方式）
    static std::vector<uint8_t> encode_varint(uint64_t value) {
        std::vector<uint8_t> bytes;
        while (value > 0x7F) {
            bytes.push_back(static_cast<uint8_t>(value & 0x7F) | 0x80);
            value >>= 7;
        }
        bytes.push_back(static_cast<uint8_t>(value));
        return bytes;
    }

    // Varint解码
    static uint64_t decode_varint(const uint8_t* data, size_t& pos) {
        uint64_t result = 0;
        int shift = 0;
        while (data[pos] & 0x80) {
            result |= static_cast<uint64_t>(data[pos] & 0x7F) << shift;
            shift += 7;
            pos++;
        }
        result |= static_cast<uint64_t>(data[pos] & 0x7F) << shift;
        pos++;
        return result;
    }

    // 编码字段标签（field_number << 3 | wire_type）
    static std::vector<uint8_t> encode_tag(uint32_t field_number,
                                             uint8_t wire_type) {
        return encode_varint((static_cast<uint64_t>(field_number) << 3) | wire_type);
    }

    // 编码字符串字段
    static std::vector<uint8_t> encode_string(uint32_t field_number,
                                                const std::string& value) {
        std::vector<uint8_t> result;
        // 标签：wire_type = 2（长度分隔）
        auto tag = encode_tag(field_number, 2);
        result.insert(result.end(), tag.begin(), tag.end());
        // 长度
        auto len = encode_varint(value.size());
        result.insert(result.end(), len.begin(), len.end());
        // 数据
        result.insert(result.end(), value.begin(), value.end());
        return result;
    }

    // 编码整数字段
    static std::vector<uint8_t> encode_int32(uint32_t field_number,
                                               int32_t value) {
        std::vector<uint8_t> result;
        // 标签：wire_type = 0（Varint）
        auto tag = encode_tag(field_number, 0);
        result.insert(result.end(), tag.begin(), tag.end());
        // 值
        auto val = encode_varint(static_cast<uint64_t>(value));
        result.insert(result.end(), val.begin(), val.end());
        return result;
    }
};

void protobuf_example() {
    // 编码一个简单的Person消息
    std::vector<uint8_t> message;

    // name = "Alice" (field 1)
    auto name_field = SimpleProtobuf::encode_string(1, "Alice");
    message.insert(message.end(), name_field.begin(), name_field.end());

    // id = 12345 (field 2)
    auto id_field = SimpleProtobuf::encode_int32(2, 12345);
    message.insert(message.end(), id_field.begin(), id_field.end());

    // email = "alice@example.com" (field 3)
    auto email_field = SimpleProtobuf::encode_string(3, "alice@example.com");
    message.insert(message.end(), email_field.begin(), email_field.end());
```

```cpp
                          reinterpret_cast<const uint8_t*>(&len),
```

## 6. 自定义协议实战

### 6.1 RPC协议设计

```cpp
#include <iostream>
#include <cstring>
#include <cstdint>
#include <vector>
#include <string>
#include <unordered_map>
#include <functional>
#include <memory>

// ==================== RPC协议定义 ====================

// 消息类型
enum class MessageType : uint8_t {
    REQUEST = 0x01,       // 请求
    RESPONSE = 0x02,      // 响应
    NOTIFICATION = 0x03,  // 通知（无需响应）
    HEARTBEAT = 0x04,     // 心跳
    ERROR = 0x05          // 错误
};

// 错误码
enum class ErrorCode : uint16_t {
    OK = 0,
    INVALID_REQUEST = 1,
    METHOD_NOT_FOUND = 2,
    INVALID_PARAMS = 3,
    INTERNAL_ERROR = 4,
    TIMEOUT = 5,
    OVERLOAD = 6
};

// RPC帧格式
// ┌──────┬──────┬──────┬──────┬──────┬──────┬──────────┬──────┐
// │ Magic│ Ver  │ Type │ Flags│ ReqId│ Len  │ Payload  │ CRC  │
// │ 2B   │ 1B   │ 1B   │ 1B   │ 4B   │ 4B   │ N B      │ 2B   │
// └──────┴──────┴──────┴──────┴──────┴──────┴──────────┴──────┘

#pragma pack(push, 1)
struct RpcHeader {
    uint16_t magic;       // 0xRPC1
    uint8_t  version;     // 协议版本
    uint8_t  msg_type;    // 消息类型
    uint8_t  flags;       // 标志位
    uint32_t request_id;  // 请求ID
    uint32_t length;      // 载荷长度
};
#pragma pack(pop)

constexpr uint16_t RPC_MAGIC = 0x5250;  // "RP"
constexpr size_t RPC_HEADER_SIZE = sizeof(RpcHeader);

// RPC请求载荷
struct RpcRequest {
    std::string method;   // 方法名
    std::string params;   // 参数（JSON/Protobuf等格式）
};

// RPC响应载荷
struct RpcResponse {
    uint16_t error_code;  // 错误码
    std::string result;   // 结果
};

// ==================== RPC编解码 ====================

class RpcCodec {
public:
    // 编码请求
    static std::vector<uint8_t> encode_request(
        uint32_t request_id,
        const std::string& method,
        const std::string& params) {

        // 构建载荷
        std::vector<uint8_t> payload;
        // 方法名（长度前缀）
        uint16_t method_len = method.size();
        payload.push_back(method_len >> 8);
        payload.push_back(method_len & 0xFF);
        payload.insert(payload.end(), method.begin(), method.end());
        // 参数
        uint32_t params_len = params.size();
        payload.push_back(params_len >> 24);
        payload.push_back((params_len >> 16) & 0xFF);
        payload.push_back((params_len >> 8) & 0xFF);
        payload.push_back(params_len & 0xFF);
        payload.insert(payload.end(), params.begin(), params.end());

        return build_frame(MessageType::REQUEST, 0, request_id, payload);
    }

    // 编码响应
    static std::vector<uint8_t> encode_response(
        uint32_t request_id,
        ErrorCode error_code,
        const std::string& result) {

        std::vector<uint8_t> payload;
        // 错误码
        payload.push_back(static_cast<uint16_t>(error_code) >> 8);
        payload.push_back(static_cast<uint16_t>(error_code) & 0xFF);
        // 结果
        uint32_t result_len = result.size();
        payload.push_back(result_len >> 24);
        payload.push_back((result_len >> 16) & 0xFF);
        payload.push_back((result_len >> 8) & 0xFF);
        payload.push_back(result_len & 0xFF);
        payload.insert(payload.end(), result.begin(), result.end());

        return build_frame(MessageType::RESPONSE, 0, request_id, payload);
    }

    // 编码心跳
    static std::vector<uint8_t> encode_heartbeat(uint32_t request_id) {
        return build_frame(MessageType::HEARTBEAT, 0, request_id, {});
    }

private:
    static std::vector<uint8_t> build_frame(
        MessageType msg_type, uint8_t flags,
        uint32_t request_id,
        const std::vector<uint8_t>& payload) {

        std::vector<uint8_t> frame;

        RpcHeader header{};
        header.magic = htons(RPC_MAGIC);
        header.version = 1;
        header.msg_type = static_cast<uint8_t>(msg_type);
        header.flags = flags;
        header.request_id = htonl(request_id);
        header.length = htonl(payload.size());

        const auto* header_bytes = reinterpret_cast<const uint8_t*>(&header);
        frame.insert(frame.end(), header_bytes,
                     header_bytes + RPC_HEADER_SIZE);
        frame.insert(frame.end(), payload.begin(), payload.end());

        return frame;
    }
};

// ==================== RPC服务器 ====================

class RpcServer {
public:
    using MethodHandler = std::function<std::string(const std::string&)>;

    void register_method(const std::string& method, MethodHandler handler) {
        methods_[method] = std::move(handler);
    }

    // 处理接收到的RPC请求
    std::vector<uint8_t> handle_request(const uint8_t* data, size_t len) {
        // 解析头部
        if (len < RPC_HEADER_SIZE) return {};

        RpcHeader header;
        memcpy(&header, data, RPC_HEADER_SIZE);
        header.magic = ntohs(header.magic);
        header.request_id = ntohl(header.request_id);
        header.length = ntohl(header.length);

        if (header.magic != RPC_MAGIC) return {};

        auto msg_type = static_cast<MessageType>(header.msg_type);

        if (msg_type == MessageType::HEARTBEAT) {
            return RpcCodec::encode_heartbeat(header.request_id);
        }

        if (msg_type == MessageType::REQUEST) {
            // 解析请求载荷
            const uint8_t* payload = data + RPC_HEADER_SIZE;

            uint16_t method_len = (payload[0] << 8) | payload[1];
            std::string method(reinterpret_cast<const char*>(payload + 2),
                               method_len);

            size_t params_offset = 2 + method_len;
            uint32_t params_len = (payload[params_offset] << 24) |
                                   (payload[params_offset + 1] << 16) |
                                   (payload[params_offset + 2] << 8) |
                                   payload[params_offset + 3];
            std::string params(
                reinterpret_cast<const char*>(payload + params_offset + 4),
                params_len);

            // 调用方法
            auto it = methods_.find(method);
            if (it != methods_.end()) {
                try {
                    std::string result = it->second(params);
                    return RpcCodec::encode_response(
                        header.request_id, ErrorCode::OK, result);
                } catch (...) {
                    return RpcCodec::encode_response(
                        header.request_id, ErrorCode::INTERNAL_ERROR, "");
                }
            } else {
                return RpcCodec::encode_response(
                    header.request_id, ErrorCode::METHOD_NOT_FOUND, "");
            }
        }

        return {};
    }

private:
    std::unordered_map<std::string, MethodHandler> methods_;
};

// 使用示例
int main() {
    RpcServer server;

    // 注册方法
    server.register_method("add", [](const std::string& params) -> std::string {
        // 简化：假设params是 "a,b" 格式
        size_t comma = params.find(',');
        int a = std::stoi(params.substr(0, comma));
        int b = std::stoi(params.substr(comma + 1));
        return std::to_string(a + b);
    });

    server.register_method("echo", [](const std::string& params) -> std::string {
        return params;
    });

    // 模拟客户端发送请求
    auto request = RpcCodec::encode_request(1, "add", "3,5");
    std::cout << "请求帧大小: " << request.size() << " 字节" << std::endl;

    // 服务器处理请求
    auto response = server.handle_request(request.data(), request.size());
    std::cout << "响应帧大小: " << response.size() << " 字节" << std::endl;

    // 心跳
    auto heartbeat = RpcCodec::encode_heartbeat(0);
    auto hb_response = server.handle_request(heartbeat.data(), heartbeat.size());
    std::cout << "心跳响应帧大小: " << hb_response.size() << " 字节" << std::endl;

    return 0;
}
```

## 7. 小结

### 核心要点回顾

| 要点 | 说明 |
|------|------|
| **简洁性** | 协议应尽可能简单，避免过度设计 |
| **高效性** | 最小化带宽和解析开销 |
| **二进制协议** | 紧凑高效，适合高性能场景 |
| **文本协议** | 可读性好，适合调试和简单场景 |
| **帧格式** | 长度前缀法最常用，TLV最灵活 |
| **版本化** | 版本号+可选字段+前向兼容 |
| **Protobuf** | 高效二进制序列化，需要Schema |
| **FlatBuffers** | 零拷贝序列化，无需解析步骤 |
| **自定义协议** | 根据业务需求设计，包含帧格式+编解码+错误处理 |

### 序列化格式选择

| 场景 | 推荐格式 | 原因 |
|------|---------|------|
| **通用RPC** | Protobuf | 高效+成熟+多语言支持 |
| **游戏** | FlatBuffers | 零拷贝，低延迟 |
| **配置文件** | JSON | 可读性好 |
| **内部通信** | 自定义二进制 | 最优性能 |
| **REST API** | JSON | 通用+可读 |
| **消息队列** | Protobuf/Avro | 高效+Schema演进 |
