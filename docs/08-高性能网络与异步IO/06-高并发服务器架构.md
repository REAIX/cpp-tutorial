### 1.2 连接管理器实现

```cpp
#include <iostream>
#include <sys/epoll.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <unordered_map>
#include <memory>

#include <functional>
#include <atomic>

// 连接信息
struct ConnectionInfo {
    int fd;
    std::chrono::steady_clock::time_point connect_time;
    std::chrono::steady_clock::time_point last_active_time;
    uint64_t bytes_read;
    uint64_t bytes_written;
    std::string client_ip;
    uint16_t client_port;
};

// 连接管理器
class ConnectionManager {
public:
    using ConnectionCallback = std::function<void(int)>;

    ConnectionManager(size_t max_connections = 100000,
                      size_t max_per_ip = 100)
        : max_connections_(max_connections),
          max_per_ip_(max_per_ip),
          current_count_(0) {}

    // 接受新连接
    bool accept_connection(int fd, const std::string& ip, uint16_t port) {
        // 检查总连接数限制
        if (current_count_ >= max_connections_) {
            std::cout << "连接数已达上限: " << max_connections_ << std::endl;
            return false;
        }

        // 检查单IP连接数限制
        if (per_ip_count_[ip] >= max_per_ip_) {
            std::cout << "IP " << ip << " 连接数达上限" << std::endl;
            return false;
        }

        // 创建连接信息
        auto info = std::make_shared<ConnectionInfo>();
        info->fd = fd;
        info->connect_time = std::chrono::steady_clock::now();
        info->last_active_time = info->connect_time;
        info->bytes_read = 0;
        info->bytes_written = 0;
        info->client_ip = ip;
        info->client_port = port;

        connections_[fd] = info;
        per_ip_count_[ip]++;
        current_count_++;

        return true;
    }

    // 关闭连接
    void close_connection(int fd) {
        auto it = connections_.find(fd);
        if (it == connections_.end()) return;

        auto& info = it->second;
        per_ip_count_[info->client_ip]--;
        current_count_--;

        close(fd);
        connections_.erase(it);
    }

    // 更新活跃时间
    void update_active_time(int fd) {
        auto it = connections_.find(fd);
        if (it != connections_.end()) {
            it->second->last_active_time = std::chrono::steady_clock::now();
        }
    }

    // 清理超时连接
    void cleanup_timeout(std::chrono::seconds timeout) {
        auto now = std::chrono::steady_clock::now();

        for (auto it = connections_.begin(); it != connections_.end();) {
            auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(
                now - it->second->last_active_time);

            if (elapsed > timeout) {
                std::cout << "清理超时连接: fd=" << it->second->fd
                          << ", 超时: " << elapsed.count() << "秒" << std::endl;
                per_ip_count_[it->second->client_ip]--;
                close(it->second->fd);
                it = connections_.erase(it);
                current_count_--;
            } else {
                ++it;
            }
        }
    }

    // 获取统计信息
    void print_stats() const {
        std::cout << "当前连接数: " << current_count_
                  << " / " << max_connections_ << std::endl;
    }

    size_t current_count() const { return current_count_; }

private:
    size_t max_connections_;
    size_t max_per_ip_;
    std::atomic<size_t> current_count_;
    std::unordered_map<int, std::shared_ptr<ConnectionInfo>> connections_;
    std::unordered_map<std::string, size_t> per_ip_count_;
};
```

```cpp
// 令牌桶限流器
class TokenBucketLimiter {
public:
    TokenBucketLimiter(double rate, double burst)
        : rate_(rate), burst_(burst), tokens_(burst),
          last_time_(std::chrono::steady_clock::now()) {}

    // 尝试获取一个令牌
    bool try_acquire(int count = 1) {
        std::lock_guard<std::mutex> lock(mutex_);

        // 补充令牌
        auto now = std::chrono::steady_clock::now();
        auto elapsed = std::chrono::duration<double>(now - last_time_).count();
        tokens_ = std::min(burst_, tokens_ + elapsed * rate_);
        last_time_ = now;

        // 尝试消费
        if (tokens_ >= count) {
            tokens_ -= count;
            return true;
        }
        return false;
    }

private:
    double rate_;        // 每秒生成令牌数
    double burst_;       // 桶容量（最大突发量）
    double tokens_;      // 当前令牌数
    std::chrono::steady_clock::time_point last_time_;
    std::mutex mutex_;
};

// 滑动窗口限流器
class SlidingWindowLimiter {
public:
    SlidingWindowLimiter(int max_requests, std::chrono::seconds window)
        : max_requests_(max_requests), window_(window) {}

    bool try_acquire() {
        auto now = std::chrono::steady_clock::now();
        auto window_start = now - window_;

        // 清理过期记录
        while (!timestamps_.empty() && timestamps_.front() < window_start) {
            timestamps_.pop_front();
        }

        // 检查是否超过限制
        if (static_cast<int>(timestamps_.size()) >= max_requests_) {
            return false;
        }

        timestamps_.push_back(now);
        return true;
    }

private:
    int max_requests_;
    std::chrono::seconds window_;
    std::deque<std::chrono::steady_clock::time_point> timestamps_;
};
```

## 2. 线程模型

### 2.1 三种线程模型

```
┌──────────────────────────────────────────────────────────────┐
│                    三种线程模型                                │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│   1:1 模型（每连接一线程）                                    │
│   ┌──────────────────────────────────────────────┐          │
│   │  连接1 ──▶ 线程1                             │          │
│   │  连接2 ──▶ 线程2                             │          │
│   │  连接3 ──▶ 线程3                             │          │
│   │  ...                                         │          │
│   │  连接N ──▶ 病程N → 线程数爆炸！              │          │
│   └──────────────────────────────────────────────┘          │
│   优点：编程简单                                              │
│   缺点：线程数过多，资源消耗大                                │
│                                                              │
│   N:1 模型（多连接一线程）                                    │
│   ┌──────────────────────────────────────────────┐          │
│   │  连接1 ─┐                                    │          │
│   │  连接2 ─┤──▶ 线程1（事件循环）                │          │
│   │  连接3 ─┘                                    │          │
│   └──────────────────────────────────────────────┘          │
│   优点：线程数少，无竞争                                      │
│   缺点：无法利用多核，业务阻塞影响所有连接                     │
│                                                              │
│   M:N 模型（多连接多线程）                                    │
│   ┌──────────────────────────────────────────────┐          │
│   │  连接1,2 ──▶ IO线程1 ──▶ 工作线程池          │          │
│   │  连接3,4 ──▶ IO线程2 ──▶ 工作线程池          │          │
│   │  连接5,6 ──▶ IO线程3 ──▶ 工作线程池          │          │
│   └──────────────────────────────────────────────┘          │
│   优点：充分利用多核，IO与业务分离                            │
│   缺点：实现复杂，需要线程间通信                              │
│   推荐：生产环境首选                                          │
```

### 2.2 M:N线程模型实现

```cpp
#include <iostream>
#include <thread>
#include <vector>
#include <queue>
#include <mutex>
#include <condition_variable>
#include <functional>
#include <atomic>
#include <sys/epoll.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <cstring>

// 工作窃取线程池
class WorkStealingPool {
public:
    WorkStealingPool(size_t num_threads) : stop_(false) {
        for (size_t i = 0; i < num_threads; ++i) {
            local_queues_.emplace_back(std::make_unique<LocalQueue>());
        }

        for (size_t i = 0; i < num_threads; ++i) {
            workers_.emplace_back([this, i] { worker_loop(i); });
        }
    }

    ~WorkStealingPool() {
        stop_ = true;
        cv_.notify_all();
        for (auto& w : workers_) {
            if (w.joinable()) w.join();
        }
    }

    // 提交任务
    void enqueue(std::function<void()> task) {
        // 随机选择一个本地队列
        size_t idx = rand() % local_queues_.size();
        {
            std::lock_guard<std::mutex> lock(local_queues_[idx]->mutex);
            local_queues_[idx]->tasks.push(std::move(task));
        }
        cv_.notify_one();
    }

private:
    struct LocalQueue {
        std::queue<std::function<void()>> tasks;
        std::mutex mutex;
    };

    void worker_loop(size_t idx) {
        while (!stop_) {
            std::function<void()> task;

            // 先从本地队列取
            {
                std::lock_guard<std::mutex> lock(local_queues_[idx]->mutex);
                if (!local_queues_[idx]->tasks.empty()) {
                    task = std::move(local_queues_[idx]->tasks.front());
                    local_queues_[idx]->tasks.pop();
                }
            }

            // 本地没有，从其他队列窃取
            if (!task) {
                for (size_t i = 0; i < local_queues_.size(); ++i) {
                    if (i == idx) continue;
                    std::lock_guard<std::mutex> lock(local_queues_[i]->mutex);
                    if (!local_queues_[i]->tasks.empty()) {
                        task = std::move(local_queues_[i]->tasks.front());
                        local_queues_[i]->tasks.pop();
                        break;
                    }
                }
            }

            if (task) {
                task();
            } else {
                std::unique_lock<std::mutex> lock(cv_mutex_);
                cv_.wait_for(lock, std::chrono::milliseconds(1));
            }
        }
    }

    std::atomic<bool> stop_;
    std::vector<std::thread> workers_;
    std::vector<std::unique_ptr<LocalQueue>> local_queues_;
    std::mutex cv_mutex_;
    std::condition_variable cv_;
};

// IO线程 + 工作线程池架构
class IOThreadPool {
public:
    IOThreadPool(size_t io_threads, size_t worker_threads)
        : pool_(worker_threads), running_(false) {
        for (size_t i = 0; i < io_threads; ++i) {
            io_loops_.push_back(std::make_unique<IOLoop>());
        }
    }

    void start() {
        running_ = true;

        // 启动IO线程
        for (size_t i = 0; i < io_loops_.size(); ++i) {
            io_threads_.emplace_back([this, i] {
                io_loops_[i]->run();
            });
        }

        std::cout << "IO线程池启动: " << io_loops_.size()
                  << " IO线程, " << "工作线程池就绪" << std::endl;
    }

    void stop() {
        running_ = false;
        for (auto& loop : io_loops_) {
            loop->stop();
        }
        for (auto& t : io_threads_) {
            if (t.joinable()) t.join();
        }
    }

    // 获取下一个IO循环（轮询）
    IOLoop* get_next_loop() {
        size_t idx = next_loop_++ % io_loops_.size();
        return io_loops_[idx].get();
    }

    WorkStealingPool& worker_pool() { return pool_; }

private:
    struct IOLoop {
        int epfd;
        std::atomic<bool> running{false};

        IOLoop() : epfd(epoll_create1(0)) {}
        ~IOLoop() { if (epfd >= 0) close(epfd); }

        void run() {
            running = true;
            std::vector<epoll_event> events(1024);

            while (running) {
                int nfds = epoll_wait(epfd, events.data(), events.size(), 100);
                for (int i = 0; i < nfds; ++i) {
                    // 处理IO事件
                }
            }
        }

        void stop() { running = false; }
    };

    std::atomic<bool> running_;
    std::vector<std::unique_ptr<IOLoop>> io_loops_;
    std::vector<std::thread> io_threads_;
    WorkStealingPool pool_;
    std::atomic<size_t> next_loop_{0};
};
```

### 3.2 最小堆定时器

```cpp
#include <iostream>
#include <vector>
#include <functional>
#include <chrono>
#include <algorithm>
#include <cstdint>

class MinHeapTimer {
public:
    using TimerId = uint64_t;
    using Callback = std::function<void()>;

    struct TimerEntry {
        TimerId id;
        std::chrono::steady_clock::time_point expire_time;
        Callback callback;
        bool cancelled = false;

        bool operator>(const TimerEntry& other) const {
            return expire_time > other.expire_time;
        }
    };

    MinHeapTimer() : next_id_(1) {}

    // 添加定时器
    TimerId add_timer(std::chrono::milliseconds delay, Callback callback) {
        TimerEntry entry;
        entry.id = next_id_++;
        entry.expire_time = std::chrono::steady_clock::now() + delay;
        entry.callback = std::move(callback);

        heap_.push_back(std::move(entry));
        sift_up(heap_.size() - 1);

        return heap_.back().id;  // 注意：move后id仍有效
    }

    // 取消定时器
    void cancel_timer(TimerId id) {
        for (auto& entry : heap_) {
            if (entry.id == id) {
                entry.cancelled = true;
                break;
            }
        }
    }

    // 处理到期定时器
    void tick() {
        auto now = std::chrono::steady_clock::now();

        while (!heap_.empty()) {
            if (heap_[0].expire_time > now) break;

            TimerEntry entry = std::move(heap_[0]);

            // 移除堆顶
            heap_[0] = std::move(heap_.back());
            heap_.pop_back();
            if (!heap_.empty()) sift_down(0);

            // 执行回调（如果未取消）
            if (!entry.cancelled && entry.callback) {
                entry.callback();
            }
        }
    }

    // 获取最近定时器的剩余时间
    std::chrono::milliseconds next_timeout() const {
        if (heap_.empty()) {
            return std::chrono::milliseconds(-1);
        }
        auto now = std::chrono::steady_clock::now();
        auto diff = std::chrono::duration_cast<std::chrono::milliseconds>(
            heap_[0].expire_time - now);
        return diff.count() > 0 ? diff : std::chrono::milliseconds(0);
    }

    bool empty() const { return heap_.empty(); }

private:
    void sift_up(size_t idx) {
        while (idx > 0) {
            size_t parent = (idx - 1) / 2;
            if (heap_[idx] > heap_[parent]) break;
            std::swap(heap_[idx], heap_[parent]);
            idx = parent;
        }
    }

    void sift_down(size_t idx) {
        size_t n = heap_.size();
        while (true) {
            size_t smallest = idx;
            size_t left = 2 * idx + 1;
            size_t right = 2 * idx + 2;

            if (left < n && heap_[left] > heap_[smallest] == false &&
                heap_[smallest] > heap_[left]) {
                smallest = left;
            }
            if (right < n && heap_[right] > heap_[smallest] == false &&
                heap_[smallest] > heap_[right]) {
                smallest = right;
            }

            if (smallest == idx) break;
            std::swap(heap_[idx], heap_[smallest]);
            idx = smallest;
        }
    }

    std::vector<TimerEntry> heap_;
    TimerId next_id_;
};
```

### 3.3 时间轮定时器

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <functional>
#include <chrono>
#include <cstdint>

// 单层时间轮
class TimingWheel {
public:
    using TimerId = uint64_t;
    using Callback = std::function<void()>;

    TimingWheel(size_t slots = 60,
                std::chrono::milliseconds tick_duration =
                    std::chrono::milliseconds(1000))
        : slots_(slots), tick_duration_(tick_duration),
          current_slot_(0), next_id_(1) {}

    // 添加定时器
    TimerId add_timer(std::chrono::milliseconds delay, Callback callback) {
        size_t ticks = delay.count() / tick_duration_.count();
        if (ticks == 0) ticks = 1;

        size_t slot = (current_slot_ + ticks) % slots_;

        TimerEntry entry;
        entry.id = next_id_++;
        entry.round = ticks / slots_;  // 需要转几圈
        entry.callback = std::move(callback);

        wheels_[slot].push_back(std::move(entry));
        return next_id_ - 1;
    }

    // 推进时间轮
    void tick() {
        auto& slot = wheels_[current_slot_];

        for (auto it = slot.begin(); it != slot.end();) {
            if (it->round == 0) {
                // 定时器到期
                if (it->callback) {
                    it->callback();
                }
                it = slot.erase(it);
            } else {
                it->round--;
                ++it;
            }
        }

        current_slot_ = (current_slot_ + 1) % slots_;
    }

private:
    struct TimerEntry {
        TimerId id;
        size_t round;     // 剩余圈数
        Callback callback;
    };

    size_t slots_;
    std::chrono::milliseconds tick_duration_;
    size_t current_slot_;
    TimerId next_id_;
    std::vector<std::list<TimerEntry>> wheels_{slots_};
};

// 分层时间轮（更精确）
class HierarchicalTimingWheel {
public:
    HierarchicalTimingWheel()
        : second_wheel_(60, std::chrono::milliseconds(1000)),
          minute_wheel_(60, std::chrono::seconds(60)),
          hour_wheel_(24, std::chrono::minutes(60)) {}

    void add_timer(std::chrono::milliseconds delay,
                   TimingWheel::Callback callback) {
        auto secs = std::chrono::duration_cast<std::chrono::seconds>(delay);

        if (delay.count() < 60000) {
            // 小于1分钟，放秒级时间轮
            second_wheel_.add_timer(delay, std::move(callback));
        } else if (secs.count() < 3600) {
            // 小于1小时，放分钟级时间轮
            minute_wheel_.add_timer(
                std::chrono::duration_cast<std::chrono::milliseconds>(secs),
                [this, cb = std::move(callback)]() {
                    // 分钟到期后，降级到秒级
                    second_wheel_.add_timer(
                        std::chrono::milliseconds(0), std::move(cb));
                });
        } else {
            // 放小时级时间轮
            hour_wheel_.add_timer(
                std::chrono::duration_cast<std::chrono::milliseconds>(secs),
                [this, secs, cb = std::move(callback)]() {
                    minute_wheel_.add_timer(
                        std::chrono::seconds(secs.count() % 3600),
                        std::move(cb));
                });
        }
    }

    void tick_seconds() { second_wheel_.tick(); }
    void tick_minutes() { minute_wheel_.tick(); }
    void tick_hours() { hour_wheel_.tick(); }

private:
    TimingWheel second_wheel_;
    TimingWheel minute_wheel_;
    TimingWheel hour_wheel_;
};
```

```cpp
#include <iostream>

// 固定大小内存池
class FixedSizeMemoryPool {
public:
    FixedSizeMemoryPool(size_t block_size, size_t block_count)
        : block_size_(block_size), block_count_(block_count) {
        // 分配大块内存
        pool_ = static_cast<uint8_t*>(malloc(block_size * block_count));

        // 初始化空闲链表
        free_list_ = nullptr;
        for (size_t i = 0; i < block_count; ++i) {
            auto* block = pool_ + i * block_size;
            *reinterpret_cast<uint8_t**>(block) = free_list_;
            free_list_ = block;
        }

        free_count_ = block_count;
    }

    ~FixedSizeMemoryPool() {
        free(pool_);
    }

    // 分配一个块
    void* allocate() {
        if (free_list_ == nullptr) return nullptr;

        void* block = free_list_;
        free_list_ = *reinterpret_cast<uint8_t**>(free_list_);
        free_count_--;
        return block;
    }

    // 释放一个块
    void deallocate(void* ptr) {
        *reinterpret_cast<uint8_t**>(ptr) = free_list_;
        free_list_ = static_cast<uint8_t*>(ptr);
        free_count_++;
    }

    size_t free_count() const { return free_count_; }

private:
    size_t block_size_;
    size_t block_count_;
    uint8_t* pool_;
    uint8_t* free_list_;
    size_t free_count_;
};

// 多级内存池
class MultiLevelMemoryPool {
public:
    MultiLevelMemoryPool() {
        // 不同大小的内存池
        // 64B, 128B, 256B, 512B, 1KB, 2KB, 4KB
        size_t sizes[] = {64, 128, 256, 512, 1024, 2048, 4096};
        for (size_t size : sizes) {
            pools_.emplace_back(size, 1024);
        }
    }

    void* allocate(size_t size) {
        // 找到合适的内存池
        for (auto& pool : pools_) {
            if (size <= pool.block_size()) {
                void* ptr = pool.allocate();
                if (ptr) return ptr;
            }
        }

        // 超出内存池范围，使用系统分配
        return malloc(size);
    }

    void deallocate(void* ptr, size_t size) {
        for (auto& pool : pools_) {
            if (size <= pool.block_size()) {
                pool.deallocate(ptr);
                return;
            }
        }

        free(ptr);
    }

private:
    struct PoolInfo {
        size_t block_size_;
        FixedSizeMemoryPool pool_;

        PoolInfo(size_t block_size, size_t count)
            : block_size_(block_size), pool_(block_size, count) {}

        size_t block_size() const { return block_size_; }
        void* allocate() { return pool_.allocate(); }
        void deallocate(void* ptr) { pool_.deallocate(ptr); }
    };

    std::vector<PoolInfo> pools_;
};
```

```cpp
#include <iostream>
#include <vector>
#include <memory>
#include <functional>
#include <mutex>

// 对象池——复用对象避免频繁构造/析构
template<typename T>
class ObjectPool {
public:
    using Deleter = std::function<void(T*)>;

    ObjectPool(size_t initial_size = 0) {
        for (size_t i = 0; i < initial_size; ++i) {
            pool_.push_back(std::make_unique<T>());
        }
    }

    // 获取一个对象
    std::unique_ptr<T, Deleter> acquire() {
        std::lock_guard<std::mutex> lock(mutex_);

        if (pool_.empty()) {
            // 池为空，创建新对象
            auto* obj = new T();
            return std::unique_ptr<T, Deleter>(obj,
                [this](T* p) { release(p); });
        }

        // 从池中取出
        auto obj = pool_.back().release();
        pool_.pop_back();

        return std::unique_ptr<T, Deleter>(obj,
            [this](T* p) { release(p); });
    }

    // 池中对象数量
    size_t size() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return pool_.size();
    }

private:
    void release(T* obj) {
        std::lock_guard<std::mutex> lock(mutex_);
        obj->reset();  // 假设T有reset方法
        pool_.push_back(std::unique_ptr<T>(obj));
    }

    mutable std::mutex mutex_;
    std::vector<std::unique_ptr<T>> pool_;
};
```

### 4.3 缓冲区管理

```cpp
#include <iostream>
#include <vector>
#include <cstring>
#include <algorithm>

// 自动增长的缓冲区
class Buffer {
public:
    Buffer(size_t initial_size = 1024)
        : buffer_(initial_size), read_index_(0), write_index_(0) {}

    // 可读字节数
    size_t readable_bytes() const {
        return write_index_ - read_index_;
    }

    // 可写字节数
    size_t writable_bytes() const {
        return buffer_.size() - write_index_;
    }

    // 读取数据
    size_t read(void* data, size_t len) {
        len = std::min(len, readable_bytes());
        memcpy(data, begin() + read_index_, len);
        read_index_ += len;
        // 如果全部读完，重置索引
        if (read_index_ == write_index_) {
            read_index_ = write_index_ = 0;
        }
        return len;
    }

    // 写入数据
    void write(const void* data, size_t len) {
        ensure_writable(len);
        memcpy(begin() + write_index_, data, len);
        write_index_ += len;
    }

    // 获取可读数据的指针
    const char* peek() const {
        return begin() + read_index_;
    }

    // 追加字符串
    void append(const std::string& str) {
        write(str.data(), str.size());
    }

    // 读取一行
    std::string read_line() {
        const char* start = begin() + read_index_;
        const char* end = begin() + write_index_;
        const char* eol = std::find(start, end, '\n');

        if (eol == end) return "";

        size_t len = eol - start + 1;
        std::string line(start, len);
        read_index_ += len;
        return line;
    }

    // 整理缓冲区（将已读部分的空间回收）
    void shrink() {
        if (read_index_ > 0) {
            std::copy(begin() + read_index_, begin() + write_index_,
                     begin());
            write_index_ -= read_index_;
            read_index_ = 0;
        }
    }

private:
    char* begin() { return buffer_.data(); }
    const char* begin() const { return buffer_.data(); }

    void ensure_writable(size_t len) {
        if (writable_bytes() < len) {
            // 需要扩容
            if (writable_bytes() + read_index_ >= len) {
                // 整理后空间足够
                shrink();
            } else {
                // 扩大缓冲区
                buffer_.resize(write_index_ + len);
            }
        }
    }

    std::vector<char> buffer_;
    size_t read_index_;
    size_t write_index_;
};
```

```cpp
class GracefulShutdown {

        signal(SIGTERM, &GracefulShutdown::signal_handler);
        signal(SIGINT, &GracefulShutdown::signal_handler);

    bool is_draining() const { draining_; }

        draining_ = true;

        // 停止接受新连接
        // close(listenfd_);

        // 设置超时
        auto deadline = std::chrono::steady_clock::now() +
                        std::chrono::seconds(30);

        // 等待现有请求完成
        while (active_connections_ > 0) {
            if (std::chrono::steady_clock::now() > deadline) {
                std::cout << "关闭超时，强制退出" << std::endl;
                break;
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }

        running_ = false;
        std::cout << "优雅关闭完成" << std::endl;
    }

    void inc_connection() { active_connections_++; }
    void dec_connection() { active_connections_--; }

private:
    static void signal_handler(int signo) {
        instance().draining_ = true;
        std::cout << "收到信号 " << signo << "，开始优雅关闭" << std::endl;
    }

    static GracefulShutdown& instance() {
        static GracefulShutdown inst;
        return inst;
    }

    std::atomic<bool> running_;
    std::atomic<bool> draining_;
    std::atomic<int> active_connections_{0};
```

```cpp
        addr.sin_family = AF_INET;

        listen(fd, 128);

    }
```

## 6. 小结

### 核心要点回顾

| 要点 | 说明 |
|------|------|
| **连接管理** | 限流、超时清理、单IP限制、半同步半异步 |
| **1:1线程模型** | 每连接一线程，简单但不可扩展 |
| **N:1线程模型** | 多连接一线程，高效但无法利用多核 |
| **M:N线程模型** | IO线程+工作线程池，生产环境首选 |
| **最小堆定时器** | O(log n)添加/删除，适合中等数量定时器 |
| **时间轮定时器** | O(1)添加/删除，适合大量定时器 |
| **内存池** | 预分配固定大小内存块，避免频繁malloc/free |
| **对象池** | 复用对象，避免频繁构造/析构 |
| **优雅关闭** | 停止接受→等待完成→关闭连接→保存状态 |
| **热更新** | fork+exec或SO_REUSEPORT，零停机更新 |

### 架构选择建议

| 规模 | 连接管理 | 线程模型 | 定时器 | 内存管理 |
|------|---------|---------|--------|---------|
| **小型(<1K)** | 简单管理 | 单Reactor | 最小堆 | 系统分配 |
| **中型(1K~10K)** | 限流+超时 | 主从Reactor | 最小堆 | 内存池 |
| **大型(10K~100K)** | 全套管理 | M:N+工作窃取 | 时间轮 | 多级内存池 |
| **超大型(100K+)** | 分布式管理 | 多进程+SO_REUSEPORT | 分层时间轮 | 自定义分配器 |
