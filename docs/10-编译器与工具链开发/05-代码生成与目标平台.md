## 1. 寄存器分配

### 1.1 寄存器分配问题

寄存器分配是编译器后端最核心的问题之一：将无限的虚拟寄存器映射到有限的物理寄存器：

```
问题：
- LLVM IR 使用无限虚拟寄存器：%0, %1, %2, ...
- 实际 CPU 只有有限物理寄存器（x86-64: 16个通用寄存器）
- 当虚拟寄存器数 > 物理寄存器数时，需要溢出（spill）到内存

目标：
- 最小化溢出次数（内存访问开销大）
- 最小化移动指令（寄存器间复制）
- 遵守调用约定（某些寄存器有特殊用途）
```

### 1.2 活跃区间分析

```cpp
#include <vector>
#include <set>
#include <map>
#include <algorithm>

// 活跃区间（Live Interval）
struct LiveInterval {
    int reg;                    // 虚拟寄存器编号
    std::vector<std::pair<int, int>> ranges;  // 活跃范围 [start, end)

    // 添加活跃范围
    void addRange(int start, int end) {
        if (ranges.empty() || ranges.back().second < start) {
            ranges.emplace_back(start, end);
        } else {
            // 合并相邻/重叠范围
            ranges.back().second = std::max(ranges.back().second, end);
        }
    }

    // 获取最早的起始位置
    int start() const {
        return ranges.empty() ? -1 : ranges.front().first;
    }

    // 获取最晚的结束位置
    int end() const {
        return ranges.empty() ? -1 : ranges.back().second;
    }

    // 检查在位置 pos 是否活跃
    bool isLiveAt(int pos) const {
        for (const auto& [s, e] : ranges) {
            if (s <= pos && pos < e) return true;
        }
        return false;
    }

    // 检查两个活跃区间是否重叠
    bool overlaps(const LiveInterval& other) const {
        for (const auto& [s1, e1] : ranges) {
            for (const auto& [s2, e2] : other.ranges) {
                if (s1 < e2 && s2 < e1) return true;
            }
        }
        return false;
    }
};

// 活跃区间分析器
class LiveIntervalAnalysis {
private:
    std::map<int, LiveInterval> intervals;  // 寄存器 → 活跃区间

public:
    // 构建活跃区间
    void buildIntervals(const std::vector<MachineInstr>& instructions) {
        // 反向遍历指令，计算活跃区间
        for (int pos = static_cast<int>(instructions.size()) - 1; pos >= 0; pos--) {
            const auto& inst = instructions[pos];

            // 定义点：结果寄存器的活跃区间开始
            if (inst.defReg >= 0) {
                intervals[inst.defReg].reg = inst.defReg;
                intervals[inst.defReg].addRange(pos, pos + 1);
            }

            // 使用点：操作数寄存器的活跃区间延伸
            for (int useReg : inst.useRegs) {
                intervals[useReg].reg = useReg;
                intervals[useReg].addRange(0, pos + 1);
            }
        }
    }

    const std::map<int, LiveInterval>& getIntervals() const {
        return intervals;
    }
};
```

### 1.3 图着色寄存器分配

```cpp
// 图着色寄存器分配算法
class GraphColoringAllocator {
private:
    int numPhysicalRegs;  // 物理寄存器数量

    // 干扰图（Interference Graph）
    // 两个虚拟寄存器如果在同一时刻活跃，则存在干扰边
    std::map<int, std::set<int>> interferenceGraph;

    // 构建干扰图
    void buildInterferenceGraph(
        const std::map<int, LiveInterval>& intervals) {
        for (const auto& [reg1, interval1] : intervals) {
            for (const auto& [reg2, interval2] : intervals) {
                if (reg1 < reg2 && interval1.overlaps(interval2)) {
                    interferenceGraph[reg1].insert(reg2);
                    interferenceGraph[reg2].insert(reg1);
                }
            }
        }
    }

public:
    GraphColoringAllocator(int numRegs) : numPhysicalRegs(numRegs) {}

    // 执行寄存器分配
    std::map<int, int> allocate(
        const std::map<int, LiveInterval>& intervals) {
        // 1. 构建干扰图
        buildInterferenceGraph(intervals);

        // 2. 简化（Simplify）：移除度数 < K 的节点
        std::vector<int> stack;
        auto workGraph = interferenceGraph;

        bool changed = true;
        while (changed) {
            changed = false;
            for (auto it = workGraph.begin(); it != workGraph.end(); ) {
                if (static_cast<int>(it->second.size()) < numPhysicalRegs) {
                    // 度数 < K，可以安全移除
                    stack.push_back(it->first);
                    // 从邻居的邻接表中移除该节点
                    for (int neighbor : it->second) {
                        workGraph[neighbor].erase(it->first);
                    }
                    it = workGraph.erase(it);
                    changed = true;
                } else {
                    ++it;
                }
            }
        }

        // 3. 选择溢出（Spill）：如果图中还有节点
        std::set<int> spilledRegs;
        for (const auto& [reg, neighbors] : workGraph) {
            // 选择溢出代价最小的寄存器
            spilledRegs.insert(reg);
            stack.push_back(reg);
        }

        // 4. 选择（Select）：从栈中弹出节点，分配颜色
        std::map<int, int> assignment;  // 虚拟寄存器 → 物理寄存器

        while (!stack.empty()) {
            int reg = stack.back();
            stack.pop_back();

            if (spilledRegs.count(reg)) {
                // 需要溢出：分配栈空间
                assignment[reg] = -1;  // -1 表示溢出到栈
                continue;
            }

            // 找到可用的颜色（物理寄存器）
            std::set<int> usedColors;
            for (int neighbor : interferenceGraph[reg]) {
                auto it = assignment.find(neighbor);
                if (it != assignment.end() && it->second >= 0) {
                    usedColors.insert(it->second);
                }
            }

            // 分配第一个可用的物理寄存器
            for (int color = 0; color < numPhysicalRegs; color++) {
                if (usedColors.find(color) == usedColors.end()) {
                    assignment[reg] = color;
                    break;
                }
            }
        }

        return assignment;
    }
};
```

### 1.4 线性扫描寄存器分配

```cpp
// 线性扫描寄存器分配（编译速度更快，适合 JIT）
class LinearScanAllocator {
private:
    int numPhysicalRegs;

public:
    LinearScanAllocator(int numRegs) : numPhysicalRegs(numRegs) {}

    std::map<int, int> allocate(std::vector<LiveInterval>& intervals) {
        // 1. 按活跃区间起始位置排序
        std::sort(intervals.begin(), intervals.end(),
                 [](const LiveInterval& a, const LiveInterval& b) {
                     return a.start() < b.start();
                 });

        std::map<int, int> assignment;
        std::vector<int> active;  // 当前活跃的区间（按结束位置排序）
        std::set<int> freeRegs;
        for (int i = 0; i < numPhysicalRegs; i++) freeRegs.insert(i);

        for (auto& interval : intervals) {
            // 2. 过期：移除已结束的活跃区间
            expireOldIntervals(interval, active, freeRegs, assignment);

            // 3. 如果没有空闲寄存器，溢出
            if (freeRegs.empty()) {
                spillAtInterval(interval, active, assignment);
            } else {
                // 4. 分配空闲寄存器
                int reg = *freeRegs.begin();
                freeRegs.erase(freeRegs.begin());
                assignment[interval.reg] = reg;
                active.push_back(interval.reg);
                // 保持 active 按结束位置排序
                std::sort(active.begin(), active.end(),
                         [&intervals](int a, int b) {
                             return intervals[a].end() < intervals[b].end();
                         });
            }
        }

        return assignment;
    }

private:
    void expireOldIntervals(const LiveInterval& current,
                          std::vector<int>& active,
                          std::set<int>& freeRegs,
                          const std::map<int, int>& assignment) {
        for (auto it = active.begin(); it != active.end(); ) {
            // 如果活跃区间已结束
            // ...（比较结束位置与当前区间的起始位置）
            if (false /* interval.end() <= current.start() */) {
                freeRegs.insert(assignment.at(*it));
                it = active.erase(it);
            } else {
                break;  // active 已排序，后面的更晚结束
            }
        }
    }

    void spillAtInterval(LiveInterval& interval,
                        std::vector<int>& active,
                        std::map<int, int>& assignment) {
        // 溢出结束位置最晚的活跃区间
        int spill = active.back();
        if (/* intervals[spill].end() > interval.end() */ true) {
            assignment[interval.reg] = assignment[spill];
            assignment[spill] = -1;  // 溢出到栈
            active.pop_back();
            active.push_back(interval.reg);
        } else {
            assignment[interval.reg] = -1;  // 溢出当前区间
        }
    }
};
```

## 2. 指令选择

### 2.1 指令选择概述

指令选择将 IR 操作映射为目标机器的指令：

```
指令选择的方法：
1. 模式匹配（Macro Expansion）
   - 简单的 IR → 机器指令映射
   - 适用于简单的 RISC 架构

2. 树模式匹配（Tree Pattern Matching）
   - 将 IR 表示为树，匹配指令模式
   - 使用动态规划选择最优模式
   - LLVM 的 SelectionDAG 使用此方法

3. DAG 模式匹配（DAG Pattern Matching）
   - 处理公共子表达式
   - LLVM 的 GlobalISel 使用此方法

示例：
IR: %r = add i32 %a, %b

x86-64:  add eax, ebx       ; 2操作数指令
ARM:     add r0, r1, r2     ; 3操作数指令
RISC-V:  add a0, a1, a2     ; 3操作数指令
```

### 2.2 LLVM SelectionDAG

```
LLVM SelectionDAG 指令选择流程：

IR → SelectionDAG → 合法化 → 指令选择 → DAG调度 → MachineInstr

1. 构建 SelectionDAG：
   将 LLVM IR 转换为有向无环图（DAG）

2. 类型合法化：
   将不支持的类型转换为支持的类型
   例：i128 → 两个 i64

3. 操作合法化：
   将不支持的操作转换为支持的操作
   例：rotl → shl + shr + or

4. 指令选择：
   使用模式匹配选择最优的机器指令

5. DAG 调度：
   将 DAG 线性化为指令序列
```

### 2.3 指令选择示例

```llvm
; LLVM IR
define i32 @add_and_shift(i32 %a, i32 %b) {
  %sum = add i32 %a, %b
  %result = shl i32 %sum, 2
  ret i32 %result
}

; SelectionDAG（简化表示）：
;     shl
;    /   \
;  add    2
;  / \
; a   b

; x86-64 机器码选择：
; lea eax, [rdi + rsi]    ; add → lea（更高效）
; shl eax, 2              ; shl 指令

; ARM 机器码选择：
; add r0, r0, r1          ; add 指令
; lsl r0, r0, #2          ; 逻辑左移

; RISC-V 机器码选择：
; add a0, a0, a1          ; add 指令
; slli a0, a0, 2          ; 立即数左移
```

## 3. 指令调度

### 3.1 指令调度的目标

指令调度重排指令顺序以提高执行效率：

```
指令调度的三个层次：

1. 前调度（Pre-RA Scheduling）
   - 在寄存器分配之前
   - 减少关键路径长度
   - 减少寄存器压力

2. 寄存器分配
   - 插入溢出/重加载指令

3. 后调度（Post-RA Scheduling）
   - 在寄存器分配之后
   - 避免流水线停顿
   - 利用指令级并行

流水线停顿示例：
add r1, r2, r3   ; 结果写入 r1
sub r4, r1, r5   ; 需要 r1，但 r1 还没写完 → 停顿！

优化后：
add r1, r2, r3   ; 结果写入 r1
mul r6, r7, r8   ; 不依赖 r1，可以执行
sub r4, r1, r5   ; 此时 r1 已就绪
```

### 3.2 列表调度算法

```cpp
#include <vector>
#include <queue>
#include <set>
#include <map>
#include <algorithm>

// 调度单元（SUnit）
struct SUnit {
    int id;
    std::vector<int> successors;    // 后继节点
    std::vector<int> predecessors;  // 前驱节点
    int latency;                     // 指令延迟
    int priority;                    // 调度优先级

    // 计算优先级：到出口的最长路径
    int computeHeight(const std::vector<SUnit>& units,
                     std::map<int, int>& cache) {
        if (cache.count(id)) return cache[id];
        int height = latency;
        for (int succ : successors) {
            height = std::max(height,
                             latency + units[succ].computeHeight(units, cache));
        }
        cache[id] = height;
        return height;
    }
};

// 列表调度器
class ListScheduler {
private:
    std::vector<SUnit> units;

public:
    ListScheduler(const std::vector<SUnit>& su) : units(su) {}

    // 执行调度
    std::vector<int> schedule() {
        std::vector<int> result;
        std::set<int> scheduled;
        std::set<int> ready;  // 就绪集合（前驱已全部调度）
        std::map<int, int> heightCache;

        // 计算所有节点的优先级
        for (auto& unit : units) {
            unit.priority = unit.computeHeight(units, heightCache);
        }

        // 初始化就绪集合
        for (const auto& unit : units) {
            if (unit.predecessors.empty()) {
                ready.insert(unit.id);
            }
        }

        // 贪心调度
        while (!ready.empty()) {
            // 选择优先级最高的就绪节点
            int best = *std::max_element(ready.begin(), ready.end(),
                [&](int a, int b) {
                    return units[a].priority < units[b].priority;
                });

            // 调度该节点
            result.push_back(best);
            scheduled.insert(best);
            ready.erase(best);

            // 更新就绪集合
            for (int succ : units[best].successors) {
                // 检查 succ 的所有前驱是否已调度
                bool allScheduled = true;
                for (int pred : units[succ].predecessors) {
                    if (scheduled.find(pred) == scheduled.end()) {
                        allScheduled = false;
                        break;
                    }
                }
                if (allScheduled) {
                    ready.insert(succ);
                }
            }
        }

        return result;
    }
};
```

## 4. 栈帧布局

### 4.1 栈帧结构

```
x86-64 栈帧布局（System V ABI）：

高地址
┌─────────────────────────────────┐
│  调用者的栈帧                    │
├─────────────────────────────────┤
│  返回地址（由 call 指令压入）    │  ← RBP + 8
├─────────────────────────────────┤
│  保存的 RBP（旧帧指针）         │  ← RBP 指向这里
├─────────────────────────────────┤
│  局部变量 1                     │  ← RBP - 8
├─────────────────────────────────┤
│  局部变量 2                     │  ← RBP - 16
├─────────────────────────────────┤
│  溢出的寄存器（spill slots）    │  ← RBP - 24
├─────────────────────────────────┤
│  对齐填充                       │
├─────────────────────────────────┤
│  参数 7（如果参数 > 6 个）      │  ← RBP + 16
├─────────────────────────────────┤
│  参数 8                         │  ← RBP + 24
└─────────────────────────────────┘
低地址
                                  ← RSP 指向这里
```

### 4.2 函数序言和结语

```asm
; x86-64 函数序言（Prologue）
; 建立栈帧
push rbp            ; 保存旧帧指针
mov rbp, rsp        ; 设置新帧指针
sub rsp, 32         ; 分配栈空间（局部变量 + 对齐）

; 函数体
; ...

; x86-64 函数结语（Epilogue）
; 恢复栈帧并返回
mov rsp, rbp        ; 恢复栈指针
pop rbp             ; 恢复旧帧指针
ret                 ; 返回

; 优化版本（省略帧指针，-fomit-frame-pointer）
sub rsp, 32         ; 直接调整栈指针
; ...
add rsp, 32         ; 恢复栈指针
ret
```

### 4.3 栈帧布局代码

```cpp
// 栈帧布局管理器
class StackFrameLayout {
private:
    int localAreaOffset;    // 局部变量区起始偏移
    int currentOffset;      // 当前偏移
    int spillAreaOffset;    // 溢出区起始偏移
    int totalFrameSize;     // 总栈帧大小
    int alignment;          // 栈对齐要求

    struct StackSlot {
        std::string name;
        int offset;
        int size;
        int alignment;
    };

    std::vector<StackSlot> slots;

public:
    StackFrameLayout(int align = 16)
        : localAreaOffset(0), currentOffset(0),
          spillAreaOffset(0), totalFrameSize(0), alignment(align) {}

    // 分配局部变量空间
    int allocateLocal(const std::string& name, int size, int align) {
        // 对齐当前偏移
        currentOffset = alignTo(currentOffset + size, align);

        slots.push_back({name, -currentOffset, size, align});
        return -currentOffset;  // 返回相对于 RBP 的偏移
    }

    // 分配溢出空间
    int allocateSpillSlot(int size, int align) {
        currentOffset = alignTo(currentOffset + size, align);
        return -currentOffset;
    }

    // 计算最终栈帧大小
    int finalize() {
        // 加上返回地址和保存的 RBP
        totalFrameSize = alignTo(currentOffset + 16, alignment);
        return totalFrameSize;
    }

    // 获取变量在栈帧中的偏移
    int getOffset(const std::string& name) const {
        for (const auto& slot : slots) {
            if (slot.name == name) return slot.offset;
        }
        return 0;
    }

    // 打印栈帧布局
    void print() const {
        std::cout << "=== 栈帧布局 ===\n";
        std::cout << "总大小: " << totalFrameSize << " 字节\n";
        for (const auto& slot : slots) {
            std::cout << "  " << slot.name
                     << ": offset=" << slot.offset
                     << ", size=" << slot.size << "\n";
        }
    }

private:
    int alignTo(int offset, int align) {
        return (offset + align - 1) & ~(align - 1);
    }
};
```

## 5. ABI约束

### 5.1 常见 ABI 对比

```
┌──────────────────────────────────────────────────────────────┐
│              x86-64 调用约定对比                              │
├──────────────────┬───────────────────┬───────────────────────┤
│     特性          │  System V (Linux) │  Microsoft (Windows)  │
├──────────────────┼───────────────────┼───────────────────────┤
│ 整数参数寄存器    │ rdi,rsi,rdx,rcx,  │ rcx,rdx,r8,r9        │
│                  │ r8,r9             │                       │
│ 浮点参数寄存器    │ xmm0-xmm7        │ xmm0-xmm3            │
│ 返回值寄存器      │ rax, rdx         │ rax                   │
│ 被调用者保存      │ rbx,rbp,r12-r15  │ rbx,rbp,rdi,rsi,     │
│                  │                   │ r12-r15               │
│ 调用者保存        │ rax,rcx,rdx,rsi, │ rax,rcx,rdx,r8-r11   │
│                  │ r8-r11            │                       │
│ 栈对齐           │ 16字节            │ 16字节                │
│ 栈清理           │ 调用者            │ 调用者                │

```

```cpp
uint32_t readBigEndian(const uint8_t* data) {

           (static_cast<uint32_t>(data[2]) << 8)  |
            static_cast<uint32_t>(data[3]);
```

## 7. JIT编译基础

### 7.1 JIT 编译原理

```
JIT（Just-In-Time）编译：在程序运行时将代码编译为机器码

JIT 的优势：
1. 运行时信息：可以利用运行时类型信息进行优化
2. 动态代码：可以在运行时生成和执行代码
3. 热点优化：只优化频繁执行的代码路径

JIT 的挑战：
1. 编译延迟：运行时编译需要时间
2. 内存开销：需要存储生成的机器码
3. 代码缓存：需要管理编译后的代码缓存
4. 安全性：需要确保生成的代码是安全的

JIT 的典型应用：
- JavaScript 引擎（V8, SpiderMonkey）
- Java 虚拟机（HotSpot）
- .NET 运行时（CoreCLR）
- Python 加速器（PyPy, Numba）
- 数据库查询引擎
```

### 7.2 简单 JIT 实现

```cpp
#include <cstdint>
#include <cstring>
#include <sys/mman.h>
#include <vector>

// 简单的 x86-64 JIT 编译器
class SimpleJIT {
private:
    std::vector<uint8_t> codeBuffer;

    // 发射单字节
    void emitByte(uint8_t byte) {
        codeBuffer.push_back(byte);
    }

    // 发射多字节
    void emitBytes(const std::vector<uint8_t>& bytes) {
        codeBuffer.insert(codeBuffer.end(), bytes.begin(), bytes.end());
    }

    // 发射 32 位立即数（小端序）
    void emitInt32(int32_t value) {
        emitByte(value & 0xFF);
        emitByte((value >> 8) & 0xFF);
        emitByte((value >> 16) & 0xFF);
        emitByte((value >> 24) & 0xFF);
    }

public:
    // 生成一个返回常数的函数
    // 对应 C 代码：int func() { return 42; }
    void emitReturnConstant(int32_t value) {
        // mov eax, value
        emitByte(0xB8);          // mov eax, imm32 的操作码
        emitInt32(value);        // 立即数
        // ret
        emitByte(0xC3);          // ret 的操作码
    }

    // 生成加法函数
    // 对应 C 代码：int add(int a, int b) { return a + b; }
    void emitAddFunction() {
        // mov eax, edi   ; 第一个参数（System V ABI: edi）
        emitBytes({0x89, 0xF8});  // mov eax, edi
        // add eax, esi   ; 加第二个参数（esi）
        emitBytes({0x01, 0xF0});  // add eax, esi
        // ret
        emitByte(0xC3);
    }

    // 生成斐波那契函数
    // int fib(int n) { if (n <= 1) return n; return fib(n-1) + fib(n-2); }
    void emitFibFunction() {
        // 这是一个更复杂的例子，需要条件跳转和递归调用
        // 简化处理：只生成迭代版本

        // fib(int n):
        //   if n <= 1: return n
        //   a = 0, b = 1
        //   for i = 2 to n:
        //     temp = a + b
        //     a = b
        //     b = temp
        //   return b

        // 使用 eax 作为返回值，edi 作为参数 n
        // ecx 作为循环计数器，r8d 作为 a，r9d 作为 b

        // cmp edi, 1
        emitBytes({0x83, 0xFF, 0x01});
        // jle .base_case
        emitBytes({0x7E});  // 占位，后面填偏移
        size_t jlePos = codeBuffer.size();
        emitByte(0x00);     // 占位

        // mov r8d, 0    ; a = 0
        emitBytes({0x41, 0xB8, 0x00, 0x00, 0x00, 0x00});
        // mov r9d, 1    ; b = 1
        emitBytes({0x41, 0xB9, 0x01, 0x00, 0x00, 0x00});
        // mov ecx, 2    ; i = 2
        emitBytes({0xB9, 0x02, 0x00, 0x00, 0x00});

        // .loop:
        size_t loopStart = codeBuffer.size();

        // mov eax, r8d
        emitBytes({0x44, 0x89, 0xC0});
        // add eax, r9d  ; temp = a + b
        emitBytes({0x44, 0x01, 0xC8});
        // mov r8d, r9d  ; a = b
        emitBytes({0x45, 0x89, 0xC8});
        // mov r9d, eax  ; b = temp
        emitBytes({0x41, 0x89, 0xC1});
        // inc ecx       ; i++
        emitBytes({0xFF, 0xC1});
        // cmp ecx, edi  ; i <= n?
        emitBytes({0x39, 0xF9});
        // jle .loop
        emitBytes({0x7E});
        int8_t loopOffset = static_cast<int8_t>(loopStart - codeBuffer.size() - 1);
        emitByte(static_cast<uint8_t>(loopOffset));

        // mov eax, r9d  ; return b
        emitBytes({0x44, 0x89, 0xC8});
        // ret
        emitByte(0xC3);

        // 回填 jle 偏移（跳到 mov eax, r9d）
        size_t retPos = codeBuffer.size() - 2;  // mov eax, r9d 的位置
        codeBuffer[jlePos] = static_cast<uint8_t>(retPos - jlePos - 1);
    }

    // 将生成的代码标记为可执行并获取函数指针
    using FuncType = int(*)();

    FuncType getFunctionPointer() {
        // 分配可执行内存
        void* execMem = mmap(nullptr, codeBuffer.size(),
                            PROT_READ | PROT_WRITE | PROT_EXEC,
                            MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
        if (execMem == MAP_FAILED) return nullptr;

        // 复制代码到可执行内存
        memcpy(execMem, codeBuffer.data(), codeBuffer.size());

        return reinterpret_cast<FuncType>(execMem);
    }

    // 释放可执行内存
    static void releaseFunctionPointer(FuncType func) {
        // 注意：这里简化了，实际需要跟踪分配大小
        munmap(reinterpret_cast<void*>(func), 4096);
    }
};

// 使用示例
int main() {
    SimpleJIT jit;

    // 生成加法函数
    jit.emitAddFunction();
    auto addFunc = jit.getFunctionPointer();
    std::cout << "3 + 4 = " << addFunc(3, 4) << std::endl;

    return 0;
}
```

## 8. LLVM JIT (ORC)

### 8.1 LLVM ORC JIT 概述

```
LLVM ORC JIT 是 LLVM 提供的现代 JIT 编译框架：

ORC 的演进：
- LLVM MCJIT（旧版，已废弃）
- LLVM ORC v1（基础版）
- LLVM ORC v2（当前推荐）

ORC v2 的核心概念：
1. JITDylib：动态库，存储编译后的代码和符号
2. ResourceTracker：跟踪资源，支持卸载代码
3. CompileOnDemandLayer：按需编译
4. IndirectStubsManager：支持热替换

ORC JIT 的分层架构：
┌──────────────────────────────────────┐
│  IRCompileLayer                      │  编译 IR
├──────────────────────────────────────┤
│  IRTransformLayer                    │  优化 IR
├──────────────────────────────────────┤
│  CompileOnDemandLayer               │  按需编译
├──────────────────────────────────────┤
│  ObjectLinkingLayer                 │  链接目标文件
├──────────────────────────────────────┤
│  ExecutorProcessControl             │  进程控制
└──────────────────────────────────────┘
```

### 8.2 使用 LLVM ORC JIT

```cpp
#include "llvm/ExecutionEngine/Orc/LLJIT.h"
#include "llvm/IR/IRBuilder.h"
#include "llvm/IR/LLVMContext.h"
#include "llvm/IR/Module.h"
#include "llvm/Support/TargetSelect.h"

using namespace llvm;
using namespace llvm::orc;

// 使用 ORC JIT 执行 LLVM IR
int runWithJIT() {
    // 初始化目标
    InitializeNativeTarget();
    InitializeNativeTargetAsmPrinter();

    // 创建 JIT 实例
    auto expectedJIT = LLJITBuilder().create();
    if (!expectedJIT) {
        errs() << "创建 JIT 失败: " << expectedJIT.takeError() << "\n";
        return 1;
    }
    auto& jit = *expectedJIT;

    // 创建 LLVM IR 模块
    auto context = std::make_unique<LLVMContext>();
    auto module = std::make_unique<Module>("jit_module", *context);

    // 创建函数：int add(int a, int b) { return a + b; }
    FunctionType* funcType = FunctionType::get(
        Type::getInt32Ty(*context),
        {Type::getInt32Ty(*context), Type::getInt32Ty(*context)},
        false
    );

    Function* addFunc = Function::Create(
        funcType, Function::ExternalLinkage, "add", module.get()
    );

    auto* arg0 = &*addFunc->arg_begin();
    auto* arg1 = &*(addFunc->arg_begin() + 1);
    arg0->setName("a");
    arg1->setName("b");

    BasicBlock* entry = BasicBlock::Create(*context, "entry", addFunc);
    IRBuilder<> builder(entry);
    Value* sum = builder.CreateAdd(arg0, arg1, "sum");
    builder.CreateRet(sum);

    // 将模块添加到 JIT
    auto err = jit->addIRModule(ThreadSafeModule(
        std::move(module), std::move(context)));
    if (err) {
        errs() << "添加模块失败\n";
        return 1;
    }

    // 查找函数符号
    auto sym = jit->lookup("add");
    if (!sym) {
        errs() << "查找符号失败\n";
        return 1;
    }

    // 获取函数指针并调用
    auto* addPtr = sym->toPtr<int(int, int)>();
    int result = addPtr(3, 4);

    outs() << "add(3, 4) = " << result << "\n";  // 输出: 7

    return 0;
}

// 更高级的 JIT 用法：动态编译和执行
class SimpleJITDriver {
private:
    std::unique_ptr<LLJIT> jit;
    std::unique_ptr<LLVMContext> context;

public:
    SimpleJITDriver() {
        InitializeNativeTarget();
        InitializeNativeTargetAsmPrinter();

        auto expectedJIT = LLJITBuilder().create();
        if (!expectedJIT) {
            errs() << "创建 JIT 失败\n";
            return;
        }
        jit = std::move(*expectedJIT);
        context = std::make_unique<LLVMContext>();
    }

    // 编译并执行表达式
    int evaluateExpression(const std::string& expr) {
        // 1. 解析表达式（简化处理，直接构建 IR）
        auto module = std::make_unique<Module>("expr", *context);
        IRBuilder<> builder(*context);

        FunctionType* funcType = FunctionType::get(
            Type::getInt32Ty(*context), false);
        Function* func = Function::Create(
            funcType, Function::ExternalLinkage, "expr", module.get());

        BasicBlock* entry = BasicBlock::Create(*context, "entry", func);
        builder.SetInsertPoint(entry);

        // 2. 生成 IR（简化：返回固定值）
        builder.CreateRet(ConstantInt::get(Type::getInt32Ty(*context), 42));

        // 3. 添加到 JIT
        jit->addIRModule(ThreadSafeModule(
            std::move(module), std::make_unique<LLVMContext>()));

        // 4. 查找并执行
        auto sym = jit->lookup("expr");
        if (!sym) return -1;

        auto* funcPtr = sym->toPtr<int()>();
        return funcPtr();
    }
};
```

## 9. 本章小结

### 核心要点回顾

| 技术 | 关键内容 |
|------|---------|
| 寄存器分配 | 图着色（最优）和线性扫描（快速），处理溢出和移动 |
| 指令选择 | SelectionDAG/GlobalISel，将 IR 映射为目标指令 |
| 指令调度 | 列表调度，减少流水线停顿和关键路径长度 |
| 栈帧布局 | 局部变量、溢出空间、参数传递的内存布局 |
| ABI约束 | 调用约定、名称修饰、结构体布局的平台差异 |
| 跨平台生成 | 固定大小类型、字节序处理、条件编译 |
| JIT编译 | 运行时编译，ORC JIT 分层架构 |

### 关键理解

1. **寄存器分配是后端最复杂的问题**：NP 完全问题，需要启发式算法
2. **指令选择决定了代码质量**：同一 IR 操作在不同平台有不同的最优指令序列
3. **ABI 是跨平台兼容的关键**：不同平台的调用约定差异巨大
4. **JIT 编译结合了解释和编译的优势**：运行时优化 + 动态代码生成
