## 1. 自定义链接器基础

### 1.1 链接器的工作流程

```
链接器的核心任务：

1. 符号解析（Symbol Resolution）
   - 收集所有目标文件的符号表
   - 将符号引用与符号定义关联
   - 检测未定义符号和重复定义

2. 重定位（Relocation）
   - 合并相同类型的段（.text, .data, .bss）
   - 计算每个符号的最终地址
   - 修正代码中的地址引用

3. 输出可执行文件
   - 生成 ELF/PE/Mach-O 格式
   - 设置程序入口点
   - 添加动态链接信息

流程图：
┌──────────┐  ┌──────────┐  ┌──────────┐
│ a.o      │  │ b.o      │  │ libc.a   │
│ .text    │  │ .text    │  │ printf   │
│ .data    │  │ .data    │  │ malloc   │
│ 符号表   │  │ 符号表   │  │ ...      │
└────┬─────┘  └────┬─────┘  └────┬─────┘
     │              │              │
     ▼              ▼              ▼
┌──────────────────────────────────────────┐
│              链接器                       │
│                                          │
│  1. 符号解析：合并符号表，解析引用       │
│  2. 段合并：合并 .text, .data, .bss     │
│  3. 重定位：修正地址引用                 │
│  4. 生成可执行文件                       │
└────────────────────┬─────────────────────┘
                     │
                     ▼
              ┌──────────┐
              │ a.out    │
              │ 可执行文件│
              └──────────┘
```

### 1.2 简易链接器实现

```cpp
#include <vector>
#include <string>
#include <map>
#include <cstdint>
#include <fstream>
#include <iostream>
#include <algorithm>

// 符号类型
enum class SymbolBinding {
    LOCAL,    // 局部符号（文件内可见）
    GLOBAL,   // 全局符号（跨文件可见）
    WEAK      // 弱符号（可被全局符号覆盖）
};

enum class SymbolType {
    NOTYPE,   // 未指定
    OBJECT,   // 数据对象
    FUNC,     // 函数
    SECTION,  // 段
    FILE      // 文件名
};

// 符号条目
struct Symbol {
    std::string name;         // 符号名
    SymbolBinding binding;    // 绑定类型
    SymbolType type;          // 符号类型
    uint64_t value;           // 符号值（地址或偏移）
    uint64_t size;            // 符号大小
    int sectionIndex;         // 所属段索引
    bool isDefined;           // 是否已定义
    std::string sourceFile;   // 来源文件

    Symbol() : binding(SymbolBinding::LOCAL), type(SymbolType::NOTYPE),
               value(0), size(0), sectionIndex(-1), isDefined(false) {}
};

// 段信息
struct Section {
    std::string name;              // 段名
    uint64_t virtualAddr;          // 虚拟地址
    uint64_t offset;               // 文件偏移
    uint64_t size;                 // 大小
    uint64_t alignment;            // 对齐
    std::vector<uint8_t> data;     // 段数据

    Section(const std::string& n) : name(n), virtualAddr(0),
        offset(0), size(0), alignment(1) {}
};

// 目标文件（简化表示）
struct ObjectFile {
    std::string filename;
    std::vector<Section> sections;
    std::vector<Symbol> symbols;

    // 重定位条目
    struct Relocation {
        uint64_t offset;           // 重定位位置
        std::string symbol;        // 引用的符号
        int type;                  // 重定位类型
        int sectionIndex;          // 所属段索引
        int64_t addend;            // 附加偏移
    };
    std::vector<Relocation> relocations;
};

// 简易链接器
class SimpleLinker {
private:
    std::vector<ObjectFile> objectFiles;
    std::map<std::string, Symbol> globalSymbols;  // 全局符号表
    std::vector<Section> mergedSections;           // 合并后的段

    // 段合并顺序
    const std::vector<std::string> sectionOrder = {
        ".text", ".rodata", ".data", ".bss"
    };

public:
    // 添加目标文件
    void addObjectFile(const ObjectFile& obj) {
        objectFiles.push_back(obj);
    }

    // 第一阶段：符号解析
    bool resolveSymbols() {
        for (const auto& obj : objectFiles) {
            for (const auto& sym : obj.symbols) {
                if (sym.binding == SymbolBinding::LOCAL) continue;

                auto it = globalSymbols.find(sym.name);
                if (it == globalSymbols.end()) {
                    // 新符号
                    globalSymbols[sym.name] = sym;
                } else {
                    // 符号已存在
                    auto& existing = it->second;

                    if (existing.isDefined && sym.isDefined) {
                        // 重复定义错误
                        if (existing.binding == SymbolBinding::GLOBAL &&
                            sym.binding == SymbolBinding::GLOBAL) {
                            std::cerr << "错误: 符号 '" << sym.name
                                     << "' 重复定义 ("
                                     << existing.sourceFile << " 和 "
                                     << sym.sourceFile << ")\n";
                            return false;
                        }
                        // 弱符号被强符号覆盖
                        if (existing.binding == SymbolBinding::WEAK &&
                            sym.binding == SymbolBinding::GLOBAL) {
                            globalSymbols[sym.name] = sym;
                        }
                    } else if (sym.isDefined) {
                        // 新定义覆盖未定义引用
                        globalSymbols[sym.name] = sym;
                    }
                }
            }
        }

        // 检查未定义符号
        for (const auto& [name, sym] : globalSymbols) {
            if (!sym.isDefined) {
                std::cerr << "错误: 未定义的符号 '" << name << "'\n";
                return false;
            }
        }

        return true;
    }

    // 第二阶段：段合并
    void mergeSections() {
        for (const auto& secName : sectionOrder) {
            Section merged(secName);
            uint64_t currentOffset = 0;

            for (const auto& obj : objectFiles) {
                for (const auto& sec : obj.sections) {
                    if (sec.name == secName) {
                        // 对齐
                        if (sec.alignment > 1) {
                            currentOffset = (currentOffset + sec.alignment - 1)
                                          & ~(sec.alignment - 1);
                        }

                        // 合并数据
                        merged.data.insert(merged.data.end(),
                                         sec.data.begin(), sec.data.end());
                        currentOffset += sec.size;
                    }
                }
            }

            merged.size = currentOffset;
            mergedSections.push_back(merged);
        }
    }

    // 第三阶段：重定位
    void relocate() {
        // 为每个段分配虚拟地址
        uint64_t baseAddr = 0x400000;  // 默认加载地址
        uint64_t currentAddr = baseAddr;

        for (auto& sec : mergedSections) {
            sec.virtualAddr = currentAddr;
            currentAddr += sec.size;
            // 页对齐
            currentAddr = (currentAddr + 0xFFF) & ~0xFFF;
        }

        // 处理重定位
        for (const auto& obj : objectFiles) {
            for (const auto& reloc : obj.relocations) {
                auto symIt = globalSymbols.find(reloc.symbol);
                if (symIt == globalSymbols.end()) continue;

                const auto& sym = symIt->second;
                uint64_t symAddr = sym.value;  // 符号的最终地址

                // 根据重定位类型修正地址
                switch (reloc.type) {
                    case 1: { // R_X86_64_64：绝对地址
                        uint64_t* target = reinterpret_cast<uint64_t*>(
                            mergedSections[sym.sectionIndex].data.data() +
                            reloc.offset);
                        *target = symAddr + reloc.addend;
                        break;
                    }
                    case 2: { // R_X86_64_PC32：相对地址
                        int32_t* target = reinterpret_cast<int32_t*>(
                            mergedSections[sym.sectionIndex].data.data() +
                            reloc.offset);
                        uint64_t relocAddr = mergedSections[sym.sectionIndex].virtualAddr
                                           + reloc.offset;
                        *target = static_cast<int32_t>(symAddr + reloc.addend - relocAddr);
                        break;
                    }
                    case 11: { // R_X86_64_32：32位绝对地址
                        uint32_t* target = reinterpret_cast<uint32_t*>(
                            mergedSections[sym.sectionIndex].data.data() +
                            reloc.offset);
                        *target = static_cast<uint32_t>(symAddr + reloc.addend);
                        break;
                    }
                }
            }
        }
    }

    // 执行链接
    bool link() {
        if (!resolveSymbols()) return false;
        mergeSections();
        relocate();
        return true;
    }

    // 打印链接结果
    void printResult() const {
        std::cout << "=== 链接结果 ===\n\n";

        std::cout << "--- 段布局 ---\n";
        for (const auto& sec : mergedSections) {
            std::cout << "  " << sec.name
                     << ": addr=0x" << std::hex << sec.virtualAddr
                     << ", size=" << std::dec << sec.size << "\n";
        }

        std::cout << "\n--- 全局符号表 ---\n";
        for (const auto& [name, sym] : globalSymbols) {
            std::cout << "  " << name
                     << ": addr=0x" << std::hex << sym.value
                     << ", size=" << std::dec << sym.size
                     << ", type=" << (sym.type == SymbolType::FUNC ? "FUNC" : "OBJECT")
                     << "\n";
        }
    }
};
```

### 2.2 自制二进制分析工具

```cpp
// 简易 ELF 文件解析器
#include <cstdint>
#include <fstream>
#include <vector>
#include <string>
#include <iostream>
#include <iomanip>

// ELF 头结构（64位）
struct Elf64_Ehdr {
    uint8_t  e_ident[16];    // 魔数和标识信息
    uint16_t e_type;         // 文件类型
    uint16_t e_machine;      // 机器架构
    uint32_t e_version;      // 版本
    uint64_t e_entry;        // 入口点地址
    uint64_t e_phoff;        // 程序头表偏移
    uint64_t e_shoff;        // 节头表偏移
    uint32_t e_flags;        // 处理器特定标志
    uint16_t e_ehsize;       // ELF头大小
    uint16_t e_phentsize;    // 程序头表项大小
    uint16_t e_phnum;        // 程序头表项数
    uint16_t e_shentsize;    // 节头表项大小
    uint16_t e_shnum;        // 节头表项数
    uint16_t e_shstrndx;     // 节名字符串表索引
};

// 节头结构（64位）
struct Elf64_Shdr {
    uint32_t sh_name;        // 节名偏移
    uint32_t sh_type;        // 节类型
    uint64_t sh_flags;       // 节标志
    uint64_t sh_addr;        // 虚拟地址
    uint64_t sh_offset;      // 文件偏移
    uint64_t sh_size;        // 节大小
    uint32_t sh_link;        // 链接信息
    uint32_t sh_info;        // 附加信息
    uint64_t sh_addralign;   // 对齐
    uint64_t sh_entsize;     // 表项大小
};

// 符号表项（64位）
struct Elf64_Sym {
    uint32_t st_name;        // 符号名偏移
    uint8_t  st_info;        // 类型和绑定
    uint8_t  st_other;       // 可见性
    uint16_t st_shndx;       // 所属节索引
    uint64_t st_value;       // 符号值
    uint64_t st_size;        // 符号大小
};

// ELF 解析器
class ELFParser {
private:
    std::vector<uint8_t> data;
    Elf64_Ehdr header;
    std::vector<Elf64_Shdr> sections;
    std::string sectionNameTable;

    // 读取数据
    template<typename T>
    T read(size_t offset) const {
        T result;
        memcpy(&result, data.data() + offset, sizeof(T));
        return result;
    }

    // 获取节名
    std::string getSectionName(uint32_t nameOffset) const {
        size_t start = nameOffset;
        while (start < sectionNameTable.size() && sectionNameTable[start] != '\0') {
            start++;
        }
        return sectionNameTable.substr(nameOffset, start - nameOffset);
    }

public:
    bool parse(const std::string& filename) {
        // 读取文件
        std::ifstream file(filename, std::ios::binary);
        if (!file) {
            std::cerr << "无法打开文件: " << filename << "\n";
            return false;
        }
        data.assign(std::istreambuf_iterator<char>(file),
                    std::istreambuf_iterator<char>());

        // 检查 ELF 魔数
        if (data[0] != 0x7F || data[1] != 'E' ||
            data[2] != 'L'  || data[3] != 'F') {
            std::cerr << "不是 ELF 文件\n";
            return false;
        }

        // 读取 ELF 头
        header = read<Elf64_Ehdr>(0);

        // 读取节头表
        for (int i = 0; i < header.e_shnum; i++) {
            sections.push_back(read<Elf64_Shdr>(
                header.e_shoff + i * header.e_shentsize));
        }

        // 读取节名字符串表
        if (header.e_shstrndx < sections.size()) {
            auto& strtab = sections[header.e_shstrndx];
            sectionNameTable.assign(
                reinterpret_cast<const char*>(data.data() + strtab.sh_offset),
                strtab.sh_size);
        }

        return true;
    }

    // 打印 ELF 头信息
    void printHeader() const {
        std::cout << "=== ELF 头 ===\n";
        std::cout << "  类别: "
                  << (header.e_ident[4] == 2 ? "64位" : "32位") << "\n";
        std::cout << "  字节序: "
                  << (header.e_ident[5] == 1 ? "小端" : "大端") << "\n";

        const char* types[] = {"无", "可重定位", "可执行", "共享对象", "核心转储"};
        std::cout << "  类型: " << types[header.e_type] << "\n";

        std::cout << "  入口点: 0x" << std::hex << header.e_entry << std::dec << "\n";
        std::cout << "  程序头: 偏移=" << header.e_phoff
                  << ", 数量=" << header.e_phnum << "\n";
        std::cout << "  节头表: 偏移=" << header.e_shoff
                  << ", 数量=" << header.e_shnum << "\n";
    }

    // 打印节信息
    void printSections() const {
        std::cout << "\n=== 节头表 ===\n";
        std::cout << std::left
                  << std::setw(20) << "名称"
                  << std::setw(12) << "类型"
                  << std::setw(12) << "地址"
                  << std::setw(10) << "偏移"
                  << std::setw(10) << "大小" << "\n";
        std::cout << std::string(64, '-') << "\n";

        for (size_t i = 0; i < sections.size(); i++) {
            std::string name = getSectionName(sections[i].sh_name);
            std::cout << std::left
                     << std::setw(20) << name
                     << std::setw(12) << sections[i].sh_type
                     << "0x" << std::hex << std::setw(10) << sections[i].sh_addr
                     << std::setw(10) << sections[i].sh_offset
                     << std::setw(10) << sections[i].sh_size
                     << std::dec << "\n";
        }
    }

    // 打印符号表
    void printSymbols() const {
        // 查找符号表节
        for (const auto& sec : sections) {
            if (sec.sh_type == 2) {  // SHT_SYMTAB
                std::string name = getSectionName(sec.sh_name);
                std::cout << "\n=== 符号表 (" << name << ") ===\n";

                // 读取关联的字符串表
                auto& strtabSec = sections[sec.sh_link];
                std::string strtab(
                    reinterpret_cast<const char*>(data.data() + strtabSec.sh_offset),
                    strtabSec.sh_size);

                // 遍历符号
                size_t count = sec.sh_size / sec.sh_entsize;
                for (size_t i = 0; i < count; i++) {
                    auto sym = read<Elf64_Sym>(
                        sec.sh_offset + i * sec.sh_entsize);

                    std::string symName;
                    if (sym.st_name < strtab.size()) {
                        symName = strtab.c_str() + sym.st_name;
                    }

                    int binding = sym.st_info >> 4;
                    int type = sym.st_info & 0xF;

                    std::cout << "  " << std::setw(30) << std::left << symName
                             << " 值=0x" << std::hex << sym.st_value
                             << " 大小=" << std::dec << sym.st_size
                             << " 绑定=" << binding
                             << " 类型=" << type << "\n";
                }
            }
        }
    }
};

// 使用示例
int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cerr << "用法: " << argv[0] << " <elf文件>\n";
        return 1;
    }

    ELFParser parser;
    if (!parser.parse(argv[1])) return 1;

    parser.printHeader();
    parser.printSections();
    parser.printSymbols();

    return 0;
}
```

## 3. 代码混淆Pass

### 3.1 代码混淆技术概述

```
代码混淆的目标：增加逆向工程的难度

常见混淆技术：
┌──────────────────┬──────────────────────────────────────────┐
│ 混淆技术          │ 描述                                     │
├──────────────────┼──────────────────────────────────────────┤
│ 控制流平坦化      │ 将 if/switch/loop 转为 switch-dispatch   │
│ 虚假控制流        │ 插入永远不会执行的基本块                  │
│ 指令替换          │ 用等价但更复杂的指令序列替换简单指令      │
│ 不透明谓词        │ 插入结果已知的复杂条件判断                │
│ 字符串加密        │ 加密字符串常量，运行时解密                │
│ 符号剥离          │ 去除函数名和变量名                        │
│ 代码虚拟化        │ 将代码转为自定义字节码，用虚拟机执行      │
└──────────────────┴──────────────────────────────────────────┘
```

### 3.2 LLVM 混淆 Pass 实现

```cpp
// LLVM Pass：控制流平坦化
// 将函数的控制流转为 switch-dispatch 结构

#include "llvm/IR/Function.h"
#include "llvm/IR/IRBuilder.h"
#include "llvm/IR/Instructions.h"
#include "llvm/Pass.h"
#include "llvm/Passes/PassBuilder.h"
#include "llvm/Passes/PassPlugin.h"
#include "llvm/Support/raw_ostream.h"

using namespace llvm;

// 控制流平坦化 Pass
struct ControlFlowFlatteningPass : public PassInfoMixin<ControlFlowFlatteningPass> {
    PreservedAnalyses run(Function& F, FunctionAnalysisManager& AM) {
        // 只混淆有多个基本块的函数
        if (F.size() <= 1) return PreservedAnalyses::all();

        // 1. 收集所有基本块（除了入口块）
        std::vector<BasicBlock*> originalBlocks;
        for (auto& BB : F) {
            if (&BB != &F.getEntryBlock()) {
                originalBlocks.push_back(&BB);
            }
        }

        // 2. 为每个基本块分配一个随机 ID
        std::map<BasicBlock*, int> blockIDs;
        int nextID = 1;
        for (auto* BB : originalBlocks) {
            blockIDs[BB] = nextID++;
        }

        // 3. 创建 dispatch 基本块
        LLVMContext& ctx = F.getContext();
        BasicBlock* entry = &F.getEntryBlock();
        BasicBlock* dispatch = BasicBlock::Create(ctx, "dispatch", &F);

        // 4. 创建状态变量
        IRBuilder<> builder(entry);
        AllocaInst* stateVar = builder.CreateAlloca(
            Type::getInt32Ty(ctx), nullptr, "state");
        builder.CreateStore(ConstantInt::get(Type::getInt32Ty(ctx), 0),
                          stateVar);

        // 5. 在 dispatch 块中创建 switch
        builder.SetInsertPoint(dispatch);
        LoadInst* currentState = builder.CreateLoad(
            Type::getInt32Ty(ctx), stateVar, "current_state");
        SwitchInst* switchInst = builder.CreateSwitch(
            currentState, dispatch, originalBlocks.size());

        // 6. 为每个原始块创建 case
        for (auto* BB : originalBlocks) {
            BasicBlock* caseBlock = BasicBlock::Create(ctx, "case_" +
                std::to_string(blockIDs[BB]), &F);
            switchInst->addCase(
                ConstantInt::get(Type::getInt32Ty(ctx), blockIDs[BB]),
                caseBlock);

            // 将原始块的内容移到 case 块
            builder.SetInsertPoint(caseBlock);
            // ... 移动指令
        }

        // 7. 更新入口块跳转到 dispatch
        builder.SetInsertPoint(entry->getTerminator());
        builder.CreateBr(dispatch);
        entry->getTerminator()->eraseFromParent();

        return PreservedAnalyses::none();
    }

    static bool isRequired() { return false; }
};

// LLVM Pass：虚假控制流
// 插入永远不会执行的基本块
struct BogusControlFlowPass : public PassInfoMixin<BogusControlFlowPass> {
    PreservedAnalyses run(Function& F, FunctionAnalysisManager& AM) {
        std::vector<BasicBlock*> blocks;
        for (auto& BB : F) {
            blocks.push_back(&BB);
        }

        for (auto* BB : blocks) {
            // 创建一个虚假的基本块
            LLVMContext& ctx = F.getContext();
            BasicBlock* bogusBlock = BasicBlock::Create(ctx, "bogus", &F);

            // 在虚假块中插入垃圾指令
            IRBuilder<> builder(bogusBlock);
            Value* v1 = builder.CreateAdd(
                ConstantInt::get(Type::getInt32Ty(ctx), 0),
                ConstantInt::get(Type::getInt32Ty(ctx), 0), "bogus1");
            Value* v2 = builder.CreateMul(v1,
                ConstantInt::get(Type::getInt32Ty(ctx), 1), "bogus2");
            builder.CreateBr(BB);  // 跳回原块

            // 创建不透明谓词（永远为 true 的复杂条件）
            // 例如：(x * x - x * x) == 0
            builder.SetInsertPoint(BB->getTerminator());
            Value* opaqueTrue = builder.CreateICmpEQ(
                ConstantInt::get(Type::getInt32Ty(ctx), 0),
                ConstantInt::get(Type::getInt32Ty(ctx), 0), "opaque");

            // 将原块的跳转改为条件跳转
            // if (opaqueTrue) goto original; else goto bogus;
            BasicBlock* originalSucc = BB->getTerminator()->getSuccessor(0);
            BB->getTerminator()->eraseFromParent();
            builder.SetInsertPoint(BB);
            builder.CreateCondBr(opaqueTrue, originalSucc, bogusBlock);
        }

        return PreservedAnalyses::none();
    }

    static bool isRequired() { return false; }
};

// 注册混淆 Pass
extern "C" LLVM_ATTRIBUTE_WEAK PassPluginLibraryInfo llvmGetPassPluginInfo() {
    return {
        LLVM_PLUGIN_API_VERSION, "ObfuscationPasses", "1.0",
        [](PassBuilder& PB) {
            PB.registerPipelineParsingCallback(
                [](StringRef Name, FunctionPassManager& FPM,
                   ArrayRef<PassBuilder::PipelineElement>) {
                    if (Name == "cff") {
                        FPM.addPass(ControlFlowFlatteningPass());
                        return true;
                    }
                    if (Name == "bcf") {
                        FPM.addPass(BogusControlFlowPass());
                        return true;
                    }
                    return false;
                }
            );
        }
    };
}
```

## 4. 插桩Pass

### 4.1 插桩技术概述

```
插桩（Instrumentation）：在代码中注入检测逻辑

常见插桩类型：
1. 代码覆盖率插桩
   - 记录哪些基本块被执行
   - 用于测试覆盖率分析

2. 性能分析插桩
   - 记录函数调用次数和时间
   - 用于性能剖析

3. 内存检测插桩
   - 检测内存泄漏、越界访问
   - 如 AddressSanitizer

4. 数据竞争检测插桩
   - 检测多线程数据竞争
   - 如 ThreadSanitizer

5. 自定义插桩
   - 根据需求注入特定逻辑
   - 如日志、追踪、安全检查
```

### 4.2 代码覆盖率插桩 Pass

```cpp
// LLVM Pass：代码覆盖率插桩
// 在每个基本块入口注入计数器

#include "llvm/IR/Function.h"
#include "llvm/IR/IRBuilder.h"
#include "llvm/IR/Module.h"
#include "llvm/Passes/PassBuilder.h"
#include "llvm/Passes/PassPlugin.h"

using namespace llvm;

struct CoverageInstrumentationPass : public PassInfoMixin<CoverageInstrumentationPass> {
    PreservedAnalyses run(Function& F, FunctionAnalysisManager& AM) {
        Module* M = F.getParent();
        LLVMContext& ctx = F.getContext();

        // 获取或创建全局计数器数组
        GlobalVariable* counters = M->getGlobalVariable("__coverage_counters");
        if (!counters) {
            // 第一次运行：创建计数器数组
            auto* counterType = ArrayType::get(Type::getInt64Ty(ctx), 10000);
            counters = new GlobalVariable(
                *M, counterType, false,
                GlobalValue::InternalLinkage,
                ConstantAggregateZero::get(counterType),
                "__coverage_counters");
        }

        // 获取或创建注册函数
        FunctionCallee registerFunc = M->getOrInsertFunction(
            "__coverage_register",
            FunctionType::get(Type::getVoidTy(ctx),
                {Type::getInt32Ty(ctx),    // 基本块索引
                 Type::getInt8PtrTy(ctx)}, // 函数名
                false));

        // 为每个基本块注入计数器
        int blockIndex = 0;
        for (auto& BB : F) {
            // 在基本块入口插入计数器递增
            IRBuilder<> builder(&*BB.getFirstInsertionPt());

            // 获取计数器地址
            Value* index = ConstantInt::get(Type::getInt32Ty(ctx), blockIndex);
            Value* counterPtr = builder.CreateInBoundsGEP(
                counters->getValueType(), counters,
                {ConstantInt::get(Type::getInt32Ty(ctx), 0), index});

            // 递增计数器
            LoadInst* currentVal = builder.CreateLoad(
                Type::getInt64Ty(ctx), counterPtr, "cov_load");
            Value* incremented = builder.CreateAdd(
                currentVal, ConstantInt::get(Type::getInt64Ty(ctx), 1), "cov_inc");
            builder.CreateStore(incremented, counterPtr);

            blockIndex++;
        }

        return PreservedAnalyses::none();
    }

    static bool isRequired() { return false; }
};
```

### 4.3 性能分析插桩 Pass

```cpp
// LLVM Pass：函数级性能分析插桩
// 在函数入口和出口注入计时逻辑

struct ProfilingInstrumentationPass : public PassInfoMixin<ProfilingInstrumentationPass> {
    PreservedAnalyses run(Function& F, FunctionAnalysisManager& AM) {
        Module* M = F.getParent();
        LLVMContext& ctx = F.getContext();

        // 声明外部计时函数
        FunctionCallee getTime = M->getOrInsertFunction(
            "__prof_get_time",
            FunctionType::get(Type::getInt64Ty(ctx), false));

        FunctionCallee recordFunc = M->getOrInsertFunction(
            "__prof_record",
            FunctionType::get(Type::getVoidTy(ctx),
                {Type::getInt8PtrTy(ctx),  // 函数名
                 Type::getInt64Ty(ctx)},   // 耗时
                false));

        // 在入口插入开始计时
        BasicBlock& entry = F.getEntryBlock();
        IRBuilder<> builder(&*entry.getFirstInsertionPt());

        // 调用 __prof_get_time()
        Value* startTime = builder.CreateCall(getTime, "prof_start");

        // 创建栈变量保存开始时间
        AllocaInst* startTimeAlloca = builder.CreateAlloca(
            Type::getInt64Ty(ctx), nullptr, "prof_start_time");
        builder.CreateStore(startTime, startTimeAlloca);

        // 在每个返回点插入结束计时和记录
        for (auto& BB : F) {
            if (auto* ret = dyn_cast<ReturnInst>(BB.getTerminator())) {
                builder.SetInsertPoint(ret);

                // 调用 __prof_get_time()
                Value* endTime = builder.CreateCall(getTime, "prof_end");

                // 计算耗时
                Value* startTimeVal = builder.CreateLoad(
                    Type::getInt64Ty(ctx), startTimeAlloca);
                Value* elapsed = builder.CreateSub(endTime, startTimeVal, "prof_elapsed");

                // 调用 __prof_record(funcName, elapsed)
                Value* funcName = builder.CreateGlobalStringPtr(F.getName());
                builder.CreateCall(recordFunc, {funcName, elapsed});
            }
        }

        return PreservedAnalyses::none();
    }

    static bool isRequired() { return false; }
};
```

```cpp
struct ProfileRecord {
    std::string funcName;
    uint64_t totalTime;
    uint64_t callCount;
};

static std::map<std::string, ProfileRecord> profileData;

// 获取当前时间（纳秒）
extern "C" uint64_t __prof_get_time() {
    auto now = std::chrono::high_resolution_clock::now();
    return std::chrono::duration_cast<std::chrono::nanoseconds>(
        now.time_since_epoch()).count();
}

// 记录函数耗时
extern "C" void __prof_record(const char* funcName, uint64_t elapsed) {
    auto& record = profileData[funcName];
    record.funcName = funcName;
    record.totalTime += elapsed;
    record.callCount++;
}

// 输出性能分析报告
extern "C" void __prof_report() {
    printf("\n=== 性能分析报告 ===\n");

    // 按总耗时排序
    std::vector<ProfileRecord> records;
    for (const auto& [name, record] : profileData) {
        records.push_back(record);
    }
    std::sort(records.begin(), records.end(),
             [](const ProfileRecord& a, const ProfileRecord& b) {
                 return a.totalTime > b.totalTime;
             });

    printf("%-30s %12s %12s %12s\n",
           "函数", "调用次数", "总耗时(ms)", "平均(μs)");
    printf("%s\n", std::string(70, '-').c_str());

    for (const auto& r : records) {
        printf("%-30s %12lu %12.3f %12.3f\n",
               r.funcName.c_str(),
               r.callCount,
               r.totalTime / 1e6,
               r.totalTime / r.callCount / 1e3);
    }
}

// 自动输出报告（atexit 注册）
static bool registered = false;
struct AutoReport {
    AutoReport() {
        if (!registered) {
            atexit([]() {
                __coverage_report();
                __prof_report();
            });
            registered = true;
        }
    }
};
```

### 6.2 编译驱动器实现

```cpp
// 简易编译驱动器
#include <iostream>
#include <string>
#include <vector>
#include <cstdlib>
#include <filesystem>

namespace fs = std::filesystem;

struct CompileOptions {
    std::vector<std::string> sourceFiles;
    std::string outputFile;
    int optLevel = 0;
    bool coverage = false;
    bool profile = false;
    bool obfuscate = false;
    std::string target;
    bool emitIR = false;
    bool emitAssembly = false;
    bool verbose = false;
};

class CompilerDriver {
private:
    CompileOptions options;
    std::string llvmConfigPath;
    std::string clangPath;

    std::string getLLVMConfigVar(const std::string& var) {
        std::string cmd = llvmConfigPath + " --" + var;
        FILE* pipe = popen(cmd.c_str(), "r");
        if (!pipe) return "";
        char buffer[256];
        std::string result;
        while (fgets(buffer, sizeof(buffer), pipe)) {
            result += buffer;
        }
        pclose(pipe);
        // 去除末尾换行
        while (!result.empty() && result.back() == '\n') {
            result.pop_back();
        }
        return result;
    }

public:
    CompilerDriver(const CompileOptions& opts) : options(opts) {
        // 查找工具路径
        llvmConfigPath = "llvm-config";
        clangPath = "clang";
    }

    // 编译单个源文件
    bool compileFile(const std::string& sourceFile) {
        std::vector<std::string> args;
        args.push_back(clangPath);

        // 优化级别
        args.push_back("-O" + std::to_string(options.optLevel));

        // 插桩选项
        if (options.coverage) {
            args.push_back("-fpass-plugin=./CoveragePass.so");
            args.push_back("-mllvm");
            args.push_back("-passes=coverage");
        }

        if (options.profile) {
            args.push_back("-fpass-plugin=./ProfilingPass.so");
            args.push_back("-mllvm");
            args.push_back("-passes=profiling");
        }

        if (options.obfuscate) {
            args.push_back("-fpass-plugin=./Obfuscation.so");
            args.push_back("-mllvm");
            args.push_back("-passes=cff,bcf");
        }

        // 目标平台
        if (!options.target.empty()) {
            args.push_back("--target=" + options.target);
        }

        // 输出类型
        if (options.emitIR) {
            args.push_back("-S");
            args.push_back("-emit-llvm");
        } else if (options.emitAssembly) {
            args.push_back("-S");
        }

        // 输入文件
        args.push_back(sourceFile);

        // 输出文件
        std::string objFile = sourceFile;
        // 替换扩展名
        auto pos = objFile.rfind('.');
        if (pos != std::string::npos) {
            objFile = objFile.substr(0, pos);
        }
        objFile += options.emitIR ? ".ll" :
                  options.emitAssembly ? ".s" : ".o";
        args.push_back("-o");
        args.push_back(objFile);

        // 构建命令
        std::string cmd;
        for (const auto& arg : args) {
            cmd += arg + " ";
        }

        if (options.verbose) {
            std::cout << "执行: " << cmd << "\n";
        }

        return system(cmd.c_str()) == 0;
    }

    // 链接目标文件
    bool linkFiles(const std::vector<std::string>& objFiles) {
        std::string cmd = clangPath;

        // 添加目标文件
        for (const auto& obj : objFiles) {
            cmd += " " + obj;
        }

        // 链接运行时库
        if (options.coverage || options.profile) {
            cmd += " -L. -lmycc_rt";
        }

        // 输出文件
        cmd += " -o " + options.outputFile;

        if (options.verbose) {
            std::cout << "链接: " << cmd << "\n";
        }

        return system(cmd.c_str()) == 0;
    }

    // 完整编译流程
    bool compile() {
        std::vector<std::string> objFiles;

        // 1. 编译每个源文件
        for (const auto& source : options.sourceFiles) {
            if (!compileFile(source)) {
                std::cerr << "编译失败: " << source << "\n";
                return false;
            }

            // 收集目标文件
            std::string objFile = source;
            auto pos = objFile.rfind('.');
            if (pos != std::string::npos) {
                objFile = objFile.substr(0, pos);
            }
            objFile += ".o";
            objFiles.push_back(objFile);
        }

        // 2. 链接
        if (!options.emitIR && !options.emitAssembly) {
            if (!linkFiles(objFiles)) {
                std::cerr << "链接失败\n";
                return false;
            }
        }

        return true;
    }
};

// 命令行解析
CompileOptions parseArgs(int argc, char* argv[]) {
    CompileOptions opts;
    for (int i = 1; i < argc; i++) {
        std::string arg = argv[i];
        if (arg == "-o" && i + 1 < argc) {
            opts.outputFile = argv[++i];
        } else if (arg == "-O0") { opts.optLevel = 0;
        } else if (arg == "-O1") { opts.optLevel = 1;
        } else if (arg == "-O2") { opts.optLevel = 2;
        } else if (arg == "-O3") { opts.optLevel = 3;
        } else if (arg == "--coverage") { opts.coverage = true;
        } else if (arg == "--profile")  { opts.profile = true;
        } else if (arg == "--obfuscate") { opts.obfuscate = true;
        } else if (arg == "--emit-ir")  { opts.emitIR = true;
        } else if (arg == "--emit-asm") { opts.emitAssembly = true;
        } else if (arg == "-v") { opts.verbose = true;
        } else if (arg.rfind("--target=", 0) == 0) {
            opts.target = arg.substr(9);
        } else if (arg.rfind("-", 0) != 0) {
            opts.sourceFiles.push_back(arg);
        }
    }
    if (opts.outputFile.empty() && !opts.sourceFiles.empty()) {
        opts.outputFile = "a.out";
    }
    return opts;
}

int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cerr << "用法: mycc [选项] 源文件...\n"
                  << "选项:\n"
                  << "  -O0/-O1/-O2/-O3  优化级别\n"
                  << "  --coverage       启用覆盖率插桩\n"
                  << "  --profile        启用性能分析插桩\n"
                  << "  --obfuscate      启用代码混淆\n"
                  << "  --emit-ir        输出 LLVM IR\n"
                  << "  --emit-asm       输出汇编\n"
                  << "  --target=TRIPLE  指定目标平台\n"
                  << "  -v               详细输出\n";
        return 1;
    }

    auto opts = parseArgs(argc, argv);
    CompilerDriver driver(opts);
    return driver.compile() ? 0 : 1;
}
```

## 7. 本章小结

### 核心要点回顾

| 技术 | 关键内容 |
|------|---------|
| 自定义链接器 | 符号解析 → 段合并 → 重定位 → 输出可执行文件 |
| 二进制工具 | objdump/readelf/nm 用于分析 ELF/PE/Mach-O 文件 |
| 代码混淆 | 控制流平坦化、虚假控制流、不透明谓词等 |
| 插桩Pass | 覆盖率插桩、性能分析插桩，配合运行时库使用 |
| 编译器测试 | 单元测试、集成测试、模糊测试、差分测试 |
| 工具链集成 | 编译驱动器 + 自定义 Pass + 运行时库 + CMake |

### 关键理解

1. **链接器是工具链的核心枢纽**：理解链接器就理解了程序从目标文件到可执行文件的完整过程
2. **二进制工具是调试的利器**：熟练使用 objdump/readelf/nm 可以快速定位链接和运行时问题
3. **LLVM Pass 是工具链扩展的标准方式**：混淆、插桩、分析都可以通过 Pass 实现
4. **工具链集成需要考虑全局**：编译驱动器、运行时库、构建系统需要协调工作
