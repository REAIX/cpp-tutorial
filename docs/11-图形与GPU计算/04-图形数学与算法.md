### 1.1 向量运算

```cpp
// 3D向量类实现
#include <cmath>
#include <algorithm>
#include <iostream>

struct Vec3 {
    float x, y, z;

    // 构造函数
    constexpr Vec3() : x(0), y(0), z(0) {}
    constexpr Vec3(float x, float y, float z) : x(x), y(y), z(z) {}

    // 基本运算
    Vec3 operator+(const Vec3& v) const { return {x + v.x, y + v.y, z + v.z}; }
    Vec3 operator-(const Vec3& v) const { return {x - v.x, y - v.y, z - v.z}; }
    Vec3 operator*(float s) const { return {x * s, y * s, z * s}; }
    Vec3 operator/(float s) const { return {x / s, y / s, z / s}; }
    Vec3 operator-() const { return {-x, -y, -z}; }
    Vec3& operator+=(const Vec3& v) { x += v.x; y += v.y; z += v.z; return *this; }
    Vec3& operator-=(const Vec3& v) { x -= v.x; y -= v.y; z -= v.z; return *this; }

    // 点积（内积）
    // 几何意义：a·b = |a||b|cos(θ)，用于计算夹角和投影
    static float dot(const Vec3& a, const Vec3& b) {
        return a.x * b.x + a.y * b.y + a.z * b.z;
    }

    // 叉积（外积）
    // 几何意义：结果向量垂直于a和b所在平面，|a×b| = |a||b|sin(θ)
    // 用于计算法线和判断方向
    static Vec3 cross(const Vec3& a, const Vec3& b) {
        return {
            a.y * b.z - a.z * b.y,
            a.z * b.x - a.x * b.z,
            a.x * b.y - a.y * b.x
        };
    }

    // 向量长度
    float length() const { return std::sqrt(x * x + y * y + z * z); }

    // 长度平方（避免开方，用于比较）
    float lengthSq() const { return x * x + y * y + z * z; }

    // 归一化
    Vec3 normalized() const {
        float len = length();
        return len > 0 ? *this / len : Vec3{};
    }

    // 距离
    static float distance(const Vec3& a, const Vec3& b) {
        return (a - b).length();
    }

    // 线性插值
    static Vec3 lerp(const Vec3& a, const Vec3& b, float t) {
        return a + (b - a) * t;
    }

    // 反射向量：v关于法线n的反射
    static Vec3 reflect(const Vec3& v, const Vec3& n) {
        return v - n * (2.0f * dot(v, n));
    }

    // 折射向量
    static Vec3 refract(const Vec3& v, const Vec3& n, float eta) {
        float cosI = -dot(v, n);
        float sinT2 = eta * eta * (1.0f - cosI * cosI);
        if (sinT2 > 1.0f) return {};  // 全反射
        float cosT = std::sqrt(1.0f - sinT2);
        return v * eta + n * (eta * cosI - cosT);
    }

    // 分量钳制
    static Vec3 clamp(const Vec3& v, const Vec3& lo, const Vec3& hi) {
        return {
            std::clamp(v.x, lo.x, hi.x),
            std::clamp(v.y, lo.y, hi.y),
            std::clamp(v.z, lo.z, hi.z)
        };
    }

    // 友元输出
    friend std::ostream& operator<<(std::ostream& os, const Vec3& v) {
        return os << "(" << v.x << ", " << v.y << ", " << v.z << ")";
    }
};

// 使用示例
void vectorExample() {
    Vec3 a(1.0f, 2.0f, 3.0f);
    Vec3 b(4.0f, 5.0f, 6.0f);

    // 基本运算
    Vec3 sum = a + b;                    // (5, 7, 9)
    Vec3 diff = b - a;                   // (3, 3, 3)
    Vec3 scaled = a * 2.0f;             // (2, 4, 6)

    // 点积和叉积
    float d = Vec3::dot(a, b);          // 32
    Vec3 c = Vec3::cross(a, b);         // (-3, 6, -3)

    // 归一化和距离
    Vec3 n = a.normalized();
    float dist = Vec3::distance(a, b);

    // 反射
    Vec3 incident(1.0f, -1.0f, 0.0f);
    Vec3 normal(0.0f, 1.0f, 0.0f);
    Vec3 reflected = Vec3::reflect(incident, normal);  // (1, 1, 0)
}
```

### 1.2 矩阵运算

```cpp
// 4x4矩阵类实现
struct Mat4 {
    float m[4][4];  // 行优先存储

    // 单位矩阵
    static constexpr Mat4 identity() {
        Mat4 result{};
        result.m[0][0] = result.m[1][1] = result.m[2][2] = result.m[3][3] = 1.0f;
        return result;
    }

    // 矩阵乘法
    Mat4 operator*(const Mat4& other) const {
        Mat4 result{};
        for (int i = 0; i < 4; ++i) {
            for (int j = 0; j < 4; ++j) {
                result.m[i][j] = 0;
                for (int k = 0; k < 4; ++k) {
                    result.m[i][j] += m[i][k] * other.m[k][j];
                }
            }
        }
        return result;
    }

    // 矩阵乘以向量（点）
    Vec3 transformPoint(const Vec3& p) const {
        float x = m[0][0] * p.x + m[0][1] * p.y + m[0][2] * p.z + m[0][3];
        float y = m[1][0] * p.x + m[1][1] * p.y + m[1][2] * p.z + m[1][3];
        float z = m[2][0] * p.x + m[2][1] * p.y + m[2][2] * p.z + m[2][3];
        float w = m[3][0] * p.x + m[3][1] * p.y + m[3][2] * p.z + m[3][3];
        if (w != 0.0f && w != 1.0f) {
            return {x / w, y / w, z / w};
        }
        return {x, y, z};
    }

    // 矩阵乘以向量（方向，忽略平移）
    Vec3 transformDirection(const Vec3& d) const {
        return {
            m[0][0] * d.x + m[0][1] * d.y + m[0][2] * d.z,
            m[1][0] * d.x + m[1][1] * d.y + m[1][2] * d.z,
            m[2][0] * d.x + m[2][1] * d.y + m[2][2] * d.z
        };
    }

    // 转置
    Mat4 transposed() const {
        Mat4 result;
        for (int i = 0; i < 4; ++i)
            for (int j = 0; j < 4; ++j)
                result.m[i][j] = m[j][i];
        return result;
    }

    // 行列式（简化版）
    float determinant() const {
        return m[0][0] * (m[1][1] * (m[2][2] * m[3][3] - m[2][3] * m[3][2]) -
                          m[1][2] * (m[2][1] * m[3][3] - m[2][3] * m[3][1]) +
                          m[1][3] * (m[2][1] * m[3][2] - m[2][2] * m[3][1]))
             - m[0][1] * (m[1][0] * (m[2][2] * m[3][3] - m[2][3] * m[3][2]) -
                          m[1][2] * (m[2][0] * m[3][3] - m[2][3] * m[3][0]) +
                          m[1][3] * (m[2][0] * m[3][2] - m[2][2] * m[3][0]))
             + m[0][2] * (m[1][0] * (m[2][1] * m[3][3] - m[2][3] * m[3][1]) -
                          m[1][1] * (m[2][0] * m[3][3] - m[2][3] * m[3][0]) +
                          m[1][3] * (m[2][0] * m[3][1] - m[2][1] * m[3][0]))
             - m[0][3] * (m[1][0] * (m[2][1] * m[3][2] - m[2][2] * m[3][1]) -
                          m[1][1] * (m[2][0] * m[3][2] - m[2][2] * m[3][0]) +
                          m[1][2] * (m[2][0] * m[3][1] - m[2][1] * m[3][0]));
    }

    // 逆矩阵（伴随矩阵法）
    Mat4 inverted() const;
};

// 变换法线需要使用逆转置矩阵
Mat4 invertTranspose(const Mat4& m);
```

### 1.3 四元数

```cpp
// 四元数实现 - 用于表示3D旋转
struct Quaternion {
    float w, x, y, z;  // w + xi + yj + zk

    // 单位四元数（无旋转）
    static constexpr Quaternion identity() {
        return {1.0f, 0.0f, 0.0f, 0.0f};
    }

    // 从轴角创建四元数
    static Quaternion fromAxisAngle(const Vec3& axis, float angle) {
        float halfAngle = angle * 0.5f;
        float s = std::sin(halfAngle);
        Vec3 n = axis.normalized();
        return {
            std::cos(halfAngle),
            n.x * s,
            n.y * s,
            n.z * s
        };
    }

    // 从欧拉角创建（YXZ顺序）
    static Quaternion fromEuler(float pitch, float yaw, float roll) {
        float cp = std::cos(pitch * 0.5f);
        float sp = std::sin(pitch * 0.5f);
        float cy = std::cos(yaw * 0.5f);
        float sy = std::sin(yaw * 0.5f);
        float cr = std::cos(roll * 0.5f);
        float sr = std::sin(roll * 0.5f);

        return {
            cr * cp * cy + sr * sp * sy,
            sr * cp * cy - cr * sp * sy,
            cr * sp * cy + sr * cp * sy,
            cr * cp * sy - sr * sp * cy
        };
    }

    // 四元数乘法（组合旋转）
    Quaternion operator*(const Quaternion& q) const {
        return {
            w * q.w - x * q.x - y * q.y - z * q.z,
            w * q.x + x * q.w + y * q.z - z * q.y,
            w * q.y - x * q.z + y * q.w + z * q.x,
            w * q.z + x * q.y - y * q.x + z * q.w
        };
    }

    // 共轭（反向旋转）
    Quaternion conjugate() const {
        return {w, -x, -y, -z};
    }

    // 模长
    float length() const {
        return std::sqrt(w * w + x * x + y * y + z * z);
    }

    // 归一化
    Quaternion normalized() const {
        float len = length();
        if (len < 1e-6f) return identity();
        float inv = 1.0f / len;
        return {w * inv, x * inv, y * inv, z * inv};
    }

    // 逆四元数
    Quaternion inverse() const {
        float lenSq = w * w + x * x + y * y + z * z;
        if (lenSq < 1e-6f) return identity();
        float inv = 1.0f / lenSq;
        return {w * inv, -x * inv, -y * inv, -z * inv};
    }

    // 旋转向量
    Vec3 rotate(const Vec3& v) const {
        // v' = q * v * q^(-1)
        Quaternion qv = {0, v.x, v.y, v.z};
        Quaternion result = (*this) * qv * inverse();
        return {result.x, result.y, result.z};
    }

    // 球面线性插值（Slerp）
    static Quaternion slerp(const Quaternion& a, const Quaternion& b, float t) {
        float dot = a.w * b.w + a.x * b.x + a.y * b.y + a.z * b.z;

        // 确保走最短路径
        Quaternion bAdj = b;
        if (dot < 0.0f) {
            bAdj = {-b.w, -b.x, -b.y, -b.z};
            dot = -dot;
        }

        // 如果四元数非常接近，使用线性插值
        if (dot > 0.9995f) {
            return Quaternion{
                a.w + t * (bAdj.w - a.w),
                a.x + t * (bAdj.x - a.x),
                a.y + t * (bAdj.y - a.y),
                a.z + t * (bAdj.z - a.z)
            }.normalized();
        }

        float theta0 = std::acos(dot);
        float theta = theta0 * t;
        float sinTheta = std::sin(theta);
        float sinTheta0 = std::sin(theta0);

        float s0 = std::cos(theta) - dot * sinTheta / sinTheta0;
        float s1 = sinTheta / sinTheta0;

        return {
            s0 * a.w + s1 * bAdj.w,
            s0 * a.x + s1 * bAdj.x,
            s0 * a.y + s1 * bAdj.y,
            s0 * a.z + s1 * bAdj.z
        }.normalized();
    }

    // 转换为旋转矩阵
    Mat4 toMatrix() const {
        Quaternion n = normalized();
        float xx = n.x * n.x, yy = n.y * n.y, zz = n.z * n.z;
        float xy = n.x * n.y, xz = n.x * n.z, yz = n.y * n.z;
        float wx = n.w * n.x, wy = n.w * n.y, wz = n.w * n.z;

        Mat4 result = Mat4::identity();
        result.m[0][0] = 1.0f - 2.0f * (yy + zz);
        result.m[0][1] = 2.0f * (xy - wz);
        result.m[0][2] = 2.0f * (xz + wy);
        result.m[1][0] = 2.0f * (xy + wz);
        result.m[1][1] = 1.0f - 2.0f * (xx + zz);
        result.m[1][2] = 2.0f * (yz - wx);
        result.m[2][0] = 2.0f * (xz - wy);
        result.m[2][1] = 2.0f * (yz + wx);
        result.m[2][2] = 1.0f - 2.0f * (xx + yy);
        return result;
    }
};
```

```cpp
    result.m[2][1] = s;  result.m[2][2] = c;
    return result;
}

Mat4 rotationY(float angle) {
    float c = std::cos(angle), s = std::sin(angle);
    Mat4 result = Mat4::identity();
    result.m[0][0] = c;  result.m[0][2] = s;
    result.m[2][0] = -s; result.m[2][2] = c;
    return result;
}

// 绕Z轴旋转矩阵
Mat4 rotationZ(float angle) {
    float c = std::cos(angle), s = std::sin(angle);
    Mat4 result = Mat4::identity();
    result.m[0][0] = c;  result.m[0][1] = -s;
    result.m[1][0] = s;  result.m[1][1] = c;
    return result;
}

// 绕任意轴旋转（Rodrigues公式）
Mat4 rotationAxis(const Vec3& axis, float angle) {
    Vec3 n = axis.normalized();
    float c = std::cos(angle), s = std::sin(angle), t = 1.0f - c;

    Mat4 result = Mat4::identity();
    result.m[0][0] = t * n.x * n.x + c;
    result.m[0][1] = t * n.x * n.y - s * n.z;
    result.m[0][2] = t * n.x * n.z + s * n.y;
    result.m[1][0] = t * n.x * n.y + s * n.z;
    result.m[1][1] = t * n.y * n.y + c;
    result.m[1][2] = t * n.y * n.z - s * n.x;
    result.m[2][0] = t * n.x * n.z - s * n.y;
    result.m[2][1] = t * n.y * n.z + s * n.x;
    result.m[2][2] = t * n.z * n.z + c;
    return result;
}
```

### 2.2 视图与投影矩阵

```cpp
// 视图矩阵（LookAt）
Mat4 lookAtMatrix(const Vec3& eye, const Vec3& center, const Vec3& up) {
    // 计算相机坐标系
    Vec3 f = (center - eye).normalized();   // 前方向
    Vec3 r = Vec3::cross(f, up).normalized(); // 右方向
    Vec3 u = Vec3::cross(r, f);              // 上方向

    Mat4 result = Mat4::identity();
    result.m[0][0] = r.x; result.m[0][1] = r.y; result.m[0][2] = r.z;
    result.m[1][0] = u.x; result.m[1][1] = u.y; result.m[1][2] = u.z;
    result.m[2][0] = -f.x; result.m[2][1] = -f.y; result.m[2][2] = -f.z;
    result.m[0][3] = -Vec3::dot(r, eye);
    result.m[1][3] = -Vec3::dot(u, eye);
    result.m[2][3] = Vec3::dot(f, eye);
    return result;
}

// 透视投影矩阵
Mat4 perspectiveMatrix(float fovY, float aspect, float near, float far) {
    float tanHalfFov = std::tan(fovY * 0.5f);

    Mat4 result{};  // 全零初始化
    result.m[0][0] = 1.0f / (aspect * tanHalfFov);
    result.m[1][1] = 1.0f / tanHalfFov;
    result.m[2][2] = -(far + near) / (far - near);
    result.m[2][3] = -(2.0f * far * near) / (far - near);
    result.m[3][2] = -1.0f;
    return result;
}

// 正交投影矩阵
Mat4 orthographicMatrix(float left, float right, float bottom, float top,
                         float near, float far) {
    Mat4 result = Mat4::identity();
    result.m[0][0] = 2.0f / (right - left);
    result.m[1][1] = 2.0f / (top - bottom);
    result.m[2][2] = -2.0f / (far - near);
    result.m[0][3] = -(right + left) / (right - left);
    result.m[1][3] = -(top + bottom) / (top - bottom);
    result.m[2][3] = -(far + near) / (far - near);
    return result;
}

// 完整的MVP变换
struct Transform {
    Mat4 model;      // 模型空间 → 世界空间
    Mat4 view;       // 世界空间 → 视图空间
    Mat4 projection; // 视图空间 → 裁剪空间

    // 模型-视图-投影组合矩阵
    Mat4 getMVP() const {
        return projection * view * model;
    }

    // 法线变换矩阵（模型矩阵的逆转置）
    Mat4 getNormalMatrix() const {
        return model.inverted().transposed();
    }
};
```

## 3. 射线检测

### 3.1 射线定义与基本求交

```cpp
// 射线定义
struct Ray {
    Vec3 origin;     // 起点
    Vec3 direction;  // 方向（归一化）

    // 沿射线方向获取点
    Vec3 at(float t) const {
        return origin + direction * t;
    }
};

// 射线与球体求交
struct HitResult {
    bool hit = false;
    float t = 0.0f;       // 交点距离
    Vec3 point;            // 交点位置
    Vec3 normal;           // 交点法线
};

HitResult raySphereIntersect(const Ray& ray, const Vec3& center, float radius) {
    HitResult result;

    // 射线：P(t) = O + t*D
    // 球体：|P - C|^2 = r^2
    // 展开：t^2(D·D) + 2t(D·(O-C)) + (O-C)·(O-C) - r^2 = 0

    Vec3 oc = ray.origin - center;
    float a = Vec3::dot(ray.direction, ray.direction);  // = 1 if normalized
    float b = 2.0f * Vec3::dot(oc, ray.direction);
    float c = Vec3::dot(oc, oc) - radius * radius;

    float discriminant = b * b - 4.0f * a * c;
    if (discriminant < 0.0f) {
        result.hit = false;
        return result;
    }

    float sqrtD = std::sqrt(discriminant);
    float t1 = (-b - sqrtD) / (2.0f * a);
    float t2 = (-b + sqrtD) / (2.0f * a);

    // 取最近的正交点
    float t = t1;
    if (t < 0.001f) t = t2;
    if (t < 0.001f) {
        result.hit = false;
        return result;
    }

    result.hit = true;
    result.t = t;
    result.point = ray.at(t);
    result.normal = (result.point - center).normalized();
    return result;
}

// 射线与AABB（轴对齐包围盒）求交
struct AABB {
    Vec3 min;  // 最小角
    Vec3 max;  // 最大角

    // Slab方法求交
    HitResult intersect(const Ray& ray) const {
        HitResult result;
        float tMin = 0.0f;
        float tMax = std::numeric_limits<float>::max();

        // 对每个轴分别计算进入和离开的t值
        for (int i = 0; i < 3; ++i) {
            float origin = (i == 0) ? ray.origin.x : (i == 1) ? ray.origin.y : ray.origin.z;
            float dir = (i == 0) ? ray.direction.x : (i == 1) ? ray.direction.y : ray.direction.z;
            float bMin = (i == 0) ? min.x : (i == 1) ? min.y : min.z;
            float bMax = (i == 0) ? max.x : (i == 1) ? max.y : max.z;

            if (std::abs(dir) < 1e-8f) {
                // 射线与slab平行
                if (origin < bMin || origin > bMax) {
                    result.hit = false;
                    return result;
                }
            } else {
                float t1 = (bMin - origin) / dir;
                float t2 = (bMax - origin) / dir;
                if (t1 > t2) std::swap(t1, t2);
                tMin = std::max(tMin, t1);
                tMax = std::min(tMax, t2);
                if (tMin > tMax) {
                    result.hit = false;
                    return result;
                }
            }
        }

        result.hit = true;
        result.t = tMin;
        result.point = ray.at(tMin);

        // 计算法线（找到最近的轴对齐面）
        Vec3 center = (min + max) * 0.5f;
        Vec3 diff = result.point - center;
        Vec3 extent = (max - min) * 0.5f;
        float minDist = std::numeric_limits<float>::max();
        Vec3 normal{};
        float signs[] = {1, -1};
        for (int i = 0; i < 3; ++i) {
            float d = std::abs(std::abs((i == 0) ? diff.x : (i == 1) ? diff.y : diff.z)
                              - ((i == 0) ? extent.x : (i == 1) ? extent.y : extent.z));
            if (d < minDist) {
                minDist = d;
                float val = (i == 0) ? diff.x : (i == 1) ? diff.y : diff.z;
                float ext = (i == 0) ? extent.x : (i == 1) ? extent.y : extent.z;
                float sign = (val > 0) ? 1.0f : -1.0f;
                if (i == 0) normal = {sign, 0, 0};
                else if (i == 1) normal = {0, sign, 0};
                else normal = {0, 0, sign};
            }
        }
        result.normal = normal;
        return result;
    }
};

// 射线与三角形求交（Möller-Trumbore算法）
HitResult rayTriangleIntersect(const Ray& ray,
                                const Vec3& v0, const Vec3& v1, const Vec3& v2) {
    HitResult result;

    Vec3 edge1 = v1 - v0;
    Vec3 edge2 = v2 - v0;
    Vec3 h = Vec3::cross(ray.direction, edge2);
    float a = Vec3::dot(edge1, h);

    // 射线与三角形平行
    if (std::abs(a) < 1e-8f) {
        result.hit = false;
        return result;
    }

    float f = 1.0f / a;
    Vec3 s = ray.origin - v0;
    float u = f * Vec3::dot(s, h);
    if (u < 0.0f || u > 1.0f) {
        result.hit = false;
        return result;
    }

    Vec3 q = Vec3::cross(s, edge1);
    float v = f * Vec3::dot(ray.direction, q);
    if (v < 0.0f || u + v > 1.0f) {
        result.hit = false;
        return result;
    }

    float t = f * Vec3::dot(edge2, q);
    if (t < 0.001f) {
        result.hit = false;
        return result;
    }

    result.hit = true;
    result.t = t;
    result.point = ray.at(t);
    result.normal = Vec3::cross(edge1, edge2).normalized();
    return result;
}
```

### 4.1 BVH（层次包围盒）

```cpp
// BVH（Bounding Volume Hierarchy）- 层次包围盒
// 用于加速射线检测和视锥裁剪

struct BVHNode {
    AABB bounds;           // 节点的包围盒
    int leftChild = -1;    // 左子节点索引
    int rightChild = -1;   // 右子节点索引
    int firstPrimitive = -1;  // 第一个图元索引（叶子节点）
    int primitiveCount = 0;   // 图元数量（叶子节点）

    bool isLeaf() const { return primitiveCount > 0; }
};

class BVH {
public:
    struct Primitive {
        AABB bounds;
        Vec3 centroid;
        int index;  // 原始图元索引
    };

    // 构建BVH
    void build(const std::vector<AABB>& primitiveBounds) {
        // 创建图元列表
        primitives_.resize(primitiveBounds.size());
        for (size_t i = 0; i < primitiveBounds.size(); ++i) {
            primitives_[i].bounds = primitiveBounds[i];
            primitives_[i].centroid = {
                (primitiveBounds[i].min.x + primitiveBounds[i].max.x) * 0.5f,
                (primitiveBounds[i].min.y + primitiveBounds[i].max.y) * 0.5f,
                (primitiveBounds[i].min.z + primitiveBounds[i].max.z) * 0.5f
            };
            primitives_[i].index = static_cast<int>(i);
        }

        // 递归构建
        nodes_.reserve(primitives_.size() * 2);
        buildRecursive(0, static_cast<int>(primitives_.size()));
    }

    // 射线检测（遍历BVH）
    std::vector<int> intersect(const Ray& ray) const {
        std::vector<int> hits;
        if (nodes_.empty()) return hits;

        // 使用栈遍历（避免递归）
        int stack[64];
        int stackPtr = 0;
        stack[stackPtr++] = 0;  // 根节点

        while (stackPtr > 0) {
            int nodeIdx = stack[--stackPtr];
            const BVHNode& node = nodes_[nodeIdx];

            // 射线与包围盒求交
            if (!node.bounds.intersect(ray).hit) continue;

            if (node.isLeaf()) {
                // 叶子节点：添加所有图元
                for (int i = 0; i < node.primitiveCount; ++i) {
                    hits.push_back(primitives_[node.firstPrimitive + i].index);
                }
            } else {
                // 内部节点：压入子节点
                stack[stackPtr++] = node.leftChild;
                stack[stackPtr++] = node.rightChild;
            }
        }

        return hits;
    }

private:
    std::vector<BVHNode> nodes_;
    std::vector<Primitive> primitives_;

    // 计算一组图元的包围盒
    AABB computeBounds(int start, int end) const {
        AABB bounds;
        bounds.min = {std::numeric_limits<float>::max(),
                      std::numeric_limits<float>::max(),
                      std::numeric_limits<float>::max()};
        bounds.max = {-std::numeric_limits<float>::max(),
                      -std::numeric_limits<float>::max(),
                      -std::numeric_limits<float>::max()};

        for (int i = start; i < end; ++i) {
            bounds.min.x = std::min(bounds.min.x, primitives_[i].bounds.min.x);
            bounds.min.y = std::min(bounds.min.y, primitives_[i].bounds.min.y);
            bounds.min.z = std::min(bounds.min.z, primitives_[i].bounds.min.z);
            bounds.max.x = std::max(bounds.max.x, primitives_[i].bounds.max.x);
            bounds.max.y = std::max(bounds.max.y, primitives_[i].bounds.max.y);
            bounds.max.z = std::max(bounds.max.z, primitives_[i].bounds.max.z);
        }
        return bounds;
    }

    // 递归构建BVH
    int buildRecursive(int start, int end) {
        int nodeIdx = static_cast<int>(nodes_.size());
        nodes_.push_back(BVHNode{});

        BVHNode& node = nodes_[nodeIdx];
        node.bounds = computeBounds(start, end);

        int count = end - start;
        if (count <= 2) {
            // 叶子节点
            node.firstPrimitive = start;
            node.primitiveCount = count;
            return nodeIdx;
        }

        // 选择分割轴（选择包围盒最长的轴）
        Vec3 extent = {
            node.bounds.max.x - node.bounds.min.x,
            node.bounds.max.y - node.bounds.min.y,
            node.bounds.max.z - node.bounds.min.z
        };
        int axis = 0;
        if (extent.y > extent.x) axis = 1;
        if (extent.z > (axis == 0 ? extent.x : extent.y)) axis = 2;

        // 按质心坐标排序
        std::sort(primitives_.begin() + start, primitives_.begin() + end,
                  [axis](const Primitive& a, const Primitive& b) {
                      return (axis == 0) ? a.centroid.x < b.centroid.x :
                             (axis == 1) ? a.centroid.y < b.centroid.y :
                                           a.centroid.z < b.centroid.z;
                  });

        // 中间分割
        int mid = start + count / 2;

        // 递归构建子树
        node.leftChild = buildRecursive(start, mid);
        node.rightChild = buildRecursive(mid, end);

        return nodeIdx;
    }
};
```

### 4.2 八叉树

```cpp
// 八叉树（Octree）- 三维空间划分数据结构

template<typename T>
class Octree {
public:
    static constexpr int MAX_OBJECTS = 8;    // 每个节点最大对象数
    static constexpr int MAX_DEPTH = 8;      // 最大深度

    struct Object {
        AABB bounds;
        T data;
    };

    struct Node {
        AABB bounds;
        std::vector<Object> objects;
        std::unique_ptr<Node> children[8];  // 8个子节点
        bool isLeaf = true;
    };

    void init(const AABB& rootBounds) {
        root_ = std::make_unique<Node>();
        root_->bounds = rootBounds;
    }

    void insert(const AABB& bounds, const T& data) {
        insertRecursive(root_.get(), {bounds, data}, 0);
    }

    // 范围查询
    std::vector<T> query(const AABB& searchBounds) const {
        std::vector<T> results;
        queryRecursive(root_.get(), searchBounds, results);
        return results;
    }

private:
    std::unique_ptr<Node> root_;

    // 判断两个AABB是否相交
    static bool intersects(const AABB& a, const AABB& b) {
        return (a.min.x <= b.max.x && a.max.x >= b.min.x) &&
               (a.min.y <= b.max.y && a.max.y >= b.min.y) &&
               (a.min.z <= b.max.z && a.max.z >= b.min.z);
    }

    // 判断AABB是否完全包含在另一个AABB内
    static bool contains(const AABB& outer, const AABB& inner) {
        return outer.min.x <= inner.min.x && outer.max.x >= inner.max.x &&
               outer.min.y <= inner.min.y && outer.max.y >= inner.max.y &&
               outer.min.z <= inner.min.z && outer.max.z >= inner.max.z;
    }

    void insertRecursive(Node* node, const Object& obj, int depth) {
        if (!intersects(node->bounds, obj.bounds)) return;

        if (node->isLeaf) {
            node->objects.push_back(obj);

            // 如果超过最大对象数且未达最大深度，则分裂
            if (static_cast<int>(node->objects.size()) > MAX_OBJECTS && depth < MAX_DEPTH) {
                splitNode(node, depth);
            }
        } else {
            // 插入到匹配的子节点
            for (auto& child : node->children) {
                if (child && intersects(child->bounds, obj.bounds)) {
                    insertRecursive(child.get(), obj, depth + 1);
                }
            }
        }
    }

    void splitNode(Node* node, int depth) {
        node->isLeaf = false;

        Vec3 mid = {
            (node->bounds.min.x + node->bounds.max.x) * 0.5f,
            (node->bounds.min.y + node->bounds.max.y) * 0.5f,
            (node->bounds.min.z + node->bounds.max.z) * 0.5f
        };

        // 创建8个子节点
        for (int i = 0; i < 8; ++i) {
            node->children[i] = std::make_unique<Node>();
            AABB& childBounds = node->children[i]->bounds;

            childBounds.min.x = (i & 1) ? mid.x : node->bounds.min.x;
            childBounds.min.y = (i & 2) ? mid.y : node->bounds.min.y;
            childBounds.min.z = (i & 4) ? mid.z : node->bounds.min.z;
            childBounds.max.x = (i & 1) ? node->bounds.max.x : mid.x;
            childBounds.max.y = (i & 2) ? node->bounds.max.y : mid.y;
            childBounds.max.z = (i & 4) ? node->bounds.max.z : mid.z;
        }

        // 将对象重新分配到子节点
        auto objects = std::move(node->objects);
        node->objects.clear();

        for (const auto& obj : objects) {
            for (auto& child : node->children) {
                if (intersects(child->bounds, obj.bounds)) {
                    insertRecursive(child.get(), obj, depth + 1);
                }
            }
        }
    }

    void queryRecursive(const Node* node, const AABB& searchBounds,
                        std::vector<T>& results) const {
        if (!intersects(node->bounds, searchBounds)) return;

        // 检查当前节点的对象
        for (const auto& obj : node->objects) {
            if (intersects(obj.bounds, searchBounds)) {
                results.push_back(obj.data);
            }
        }

        // 递归查询子节点
        if (!node->isLeaf) {
            for (const auto& child : node->children) {
                if (child) {
                    queryRecursive(child.get(), searchBounds, results);
                }
            }
        }
    }
};
```

### 5.1 CPU端粒子系统

```cpp
// 基于CPU的粒子系统
#include <vector>
#include <random>
#include <functional>

struct Particle {
    Vec3 position;       // 位置
    Vec3 velocity;       // 速度
    Vec3 acceleration;   // 加速度
    Vec3 color;          // 颜色
    float life;          // 剩余生命
    float maxLife;       // 最大生命
    float size;          // 大小
    float mass;          // 质量
};

class ParticleSystem {
public:
    using ForceFunc = std::function<Vec3(const Particle&, float)>;

    void init(int maxParticles) {
        particles_.resize(maxParticles);
        maxParticles_ = maxParticles;
        aliveCount_ = 0;
    }

    // 设置发射器参数
    void setEmitter(const Vec3& position, const Vec3& direction,
                    float speed, float spread, float rate) {
        emitterPos_ = position;
        emitterDir_ = direction.normalized();
        emitterSpeed_ = speed;
        emitterSpread_ = spread;
        emitRate_ = rate;
    }

    // 添加力函数
    void addForce(ForceFunc force) {
        forces_.push_back(std::move(force));
    }

    void update(float dt) {
        // 发射新粒子
        emit(dt);

        // 更新所有活跃粒子
        for (int i = 0; i < aliveCount_; ++i) {
            Particle& p = particles_[i];

            // 累加所有力
            Vec3 totalForce{};
            for (const auto& force : forces_) {
                totalForce += force(p, dt);
            }

            // 更新加速度（F = ma）
            p.acceleration = totalForce / p.mass;

            // 半隐式欧拉积分
            p.velocity += p.acceleration * dt;
            p.position += p.velocity * dt;

            // 更新生命值
            p.life -= dt;

            // 大小随生命衰减
            float lifeRatio = p.life / p.maxLife;
            p.size = p.size * lifeRatio;
        }

        // 移除死亡粒子（与末尾交换）
        int i = 0;
        while (i < aliveCount_) {
            if (particles_[i].life <= 0.0f) {
                particles_[i] = particles_[aliveCount_ - 1];
                --aliveCount_;
            } else {
                ++i;
            }
        }
    }

    const std::vector<Particle>& getParticles() const { return particles_; }
    int getAliveCount() const { return aliveCount_; }

private:
    std::vector<Particle> particles_;
    std::vector<ForceFunc> forces_;
    int maxParticles_ = 0;
    int aliveCount_ = 0;

    Vec3 emitterPos_;
    Vec3 emitterDir_;
    float emitterSpeed_ = 1.0f;
    float emitterSpread_ = 0.5f;
    float emitRate_ = 100.0f;
    float emitAccumulator_ = 0.0f;

    void emit(float dt) {
        emitAccumulator_ += emitRate_ * dt;
        int count = static_cast<int>(emitAccumulator_);
        emitAccumulator_ -= count;

        for (int i = 0; i < count && aliveCount_ < maxParticles_; ++i) {
            Particle& p = particles_[aliveCount_++];
            p.position = emitterPos_;

            // 随机方向（在发射方向周围扩散）
            float theta = randomFloat(0, 2.0f * 3.14159f);
            float phi = randomFloat(0, emitterSpread_);
            Vec3 offset = {
                std::sin(phi) * std::cos(theta),
                std::sin(phi) * std::sin(theta),
                std::cos(phi)
            };
            p.velocity = (emitterDir_ + offset) * emitterSpeed_;

            p.acceleration = {};
            p.color = {1.0f, 0.8f, 0.2f};
            p.life = randomFloat(1.0f, 3.0f);
            p.maxLife = p.life;
            p.size = randomFloat(0.1f, 0.3f);
            p.mass = 1.0f;
        }
    }

    static float randomFloat(float min, float max) {
        static std::mt19937 gen(42);
        std::uniform_real_distribution<float> dist(min, max);
        return dist(gen);
    }
};

// 常用力函数
namespace Forces {
    // 重力
    ParticleSystem::ForceFunc gravity(float g = 9.8f) {
        return [g](const Particle&, float) -> Vec3 {
            return {0.0f, -g, 0.0f};
        };
    }

    // 阻力
    ParticleSystem::ForceFunc drag(float coefficient) {
        return [coefficient](const Particle& p, float) -> Vec3 {
            return p.velocity * (-coefficient);
        };
    }

    // 风力
    ParticleSystem::ForceFunc wind(const Vec3& direction, float strength) {
        return [direction, strength](const Particle&, float) -> Vec3 {
            return direction * strength;
        };
    }

    // 吸引力（向某点）
    ParticleSystem::ForceFunc attraction(const Vec3& center, float strength) {
        return [center, strength](const Particle& p, float) -> Vec3 {
            Vec3 dir = center - p.position;
            float dist = dir.length();
            if (dist < 0.01f) return {};
            return dir.normalized() * strength / (dist * dist);
        };
    }
}

// 使用示例
void particleSystemExample() {
    ParticleSystem ps;
    ps.init(10000);

    // 设置发射器
    ps.setEmitter({0, 0, 0}, {0, 1, 0}, 5.0f, 0.3f, 500.0f);

    // 添加力
    ps.addForce(Forces::gravity(9.8f));
    ps.addForce(Forces::drag(0.1f));
    ps.addForce(Forces::wind({1, 0, 0}, 2.0f));

    // 模拟循环
    float dt = 1.0f / 60.0f;
    for (int frame = 0; frame < 600; ++frame) {
        ps.update(dt);
        // 渲染粒子...
    }
}
```

### 5.2 GPU加速粒子系统

```glsl
// GPU粒子更新计算着色器
#version 450 core

layout(local_size_x = 256) in;

struct Particle {
    vec3 position;
    float life;
    vec3 velocity;
    float maxLife;
    vec4 color;
    float size;
    float mass;
    vec2 padding;
};

layout(std430, binding = 0) buffer ParticleBuffer {
    Particle particles[];
};

uniform float uDeltaTime;
uniform vec3 uGravity;
uniform float uDrag;

// 伪随机数
float hash(uint n) {
    n = (n << 13u) ^ n;
    return float((n * (n * n * 15731u + 789221u) + 1376312589u) & 0x7fffffffu) / 1073741824.0;
}

void main() {
    uint idx = gl_GlobalInvocationID.x;
    if (idx >= particles.length()) return;

    Particle p = particles[idx];

    // 更新生命值
    p.life -= uDeltaTime;

    // 如果粒子死亡，重新发射
    if (p.life <= 0.0) {
        float r1 = hash(idx * 3u + uint(uDeltaTime * 1000.0));
        float r2 = hash(idx * 7u + uint(uDeltaTime * 1000.0));
        float r3 = hash(idx * 13u + uint(uDeltaTime * 1000.0));

        p.position = vec3(0.0);
        p.velocity = vec3(
            (r1 - 0.5) * 2.0,
            r2 * 3.0 + 2.0,
            (r3 - 0.5) * 2.0
        );
        p.life = r1 * 2.0 + 1.0;
        p.maxLife = p.life;
        p.color = vec4(1.0, 0.8, 0.2, 1.0);
        p.size = 0.2;
    }

    // 应用力
    vec3 force = uGravity * p.mass;  // 重力
    force += -p.velocity * uDrag;     // 阻力

    // 更新速度和位置
    vec3 acceleration = force / p.mass;
    p.velocity += acceleration * uDeltaTime;
    p.position += p.velocity * uDeltaTime;

    // 更新颜色（随生命衰减）
    float lifeRatio = clamp(p.life / p.maxLife, 0.0, 1.0);
    p.color.a = lifeRatio;
    p.size = 0.2 * lifeRatio;

    particles[idx] = p;
}
```

## 6. 本章小结

本章讲解了图形编程所需的数学基础和算法，核心要点如下：

| 主题 | 核心要点 |
|------|---------|
| 线性代数 | 向量运算、4x4矩阵、四元数旋转、Slerp插值 |
| 坐标变换 | 平移/旋转/缩放矩阵、LookAt视图、透视/正交投影 |
| 射线检测 | 射线-球体、射线-AABB、射线-三角形（Möller-Trumbore） |
| 空间数据结构 | BVH层次包围盒、八叉树空间划分 |
| 粒子系统 | CPU/GPU粒子模拟、力函数、半隐式欧拉积分 |

**关键理解**：

1. **向量与矩阵是图形学的语言**：所有3D变换都可以用矩阵乘法表示
2. **四元数优于欧拉角**：避免万向锁，插值更平滑（Slerp）
3. **射线检测是交互基础**：鼠标拾取、碰撞检测都依赖射线-几何体求交
4. **空间数据结构加速查询**：BVH和八叉树将O(n)查询降为O(log n)
5. **粒子系统需要GPU加速**：大规模粒子（>10万）必须使用计算着色器
