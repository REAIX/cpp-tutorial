> 💡 **通俗理解 - 什么是裸机编程？**

想象一下：

- **有操作系统的编程** 就像"住酒店"，有人帮你打扫房间、管理水电
- **裸机编程** 就像"住荒岛"，一切都要自己动手——自己发电、自己打水、自己建房子
- **启动代码** 就像"荒岛上的第一把火"，没有它什么都做不了
- **中断** 就像"荒岛上的警报器"，有紧急情况时立刻响应

裸机编程就是没有任何操作系统支持，程序直接与硬件对话。

> 🔬 **抽象理解 - 裸机编程层次**：
>
> - **启动代码**：从复位向量到main函数的桥梁
> - **链接脚本**：定义内存布局和段放置规则
> - **内存映射**：理解芯片的地址空间分配
> - **中断与异常**：硬件事件的响应机制
> - **寄存器操作**：通过读写寄存器控制外设

## 1. 启动代码

### 1.1 启动流程

```
上电/复位
    │
    ↓
CPU从复位向量(0x00000000)取初始SP和PC
    │
    ↓
执行Reset_Handler
    │
    ├── 初始化.data段（从Flash复制到RAM）
    ├── 初始化.bss段（清零RAM）
    ├── 调用SystemInit()（时钟配置）
    ├── 调用__libc_init_array()（C++全局构造）
    └── 调用main()
```

### 1.2 启动代码实现

```assembly
// startup_stm32f407.s - STM32F407启动文件（ARM汇编）

// 栈定义
.section .isr_vector, "a", %progbits
.type g_pfnVectors, %object
g_pfnVectors:
    .word _estack               // 初始栈指针（栈顶）
    .word Reset_Handler         // 复位向量
    .word NMI_Handler           // 不可屏蔽中断
    .word HardFault_Handler     // 硬件故障
    .word MemManage_Handler     // 内存管理故障
    .word BusFault_Handler      // 总线故障
    .word UsageFault_Handler    // 用法故障
    .word 0                     // 保留
    .word 0                     // 保留
    .word 0                     // 保留
    .word 0                     // 保留
    .word SVC_Handler           // 系统服务调用
    .word DebugMon_Handler      // 调试监控
    .word 0                     // 保留
    .word PendSV_Handler        // PendSV（任务切换用）
    .word SysTick_Handler       // 系统滴答定时器
    // 外部中断向量...
    .word WWDG_IRQHandler       // 窗口看门狗
    .word PVD_IRQHandler        // 电源电压检测
    // ... 更多中断向量

.size g_pfnVectors, . - g_pfnVectors

// 复位处理函数
.section .text
.type Reset_Handler, %function
Reset_Handler:
    // 1. 将.data段从Flash复制到RAM
    ldr r0, =_sdata             // 数据段起始地址（RAM）
    ldr r1, =_edata             // 数据段结束地址（RAM）
    ldr r2, =_sidata            // 数据段在Flash中的起始地址
copy_data:
    cmp r0, r1
    bcc copy_data_loop
    b zero_bss
copy_data_loop:
    ldr r3, [r2]                // 从Flash读取
    str r3, [r0]                // 写入RAM
    adds r0, r0, #4
    adds r2, r2, #4
    b copy_data

    // 2. 清零.bss段
zero_bss:
    ldr r0, =_sbss              // bss段起始地址
    ldr r1, =_ebss              // bss段结束地址
    mov r2, #0
zero_bss_loop:
    cmp r0, r1
    bcc zero_bss_continue
    b call_init
zero_bss_continue:
    str r2, [r0]                // 写零
    adds r0, r0, #4
    b zero_bss_loop

    // 3. 调用SystemInit
call_init:
    bl SystemInit

    // 4. 调用C++全局构造函数
    bl __libc_init_array

    // 5. 调用main
    bl main

    // main不应返回，但如果返回则进入死循环
    b .

// 默认中断处理函数（无限循环）
.weak NMI_Handler
.thumb_set NMI_Handler, Default_Handler

.weak HardFault_Handler
.thumb_set HardFault_Handler, Default_Handler

// ... 其他弱定义

Default_Handler:
    b .
```

### 1.3 C语言版启动代码

```c
// startup.c - C语言版启动代码

#include <stdint.h>

// 外部符号（由链接脚本定义）
extern uint32_t _estack;       // 栈顶
extern uint32_t _sidata;       // .data在Flash中的加载地址
extern uint32_t _sdata;        // .data在RAM中的起始地址
extern uint32_t _edata;        // .data在RAM中的结束地址
extern uint32_t _sbss;         // .bss起始地址
extern uint32_t _ebss;         // .bss结束地址

// 外部函数
extern int main(void);
extern void SystemInit(void);

// 默认中断处理函数
void Default_Handler(void) {
    while (1);
}

// 中断向量表
// 必须放在特殊的段中，确保位于Flash起始位置
__attribute__((section(".isr_vector")))
void (* const g_pfnVectors[])(void) = {
    (void (*)(void))((uint32_t)&_estack),  // 初始栈指针
    Reset_Handler,                          // 复位
    NMI_Handler,                            // NMI
    HardFault_Handler,                      // 硬件故障
    MemManage_Handler,                      // 内存管理
    BusFault_Handler,                       // 总线故障
    UsageFault_Handler,                     // 用法故障
    0, 0, 0, 0,                             // 保留
    SVC_Handler,                            // SVC
    DebugMon_Handler,                       // 调试监控
    0,                                      // 保留
    PendSV_Handler,                         // PendSV
    SysTick_Handler,                        // SysTick
    // 外部中断...
};

// 复位处理函数
void Reset_Handler(void) {
    // 1. 复制.data段
    uint32_t* src = &_sidata;
    uint32_t* dst = &_sdata;
    while (dst < &_edata) {
        *dst++ = *src++;
    }

    // 2. 清零.bss段
    dst = &_sbss;
    while (dst < &_ebss) {
        *dst++ = 0;
    }

    // 3. 系统初始化
    SystemInit();

    // 4. C++全局构造
    // __libc_init_array();  // 如果使用newlib

    // 5. 调用main
    main();

    // main不应返回
    while (1);
}

// 弱定义中断处理函数
void NMI_Handler(void) __attribute__((weak, alias("Default_Handler")));
void HardFault_Handler(void) __attribute__((weak, alias("Default_Handler")));
void MemManage_Handler(void) __attribute__((weak, alias("Default_Handler")));
void BusFault_Handler(void) __attribute__((weak, alias("Default_Handler")));
void UsageFault_Handler(void) __attribute__((weak, alias("Default_Handler")));
void SVC_Handler(void) __attribute__((weak, alias("Default_Handler")));
void DebugMon_Handler(void) __attribute__((weak, alias("Default_Handler")));
void PendSV_Handler(void) __attribute__((weak, alias("Default_Handler")));
void SysTick_Handler(void) __attribute__((weak, alias("Default_Handler")));
```

## 2. 链接脚本

### 2.1 链接脚本基础

```ld
/* STM32F407链接脚本 */

/* 内存定义 */
MEMORY
{
    FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 1024K  /* Flash存储器 */
    RAM (rwx)   : ORIGIN = 0x20000000, LENGTH = 128K   /* 主SRAM */
    CCMRAM (rw) : ORIGIN = 0x10000000, LENGTH = 64K    /* CCM数据RAM */
}

/* 栈大小 */
_estack = ORIGIN(RAM) + LENGTH(RAM);  /* 栈顶 = RAM末尾，向下增长 */

/* 堆大小（裸机通常不用） */
_Min_Heap_Size = 0x0;
_Min_Stack_Size = 0x400;  /* 最小1KB栈 */

/* 段定义 */
SECTIONS
{
    /* 中断向量表 - 必须放在Flash最前面 */
    .isr_vector :
    {
        . = ALIGN(4);
        KEEP(*(.isr_vector))
        . = ALIGN(4);
    } > FLASH

    /* 代码段 */
    .text :
    {
        . = ALIGN(4);
        *(.text)           /* 普通代码 */
        *(.text*)          /* 模板实例化代码等 */
        *(.glue_7)         /* ARM/Thumb互调用代码 */
        *(.glue_7t)
        KEEP(*(.init))     /* 初始化代码 */
        KEEP(*(.fini))     /* 终止代码 */
        . = ALIGN(4);
        _etext = .;        /* 代码段结束标记 */
    } > FLASH

    /* 只读数据段 */
    .rodata :
    {
        . = ALIGN(4);
        *(.rodata)         /* 只读数据 */
        *(.rodata*)
        . = ALIGN(4);
    } > FLASH

    /* ARM特定属性 */
    .ARM.extab :
    {
        *(.ARM.extab* .gnu.linkonce.armextab.*)
    } > FLASH

    .ARM :
    {
        __exidx_start = .;
        *(.ARM.exidx*)
        __exidx_end = .;
    } > FLASH

    /* .data段的加载地址（在Flash中） */
    _sidata = LOADADDR(.data);

    /* 已初始化数据段 - 运行时在RAM，加载时在Flash */
    .data :
    {
        . = ALIGN(4);
        _sdata = .;        /* 数据段起始（RAM地址） */
        *(.data)           /* 已初始化数据 */
        *(.data*)
        . = ALIGN(4);
        _edata = .;        /* 数据段结束（RAM地址） */
    } > RAM AT > FLASH     /* 运行在RAM，加载在FLASH */

    /* 未初始化数据段 */
    .bss :
    {
        . = ALIGN(4);
        _sbss = .;         /* bss段起始 */
        __bss_start__ = _sbss;
        *(.bss)            /* 未初始化数据 */
        *(.bss*)
        *(COMMON)
        . = ALIGN(4);
        _ebss = .;         /* bss段结束 */
        __bss_end__ = _ebss;
    } > RAM

    /* 用户堆 */
    ._user_heap_stack :
    {
        . = ALIGN(8);
        PROVIDE(end = .);
        PROVIDE(_end = .);
        . = . + _Min_Heap_Size;
        . = . + _Min_Stack_Size;
        . = ALIGN(8);
    } > RAM

    /* 丢弃不需要的段 */
    /DISCARD/ :
    {
        libc.a(*)          /* 不链接标准C库 */
        libm.a(*)          /* 不链接数学库 */
        libgcc.a(*)        /* 不链接GCC运行时库 */
    }

    /* 检查内存溢出 */
    ASSERT(_ebss <= ORIGIN(RAM) + LENGTH(RAM), "RAM区域溢出")
}
```

### 3.2 内存映射寄存器访问

```cpp
// 通过内存映射访问外设寄存器

// 方法1：直接使用指针（最直接但不安全）
#define GPIOA_BASE  0x40020000UL
#define GPIOA_MODER  (*(volatile uint32_t*)(GPIOA_BASE + 0x00))
#define GPIOA_ODR    (*(volatile uint32_t*)(GPIOA_BASE + 0x14))
#define GPIOA_IDR    (*(volatile uint32_t*)(GPIOA_BASE + 0x10))

void toggleLed() {
    // 读取-修改-写入
    GPIOA_ODR ^= (1 << 5);  // 翻转PA5
}

// 方法2：使用结构体映射（更清晰）
typedef struct {
    volatile uint32_t MODER;     // 模式寄存器，偏移0x00
    volatile uint32_t OTYPER;    // 输出类型寄存器，偏移0x04
    volatile uint32_t OSPEEDR;   // 输出速度寄存器，偏移0x08
    volatile uint32_t PUPDR;     // 上下拉寄存器，偏移0x0C
    volatile uint32_t IDR;       // 输入数据寄存器，偏移0x10
    volatile uint32_t ODR;       // 输出数据寄存器，偏移0x14
    volatile uint32_t BSRR;      // 置位/复位寄存器，偏移0x18
    volatile uint32_t LCKR;      // 配置锁定寄存器，偏移0x1C
    volatile uint32_t AFRL;      // 复用功能低位寄存器，偏移0x20
    volatile uint32_t AFRH;      // 复用功能高位寄存器，偏移0x24
} GPIO_TypeDef;

#define GPIOA ((GPIO_TypeDef*)GPIOA_BASE)
#define GPIOB ((GPIO_TypeDef*)(GPIOA_BASE + 0x0400))
#define GPIOC ((GPIO_TypeDef*)(GPIOA_BASE + 0x0800))

void initGpio() {
    // 设置PA5为输出模式（LED引脚）
    GPIOA->MODER &= ~(3U << (5 * 2));    // 清除PA5的模式位
    GPIOA->MODER |= (1U << (5 * 2));     // 设置为通用输出模式

    // 设置PA5为推挽输出
    GPIOA->OTYPER &= ~(1U << 5);         // 推挽输出

    // 设置PA5为高速
    GPIOA->OSPEEDR |= (3U << (5 * 2));   // 高速

    // 不使用上下拉
    GPIOA->PUPDR &= ~(3U << (5 * 2));    // 无上下拉
}

void setLed(bool on) {
    if (on) {
        GPIOA->BSRR = (1U << 5);    // 置位PA5（原子操作）
    } else {
        GPIOA->BSRR = (1U << (5 + 16));  // 复位PA5（原子操作）
    }
}

// 方法3：使用C++类型安全的寄存器访问
template<uint32_t BaseAddr>
class GpioPort {
public:
    static constexpr GPIO_TypeDef* registers =
        reinterpret_cast<GPIO_TypeDef*>(BaseAddr);

    enum class Mode : uint8_t {
        Input = 0,
        Output = 1,
        Alternate = 2,
        Analog = 3
    };

    static void setMode(uint8_t pin, Mode mode) {
        registers->MODER &= ~(3U << (pin * 2));
        registers->MODER |= (static_cast<uint32_t>(mode) << (pin * 2));
    }

    static void setOutput(uint8_t pin, bool high) {
        if (high) {
            registers->BSRR = (1U << pin);
        } else {
            registers->BSRR = (1U << (pin + 16));
        }
    }

    static bool readInput(uint8_t pin) {
        return (registers->IDR & (1U << pin)) != 0;
    }
};

using PortA = GpioPort<0x40020000>;
using PortB = GpioPort<0x40020400>;

void cppGpioExample() {
    PortA::setMode(5, PortA::Mode::Output);
    PortA::setOutput(5, true);
}
```

## 4. 中断与异常处理

### 4.1 Cortex-M异常模型

```
异常优先级（从高到低）：
1. Reset（复位）
2. NMI（不可屏蔽中断）
3. HardFault（硬件故障）
4. MemManage（内存管理故障）
5. BusFault（总线故障）
6. UsageFault（用法故障）
7. SVCall（系统服务调用）
8. PendSV（可挂起系统服务）
9. SysTick（系统滴答）
10. 外部中断（IRQ0 ~ IRQn）
```

### 4.2 NVIC配置

```cpp
// NVIC（嵌套向量中断控制器）配置

#include <cstdint>

// NVIC寄存器结构体
typedef struct {
    volatile uint32_t ISER[8];    // 中断使能寄存器
    uint32_t RESERVED0[24];
    volatile uint32_t ICER[8];    // 中断清除使能寄存器
    uint32_t RESERVED1[24];
    volatile uint32_t ISPR[8];    // 中断挂起寄存器
    uint32_t RESERVED2[24];
    volatile uint32_t ICPR[8];    // 中断清除挂起寄存器
    uint32_t RESERVED3[24];
    volatile uint32_t IABR[8];    // 中断活动位寄存器
    uint32_t RESERVED4[56];
    volatile uint8_t  IP[240];    // 中断优先级寄存器
    uint32_t RESERVED5[644];
    volatile uint32_t STIR;       // 软件触发中断寄存器
} NVIC_TypeDef;

#define NVIC ((NVIC_TypeDef*)0xE000E100)

// 使能中断
void NVIC_EnableIRQ(uint32_t irq) {
    NVIC->ISER[irq >> 5] = (1U << (irq & 0x1F));
}

// 禁用中断
void NVIC_DisableIRQ(uint32_t irq) {
    NVIC->ICER[irq >> 5] = (1U << (irq & 0x1F));
}

// 设置中断优先级
void NVIC_SetPriority(uint32_t irq, uint32_t priority) {
    // STM32F4使用4位优先级（0最高，15最低）
    NVIC->IP[irq] = (uint8_t)((priority & 0x0F) << 4);
}

// 配置外部中断示例
void configureExternalInterrupt() {
    // 1. 配置GPIO为中断输入
    // SYSCFG外部中断配置
    RCC->APB2ENR |= RCC_APB2ENR_SYSCFGEN;  // 使能SYSCFG时钟

    // 将PA0映射到EXTI0
    SYSCFG->EXTICR[0] &= ~SYSCFG_EXTICR1_EXTI0;
    SYSCFG->EXTICR[0] |= SYSCFG_EXTICR1_EXTI0_PA;

    // 2. 配置EXTI
    EXTI->IMR |= EXTI_IMR_MR0;     // 使能EXTI0中断
    EXTI->RTSR |= EXTI_RTSR_TR0;   // 上升沿触发
    EXTI->FTSR |= EXTI_FTSR_TR0;   // 下降沿触发

    // 3. 配置NVIC
    NVIC_SetPriority(EXTI0_IRQn, 5);   // 设置优先级
    NVIC_EnableIRQ(EXTI0_IRQn);         // 使能中断
}

// 中断服务例程
extern "C" void EXTI0_IRQHandler(void) {
    // 检查中断标志
    if (EXTI->PR & EXTI_PR_PR0) {
        EXTI->PR = EXTI_PR_PR0;  // 清除中断标志（写1清除）

        // 处理中断事件
        buttonPressed();
    }
}
```

### 4.3 临界区保护

```cpp
// 临界区保护 - 禁用中断

class CriticalSection {
public:
    // 进入临界区（禁用中断）
    static inline uint32_t enter() {
        uint32_t primask = __get_PRIMASK();  // 保存当前中断状态
        __disable_irq();                      // 禁用中断
        return primask;
    }

    // 退出临界区（恢复中断状态）
    static inline void exit(uint32_t primask) {
        __set_PRIMASK(primask);  // 恢复之前的中断状态
    }
};

// RAII方式的临界区保护
class ScopedCriticalSection {
public:
    ScopedCriticalSection() : primask_(CriticalSection::enter()) {}
    ~ScopedCriticalSection() { CriticalSection::exit(primask_); }

    // 禁止拷贝
    ScopedCriticalSection(const ScopedCriticalSection&) = delete;
    ScopedCriticalSection& operator=(const ScopedCriticalSection&) = delete;

private:
    uint32_t primask_;
};

// 使用示例
volatile uint32_t sharedCounter = 0;

void incrementSharedCounter() {
    ScopedCriticalSection cs;  // 自动进入/退出临界区
    ++sharedCounter;
}  // 离开作用域时自动恢复中断

void manualCriticalSection() {
    uint32_t primask = CriticalSection::enter();
    // 临界区代码
    sharedCounter += 10;
    CriticalSection::exit(primask);
```

```cpp
// 嵌入式常用位操作

// 置位（设置某一位为1）
#define SET_BIT(reg, bit)       ((reg) |= (1U << (bit)))

// 清位（设置某一位为0）
#define CLEAR_BIT(reg, bit)     ((reg) &= ~(1U << (bit)))

// 翻转位
#define TOGGLE_BIT(reg, bit)    ((reg) ^= (1U << (bit)))

// 读取位
#define READ_BIT(reg, bit)      (((reg) >> (bit)) & 1U)

// 修改多位字段
#define MODIFY_REG(reg, clearmask, setmask) \
    ((reg) = ((reg) & ~(clearmask)) | (setmask))

// C++类型安全版本
template<typename T>
class BitField {
public:
    static constexpr void set(volatile T& reg, uint8_t bit) {
        reg |= (static_cast<T>(1) << bit);
    }

    static constexpr void clear(volatile T& reg, uint8_t bit) {
        reg &= ~(static_cast<T>(1) << bit);
    }

    static constexpr void toggle(volatile T& reg, uint8_t bit) {
        reg ^= (static_cast<T>(1) << bit);
    }

    static constexpr bool read(volatile T& reg, uint8_t bit) {
        return (reg >> bit) & 1;
    }

    // 修改多位的字段
    static constexpr void writeField(volatile T& reg, uint8_t pos,
                                      uint8_t width, T value) {
        T mask = ((static_cast<T>(1) << width) - 1) << pos;
        reg = (reg & ~mask) | ((value << pos) & mask);
    }

    static constexpr T readField(volatile T& reg, uint8_t pos, uint8_t width) {
        T mask = (static_cast<T>(1) << width) - 1;
        return (reg >> pos) & mask;
    }
};

// 使用示例
void bitOperationExample() {
    volatile uint32_t reg = 0;

    BitField<uint32_t>::set(reg, 5);       // 置位第5位
    BitField<uint32_t>::clear(reg, 3);     // 清除第3位
    BitField<uint32_t>::toggle(reg, 7);    // 翻转第7位
    bool bit5 = BitField<uint32_t>::read(reg, 5);  // 读取第5位

    // 写入4位宽的字段到位置8
    BitField<uint32_t>::writeField(reg, 8, 4, 0xA);
    uint32_t field = BitField<uint32_t>::readField(reg, 8, 4);
}
```

```cpp
// UART驱动实现
#include <cstdint>

class UartDriver {
public:
    enum class BaudRate : uint32_t {
        B9600 = 9600,
        B19200 = 19200,
        B115200 = 115200,
        B921600 = 921600
    };

    enum class Parity : uint8_t {
        None = 0,
        Even = 1,
        Odd = 2
    };

    struct Config {
        BaudRate baudRate = BaudRate::B115200;
        Parity parity = Parity::None;
        uint8_t dataBits = 8;
        uint8_t stopBits = 1;
    };

    void init(const Config& config) {
        // 使能时钟
        RCC->APB1ENR |= RCC_APB1ENR_USART2EN;
        RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;

        // 配置GPIO（PA2=TX, PA3=RX，复用功能）
        GPIOA->MODER &= ~(3U << (2 * 2) | 3U << (3 * 2));
        GPIOA->MODER |= (2U << (2 * 2) | 2U << (3 * 2));  // 复用模式
        GPIOA->AFRL &= ~(0xFU << (2 * 4) | 0xFU << (3 * 4));
        GPIOA->AFRL |= (7U << (2 * 4) | 7U << (3 * 4));   // AF7 = USART2

        // 配置波特率
        uint32_t clockFreq = 84000000;  // APB1时钟84MHz
        uint32_t baud = static_cast<uint32_t>(config.baudRate);
        USART2->BRR = clockFreq / baud;

        // 配置数据格式
        uint32_t cr1 = USART_CR1_TE | USART_CR1_RE;  // 使能发送和接收
        if (config.dataBits == 9) cr1 |= USART_CR1_M;

        switch (config.parity) {
            case Parity::Even: cr1 |= USART_CR1_PCE; break;
            case Parity::Odd:  cr1 |= USART_CR1_PCE | USART_CR1_PS; break;
            default: break;
        }

        USART2->CR1 = cr1;

        // 使能接收中断
        USART2->CR1 |= USART_CR1_RXNEIE;
        NVIC_SetPriority(USART2_IRQn, 6);
        NVIC_EnableIRQ(USART2_IRQn);

        // 使能UART
        USART2->CR1 |= USART_CR1_UE;
    }

    // 发送一个字节（阻塞）
    void sendByte(uint8_t byte) {
        while (!(USART2->SR & USART_SR_TXE));  // 等待发送缓冲区空
        USART2->DR = byte;
    }

    // 发送字符串
    void sendString(const char* str) {
        while (*str) {
            sendByte(*str++);
        }
    }

    // 接收一个字节（阻塞）
    uint8_t receiveByte() {
        while (!(USART2->SR & USART_SR_RXNE));  // 等待接收缓冲区非空
        return static_cast<uint8_t>(USART2->DR);
    }

    // 中断驱动的接收缓冲区
    static constexpr size_t RX_BUFFER_SIZE = 256;
    CircularBuffer<uint8_t, RX_BUFFER_SIZE> rxBuffer;

    // 中断服务例程
    void irqHandler() {
        if (USART2->SR & USART_SR_RXNE) {
            uint8_t byte = static_cast<uint8_t>(USART2->DR);
            rxBuffer.push(byte);  // 存入环形缓冲区
        }
    }

private:
    // UART寄存器结构体（由CMSIS提供）
    // 这里简化表示
    struct USART_TypeDef {
        volatile uint32_t SR;    // 状态寄存器
        volatile uint32_t DR;    // 数据寄存器
        volatile uint32_t BRR;   // 波特率寄存器
        volatile uint32_t CR1;   // 控制寄存器1
        volatile uint32_t CR2;   // 控制寄存器2
        volatile uint32_t CR3;   // 控制寄存器3
    };
};

// 全局UART实例
UartDriver uart;

// 中断服务例程
extern "C" void USART2_IRQHandler(void) {
```

```cpp
// 通用定时器驱动（以TIM2为例）
// STM32F407 TIM2是32位通用定时器

class TimerDriver {
public:
    struct Config {
        uint32_t prescaler;      // 预分频值
        uint32_t period;         // 自动重载值（周期）
        uint32_t counterMode;    // 计数模式
        bool autoReloadPreload;  // 自动重载预装载
    };

    static void init(const Config& config) {
        // 使能TIM2时钟
        RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;

        // 设置预分频器（将定时器时钟分频）
        // 例如：84MHz / (prescaler+1) = 定时器计数频率
        TIM2->PSC = config.prescaler;

        // 设置自动重载值（计数到这个值后溢出）
        // 计数频率 / (period+1) = 中断频率
        TIM2->ARR = config.period;

        // 计数模式：向上计数
        TIM2->CR1 &= ~TIM_CR1_DIR;

        // 自动重载预装载使能
        if (config.autoReloadPreload) {
            TIM2->CR1 |= TIM_CR1_ARPE;
        }

        // 使能更新中断
        TIM2->DIER |= TIM_DIER_UIE;

        // 配置NVIC
        NVIC_SetPriority(TIM2_IRQn, 6);
        NVIC_EnableIRQ(TIM2_IRQn);

        // 启动定时器
        TIM2->CR1 |= TIM_CR1_CEN;
    }

    // PWM输出配置
    static void initPwm(uint32_t channel, uint32_t dutyCycle) {
        // 使能TIM2时钟
        RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;

        // 配置GPIO为复用功能（PA0 = TIM2_CH1）
        GPIOA->MODER &= ~(3U << (0 * 2));
        GPIOA->MODER |= (2U << (0 * 2));  // 复用模式
        GPIOA->AFRL &= ~(0xFU << (0 * 4));
        GPIOA->AFRL |= (1U << (0 * 4));   // AF1 = TIM2

        // 配置预分频和周期
        TIM2->PSC = 83;    // 84MHz / 84 = 1MHz
        TIM2->ARR = 999;   // 1MHz / 1000 = 1kHz PWM频率

        // PWM模式1配置
        uint32_t ccmrOffset = (channel <= 2) ? 0 : 1;
        uint32_t ccmrShift = ((channel - 1) % 2) * 8;

        volatile uint32_t* ccmr = (channel <= 2) ? &TIM2->CCMR1 : &TIM2->CCMR2;

        // 设置为PWM模式1（OCxM = 110）
        *ccmr &= ~(7U << (ccmrShift + 4));
        *ccmr |= (6U << (ccmrShift + 4));

        // 使能预装载
        *ccmr |= (1U << (ccmrShift + 3));

        // 使能输出
        TIM2->CCER |= (1U << ((channel - 1) * 4));

        // 设置占空比
        switch (channel) {
            case 1: TIM2->CCR1 = dutyCycle; break;
            case 2: TIM2->CCR2 = dutyCycle; break;
            case 3: TIM2->CCR3 = dutyCycle; break;
            case 4: TIM2->CCR4 = dutyCycle; break;
        }

        // 启动定时器
        TIM2->CR1 |= TIM_CR1_CEN;
    }

    // 输入捕获配置（测量脉宽）
    static void initInputCapture(uint32_t channel) {
        // 配置GPIO为复用输入
        GPIOA->MODER &= ~(3U << (0 * 2));
        GPIOA->MODER |= (2U << (0 * 2));
        GPIOA->AFRL &= ~(0xFU << (0 * 4));
        GPIOA->AFRL |= (1U << (0 * 4));

        // 配置输入捕获
        volatile uint32_t* ccmr = &TIM2->CCMR1;
        *ccmr &= ~(3U);          // CC1通道配置为输入
        *ccmr |= (1U << 0);      // IC1映射到TI1
        *ccmr &= ~(7U << 4);     // 输入滤波器：无滤波
        *ccmr &= ~(3U << 2);     // 无预分频

        // 上升沿触发
        TIM2->CCER &= ~(1U << 1);  // 非反相
        TIM2->CCER &= ~(1U << 3);  // 上升沿

        // 使能捕获
        TIM2->CCER |= (1U << 0);

        // 使能捕获/比较中断
        TIM2->DIER |= TIM_DIER_CC1IE;
        NVIC_EnableIRQ(TIM2_IRQn);

        // 启动定时器
        TIM2->CR1 |= TIM_CR1_CEN;
    }
};

// 定时器中断处理
volatile uint32_t timer2Count = 0;

extern "C" void TIM2_IRQHandler(void) {
    if (TIM2->SR & TIM_SR_UIF) {
        TIM2->SR &= ~TIM_SR_UIF;  // 清除更新中断标志
        ++timer2Count;
        // 执行周期性任务
        periodicTask();
    }
}
```

```cpp
// ADC驱动实现（STM32F407 ADC1）
class AdcDriver {
public:
    struct Config {
        uint32_t channel;       // ADC通道
        uint32_t sampleTime;    // 采样时间
        uint32_t resolution;    // 分辨率（12/10/8/6位）
    };

    static void init() {
        // 使能ADC1时钟
        RCC->APB2ENR |= RCC_APB2ENR_ADC1EN;

        // 配置ADC时钟分频（APB2 / 4 = 21MHz）
        ADC->CCR &= ~ADC_CCR_ADCPRE;
        ADC->CCR |= ADC_CCR_ADCPRE_1;

        // 12位分辨率
        ADC1->CR1 &= ~ADC_CR1_RES;

        // 右对齐
        ADC1->CR2 &= ~ADC_CR2_ALIGN;

        // 使能扫描模式（多通道时使用）
        ADC1->CR1 &= ~ADC_CR1_SCAN;

        // 关闭连续转换
        ADC1->CR2 &= ~ADC_CR2_CONT;

        // 关闭DMA
        ADC1->CR2 &= ~ADC_CR2_DMA;

        // 使能ADC
        ADC1->CR2 |= ADC_CR2_ADON;
    }

    // 单次转换读取ADC值
    static uint16_t readChannel(uint32_t channel) {
        // 设置采样时间（通道10，采样时间56周期）
        ADC1->SMPR1 &= ~(7U << 0);
        ADC1->SMPR1 |= (3U << 0);  // 56周期

        // 设置常规序列长度为1
        ADC1->SQR1 &= ~ADC_SQR1_L;
        ADC1->SQR1 &= ~(0x1F << 0);

        // 设置第一个转换为指定通道
        ADC1->SQR3 = channel & 0x1F;

        // 启动软件转换
        ADC1->CR2 |= ADC_CR2_SWSTART;

        // 等待转换完成
        while (!(ADC1->SR & ADC_SR_EOC));

        // 读取转换结果
        return static_cast<uint16_t>(ADC1->DR);
    }

    // DMA连续转换（多通道扫描）
    static void initDmaScan(uint32_t* channels, uint32_t count,
                            uint16_t* resultBuffer) {
        // 使能DMA2时钟
        RCC->AHB1ENR |= RCC_AHB1ENR_DMA2EN;

        // 配置DMA2 Stream0 Channel0（ADC1）
        DMA2_Stream0->CR &= ~DMA_SxCR_EN;
        while (DMA2_Stream0->CR & DMA_SxCR_EN);

        DMA2_Stream0->PAR  = reinterpret_cast<uint32_t>(&(ADC1->DR));
        DMA2_Stream0->M0AR = reinterpret_cast<uint32_t>(resultBuffer);
        DMA2_Stream0->NDTR = count;

        DMA2_Stream0->CR = DMA_SxCR_CHSEL_0   |  // Channel 0
                           DMA_SxCR_MINC       |  // 内存地址自增
                           DMA_SxCR_CIRC       |  // 循环模式
                           DMA_SxCR_PL_0;         // 中优先级

        DMA2_Stream0->CR |= DMA_SxCR_EN;

        // 配置ADC
        ADC1->CR1 |= ADC_CR1_SCAN;       // 扫描模式
        ADC1->CR2 |= ADC_CR2_DMA;        // DMA模式
        ADC1->CR2 |= ADC_CR2_CONT;       // 连续转换
        ADC1->CR2 |= ADC_CR2_DDS;        // DMA不停止

        // 设置序列长度
        ADC1->SQR1 &= ~ADC_SQR1_L;
        ADC1->SQR1 |= ((count - 1) << 20);

        // 设置各通道
        for (uint32_t i = 0; i < count && i < 16; ++i) {
            volatile uint32_t* sqr = (i < 6) ? &ADC1->SQR3 :
                                     (i < 12) ? &ADC1->SQR2 : &ADC1->SQR1;
            uint32_t shift = (i % 6) * 5;
            *sqr &= ~(0x1FU << shift);
            *sqr |= ((channels[i] & 0x1F) << shift);
        }

        // 启动转换
        ADC1->CR2 |= ADC_CR2_SWSTART;
    }

    // 将ADC值转换为电压
    static float toVoltage(uint16_t adcValue, float vref = 3.3f) {
        return static_cast<float>(adcValue) * vref / 4096.0f;
    }

    // 将ADC值转换为温度（内部温度传感器）
    static float toTemperature(uint16_t adcValue) {
        // STM32F407内部温度传感器公式
        // V25 = 0.76V（25°C时的电压）
        // Avg_Slope = 2.5mV/°C
        float vSense = toVoltage(adcValue);
        float temperature = (vSense - 0.76f) / 0.0025f + 25.0f;
        return temperature;
    }
};
```

## 6. CMSIS标准

### 6.1 CMSIS概述

CMSIS（Cortex Microcontroller Software Interface Standard）是ARM制定的Cortex-M微控制器软件接口标准，提供统一的硬件抽象层。

```
CMSIS组成：
├── CMSIS-Core    → 核心寄存器访问、内在函数
├── CMSIS-Driver  → 统一的外设驱动接口
├── CMSIS-DSP     → 数字信号处理库
├── CMSIS-NN      → 神经网络推理库
└── CMSIS-RTOS    → 实时操作系统标准接口
```

```cpp
// ❌ 使用ODR进行位操作（非原子，可能被中断打断）
void toggleLedBad() {
    GPIOA->ODR ^= (1U << 5);  // 读-改-写，非原子操作
    // 如果在读取ODR和写回之间发生中断，中断中修改了ODR，
    // 则中断的修改会被覆盖
}

// ✅ 使用BSRR进行原子位操作
void toggleLedGood() {
    // 使用BSRR的置位和复位功能
    if (GPIOA->ODR & (1U << 5)) {
        GPIOA->BSRR = (1U << (5 + 16));  // 复位PA5
    } else {
        GPIOA->BSRR = (1U << 5);          // 置位PA5
    }
}

// 陷阱5：忘记清除中断标志
// ❌ 未清除中断标志导致无限重入
extern "C" void USART2_IRQHandler(void) {
    uint8_t data = USART2->DR;  // 读取DR会清除RXNE标志
    // 但如果还有其他中断标志未清除，会立即再次进入
}

// ✅ 显式清除所有相关中断标志
extern "C" void USART2_IRQHandler(void) {
    if (USART2->SR & USART_SR_RXNE) {
        uint8_t data = USART2->DR;  // 读取DR清除RXNE
        processReceivedData(data);
    }
    if (USART2->SR & USART_SR_ORE) {
        (void)USART2->DR;  // 读取DR清除ORE
        handleOverrunError();
    }
}

// 陷阱6：Cache一致性问题（Cortex-M7）
// ❌ DMA写入的内存区域被CPU缓存，CPU读到的是旧数据
// Cortex-M7有D-Cache，DMA写入RAM后CPU可能读到缓存中的旧值

// ✅ 解决方案：
// 1. 将DMA缓冲区放在非缓存区域（如STM32H7的0x24000000区域）
// 2. 在读取DMA数据前手动无效化D-Cache
//    SCB_InvalidateDCache_by_Addr(buffer, size);
// 3. 禁用DMA缓冲区区域的D-Cache（通过MPU配置）
```

```cpp
// 最小化的串口输出，不使用printf
class DebugUart {
public:
    static void init() {
        RCC->APB1ENR |= RCC_APB1ENR_USART2EN;
        RCC->AHB1ENR |= RCC_AHB1ENR_GPIOAEN;

        GPIOA->MODER |= (2U << (2 * 2)) | (2U << (3 * 2));
        GPIOA->AFRL |= (7U << (2 * 4)) | (7U << (3 * 4));

        USART2->BRR = 84000000 / 115200;
        USART2->CR1 = USART_CR1_TE | USART_CR1_UE;
    }

    static void send(const char* str) {
        while (*str) {
            while (!(USART2->SR & USART_SR_TXE));
            USART2->DR = *str++;
        }
    }

    static void sendHex(uint32_t value) {
        const char hex[] = "0123456789ABCDEF";
        char buf[11] = "0x";
        for (int i = 7; i >= 0; --i) {
            buf[2 + (7 - i)] = hex[(value >> (i * 4)) & 0xF];
        }
        buf[10] = '\0';
        send(buf);
    }

    static void sendDec(int32_t value) {
        char buf[12];
        int i = 10;
        bool negative = value < 0;
        if (negative) value = -value;

        buf[10] = '\0';
        do {
            buf[--i] = '0' + (value % 10);
            value /= 10;
        } while (value > 0 && i > 0);

        if (negative) buf[--i] = '-';
        send(&buf[i]);
    }
};
```

```cpp
// 实践1：时钟树配置
// 不要依赖SystemInit()的默认配置，根据实际需求精确配置
void customClockConfig() {
    // 目标：168MHz主频（STM32F407最大频率）
    // 外部晶振8MHz → PLL → 168MHz

    // 1. 使能HSE（外部高速晶振）
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY));  // 等待HSE就绪

    // 2. 配置PLL
    // PLL_VCO = HSE * (PLLN / PLLM) = 8MHz * (336 / 8) = 336MHz
    // SYSCLK = PLL_VCO / PLLP = 336MHz / 2 = 168MHz
    // USB OTG SDIO RNG = PLL_VCO / PLLQ = 336MHz / 7 = 48MHz
    RCC->PLLCFGR = (8U << 0)    |  // PLLM = 8
                   (336U << 6)  |  // PLLN = 336
                   (0U << 16)   |  // PLLP = 2
                   (7U << 24)   |  // PLLQ = 7
                   RCC_PLLCFGR_PLLSRC_HSE;

    // 3. 使能PLL
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY));

    // 4. 配置Flash预取、指令缓存、数据缓存、等待周期
    // 168MHz需要5个等待周期
    FLASH->ACR = FLASH_ACR_ICEN | FLASH_ACR_DCEN |
                 FLASH_ACR_PRFTEN | FLASH_ACR_LATENCY_5WS;

    // 5. 切换系统时钟到PLL
    RCC->CFGR &= ~RCC_CFGR_SW;
    RCC->CFGR |= RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL);

    // 更新SystemCoreClock变量
    SystemCoreClock = 168000000;
}

// 实践2：低功耗设计
// 在电池供电设备中，降低功耗至关重要
void enterStopMode() {
    // 1. 关闭不需要的外设时钟
    RCC->AHB1ENR &= ~(RCC_AHB1ENR_GPIOBEN | RCC_AHB1ENR_GPIOCEN);
    RCC->APB1ENR &= ~RCC_APB1ENR_USART2EN;
    RCC->APB2ENR &= ~RCC_APB2ENR_ADC1EN;

    // 2. 配置唤醒源（EXTI中断）
    configureWakeupInterrupt();

    // 3. 进入Stop模式
    // Stop模式：所有时钟停止，SRAM和寄存器内容保持
    // 功耗：约20-100uA（取决于外设配置）
    PWR->CR |= PWR_CR_FPDS;  // Flash进入深度掉电模式
    SCB->SCR |= SCB_SCR_SLEEPDEEP_Msk;
    __WFI();  // 等待中断唤醒

    // 唤醒后继续执行
    SCB->SCR &= ~SCB_SCR_SLEEPDEEP_Msk;

    // 4. 重新使能外设时钟
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIOBEN | RCC_AHB1ENR_GPIOCEN;
    RCC->APB1ENR |= RCC_APB1ENR_USART2EN;

    // 5. 重新配置时钟（Stop模式唤醒后使用HSI）
    customClockConfig();
}

// 实践3：固件升级支持（Bootloader）
// 在Flash中预留Bootloader区域
// 0x08000000 - 0x0800FFFF: Bootloader (64KB)
// 0x08010000 - 0x080FFFFF: Application (960KB)

// 应用程序的链接脚本需要修改起始地址
// FLASH (rx) : ORIGIN = 0x08010000, LENGTH = 960K

// 应用程序启动时需要重定位中断向量表
void relocateVectorTable() {
    // 将中断向量表重定位到应用程序起始位置
    SCB->VTOR = 0x08010000;
}

// Bootloader检查是否需要升级
void bootloaderCheck() {
    // 检查升级标志（存储在RTC备份寄存器或特定Flash区域）
    if (upgradeFlagIsSet()) {
        // 进入升级模式
        enterUpgradeMode();
    } else {
        // 跳转到应用程序
        jumpToApplication(0x08010000);
    }
}

void jumpToApplication(uint32_t address) {
    // 检查应用程序是否有效
    uint32_t appStack = *reinterpret_cast<uint32_t*>(address);
    uint32_t appEntry = *reinterpret_cast<uint32_t*>(address + 4);

    // 验证栈指针值是否在RAM范围内
    if (appStack < 0x20000000 || appStack > 0x20020000) {
        return;  // 无效应用程序
    }

    // 禁用所有中断
    __disable_irq();

    // 重定位中断向量表
    SCB->VTOR = address;

    // 设置栈指针
    __set_MSP(appStack);

    // 跳转到应用程序的Reset_Handler
    auto resetHandler = reinterpret_cast<void(*)(void)>(appEntry);
    resetHandler();
}
```

## 9. 本章小结

本章深入讲解了裸机编程的核心技术，要点如下：

| 主题 | 核心要点 |
|------|---------|
| 启动代码 | 复位向量、.data复制、.bss清零、调用main |
| 链接脚本 | MEMORY区域定义、SECTIONS段放置、符号导出 |
| 内存映射 | Flash/RAM/外设地址空间、寄存器结构体映射 |
| 中断与异常 | NVIC配置、中断优先级、临界区保护 |
| 寄存器操作 | 位操作宏、类型安全封装、UART/Timer/ADC驱动 |
| CMSIS标准 | Core内在函数、DSP库、驱动接口 |
| 常见陷阱 | .data段未复制、栈溢出、ISR中阻塞、BSRR误用、Cache一致性 |
| 调试技巧 | GPIO翻转、DWT观察点、轻量级串口输出 |
| 项目实践 | 时钟树配置、低功耗设计、Bootloader固件升级 |

**关键理解**：

1. **启动代码是系统的第一步**：从复位到main函数，每一步都必须正确
2. **链接脚本定义内存布局**：理解内存映射是裸机编程的基础
3. **volatile是关键**：所有外设寄存器访问必须使用volatile
4. **中断是实时响应的核心**：正确配置NVIC和优先级至关重要
5. **CMSIS提供标准抽象**：使用CMSIS可以写出可移植的代码
6. **ISR中只做最少的工作**：设置标志，在主循环中处理
7. **Bootloader是固件升级的基础**：理解向量表重定位和跳转机制
