# 嵌入式C++实践

> 掌握嵌入式C++约束、静态分配模式、HAL设计与驱动开发

***

> **The most important property of a program is whether it accomplishes the intention of its user.** — C.A.R. Hoare
> （程序最重要的属性是它是否完成了用户的意图。）

> **In embedded C++, you pay only for what you use, and you must know exactly what you're paying.** — Unknown
> （在嵌入式C++中，你只为使用的东西付费，而且必须确切知道你在付什么。）

***

> **🎯 因地制宜，量体裁衣。**
>
> （嵌入式C++需要根据硬件约束量身定制，不能照搬桌面开发的习惯。）

> 💡 **通俗理解 - 嵌入式C++有什么不同？**

想象一下：

- **桌面C++** 就像"开SUV"，空间大、动力足，可以随意装东西
- **嵌入式C++** 就像"开卡丁车"，空间小、资源少，每克重量都要精打细算
- **异常处理** 就像"安全气囊"，虽然安全但太重了，卡丁车装不起
- **动态内存** 就像"加油卡"，方便但不知道什么时候会用完

嵌入式C++需要砍掉"奢侈"特性，保留"实用"特性，在有限的资源上写出高效的代码。

> 🔬 **抽象理解 - 嵌入式C++约束**：
>
> - **无异常**：异常处理需要额外栈空间和代码，延迟不可预测
> - **无RTTI**：运行时类型识别需要额外内存存储类型信息
> - **受限堆**：动态内存分配导致碎片化和不确定延迟
> - **静态分配**：所有内存在编译期确定，运行时零分配

## 1. 嵌入式C++约束

### 1.1 禁用异常

```cpp
// 编译选项：-fno-exceptions

// ❌ 使用异常的代码
int divide(int a, int b) {
    if (b == 0) {
        throw std::runtime_error("除零错误");  // ❌ 嵌入式禁用异常
    }
    return a / b;
}

// ✅ 使用错误码的替代方案
enum class Error : uint8_t {
    Ok = 0,
    InvalidParam,
    DivideByZero,
    Timeout,
    HardwareFault
};

struct Result {
    int value;
    Error error;
};

Result divide(int a, int b) {
    if (b == 0) {
        return {0, Error::DivideByZero};
    }
    return {a / b, Error::Ok};
}

// ✅ 使用std::expected风格（C++23，可自行实现）
template<typename T, typename E = Error>
class Expected {
public:
    constexpr Expected(const T& value) : hasValue_(true), value_(value) {}
    constexpr Expected(const E& error) : hasValue_(false), error_(error) {}

    constexpr bool hasValue() const { return hasValue_; }
    constexpr explicit operator bool() const { return hasValue_; }

    constexpr const T& value() const { return value_; }
    constexpr const E& error() const { return error_; }

    constexpr T valueOr(const T& defaultValue) const {
        return hasValue_ ? value_ : defaultValue;
    }

private:
    bool hasValue_;
    union {
        T value_;
        E error_;
    };
};

// 使用Expected
Expected<int> safeDivide(int a, int b) {
    if (b == 0) return Expected<int>(Error::DivideByZero);
    return Expected<int>(a / b);
}

void example() {
    auto result = safeDivide(10, 0);
    if (result) {
        int val = result.value();
    } else {
        Error err = result.error();
        // 处理错误
    }
}
```

### 1.3 受限堆与替代方案

```cpp
// ❌ 避免使用动态内存
#include <vector>
#include <string>

void badExample() {
    std::vector<int> data;         // ❌ 使用堆分配
    data.push_back(42);           // ❌ 不确定延迟

    std::string msg = "Hello";    // ❌ 可能使用堆分配
}

// ✅ 使用静态容器
template<typename T, size_t N>
class StaticVector {
public:
    constexpr bool push_back(const T& item) {
        if (size_ >= N) return false;
        data_[size_] = item;
        ++size_;
        return true;
    }

    constexpr T& operator[](size_t i) { return data_[i]; }
    constexpr const T& operator[](size_t i) const { return data_[i]; }
    constexpr T* begin() { return data_.data(); }
    constexpr T* end() { return data_.data() + size_; }
    constexpr size_t size() const { return size_; }
    static constexpr size_t capacity() { return N; }

private:
    std::array<T, N> data_{};
    size_t size_ = 0;
};

// ✅ 使用静态字符串
template<size_t N>
class StaticString {
public:
    constexpr StaticString() : len_(0) { data_[0] = '\0'; }

    constexpr StaticString(const char* str) {
        size_t i = 0;
        while (str[i] && i < N - 1) {
            data_[i] = str[i];
            ++i;
        }
        data_[i] = '\0';
        len_ = i;
    }

    constexpr bool append(const char* str) {
        size_t i = 0;
        while (str[i] && len_ < N - 1) {
            data_[len_++] = str[i++];
        }
        data_[len_] = '\0';
        return str[i] == '\0';
    }

    constexpr const char* c_str() const { return data_.data(); }
    constexpr size_t length() const { return len_; }

private:
    std::array<char, N + 1> data_{};
    size_t len_ = 0;
};

void goodExample() {
    StaticVector<int, 32> data;     // ✅ 栈上分配，最大32个元素
    data.push_back(42);

    StaticString<64> msg("Hello");  // ✅ 栈上分配，最大64字符
    msg.append(" World");
}
```

### 2.2 静态对象管理

```cpp
// 嵌入式系统中全局对象的构造和析构需要特别注意

// 问题：C++全局对象在main之前构造，但此时硬件可能未初始化

// 解决方案1：延迟构造（使用placement new）
template<typename T>
class LazyStatic {
public:
    // 不自动构造
    LazyStatic() = default;

    // 手动构造
    template<typename... Args>
    void construct(Args&&... args) {
        new(storage_) T(std::forward<Args>(args)...);
        constructed_ = true;
    }

    // 手动析构
    void destruct() {
        if (constructed_) {
            get().~T();
            constructed_ = false;
        }
    }

    T& get() {
        return *reinterpret_cast<T*>(storage_);
    }

    const T& get() const {
        return *reinterpret_cast<const T*>(storage_);
    }

    T* operator->() { return &get(); }
    const T* operator->() const { return &get(); }

    bool isConstructed() const { return constructed_; }

private:
    alignas(T) uint8_t storage_[sizeof(T)];
    bool constructed_ = false;
};

// 使用示例
LazyStatic<SensorDriver> g_sensor;

int main() {
    // 先初始化硬件
    hardware_init();

    // 再构造全局对象
    g_sensor.construct(Uart1Config{});

    // 正常使用
    g_sensor->read();

    // 不需要时手动析构
    g_sensor.destruct();

    return 0;
}
```

```cpp
// 通用GPIO接口
class IGpio {
public:
    enum class Mode : uint8_t {
        Input,
        Output,
        OpenDrain,
        Alternate,
        Analog
    };

    enum class Pull : uint8_t {
        None,
        Up,
        Down
    };

    enum class Speed : uint8_t {
        Low,
        Medium,
        High,
        VeryHigh
    };

    virtual ~IGpio() = default;

    virtual void setMode(Mode mode) = 0;
    virtual void setPull(Pull pull) = 0;
    virtual void setSpeed(Speed speed) = 0;

    virtual void write(bool high) = 0;
    virtual bool read() const = 0;
    virtual void toggle() = 0;
};

// STM32实现
class Stm32Gpio : public IGpio {
public:
    constexpr Stm32Gpio(GPIO_TypeDef* port, uint8_t pin)
        : port_(port), pin_(pin) {}

    void setMode(Mode mode) override {
        uint32_t modeVal = static_cast<uint32_t>(mode);
        port_->MODER &= ~(3U << (pin_ * 2));
        port_->MODER |= (modeVal << (pin_ * 2));
    }

    void setPull(Pull pull) override {
        uint32_t pullVal = static_cast<uint32_t>(pull);
        port_->PUPDR &= ~(3U << (pin_ * 2));
        port_->PUPDR |= (pullVal << (pin_ * 2));
    }

    void setSpeed(Speed speed) override {
        uint32_t speedVal = static_cast<uint32_t>(speed);
        port_->OSPEEDR &= ~(3U << (pin_ * 2));
        port_->OSPEEDR |= (speedVal << (pin_ * 2));
    }

    void write(bool high) override {
        if (high) {
            port_->BSRR = (1U << pin_);
        } else {
            port_->BSRR = (1U << (pin_ + 16));
        }
    }

    bool read() const override {
        return (port_->IDR & (1U << pin_)) != 0;
    }

    void toggle() override {
        port_->ODR ^= (1U << pin_);
    }

private:
    GPIO_TypeDef* port_;
    uint8_t pin_;
};

// 通用UART接口
class IUart {
public:
    struct Config {
        uint32_t baudRate = 115200;
        uint8_t dataBits = 8;
        uint8_t stopBits = 1;
        enum class Parity : uint8_t { None, Even, Odd } parity = Parity::None;
    };

    virtual ~IUart() = default;

    virtual bool init(const Config& config) = 0;
    virtual bool sendByte(uint8_t byte) = 0;
    virtual bool receiveByte(uint8_t& byte) = 0;
    virtual size_t send(const uint8_t* data, size_t len) = 0;
    virtual size_t receive(uint8_t* data, size_t maxLen) = 0;
    virtual bool setBaudRate(uint32_t baud) = 0;
};
```

```cpp
// GPIO引脚类型（编译期确定）
template<uint32_t PortAddr, uint8_t Pin>
class GpioPinStatic {
public:
    static constexpr GPIO_TypeDef* port = reinterpret_cast<GPIO_TypeDef*>(PortAddr);
    static constexpr uint8_t pin = Pin;
    static constexpr uint32_t bitMask = 1U << Pin;

    static void setMode(IGpio::Mode mode) {
        port->MODER &= ~(3U << (pin * 2));
        port->MODER |= (static_cast<uint32_t>(mode) << (pin * 2));
    }

    static void write(bool high) {
        if (high) {
            port->BSRR = bitMask;
        } else {
            port->BSRR = bitMask << 16;
        }
    }

    static bool read() {
        return (port->IDR & bitMask) != 0;
    }

    static void toggle() {
        port->ODR ^= bitMask;
    }
};

// LED模板（组合GPIO引脚）
template<typename Pin>
class Led {
public:
    static void init() {
        Pin::setMode(IGpio::Mode::Output);
    }

    static void on()  { Pin::write(true); }
    static void off() { Pin::write(false); }
    static void toggle() { Pin::toggle(); }
};

// 按钮模板
template<typename Pin>
class Button {
public:
    static void init() {
        Pin::setMode(IGpio::Mode::Input);
    }

    static bool isPressed() {
        return !Pin::read();  // 假设低电平表示按下
    }

    static bool debounce() {
        // 简单的软件消抖
        if (!isPressed()) return false;
        delayMs(20);
        return isPressed();
    }
};

// 板级定义

    using LedRed   = Led<GpioPinStatic<0x40020000, 5>>;
    using LedGreen = Led<GpioPinStatic<0x40020000, 6>>;
    using UserBtn  = Button<GpioPinStatic<0x40020400, 0>>;
}

// 使用示例
void boardInit() {
    board::LedRed::init();
    board::LedGreen::init();
    board::UserBtn::init();
}

void applicationLoop() {
    if (board::UserBtn::debounce()) {
        board::LedRed::toggle();
    }
    board::LedGreen::on();
}
```

## 4. 驱动开发模式

### 4.1 SPI驱动

```cpp
// SPI驱动实现

template<typename SpiInstance>
class SpiDriver {
public:
    struct Config {
        uint32_t clockSpeed = 1000000;  // 1MHz
        uint8_t mode = 0;               // SPI模式0-3
        bool lsbFirst = false;
        uint8_t dataBits = 8;
    };

    static bool init(const Config& config) {
        // 使能SPI时钟
        enableClock();

        // 配置SPI
        uint32_t cr1 = SPI_CR1_SSM | SPI_CR1_SSI | SPI_CR1_MSTR;

        // 时钟分频
        uint32_t div = SystemCoreClock / config.clockSpeed;
        uint32_t br = 0;
        while (div > 2 && br < 7) { div >>= 1; ++br; }
        cr1 |= (br << 3);

        // SPI模式
        if (config.mode & 0x01) cr1 |= SPI_CR1_CPHA;
        if (config.mode & 0x02) cr1 |= SPI_CR1_CPOL;

        // 位序
        if (config.lsbFirst) cr1 |= SPI_CR1_LSBFIRST;

        // 数据位数
        if (config.dataBits == 16) cr1 |= SPI_CR1_DFF;

        SpiInstance::regs->CR1 = cr1;

        // 使能SPI
        SpiInstance::regs->CR1 |= SPI_CR1_SPE;

        return true;
    }

    static uint8_t transfer(uint8_t txData) {
        // 等待发送缓冲区空
        while (!(SpiInstance::regs->SR & SPI_SR_TXE));

        // 发送数据
        SpiInstance::regs->DR = txData;

        // 等待接收缓冲区非空
        while (!(SpiInstance::regs->SR & SPI_SR_RXNE));

        // 返回接收数据
        return static_cast<uint8_t>(SpiInstance::regs->DR);
    }

    static void transfer(const uint8_t* txData, uint8_t* rxData, size_t len) {
        for (size_t i = 0; i < len; ++i) {
            uint8_t tx = txData ? txData[i] : 0xFF;
            uint8_t rx = transfer(tx);
            if (rxData) rxData[i] = rx;
        }
    }

    // DMA传输（高效）
    static void transferDma(const uint8_t* txData, uint8_t* rxData, size_t len) {
        // 配置DMA通道
        configureDma(txData, rxData, len);

        // 使能DMA请求
        SpiInstance::regs->CR2 |= SPI_CR2_TXDMAEN | SPI_CR2_RXDMAEN;

        // 等待DMA完成
        waitForDmaComplete();
    }

private:
    static void enableClock();
    static void configureDma(const uint8_t* tx, uint8_t* rx, size_t len);
    static void waitForDmaComplete();
};

// SPI设备驱动（以Flash存储器为例）
template<typename Spi, typename CsPin>
class W25qFlash {
public:
    static void init() {
        CsPin::setMode(IGpio::Mode::Output);
        CsPin::write(true);  // CS高电平（未选中）

        typename SpiDriver<Spi>::Config config;
        config.clockSpeed = 1000000;  // 1MHz（初始化用低速）
        config.mode = 0;
        SpiDriver<Spi>::init(config);
    }

    static uint32_t readId() {
        csLow();
        SpiDriver<Spi>::transfer(0x9F);  // Read ID命令
        uint8_t manufacturer = SpiDriver<Spi>::transfer(0xFF);
        uint8_t memoryType = SpiDriver<Spi>::transfer(0xFF);
        uint8_t capacity = SpiDriver<Spi>::transfer(0xFF);
        csHigh();

        return (static_cast<uint32_t>(manufacturer) << 16) |
               (static_cast<uint32_t>(memoryType) << 8) |
               capacity;
    }

    static void readData(uint32_t address, uint8_t* buffer, size_t len) {
        csLow();
        SpiDriver<Spi>::transfer(0x03);  // Read Data命令
        SpiDriver<Spi>::transfer((address >> 16) & 0xFF);
        SpiDriver<Spi>::transfer((address >> 8) & 0xFF);
        SpiDriver<Spi>::transfer(address & 0xFF);
        SpiDriver<Spi>::transfer(nullptr, buffer, len);
        csHigh();
    }

    static void writeEnable() {
        csLow();
        SpiDriver<Spi>::transfer(0x06);  // Write Enable命令
        csHigh();
    }

    static void sectorErase(uint32_t address) {
        writeEnable();
        csLow();
        SpiDriver<Spi>::transfer(0x20);  // Sector Erase命令
        SpiDriver<Spi>::transfer((address >> 16) & 0xFF);
        SpiDriver<Spi>::transfer((address >> 8) & 0xFF);
        SpiDriver<Spi>::transfer(address & 0xFF);
        csHigh();
        waitForReady();
    }

private:
    static void csLow() { CsPin::write(false); }
    static void csHigh() { CsPin::write(true); }

    static void waitForReady() {
        csLow();
        SpiDriver<Spi>::transfer(0x05);  // Read Status Register
        while (SpiDriver<Spi>::transfer(0xFF) & 0x01);  // BUSY位
        csHigh();
    }
};
```

```cpp
        Ack = 0,
        Nack = 1
    };

    static bool init(uint32_t clockSpeed) {
        // 使能I2C时钟
        RCC->APB1ENR |= RCC_APB1ENR_I2C1EN;

        // 复位I2C
        I2C1->CR1 = I2C_CR1_SWRST;
        I2C1->CR1 = 0;

        // 配置时钟
        uint32_t ccr = SystemCoreClock / (2 * clockSpeed);
        I2C1->CCR = ccr;
        I2C1->TRISE = SystemCoreClock / 1000000 + 1;

        // 使能I2C
        I2C1->CR1 |= I2C_CR1_PE;

        return true;
    }

    // 写入数据到I2C设备
    static bool write(uint8_t deviceAddr, const uint8_t* data, size_t len) {
        // 发送起始条件
        sendStart();
        if (!waitFlag(I2C_SR1_SB)) return false;

        // 发送设备地址（写模式）
        I2C1->DR = (deviceAddr << 1);
        if (!waitFlag(I2C_SR1_ADDR)) return false;
        clearAddrFlag();

        // 发送数据
        for (size_t i = 0; i < len; ++i) {
            I2C1->DR = data[i];
            if (!waitFlag(I2C_SR1_TXE)) return false;
        }

        // 发送停止条件
        sendStop();
        return true;
    }

    // 从I2C设备读取数据
    static bool read(uint8_t deviceAddr, uint8_t* data, size_t len) {
        // 发送起始条件
        sendStart();
        if (!waitFlag(I2C_SR1_SB)) return false;

        // 发送设备地址（读模式）
        I2C1->DR = (deviceAddr << 1) | 1;
        if (!waitFlag(I2C_SR1_ADDR)) return false;

        if (len == 1) {
            // 单字节读取
            I2C1->CR1 &= ~I2C_CR1_ACK;  // 发送NACK
            clearAddrFlag();
            sendStop();
            if (!waitFlag(I2C_SR1_RXNE)) return false;
            data[0] = static_cast<uint8_t>(I2C1->DR);
        } else {
            // 多字节读取
            I2C1->CR1 |= I2C_CR1_ACK;
            clearAddrFlag();

            for (size_t i = 0; i < len; ++i) {
                if (i == len - 1) {
                    I2C1->CR1 &= ~I2C_CR1_ACK;  // 最后一个字节发NACK
                    sendStop();
                }
                if (!waitFlag(I2C_SR1_RXNE)) return false;
                data[i] = static_cast<uint8_t>(I2C1->DR);
            }
        }

        I2C1->CR1 |= I2C_CR1_ACK;  // 恢复ACK
        return true;
    }

private:
    static void sendStart() { I2C1->CR1 |= I2C_CR1_START; }
    static void sendStop()  { I2C1->CR1 |= I2C_CR1_STOP; }
    static void clearAddrFlag() {
        (void)I2C1->SR1;
        (void)I2C1->SR2;
    }

    static bool waitFlag(uint32_t flag, uint32_t timeout = 10000) {
        while (timeout--) {
            if (I2C1->SR1 & flag) return true;
        }
        return false;
    }
};
```

```cpp
template<typename T, typename E = Error>

public:
    constexpr Result(const T& value) : hasValue_(true), value_(value) {}
    constexpr Result(const E& error) : hasValue_(false), error_(error) {}

    constexpr bool ok() const { return hasValue_; }
    constexpr const T& value() const { return value_; }
    constexpr const E& error() const { return error_; }

    // 链式操作
    template<typename Func>
    auto map(Func f) -> Result<decltype(f(value_)), E> {
        if (hasValue_) return Result<decltype(f(value_)), E>(f(value_));
        return Result<decltype(f(value_)), E>(error_);
    }

    template<typename Func>
    auto andThen(Func f) -> decltype(f(value_)) {
        if (hasValue_) return f(value_);
        return decltype(f(value_))(error_);
    }

private:
    bool hasValue_;
    union {
        T value_;
        E error_;
    };
```

```cpp
// 实践1：输入验证
// 所有来自外部的输入都必须验证
class SafeSensorReader {
public:
    Expected<float, Error> read() {
        // 1. 读取原始值
        uint16_t raw = AdcDriver::readChannel(channel_);

        // 2. 范围检查
        if (raw == 0 || raw >= 4095) {
            return Expected<float, Error>(Error::SensorFault);
        }

        // 3. 转换
        float value = AdcDriver::toVoltage(raw);

        // 4. 合理性检查
        if (value < minValid_ || value > maxValid_) {
            return Expected<float, Error>(Error::OutOfRange);
        }

        // 5. 变化率检查
        if (lastValid_.hasValue()) {
            float delta = std::abs(value - lastValid_.value());
            if (delta > maxRateOfChange_) {
                return Expected<float, Error>(Error::RateExceeded);
            }
        }

        lastValid_ = Expected<float, Error>(value);
        return Expected<float, Error>(value);
    }

private:
    uint32_t channel_;
    float minValid_;
    float maxValid_;
    float maxRateOfChange_;
    Expected<float, Error> lastValid_;
};

// 实践2：安全的状态转换
// 使用枚举类和显式转换函数，避免非法状态
class SafeStateHandler {
public:
    enum class State : uint8_t {
        Init,
        Idle,
        Running,
        Error,
        Shutdown
    };

    // 状态转换必须通过此函数，不允许直接修改
    bool transition(State newState) {
        if (!isValidTransition(currentState_, newState)) {
            return false;  // 非法转换被拒绝
        }
        currentState_ = newState;
        onStateEntry(newState);
        return true;
    }

    State current() const { return currentState_; }

private:
    State currentState_ = State::Init;

    bool isValidTransition(State from, State to) {
        switch (from) {
            case State::Init:    return to == State::Idle || to == State::Error;
            case State::Idle:    return to == State::Running || to == State::Shutdown;
            case State::Running: return to == State::Idle || to == State::Error;
            case State::Error:   return to == State::Shutdown;
            case State::Shutdown: return false;
            default: return false;
        }
    }

    void onStateEntry(State state);
};

// 实践3：资源管理RAII模式
// 确保资源在任何情况下都能正确释放
class ScopedGpio {
public:
    ScopedGpio(uint32_t portBase, uint8_t pin, bool initialState = false)
        : port_(reinterpret_cast<GPIO_TypeDef*>(portBase)), pin_(pin) {
        // 配置GPIO为输出
        port_->MODER &= ~(3U << (pin_ * 2));
        port_->MODER |= (1U << (pin_ * 2));
        write(initialState);
    }

    ~ScopedGpio() {
        // 析构时自动将GPIO设为安全状态（低电平）
        port_->BSRR = (1U << (pin_ + 16));
    }

    void write(bool high) {
        if (high) port_->BSRR = (1U << pin_);
        else port_->BSRR = (1U << (pin_ + 16));
    }

    // 禁止拷贝和移动
    ScopedGpio(const ScopedGpio&) = delete;
    ScopedGpio& operator=(const ScopedGpio&) = delete;

private:
    GPIO_TypeDef* port_;
    uint8_t pin_;
};

// 使用示例：即使函数异常退出（虽然嵌入式禁用异常），
// 或提前return，GPIO也会被正确释放
void controlActuator() {
    ScopedGpio enablePin(0x40020000, 8, true);  // PA8使能
    ScopedGpio dirPin(0x40020000, 9, false);    // PA9方向

    // 执行控制逻辑...
    if (sensorFault) return;  // 提前返回，enablePin和dirPin自动析构

    // 正常结束也会自动析构
}
```

## 8. 本章小结

本章讲解了嵌入式C++的具体实践，核心要点如下：

| 主题 | 核心要点 |
|------|---------|
| C++约束 | 禁用异常/RTTI/堆、使用错误码/Expected、静态容器 |
| 静态分配 | 编译期配置、延迟构造、constexpr构造 |
| HAL设计 | 接口抽象、模板零开销、板级定义 |
| 驱动开发 | SPI/I2C驱动、DMA传输、设备驱动模板 |
| 测试策略 | HIL测试、模拟层测试、测试宏 |
| 常见陷阱 | 全局构造顺序、虚函数开销、模板膨胀、placement new |
| 调试技巧 | static_assert、constexpr验证、分支预测优化、ITCM代码段 |
| 项目实践 | 强类型、CRTP编译期多态、Result错误处理链 |

**关键理解**：

1. **禁用特性不等于放弃C++**：保留类、模板、constexpr等零开销特性
2. **静态分配是首选**：所有内存在编译期确定，避免运行时不确定性
3. **HAL是可移植性的关键**：良好的HAL设计让代码可以在不同平台间迁移
4. **模板实现零开销抽象**：编译期确定所有参数，运行时无额外开销
5. **测试需要特殊策略**：HIL测试和模拟层测试互补
6. **CRTP替代虚函数**：在性能关键路径上使用编译期多态
7. **强类型防止配置错误**：用类型系统在编译期捕获参数顺序错误
