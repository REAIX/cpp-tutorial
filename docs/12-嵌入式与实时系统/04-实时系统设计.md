> 💡 **通俗理解 - 什么是实时系统设计？**

想象一下：

- **普通软件** 就像"写文章"，写错了可以修改，没什么大不了
- **实时系统** 就像"造飞机"，一个螺丝松了可能导致坠机
- **WCET** 就像"最慢厨师做一道菜的时间"，必须保证不超过这个时间
- **看门狗** 就像"定时签到"，如果厨师忘了签到，系统就认为出问题了

实时系统设计的核心：**在最坏情况下也能保证系统正确运行**。

> 🔬 **抽象理解 - 实时系统设计要素**：
>
> - **WCET分析**：最坏执行时间，实时性的数学保证
> - **调度理论**：速率单调调度、EDF等，保证任务可调度
> - **容错设计**：看门狗、冗余、降级，确保系统不崩溃
> - **安全标准**：IEC 61508、DO-178C，合规性要求

## 1. 实时性分析

### 1.1 WCET（最坏执行时间）

```cpp
// WCET分析 - 确定任务在最坏情况下的执行时间

// WCET分析方法：
// 1. 静态分析：通过代码分析确定WCET
// 2. 测量法：在实际硬件上测量最坏情况
// 3. 混合法：结合静态分析和测量

// 影响WCET的因素：
// - 流水线效应
// - 缓存命中/未命中
// - 分支预测
// - 中断干扰
// - 总线竞争

// 示例：测量WCET
#include <cstdint>

class WcetMeasurement {
public:
    static void begin() {
        // 使用DWT（Data Watchpoint and Trace）周期计数器
        CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
        DWT->CYCCNT = 0;
        DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
        startCycles_ = DWT->CYCCNT;
    }

    static uint32_t end() {
        uint32_t endCycles = DWT->CYCCNT;
        uint32_t elapsed = endCycles - startCycles_;

        // 更新最大值
        if (elapsed > maxCycles_) {
            maxCycles_ = elapsed;
        }

        return elapsed;
    }

    static uint32_t getMaxCycles() { return maxCycles_; }
    static uint32_t getMaxMicroseconds() {
        return maxCycles_ / (SystemCoreClock / 1000000);
    }

    static void reset() { maxCycles_ = 0; }

private:
    static uint32_t startCycles_;
    static uint32_t maxCycles_;
};

uint32_t WcetMeasurement::startCycles_ = 0;
uint32_t WcetMeasurement::maxCycles_ = 0;

// 测量任务WCET
void measureTaskWcet() {
    WcetMeasurement::begin();
    criticalTask();
    uint32_t cycles = WcetMeasurement::end();
    printf("本次执行: %lu 周期\n", cycles);
    printf("最坏情况: %lu 周期 (%lu us)\n",
           WcetMeasurement::getMaxCycles(),
           WcetMeasurement::getMaxMicroseconds());
}

// WCET标注（用于静态分析工具）
// 使用注解标记循环边界
void annotatedFunction() {
    // @LOOP_BOUND max=100
    for (int i = 0; i < 100; ++i) {
        // @WCET 50 cycles
        processItem(i);
    }
}
```

### 1.2 时序分析

```cpp
// 任务时序分析

struct TaskTiming {
    const char* name;
    uint32_t period;        // 周期（ms）
    uint32_t wcet;          // 最坏执行时间（ms）
    uint32_t deadline;      // 截止时间（ms）
    uint32_t priority;      // 优先级
    uint32_t blockingTime;  // 最大阻塞时间（ms）
};

// 响应时间分析（RMA理论）
// R_i = C_i + B_i + Σ(ceil(R_i / T_j) * C_j)
// 其中 j 是所有优先级高于 i 的任务

uint32_t calculateResponseTime(const TaskTiming* tasks, size_t count, size_t taskIndex) {
    const TaskTiming& task = tasks[taskIndex];

    // 初始值：自身WCET + 最大阻塞时间
    uint32_t R = task.wcet + task.blockingTime;

    // 迭代计算直到收敛
    for (int iter = 0; iter < 100; ++iter) {
        uint32_t R_new = task.wcet + task.blockingTime;

        // 加上所有高优先级任务的干扰
        for (size_t j = 0; j < count; ++j) {
            if (tasks[j].priority > task.priority) {
                // 高优先级任务j在R时间内的干扰
                uint32_t numInvocations = (R + tasks[j].period - 1) / tasks[j].period;
                R_new += numInvocations * tasks[j].wcet;
            }
        }

        if (R_new == R) break;  // 收敛
        R = R_new;

        if (R > task.deadline) {
            // 不可调度！响应时间超过截止时间
            return UINT32_MAX;
        }
    }

    return R;
}

// 分析所有任务的可调度性
void analyzeSchedulability() {
    TaskTiming tasks[] = {
        {"SensorRead",  10, 2, 10, 3, 0},   // 10ms周期，2ms WCET
        {"DataProcess", 20, 3, 20, 2, 1},   // 20ms周期，3ms WCET
        {"Display",     50, 5, 50, 1, 2},   // 50ms周期，5ms WCET
    };

    printf("=== 时序分析结果 ===\n");

    for (size_t i = 0; i < 3; ++i) {
        uint32_t R = calculateResponseTime(tasks, 3, i);
        bool schedulable = R <= tasks[i].deadline;

        printf("任务: %-12s 周期:%3ums WCET:%2ums "
               "截止时间:%3ums 响应时间:%3ums %s\n",
               tasks[i].name, tasks[i].period, tasks[i].wcet,
               tasks[i].deadline, R,
               schedulable ? "✓可调度" : "✗不可调度");
    }

    // CPU利用率
    float utilization = 0.0f;
    for (const auto& t : tasks) {
        utilization += static_cast<float>(t.wcet) / static_cast<float>(t.period);
    }
    printf("CPU利用率: %.1f%%\n", utilization * 100.0f);
}
```

## 2. 速率单调调度

### 2.1 RMS理论与实践

```cpp
// 速率单调调度（Rate Monotonic Scheduling）

// RMS定理：
// 1. 固定优先级调度中，RMS是最优的
// 2. 优先级分配：周期越短，优先级越高
// 3. 可调度充分条件：U ≤ n(2^(1/n) - 1)

class RmsAnalyzer {
public:
    struct Task {
        const char* name;
        uint32_t period;    // 周期（ms）
        uint32_t wcet;      // 最坏执行时间（ms）
    };

    // 检查RMS可调度性
    static bool isSchedulable(const Task* tasks, size_t count) {
        // 方法1：Liu & Layland充分条件
        float utilization = totalUtilization(tasks, count);
        float rmsBound = count * (std::pow(2.0f, 1.0f / count) - 1.0f);

        printf("CPU利用率: %.2f%%\n", utilization * 100);
        printf("RMS上限: %.2f%%\n", rmsBound * 100);

        if (utilization <= rmsBound) {
            printf("满足Liu & Layland条件 → 一定可调度\n");
            return true;
        }

        // 方法2：响应时间分析（精确判断）
        printf("不满足充分条件，进行响应时间分析...\n");
        return checkResponseTime(tasks, count);
    }

private:
    static float totalUtilization(const Task* tasks, size_t count) {
        float U = 0.0f;
        for (size_t i = 0; i < count; ++i) {
            U += static_cast<float>(tasks[i].wcet) /
                 static_cast<float>(tasks[i].period);
        }
        return U;
    }

    static bool checkResponseTime(const Task* tasks, size_t count) {
        // 按周期排序（短周期 = 高优先级）
        // 假设已排序

        bool allSchedulable = true;
        for (size_t i = 0; i < count; ++i) {
            uint32_t R = tasks[i].wcet;  // 初始值

            for (int iter = 0; iter < 1000; ++iter) {
                uint32_t R_new = tasks[i].wcet;

                // 高优先级任务的干扰
                for (size_t j = 0; j < i; ++j) {
                    uint32_t invocations = (R + tasks[j].period - 1) / tasks[j].period;
                    R_new += invocations * tasks[j].wcet;
                }

                if (R_new == R) break;
                R = R_new;

                if (R > tasks[i].period) {
                    printf("任务 %s 不可调度: R=%u > T=%u\n",
                           tasks[i].name, R, tasks[i].period);
                    allSchedulable = false;
                    break;
                }
            }

            if (R <= tasks[i].period) {
                printf("任务 %s 可调度: R=%u <= T=%u\n",
                       tasks[i].name, R, tasks[i].period);
            }
        }

        return allSchedulable;
    }
};

// 示例分析
void rmsAnalysisExample() {
    RmsAnalyzer::Task tasks[] = {
        {"MotorControl", 5,  1},    // 5ms周期，1ms执行
        {"SensorRead",   10, 2},    // 10ms周期，2ms执行
        {"Communication",20, 3},    // 20ms周期，3ms执行
        {"Logging",      100, 5},   // 100ms周期，5ms执行
    };

    // 利用率 = 1/5 + 2/10 + 3/20 + 5/100 = 0.2 + 0.2 + 0.15 + 0.05 = 0.6
    // RMS上限(4任务) = 4*(2^0.25-1) ≈ 0.757

    bool schedulable = RmsAnalyzer::isSchedulable(tasks, 4);
    printf("系统可调度性: %s\n", schedulable ? "是" : "否");
}
```

### 2.2 EDF调度

```cpp
// 最早截止时间优先（Earliest Deadline First, EDF）

// EDF特点：
// - 动态优先级：截止时间最近的任务优先
// - 理论上最优：U ≤ 1 即可调度
// - 实现复杂：需要动态排序

class EdfScheduler {
public:
    struct Task {
        const char* name;
        uint32_t period;
        uint32_t wcet;
        uint32_t nextDeadline;  // 下一个截止时间
        uint32_t remaining;     // 剩余执行时间
    };

    void addTask(const Task& task) {
        tasks_.push_back(task);
        // 初始化截止时间
        tasks_.back().nextDeadline = task.period;
        tasks_.back().remaining = task.wcet;
    }

    // 每个时钟滴答调用
    void tick() {
        // 找到截止时间最早的任务
        Task* earliest = nullptr;
        uint32_t minDeadline = UINT32_MAX;

        for (auto& task : tasks_) {
            if (task.remaining > 0 && task.nextDeadline < minDeadline) {
                minDeadline = task.nextDeadline;
                earliest = &task;
            }
        }

        // 执行一个时间单位
        if (earliest) {
            --earliest->remaining;
        }

        // 更新截止时间
        for (auto& task : tasks_) {
            if (currentTime_ == task.nextDeadline - task.period) {
                // 新周期开始
                if (task.remaining > 0) {
                    // 错过了截止时间！
                    handleDeadlineMiss(task);
                }
                task.nextDeadline += task.period;
                task.remaining = task.wcet;
            }
        }

        ++currentTime_;
    }

    // EDF可调度性检查
    static bool isSchedulable(const Task* tasks, size_t count) {
        float utilization = 0.0f;
        for (size_t i = 0; i < count; ++i) {
            utilization += static_cast<float>(tasks[i].wcet) /
                          static_cast<float>(tasks[i].period);
        }
        return utilization <= 1.0f;  // EDF条件：U ≤ 1
    }

private:
    StaticVector<Task, 16> tasks_;
    uint32_t currentTime_ = 0;

    void handleDeadlineMiss(const Task& task) {
        printf("截止时间错过! 任务: %s\n", task.name);
    }
};
```

## 3. 看门狗与容错

### 3.1 看门狗定时器

```cpp
// 看门狗定时器（Watchdog Timer, WDT）

// 独立看门狗（IWDG）- 使用独立LSI时钟，不受主时钟影响
class IndependentWatchdog {
public:
    static void init(uint32_t timeoutMs) {
        // 使能IWDG寄存器访问
        IWDG->KR = 0x5555;

        // 设置预分频器
        // LSI时钟约32KHz
        uint32_t prescaler = 4;  // 分频后约8KHz
        IWDG->PR = prescaler - 4;

        // 设置重载值
        uint32_t reload = (32000 / prescaler) * timeoutMs / 1000;
        if (reload > 0xFFF) reload = 0xFFF;
        IWDG->RLR = reload;

        // 启动看门狗
        IWDG->KR = 0xCCCC;
    }

    // 喂狗（重置计数器）
    static void feed() {
        IWDG->KR = 0xAAAA;
    }
};

// 窗口看门狗（WWDG）- 必须在窗口内喂狗
class WindowWatchdog {
public:
    static void init(uint32_t window, uint32_t counter) {
        // 使能WWDG时钟
        RCC->APB1ENR |= RCC_APB1ENR_WWDGEN;

        // 设置计数器和窗口值
        WWDG->CFR = (window & 0x7F) | WWDG_CFR_WDGTB_0;  // 分频
        WWDG->CR = WWDG_CR_WDGA | (counter & 0x7F);       // 使能并设置计数器
    }

    static void feed(uint8_t counter) {
        WWDG->CR = (counter & 0x7F);
    }
};

// 多任务看门狗管理
class WatchdogManager {
public:
    struct TaskWatchdog {
        TaskHandle_t taskHandle;
        uint32_t timeout;       // 超时时间（ms）
        uint32_t lastFeed;      // 上次喂狗时间
        bool enabled;
    };

    static void init() {
        // 初始化独立看门狗，超时2秒
        IndependentWatchdog::init(2000);

        // 创建监控任务
        xTaskCreate(monitorTask, "WDT_Mon", 128, nullptr, 7, nullptr);
    }

    static void registerTask(TaskHandle_t handle, uint32_t timeoutMs) {
        ScopedCriticalSection cs;
        if (taskCount_ < MAX_TASKS) {
            tasks_[taskCount_++] = {handle, timeoutMs, 0, true};
        }
    }

    static void feedTask(TaskHandle_t handle) {
        ScopedCriticalSection cs;
        for (uint32_t i = 0; i < taskCount_; ++i) {
            if (tasks_[i].taskHandle == handle) {
                tasks_[i].lastFeed = xTaskGetTickCount();
                break;
            }
        }
    }

private:
    static constexpr uint32_t MAX_TASKS = 8;
    static TaskWatchdog tasks_[MAX_TASKS];
    static uint32_t taskCount_;

    static void monitorTask(void* pvParameters) {
        while (1) {
            bool allOk = true;
            uint32_t now = xTaskGetTickCount();

            {
                ScopedCriticalSection cs;
                for (uint32_t i = 0; i < taskCount_; ++i) {
                    if (!tasks_[i].enabled) continue;

                    uint32_t elapsed = (now - tasks_[i].lastFeed) * portTICK_PERIOD_MS;
                    if (elapsed > tasks_[i].timeout) {
                        // 任务超时
                        printf("看门狗超时! 任务: %s\n",
                               pcTaskGetName(tasks_[i].taskHandle));
                        allOk = false;
                    }
                }
            }

            if (allOk) {
                IndependentWatchdog::feed();
            }
            // 如果有任务超时不喂狗，系统将被看门狗复位

            vTaskDelay(pdMS_TO_TICKS(100));
        }
    }
};
```

### 3.2 容错设计模式

```cpp
// 容错设计模式

// 1. 冗余计算模式
template<typename T, size_t N>
class RedundantComputation {
public:
    T compute(std::function<T()> func) {
        T results[N];

        // 执行N次计算
        for (size_t i = 0; i < N; ++i) {
            results[i] = func();
        }

        // 多数表决
        return majorityVote(results);
    }

private:
    T majorityVote(const T results[N]) {
        // 找出出现次数最多的结果
        size_t maxCount = 0;
        T majority = results[0];

        for (size_t i = 0; i < N; ++i) {
            size_t count = 0;
            for (size_t j = 0; j < N; ++j) {
                if (results[j] == results[i]) ++count;
            }
            if (count > maxCount) {
                maxCount = count;
                majority = results[i];
            }
        }

        return majority;
    }
};

// 2. 降级模式
class DegradableSystem {
public:
    enum class Level : uint8_t {
        Full,          // 完全功能
        Reduced,       // 降级功能
        Minimal,       // 最小功能
        Safe           // 安全模式
    };

    void checkAndDegrade() {
        if (memoryUsage() > 90) {
            currentLevel_ = Level::Reduced;
            disableNonEssentialFeatures();
        }

        if (cpuUsage() > 95) {
            currentLevel_ = Level::Minimal;
            disableAllButCritical();
        }

        if (hasCriticalError()) {
            currentLevel_ = Level::Safe;
            enterSafeMode();
        }
    }

    Level getCurrentLevel() const { return currentLevel_; }

private:
    Level currentLevel_ = Level::Full;

    uint32_t memoryUsage();
    uint32_t cpuUsage();
    bool hasCriticalError();
    void disableNonEssentialFeatures();
    void disableAllButCritical();
    void enterSafeMode();
};

// 3. 心跳检测模式
class HeartbeatMonitor {
public:
    struct Node {
        uint32_t id;
        uint32_t timeout;     // 超时时间
        uint32_t lastBeat;    // 上次心跳时间
        bool alive;
    };

    void registerNode(uint32_t id, uint32_t timeoutMs) {
        nodes_.push_back({id, timeoutMs, 0, true});
    }

    void receiveHeartbeat(uint32_t id) {
        for (auto& node : nodes_) {
            if (node.id == id) {
                node.lastBeat = getCurrentTime();
                node.alive = true;
                break;
            }
        }
    }

    void check() {
        uint32_t now = getCurrentTime();
        for (auto& node : nodes_) {
            if (now - node.lastBeat > node.timeout) {
                if (node.alive) {
                    node.alive = false;
                    onNodeFailure(node.id);
                }
            }
        }
    }

private:
    StaticVector<Node, 16> nodes_;

    uint32_t getCurrentTime();
    void onNodeFailure(uint32_t id);
};

// 4. 安全状态机
class SafetyStateMachine {
public:
    enum class State : uint8_t {
        Init,
        Normal,
        Warning,
        Error,
        SafeShutdown
    };

    void transition(State newState) {
        // 只允许合法的状态转换
        if (isValidTransition(currentState_, newState)) {
            currentState_ = newState;
            onStateEntry(newState);
        } else {
            // 非法转换 → 进入安全状态
            currentState_ = State::SafeShutdown;
            onStateEntry(State::SafeShutdown);
        }
    }

    State getCurrentState() const { return currentState_; }

private:
    State currentState_ = State::Init;

    bool isValidTransition(State from, State to) {
        // 定义合法的状态转换
        switch (from) {
            case State::Init:
                return to == State::Normal || to == State::SafeShutdown;
            case State::Normal:
                return to == State::Warning || to == State::SafeShutdown;
            case State::Warning:
                return to == State::Normal || to == State::Error || to == State::SafeShutdown;
            case State::Error:

                break;
            case State::Error:
                disableNonCritical();
                break;
            case State::SafeShutdown:
                performSafeShutdown();
                break;
            default:
                break;
        }
    }
```

```cpp
        // 多次读取取中值（防止单点故障）
        float readings[3];
        for (int i = 0; i < 3; ++i) {
            auto result = readRawTemperature();
            if (!result) return Expected<float, Error>(result.error());
            readings[i] = result.value();

        float median = medianOf3(readings[0], readings[1], readings[2]);

        // 范围检查
        if (median < -50.0f || median > 150.0f) {
            return Expected<float, Error>(Error::SensorFault);

        if (std::abs(median - lastValidReading_) > MAX_RATE_OF_CHANGE) {
            return Expected<float, Error>(Error::SensorFault);
        }

        lastValidReading_ = median;
        return Expected<float, Error>(median);

        data_ = value;
        checksum_ = computeChecksum();
    }

            // 数据损坏！
```

```cpp
        if (taskCount_ < MAX_MONITORED_TASKS) {
            checkpoints_[taskCount_] = {
                taskId, maxIntervalMs, getCurrentTime(), false

        for (uint32_t i = 0; i < taskCount_; ++i) {

                checkpoints_[i].checkedIn = true;

class ProtectedDataBlock {
public:
    void write(const void* data, size_t len) {
        ScopedCriticalSection cs;
        memcpy(data_, data, len);
        crc_ = calculateCrc32(data_, len);
    }

    bool read(void* data, size_t len) {
        ScopedCriticalSection cs;
        uint32_t expectedCrc = calculateCrc32(data_, len);
        if (expectedCrc != crc_) {
            // 数据损坏！
            return false;
        }
        memcpy(data, data_, len);
        return true;
    }

private:
    uint8_t data_[256];
    uint32_t crc_;

    static uint32_t calculateCrc32(const uint8_t* data, size_t len) {
        uint32_t crc = 0xFFFFFFFF;
        for (size_t i = 0; i < len; ++i) {
            crc ^= data[i];
            for (int j = 0; j < 8; ++j) {
                if (crc & 1) crc = (crc >> 1) ^ 0xEDB88320;
                else crc >>= 1;
            }
        }
        return ~crc;
    }
};

class CertifiedStateMachine {
public:
    enum class State : uint8_t {
        PowerOn,
        SelfTest,
        Normal,
        Degraded,
        SafeState,
        FatalError
    };

    // 状态转换表（编译期确定，不可修改）
    struct Transition {
        State from;
        State to;
        bool (*guard)();          // 转换条件
        void (*action)();         // 转换动作
    };

    static constexpr Transition transitions[] = {
        {State::PowerOn,  State::SelfTest,  nullptr,        startSelfTest},
        {State::SelfTest, State::Normal,    selfTestPassed, enableAllSystems},
        {State::SelfTest, State::FatalError,selfTestFailed, enterFatalError},
        {State::Normal,   State::Degraded,  degradationDetected, reduceFunctionality},
        {State::Normal,   State::SafeState, safetyViolation, enterSafeState},
        {State::Degraded, State::Normal,    systemRecovered, enableAllSystems},
        {State::Degraded, State::SafeState, safetyViolation, enterSafeState},
        {State::SafeState,State::FatalError, recoveryFailed, enterFatalError},
    };

    void processEvent() {
        for (const auto& t : transitions) {
            if (currentState_ == t.from) {
                if (t.guard == nullptr || t.guard()) {
                    currentState_ = t.to;
                    if (t.action) t.action();
                    return;
                }
            }
        }
        // 未找到合法转换 → 安全状态
        currentState_ = State::SafeState;
        enterSafeState();
    }
```

```cpp
// - 任务切换时机
// - 优先级抢占

// 技巧2：运行时WCET监控
// 持续监控关键任务的执行时间
class WcetMonitor {
public:
    static void begin(uint32_t taskId) {
        startTimes_[taskId] = DWT->CYCCNT;
    }

    static void end(uint32_t taskId) {
        uint32_t elapsed = DWT->CYCCNT - startTimes_[taskId];
        if (elapsed > maxCycles_[taskId]) {
            maxCycles_[taskId] = elapsed;
        }
        // 如果超过预设阈值，记录违规
        if (elapsed > thresholds_[taskId]) {
            logWcetViolation(taskId, elapsed);
        }
    }

private:
    static uint32_t startTimes_[MAX_TASKS];
    static uint32_t maxCycles_[MAX_TASKS];
    static uint32_t thresholds_[MAX_TASKS];
    static constexpr uint32_t MAX_TASKS = 8;
    static void logWcetViolation(uint32_t taskId, uint32_t cycles);
};

// 技巧3：死锁检测
// 使用超时机制检测死锁
bool tryAcquireMultipleMutexes(SemaphoreHandle_t* mutexes, size_t count,
                                TickType_t timeout) {
    TickType_t startTime = xTaskGetTickCount();

    for (size_t i = 0; i < count; ++i) {
        TickType_t remaining = timeout - (xTaskGetTickCount() - startTime);
        if (remaining <= 0) {
            // 超时，释放已获取的互斥量
            for (size_t j = 0; j < i; ++j) {
                xSemaphoreGive(mutexes[j]);
            }
            return false;  // 可能是死锁
        }

        if (xSemaphoreTake(mutexes[i], remaining) != pdTRUE) {
            // 获取失败，释放已获取的互斥量
            for (size_t j = 0; j < i; ++j) {
                xSemaphoreGive(mutexes[j]);
            }
            return false;
        }
    }
    return true;
}
```

```cpp
// 实践1：分层容错架构
// 在不同层级实现不同的容错机制

// 硬件层：看门狗、ECC内存、电源监控
// 驱动层：超时检测、重试机制、错误计数
// 应用层：数据校验、降级模式、安全状态

// 驱动层容错：通信重试机制
template<typename Driver>
class RetryWrapper {
public:
    static constexpr uint32_t MAX_RETRIES = 3;
    static constexpr uint32_t RETRY_DELAY_MS = 10;

    static Expected<uint8_t, Error> read(uint8_t addr) {
        uint32_t attempts = 0;
        Error lastError = Error::Ok;

        while (attempts < MAX_RETRIES) {
            auto result = Driver::read(addr);
            if (result.hasValue()) {
                // 读取成功，重置错误计数
                consecutiveErrors_ = 0;
                return result;
            }

            lastError = result.error();
            ++attempts;
            ++consecutiveErrors_;

            if (consecutiveErrors_ > ERROR_THRESHOLD) {
                // 连续错误过多，标记设备为故障
                markDeviceFaulty();
                break;
            }

            delayMs(RETRY_DELAY_MS);
        }

        return Expected<uint8_t, Error>(lastError);
    }

private:
    static constexpr uint32_t ERROR_THRESHOLD = 10;
    static uint32_t consecutiveErrors_;
    static void markDeviceFaulty();
};

// 实践2：安全关键数据的双备份存储
// 将关键配置数据存储在Flash的两个不同区域
// 读取时比较两份数据，不一致则使用默认值
class DualBackupConfig {
public:
    struct Config {
        uint32_t magic;       // 魔数验证
        uint32_t version;     // 版本号
        uint8_t  data[240];   // 配置数据
        uint32_t crc;         // CRC校验
    };

    static bool load(Config& config) {
        Config configA, configB;

        // 从Flash区域A读取
        memcpy(&configA, reinterpret_cast<const void*>(FLASH_AREA_A), sizeof(Config));
        // 从Flash区域B读取
        memcpy(&configB, reinterpret_cast<const void*>(FLASH_AREA_B), sizeof(Config));

        bool aValid = validateConfig(configA);
        bool bValid = validateConfig(configB);

        if (aValid && bValid) {
            // 两份都有效，使用较新的版本
            config = (configA.version >= configB.version) ? configA : configB;
            return true;
        } else if (aValid) {
            config = configA;
            // 修复区域B
            saveToFlash(FLASH_AREA_B, configA);
            return true;
        } else if (bValid) {
            config = configB;
            // 修复区域A
            saveToFlash(FLASH_AREA_A, configB);
            return true;
        }

        // 两份都无效，使用默认配置
        loadDefaultConfig(config);
        return false;
    }

    static bool save(const Config& config) {
        // 交替写入两个区域
        Config mutableConfig = config;
        mutableConfig.magic = CONFIG_MAGIC;
        mutableConfig.version = nextVersion();
        mutableConfig.crc = calculateCrc(&mutableConfig, sizeof(Config) - 4);

        bool aOk = saveToFlash(FLASH_AREA_A, mutableConfig);
        bool bOk = saveToFlash(FLASH_AREA_B, mutableConfig);

        return aOk || bOk;
    }

private:
    static constexpr uint32_t FLASH_AREA_A = 0x08060000;
    static constexpr uint32_t FLASH_AREA_B = 0x08060800;
    static constexpr uint32_t CONFIG_MAGIC = 0xCAFEBABE;

    static bool validateConfig(const Config& cfg) {
        if (cfg.magic != CONFIG_MAGIC) return false;
        uint32_t expectedCrc = calculateCrc(&cfg, sizeof(Config) - 4);
        return cfg.crc == expectedCrc;
    }

    static uint32_t calculateCrc(const void* data, size_t len);
    static uint32_t nextVersion();
    static bool saveToFlash(uint32_t address, const Config& cfg);
    static void loadDefaultConfig(Config& cfg);
};
```

## 8. 本章小结

本章讲解了实时系统设计的核心知识，要点如下：

| 主题 | 核心要点 |
|------|---------|
| 实时性分析 | WCET测量、响应时间分析、抖动分析 |
| 速率单调调度 | RMS理论、EDF调度、可调度性分析 |
| 看门狗与容错 | IWDG/WWDG、冗余计算、降级模式、安全状态机 |
| 安全关键系统 | SIL等级、DO-178C、安全编码、数据完整性 |
| 形式化验证 | 模型检验、定理证明、抽象解释、Frama-C |
| 安全认证实践 | IEC 61508技术措施表、安全看门狗、CRC数据保护、认证状态机 |
| DO-178C认证 | 计划/开发/验证/完成四阶段、文档体系、认证周期与成本 |
| 常见陷阱 | WCET估计不足、优先级分配不当、看门狗过于简单、安全状态设计不当 |
| 调试技巧 | 逻辑分析仪观察调度、运行时WCET监控、超时死锁检测 |

**关键理解**：

1. **WCET是实时性的基础**：必须知道最坏执行时间才能保证实时性
2. **RMS是最优固定优先级调度**：周期越短优先级越高
3. **看门狗是最后防线**：软件出错时硬件看门狗可以复位系统
4. **安全标准是法律要求**：航空、汽车、医疗等领域必须遵守相应标准
5. **形式化验证提供数学保证**：比测试更可靠，但成本也更高
6. **安全认证需要系统化方法**：从计划到验证，每个阶段都有严格要求
7. **条件式看门狗比简单喂狗更可靠**：检查系统健康状态后才喂狗
