> 💡 **通俗理解 - 什么是密码学？**

想象一下：

- **对称加密** 就像"同一把钥匙锁箱子和开箱子"，加密解密用同一个密钥
- **非对称加密** 就像"信箱"，任何人可以投信（公钥加密），只有主人能取信（私钥解密）
- **哈希** 就像"指纹"，每个人（数据）都有唯一的指纹，但不能从指纹还原人
- **TLS** 就像"密封的快递"，保证数据在传输中不被偷看和篡改

密码学是信息安全的数学基础，但**错误使用密码学比不使用更危险**。

> 🔬 **抽象理解 - 密码学体系**：
>
> - **对称加密**：AES，快速但密钥分发困难
> - **非对称加密**：RSA/ECC，解决密钥分发但速度慢
> - **哈希与MAC**：SHA-256/HMAC，保证完整性和认证
> - **TLS/SSL**：综合运用以上技术实现安全通信

## 1. 对称加密

### 1.1 AES加密

```cpp
// AES（Advanced Encryption Standard）- 最广泛使用的对称加密算法

#include <openssl/evp.h>
#include <openssl/rand.h>
#include <vector>
#include <cstdint>

class AesCipher {
public:
    enum class Mode {
        CBC,    // 密码分组链接模式
        CTR,    // 计数器模式
        GCM     // 伽罗瓦/计数器模式（推荐）
    };

    // AES-256-GCM加密（推荐模式）
    static bool encrypt(const std::vector<uint8_t>& plaintext,
                        const std::vector<uint8_t>& key,
                        const std::vector<uint8_t>& aad,
                        std::vector<uint8_t>& ciphertext,
                        std::vector<uint8_t>& iv,
                        std::vector<uint8_t>& tag) {
        // 参数验证
        if (key.size() != 32) return false;  // AES-256需要32字节密钥

        // 生成随机IV（初始化向量）
        iv.resize(12);  // GCM推荐12字节IV
        if (RAND_bytes(iv.data(), iv.size()) != 1) return false;

        // 创建加密上下文
        EVP_CIPHER_CTX* ctx = EVP_CIPHER_CTX_new();
        if (!ctx) return false;

        bool success = false;
        do {
            // 初始化加密操作
            if (EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), nullptr,
                                    nullptr, nullptr) != 1) break;

            // 设置IV长度
            if (EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN,
                                     iv.size(), nullptr) != 1) break;

            // 设置密钥和IV
            if (EVP_EncryptInit_ex(ctx, nullptr, nullptr,
                                    key.data(), iv.data()) != 1) break;

            // 提供AAD（附加认证数据，不加密但认证）
            int len;
            if (!aad.empty()) {
                if (EVP_EncryptUpdate(ctx, nullptr, &len,
                                       aad.data(), aad.size()) != 1) break;
            }

            // 加密数据
            ciphertext.resize(plaintext.size());
            if (EVP_EncryptUpdate(ctx, ciphertext.data(), &len,
                                   plaintext.data(), plaintext.size()) != 1) break;
            ciphertext.resize(len);

            // 完成加密
            if (EVP_EncryptFinal_ex(ctx, ciphertext.data() + len, &len) != 1) break;

            // 获取认证标签
            tag.resize(16);
            if (EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_GET_TAG,
                                     tag.size(), tag.data()) != 1) break;

            success = true;
        } while (0);

        EVP_CIPHER_CTX_free(ctx);
        return success;
    }

    // AES-256-GCM解密
    static bool decrypt(const std::vector<uint8_t>& ciphertext,
                        const std::vector<uint8_t>& key,
                        const std::vector<uint8_t>& iv,
                        const std::vector<uint8_t>& aad,
                        const std::vector<uint8_t>& tag,
                        std::vector<uint8_t>& plaintext) {
        if (key.size() != 32 || iv.size() != 12) return false;

        EVP_CIPHER_CTX* ctx = EVP_CIPHER_CTX_new();
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_DecryptInit_ex(ctx, EVP_aes_256_gcm(), nullptr,
                                    nullptr, nullptr) != 1) break;
            if (EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_IVLEN,
                                     iv.size(), nullptr) != 1) break;
            if (EVP_DecryptInit_ex(ctx, nullptr, nullptr,
                                    key.data(), iv.data()) != 1) break;

            // 提供AAD
            int len;
            if (!aad.empty()) {
                if (EVP_DecryptUpdate(ctx, nullptr, &len,
                                       aad.data(), aad.size()) != 1) break;
            }

            // 解密数据
            plaintext.resize(ciphertext.size());
            if (EVP_DecryptUpdate(ctx, plaintext.data(), &len,
                                   ciphertext.data(), ciphertext.size()) != 1) break;
            int plaintextLen = len;

            // 设置认证标签
            if (EVP_CIPHER_CTX_ctrl(ctx, EVP_CTRL_GCM_SET_TAG,
                                     tag.size(), const_cast<uint8_t*>(tag.data())) != 1) break;

            // 验证标签并完成解密
            if (EVP_DecryptFinal_ex(ctx, plaintext.data() + plaintextLen, &len) != 1) {
                // 认证失败！数据被篡改或密钥错误
                break;
            }
            plaintextLen += len;
            plaintext.resize(plaintextLen);

            success = true;
        } while (0);

        EVP_CIPHER_CTX_free(ctx);
        return success;
    }
};

// 使用示例
void aesExample() {
    // 生成随机密钥
    std::vector<uint8_t> key(32);
    RAND_bytes(key.data(), key.size());

    std::vector<uint8_t> plaintext = {'H', 'e', 'l', 'l', 'o', '!', 'W', 'o', 'r', 'l', 'd'};
    std::vector<uint8_t> aad = {'m', 'y', 'a', 'p', 'p'};  // 附加认证数据

    std::vector<uint8_t> ciphertext, iv, tag;
    if (AesCipher::encrypt(plaintext, key, aad, ciphertext, iv, tag)) {
        printf("加密成功，密文长度: %zu\n", ciphertext.size());

        std::vector<uint8_t> decrypted;
        if (AesCipher::decrypt(ciphertext, key, iv, aad, tag, decrypted)) {
            printf("解密成功: %.*s\n", (int)decrypted.size(), decrypted.data());
        } else {
            printf("解密失败（认证失败）\n");
        }
    }
}
```

### 1.2 密钥派生

```cpp
// 密钥派生函数（KDF）- 从密码生成安全的加密密钥

#include <openssl/evp.h>
#include <openssl/rand.h>

class KeyDerivation {
public:
    // PBKDF2 - 从密码派生密钥
    static bool pbkdf2(const std::string& password,
                       const std::vector<uint8_t>& salt,
                       int iterations,
                       std::vector<uint8_t>& derivedKey,
                       size_t keyLength = 32) {
        derivedKey.resize(keyLength);

        int result = PKCS5_PBKDF2_HMAC(
            password.c_str(), password.size(),
            salt.data(), salt.size(),
            iterations,
            EVP_sha256(),
            keyLength,
            derivedKey.data()
        );

        return result == 1;
    }

    // HKDF - 基于HMAC的密钥派生
    static bool hkdf(const std::vector<uint8_t>& inputKey,
                     const std::vector<uint8_t>& salt,
                     const std::string& info,
                     std::vector<uint8_t>& derivedKey,
                     size_t keyLength = 32) {
        derivedKey.resize(keyLength);

        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new_id(EVP_PKEY_HKDF, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_derive_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_hkdf_md(ctx, EVP_sha256()) != 1) break;
            if (EVP_PKEY_CTX_set1_hkdf_salt(ctx, salt.data(), salt.size()) != 1) break;
            if (EVP_PKEY_CTX_set1_hkdf_key(ctx, inputKey.data(), inputKey.size()) != 1) break;
            if (EVP_PKEY_CTX_add1_hkdf_info(ctx,
                    reinterpret_cast<const unsigned char*>(info.c_str()),
                    info.size()) != 1) break;

            size_t outLen = keyLength;
            if (EVP_PKEY_derive(ctx, derivedKey.data(), &outLen) != 1) break;

            success = true;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }

    // 生成随机盐
    static std::vector<uint8_t> generateSalt(size_t length = 16) {
        std::vector<uint8_t> salt(length);
        RAND_bytes(salt.data(), salt.size());
        return salt;
    }
};
```

## 2. 非对称加密

### 2.0 密码学数学基础

```
对称加密的数学基础：

1. 有限域（Galois Field）
   - AES在GF(2^8)上进行运算
   - 加法 = XOR运算
   - 乘法 = 模不可约多项式的乘法
   - 不可约多项式: x^8 + x^4 + x^3 + x + 1 (0x11B)

2. AES的数学运算
   ┌──────────────┬──────────────────────────────────────┐
   │ 操作          │ 数学描述                              │
   ├──────────────┼──────────────────────────────────────┤
   │ SubBytes     │ 在GF(2^8)中求乘法逆元，再仿射变换     │
   │ ShiftRows    │ 行循环移位（置换）                     │
   │ MixColumns   │ GF(2^8)上的矩阵乘法                   │
   │ AddRoundKey  │ GF(2^8)上的加法（XOR）                │
   └──────────────┴──────────────────────────────────────┘

3. 分组密码工作模式
   ┌──────────┬──────────────────────────────────────┐
   │ 模式      │ 特点                                  │
   ├──────────┼──────────────────────────────────────┤
   │ ECB      │ 相同明文→相同密文，不安全               │
   │ CBC      │ 链式依赖，需要IV，串行加密              │
   │ CTR      │ 计数器模式，可并行，无填充              │
   │ GCM      │ CTR+认证，推荐模式                     │
   │ CFB      │ 类似流密码，可并行解密                  │
   │ OFB      │ 输出反馈模式，无错误传播                │
   └──────────┴──────────────────────────────────────┘

   ECB模式的问题（企鹅图）：
   相同的明文块产生相同的密文块
   ┌──────┐  ┌──────┐  ┌──────┐
   │ AAAA │→│ ████ │  │ AAAA │→│ ████ │  (相同!)
   └──────┘  └──────┘  └──────┘  └──────┘
   ┌──────┐  ┌──────┐  ┌──────┐  ┌──────┐
   │ BBBB │→│ ████ │  │ BBBB │→│ ████ │  (相同!)
   └──────┘  └──────┘  └──────┘  └──────┘

   CBC模式：
   C₀ = IV
   Cᵢ = E_K(Pᵢ ⊕ Cᵢ₋₁)
   解密: Pᵢ = D_K(Cᵢ) ⊕ Cᵢ₋₁
   优点：相同明文块产生不同密文块
   缺点：加密必须串行，需要填充

   GCM模式（推荐）：
   加密：使用CTR模式加密数据
   认证：使用GHASH函数计算认证标签
   优点：可并行、无填充、同时加密+认证
```

```
非对称加密的数学基础：

1. RSA数学原理
   - 基于大整数分解难题
   - 密钥生成：
     1) 选择两个大素数 p, q
     2) n = p × q
     3) φ(n) = (p-1)(q-1)  （欧拉函数）
     4) 选择 e，满足 1 < e < φ(n) 且 gcd(e, φ(n)) = 1
     5) 计算 d，满足 e × d ≡ 1 (mod φ(n))
   - 公钥: (n, e)
   - 私钥: (n, d)
   - 加密: c = m^e mod n
   - 解密: m = c^d mod n
   - 正确性证明：m^(ed) ≡ m (mod n)  （欧拉定理）

2. ECC数学原理
   - 基于椭圆曲线离散对数问题(ECDLP)
   - 椭圆曲线方程: y² = x³ + ax + b (mod p)
   - 点加法：P + Q = R（几何方法：连线交曲线的第三点的负值）
   - 标量乘法：k × P = P + P + ... + P（k次）
   - 困难问题：已知P和k×P，求k（离散对数问题）
   - 密钥生成：
     1) 选择基点G
     2) 选择私钥 d（随机数）
     3) 计算公钥 Q = d × G
   - ECDH：双方交换公钥，计算共享密钥
     Alice: d_A × Q_B = d_A × d_B × G
     Bob:   d_B × Q_A = d_B × d_A × G
     结果相同 = 共享密钥

3. 安全等级对比
   ┌──────────────┬──────────────┬──────────────┐
   │ 对称密钥位数  │ RSA密钥位数   │ ECC密钥位数   │
   ├──────────────┼──────────────┼──────────────┤
   │ 80           │ 1024         │ 160          │
   │ 112          │ 2048         │ 224          │
   │ 128          │ 3072         │ 256          │
   │ 192          │ 7680         │ 384          │
   │ 256          │ 15360        │ 521          │
   └──────────────┴──────────────┴──────────────┘
   ECC的优势：更短的密钥，更快的运算，更少的带宽
```

### 2.1 RSA加密

```cpp
// RSA加密与签名

#include <openssl/rsa.h>
#include <openssl/pem.h>
#include <openssl/err.h>

class RsaCrypto {
public:
    // 生成RSA密钥对
    static EVP_PKEY* generateKeyPair(int bits = 2048) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new_id(EVP_PKEY_RSA, nullptr);
        if (!ctx) return nullptr;

        EVP_PKEY* pkey = nullptr;
        do {
            if (EVP_PKEY_keygen_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_keygen_bits(ctx, bits) != 1) break;
            if (EVP_PKEY_keygen(ctx, &pkey) != 1) break;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return pkey;
    }

    // RSA加密（使用OAEP填充）
    static bool encrypt(EVP_PKEY* publicKey,
                        const std::vector<uint8_t>& plaintext,
                        std::vector<uint8_t>& ciphertext) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(publicKey, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_encrypt_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_OAEP_PADDING) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_oaep_md(ctx, EVP_sha256()) != 1) break;

            // 确定输出大小
            size_t outLen;
            if (EVP_PKEY_encrypt(ctx, nullptr, &outLen,
                                  plaintext.data(), plaintext.size()) != 1) break;

            ciphertext.resize(outLen);
            if (EVP_PKEY_encrypt(ctx, ciphertext.data(), &outLen,
                                  plaintext.data(), plaintext.size()) != 1) break;
            ciphertext.resize(outLen);

            success = true;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }

    // RSA解密
    static bool decrypt(EVP_PKEY* privateKey,
                        const std::vector<uint8_t>& ciphertext,
                        std::vector<uint8_t>& plaintext) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(privateKey, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_decrypt_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_OAEP_PADDING) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_oaep_md(ctx, EVP_sha256()) != 1) break;

            size_t outLen;
            if (EVP_PKEY_decrypt(ctx, nullptr, &outLen,
                                  ciphertext.data(), ciphertext.size()) != 1) break;

            plaintext.resize(outLen);
            if (EVP_PKEY_decrypt(ctx, plaintext.data(), &outLen,
                                  ciphertext.data(), ciphertext.size()) != 1) break;
            plaintext.resize(outLen);

            success = true;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }

    // RSA签名（使用PSS填充）
    static bool sign(EVP_PKEY* privateKey,
                     const std::vector<uint8_t>& message,
                     std::vector<uint8_t>& signature) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(privateKey, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_sign_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PSS_PADDING) != 1) break;
            if (EVP_PKEY_CTX_set_signature_md(ctx, EVP_sha256()) != 1) break;

            // 先计算消息摘要
            unsigned char md[32];
            unsigned int mdLen;
            EVP_Digest(message.data(), message.size(), md, &mdLen, EVP_sha256(), nullptr);

            size_t sigLen;
            if (EVP_PKEY_sign(ctx, nullptr, &sigLen, md, mdLen) != 1) break;

            signature.resize(sigLen);
            if (EVP_PKEY_sign(ctx, signature.data(), &sigLen, md, mdLen) != 1) break;
            signature.resize(sigLen);

            success = true;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }

    // RSA验签
    static bool verify(EVP_PKEY* publicKey,
                       const std::vector<uint8_t>& message,
                       const std::vector<uint8_t>& signature) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(publicKey, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_verify_init(ctx) != 1) break;
            if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PSS_PADDING) != 1) break;
            if (EVP_PKEY_CTX_set_signature_md(ctx, EVP_sha256()) != 1) break;

            unsigned char md[32];
            unsigned int mdLen;
            EVP_Digest(message.data(), message.size(), md, &mdLen, EVP_sha256(), nullptr);

            if (EVP_PKEY_verify(ctx, signature.data(), signature.size(),
                                md, mdLen) == 1) {
                success = true;
            }
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }
};
```

### 2.2 ECC（椭圆曲线加密）

```cpp
// ECC椭圆曲线加密 - 更短的密钥，同等安全

class EccCrypto {
public:
    // 生成ECC密钥对
    static EVP_PKEY* generateKeyPair() {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new_id(EVP_PKEY_EC, nullptr);
        if (!ctx) return nullptr;

        EVP_PKEY* pkey = nullptr;
        do {
            if (EVP_PKEY_keygen_init(ctx) != 1) break;
            // 使用prime256v1曲线（等同于P-256）
            if (EVP_PKEY_CTX_set_ec_paramgen_curve_nid(ctx, NID_X9_62_prime256v1) != 1) break;
            if (EVP_PKEY_keygen(ctx, &pkey) != 1) break;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return pkey;
    }

    // ECDH密钥交换
    static bool deriveSharedSecret(EVP_PKEY* myKey,
                                    EVP_PKEY* peerKey,
                                    std::vector<uint8_t>& sharedSecret) {
        EVP_PKEY_CTX* ctx = EVP_PKEY_CTX_new(myKey, nullptr);
        if (!ctx) return false;

        bool success = false;
        do {
            if (EVP_PKEY_derive_init(ctx) != 1) break;
            if (EVP_PKEY_derive_set_peer(ctx, peerKey) != 1) break;

            size_t secretLen;
            if (EVP_PKEY_derive(ctx, nullptr, &secretLen) != 1) break;

            sharedSecret.resize(secretLen);
            if (EVP_PKEY_derive(ctx, sharedSecret.data(), &secretLen) != 1) break;
            sharedSecret.resize(secretLen);

            success = true;
        } while (0);

        EVP_PKEY_CTX_free(ctx);
        return success;
    }

    // ECDSA签名
    static bool sign(EVP_PKEY* privateKey,
                     const std::vector<uint8_t>& message,
                     std::vector<uint8_t>& signature) {
        EVP_MD_CTX* mdCtx = EVP_MD_CTX_new();
        if (!mdCtx) return false;

        bool success = false;
        do {
            if (EVP_DigestSignInit(mdCtx, nullptr, EVP_sha256(),
                                    nullptr, privateKey) != 1) break;
            if (EVP_DigestSignUpdate(mdCtx, message.data(), message.size()) != 1) break;

            size_t sigLen;
            if (EVP_DigestSignFinal(mdCtx, nullptr, &sigLen) != 1) break;

            signature.resize(sigLen);
            if (EVP_DigestSignFinal(mdCtx, signature.data(), &sigLen) != 1) break;
            signature.resize(sigLen);

            success = true;
        } while (0);

        EVP_MD_CTX_free(mdCtx);
        return success;
    }

    // ECDSA验签
    static bool verify(EVP_PKEY* publicKey,
                       const std::vector<uint8_t>& message,
                       const std::vector<uint8_t>& signature) {
        EVP_MD_CTX* mdCtx = EVP_MD_CTX_new();
        if (!mdCtx) return false;

        bool success = false;
        do {
            if (EVP_DigestVerifyInit(mdCtx, nullptr, EVP_sha256(),
                                      nullptr, publicKey) != 1) break;
            if (EVP_DigestVerifyUpdate(mdCtx, message.data(), message.size()) != 1) break;

            if (EVP_DigestVerifyFinal(mdCtx, signature.data(), signature.size()) == 1) {
                success = true;
            }
        } while (0);

        EVP_MD_CTX_free(mdCtx);
        return success;
    }
};

// 安全等级对比
// RSA-2048 ≈ ECC-224  (112位安全)
// RSA-3072 ≈ ECC-256  (128位安全)
// RSA-7680 ≈ ECC-384  (192位安全)
// RSA-15360 ≈ ECC-521 (256位安全)
```

```cpp
class HashFunction {
public:
    // SHA-256哈希
    static std::vector<uint8_t> sha256(const std::vector<uint8_t>& data) {
        std::vector<uint8_t> digest(32);
        unsigned int digestLen;

        EVP_Digest(data.data(), data.size(),
                   digest.data(), &digestLen,
                   EVP_sha256(), nullptr);

        return digest;
    }

    // SHA-512哈希
    static std::vector<uint8_t> sha512(const std::vector<uint8_t>& data) {
        std::vector<uint8_t> digest(64);
        unsigned int digestLen;

        EVP_Digest(data.data(), data.size(),
                   digest.data(), &digestLen,
                   EVP_sha512(), nullptr);

        return digest;
    }

    // 流式哈希（处理大文件）
    static std::vector<uint8_t> sha256File(const std::string& filePath) {
        EVP_MD_CTX* ctx = EVP_MD_CTX_new();
        EVP_DigestInit_ex(ctx, EVP_sha256(), nullptr);

        std::ifstream file(filePath, std::ios::binary);
        char buffer[8192];

        while (file.read(buffer, sizeof(buffer))) {
            EVP_DigestUpdate(ctx, buffer, file.gcount());
        }
        if (file.gcount() > 0) {
            EVP_DigestUpdate(ctx, buffer, file.gcount());
        }

        std::vector<uint8_t> digest(32);
        unsigned int digestLen;
        EVP_DigestFinal_ex(ctx, digest.data(), &digestLen);

        EVP_MD_CTX_free(ctx);
        return digest;
    }

    // 转换为十六进制字符串
    static std::string toHex(const std::vector<uint8_t>& data) {
        std::string hex;
        hex.reserve(data.size() * 2);
        for (uint8_t byte : data) {
            char buf[3];
            snprintf(buf, sizeof(buf), "%02x", byte);
            hex += buf;
        }
        return hex;
    }
};
```

```cpp
class Hmac {
public:
    // 计算HMAC-SHA256
    static std::vector<uint8_t> compute(const std::vector<uint8_t>& key,
                                         const std::vector<uint8_t>& message) {
        std::vector<uint8_t> mac(32);
        unsigned int macLen;

        HMAC(EVP_sha256(), key.data(), key.size(),
             message.data(), message.size(),
             mac.data(), &macLen);

        return mac;
    }

    // 验证HMAC（常量时间比较，防止时序攻击）
    static bool verify(const std::vector<uint8_t>& key,
                       const std::vector<uint8_t>& message,
                       const std::vector<uint8_t>& expectedMac) {
        auto computedMac = compute(key, message);

        if (computedMac.size() != expectedMac.size()) return false;

        // ❌ 不安全的比较（时序攻击）
        // return computedMac == expectedMac;

        // ✅ 常量时间比较
        return CRYPTO_memcmp(computedMac.data(), expectedMac.data(),
                             computedMac.size()) == 0;
    }
};
```

```cpp
// TLS客户端实现

#include <openssl/ssl.h>
#include <openssl/err.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <netdb.h>

class TlsClient {
public:
    bool connect(const std::string& host, uint16_t port) {
        // 1. 初始化OpenSSL
        SSL_library_init();
        SSL_load_error_strings();
        OpenSSL_add_all_algorithms();

        // 2. 创建SSL上下文
        ctx_ = SSL_CTX_new(TLS_client_method());
        if (!ctx_) return false;

        // 3. 配置安全参数
        if (!TlsConfig::configureContext(ctx_)) return false;

        // 4. 创建TCP连接
        int sock = createTcpConnection(host, port);
        if (sock < 0) return false;

        // 5. 创建SSL连接
        ssl_ = SSL_new(ctx_);
        if (!ssl_) return false;

        SSL_set_fd(ssl_, sock);

        // 设置SNI（Server Name Indication）
        SSL_set_tlsext_host_name(ssl_, host.c_str());

        // 6. TLS握手
        if (SSL_connect(ssl_) != 1) {
            ERR_print_errors_fp(stderr);
            return false;
        }

        // 7. 验证证书
        if (!verifyCertificate(host)) {
            printf("证书验证失败!\n");
            return false;
        }

        printf("TLS连接建立成功，协议: %s，密码套件: %s\n",
               SSL_get_version(ssl_),
               SSL_get_cipher_name(ssl_));

        return true;
    }

    bool send(const std::string& data) {
        int result = SSL_write(ssl_, data.c_str(), data.size());
        return result > 0;
    }

    std::string receive(size_t maxLen = 4096) {
        std::vector<char> buffer(maxLen);
        int len = SSL_read(ssl_, buffer.data(), buffer.size());
        if (len <= 0) return "";
        return std::string(buffer.data(), len);
    }

    void disconnect() {
        if (ssl_) {
            SSL_shutdown(ssl_);
            SSL_free(ssl_);
            ssl_ = nullptr;
        }
        if (ctx_) {
            SSL_CTX_free(ctx_);
            ctx_ = nullptr;
        }
    }

private:
    SSL_CTX* ctx_ = nullptr;
    SSL* ssl_ = nullptr;

    int createTcpConnection(const std::string& host, uint16_t port) {
        struct hostent* server = gethostbyname(host.c_str());
        if (!server) return -1;

        int sock = socket(AF_INET, SOCK_STREAM, 0);
        if (sock < 0) return -1;

        struct sockaddr_in addr{};
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        memcpy(&addr.sin_addr.s_addr, server->h_addr, server->h_length);

        if (::connect(sock, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
            close(sock);
            return -1;
        }

        return sock;
    }

    bool verifyCertificate(const std::string& expectedHost) {
        X509* cert = SSL_get_peer_certificate(ssl_);
        if (!cert) return false;

        // 验证主机名
        bool match = false;
        char* subject = X509_NAME_oneline(X509_get_subject_name(cert), nullptr, 0);

        // 简化的主机名验证（实际应使用X509_check_host）
        match = (strstr(subject, expectedHost.c_str()) != nullptr);

        OPENSSL_free(subject);
        X509_free(cert);

        return match;
    }
};
```

```cpp
// TLS服务端实现

class TlsServer {
public:
    bool init(const std::string& certFile, const std::string& keyFile, uint16_t port) {
        SSL_library_init();
        SSL_load_error_strings();

        ctx_ = SSL_CTX_new(TLS_server_method());
        if (!ctx_) return false;

        if (!TlsConfig::configureContext(ctx_)) return false;

        // 加载证书和私钥
        if (SSL_CTX_use_certificate_file(ctx_, certFile.c_str(), SSL_FILETYPE_PEM) != 1) {
            ERR_print_errors_fp(stderr);
            return false;
        }

        if (SSL_CTX_use_PrivateKey_file(ctx_, keyFile.c_str(), SSL_FILETYPE_PEM) != 1) {
            ERR_print_errors_fp(stderr);
            return false;
        }

        // 验证私钥与证书匹配
        if (SSL_CTX_check_private_key(ctx_) != 1) {
            printf("私钥与证书不匹配!\n");
            return false;
        }

        // 创建监听套接字
        listenSock_ = socket(AF_INET, SOCK_STREAM, 0);
        if (listenSock_ < 0) return false;

        int opt = 1;
        setsockopt(listenSock_, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

        struct sockaddr_in addr{};
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        addr.sin_addr.s_addr = INADDR_ANY;

        if (bind(listenSock_, (struct sockaddr*)&addr, sizeof(addr)) < 0) return false;
        if (listen(listenSock_, 5) < 0) return false;

        return true;
    }

    void run() {
        printf("TLS服务器启动，等待连接...\n");

        while (true) {
            struct sockaddr_in clientAddr{};
            socklen_t len = sizeof(clientAddr);
            int clientSock = accept(listenSock_, (struct sockaddr*)&clientAddr, &len);

            if (clientSock < 0) continue;

            // 为每个连接创建SSL对象
            SSL* ssl = SSL_new(ctx_);
            SSL_set_fd(ssl, clientSock);

            if (SSL_accept(ssl) == 1) {
                handleClient(ssl);
            }

            SSL_shutdown(ssl);
            SSL_free(ssl);
            close(clientSock);
        }
    }

private:
    SSL_CTX* ctx_ = nullptr;
    int listenSock_ = -1;

    void handleClient(SSL* ssl) {
        char buffer[4096];
        int bytes = SSL_read(ssl, buffer, sizeof(buffer) - 1);
        if (bytes > 0) {
            buffer[bytes] = '\0';
            printf("收到: %s\n", buffer);

            std::string response = "HTTP/1.1 200 OK\r\n"
                                   "Content-Type: text/plain\r\n"
                                   "Content-Length: 13\r\n"
                                   "\r\n"
                                   "Hello, TLS!\r\n";
            SSL_write(ssl, response.c_str(), response.size());
        }
    }
};
```

## 7. 密码学攻击与防御

### 7.1 常见密码学攻击

```cpp
// 1. 填充预言攻击（Padding Oracle Attack）
// 针对CBC模式的攻击，通过服务器对填充错误的响应来解密密文

// 攻击原理：
// CBC解密：Pᵢ = D_K(Cᵢ) ⊕ Cᵢ₋₁
// 服务器返回"填充错误"vs"MAC错误"的不同响应
// 攻击者可以逐字节推断中间值 Iᵢ = D_K(Cᵢ)
// 然后 Pᵢ = Iᵢ ⊕ Cᵢ₋₁

// 防御：
// - 使用AEAD模式（GCM）而非CBC+MAC
// - 先验证MAC再解密（Encrypt-then-MAC）
// - 统一错误信息（不区分填充错误和MAC错误）

// 2. BEAST攻击（CVE-2011-3389）
// 针对TLS 1.0的CBC模式攻击
// TLS 1.0使用前一个密文块作为下一个块的IV（可预测）
// 攻击者可以预测IV，选择明文攻击

// 防御：使用TLS 1.1+（使用显式IV）

// 3. Lucky 13攻击
// 针对TLS CBC模式实现的时序攻击
// 通过测量MAC验证的时间来推断密文内容

// 防御：使用常量时间的MAC验证

// 4. POODLE攻击（CVE-2014-3566）
// 针对SSLv3的攻击
// SSLv3的填充只检查最后一个字节
// 攻击者可以降级协议版本到SSLv3

// 防御：禁用SSLv3

// 5. ROBOT攻击（Return Of Bleichenbacher's Oracle Threat）
// 针对RSA PKCS#1 v1.5填充的攻击
// 通过服务器的错误响应来解密RSA密文

// 防御：使用RSA-OAEP填充替代PKCS#1 v1.5
```

### 7.2 时序攻击与防御

```cpp
// 时序攻击（Timing Attack）

// ❌ 不安全的比较（时序泄露）
bool insecureCompare(const std::vector<uint8_t>& a,
                     const std::vector<uint8_t>& b) {
    if (a.size() != b.size()) return false;
    for (size_t i = 0; i < a.size(); i++) {
        if (a[i] != b[i]) return false;  // 第一个不同的字节就返回
        // 攻击者可以通过响应时间推断前多少字节匹配
    }
    return true;
}

// ✅ 常量时间比较
bool constantTimeCompare(const uint8_t* a, const uint8_t* b, size_t len) {
    uint8_t result = 0;
    for (size_t i = 0; i < len; i++) {
        result |= a[i] ^ b[i];  // XOR所有差异，不提前返回
    }
    return result == 0;
}

// ✅ 使用OpenSSL的常量时间比较
bool safeCompare(const std::vector<uint8_t>& a,
                 const std::vector<uint8_t>& b) {
    if (a.size() != b.size()) return false;
    return CRYPTO_memcmp(a.data(), b.data(), a.size()) == 0;
}

// 时序攻击实战：HMAC签名验证
class SafeHmacVerification {
public:
    // ❌ 不安全：字符串比较不是常量时间
    static bool verifyInsecure(const std::string& received,
                               const std::string& computed) {
        return received == computed;  // std::string::operator== 不是常量时间
    }

    // ✅ 安全：使用常量时间比较
    static bool verifySecure(const std::string& received,
                             const std::string& computed) {
        if (received.size() != computed.size()) {
            // 仍然执行比较以保持常量时间
            // 但结果一定是false
            uint8_t dummy = 0;
            for (size_t i = 0; i < computed.size(); i++) {
                dummy |= computed[i] ^ (i < received.size() ? received[i] : 0);
            }
            (void)dummy;
            return false;
        }
        return CRYPTO_memcmp(received.data(), computed.data(),
                            received.size()) == 0;
    }
};
```

```cpp
// CVE-2016-0800 (DROWN) - SSLv2跨协议攻击
// 漏洞原理：服务器同时支持SSLv2和TLS时，
// 攻击者可以利用SSLv2的弱点解密TLS连接

// 攻击者可以通过SSLv2连接暴力破解会话密钥
// 然后利用Bleichenbacher攻击解密TLS流量

// 防御：完全禁用SSLv2

// CVE-2014-3566 (POODLE) - SSLv3填充攻击
// 漏洞原理：SSLv3的CBC填充只验证最后一个字节
// 攻击者可以修改密文块，使填充验证通过但明文改变

// 防御：禁用SSLv3，使用TLS 1.2+

// CVE-2017-7657 - OpenSSL拒绝服务
// 漏洞原理：特殊的ALPN值导致空指针解引用
// 教训：所有输入都必须验证，即使是TLS握手数据

// CVE-2022-0778 - OpenSSL无限循环DoS
// 漏洞原理：解析特制的证书时，椭圆曲线参数导致无限循环
// 具体是 p = 0 时的模逆运算
// 教训：密码学实现必须处理所有边界情况

// CVE-2023-0286 - OpenSSL类型混淆
// 漏洞原理：EVP_PKEY_set1_RSA()等函数的类型检查不足
// 可能导致使用错误的密钥类型进行操作
```

### 8.1 libsodium使用示例

```cpp
// libsodium - 现代密码学库（比OpenSSL更易用更安全）

// 安装：vcpkg install libsodium 或 apt install libsodium-dev

#include <sodium.h>

class SodiumCrypto {
public:
    static bool init() {
        return sodium_init() >= 0;
    }

    // ✅ 对称加密（XChaCha20-Poly1305）
    static bool encrypt(const std::string& plaintext,
                       const unsigned char key[crypto_aead_xchacha20poly1305_ietf_KEYBYTES],
                       std::string& ciphertext) {
        unsigned char nonce[crypto_aead_xchacha20poly1305_ietf_NPUBBYTES];
        randombytes_buf(nonce, sizeof(nonce));

        ciphertext.resize(sizeof(nonce) + plaintext.size() +
                         crypto_aead_xchacha20poly1305_ietf_ABYTES);

        // 前面放nonce
        std::memcpy(&ciphertext[0], nonce, sizeof(nonce));

        unsigned long long ciphertextLen;
        crypto_aead_xchacha20poly1305_ietf_encrypt(
            reinterpret_cast<unsigned char*>(&ciphertext[sizeof(nonce)]),
            &ciphertextLen,
            reinterpret_cast<const unsigned char*>(plaintext.data()),
            plaintext.size(),
            nullptr, 0,  // 附加数据
            nullptr,     // 额外nonce
            nonce, key
        );

        ciphertext.resize(sizeof(nonce) + ciphertextLen);
        return true;
    }

    // ✅ 密码哈希（Argon2id）
    static std::string hashPassword(const std::string& password) {
        char hash[crypto_pwhash_STRBYTES];
        if (crypto_pwhash_str(hash, password.c_str(), password.size(),
                              crypto_pwhash_OPSLIMIT_MODERATE,
                              crypto_pwhash_MEMLIMIT_MODERATE) != 0) {
            return "";
        }
        return std::string(hash);
    }

    // ✅ 密码验证
    static bool verifyPassword(const std::string& password,
                               const std::string& hash) {
        return crypto_pwhash_str_verify(hash.c_str(),
                                        password.c_str(),
                                        password.size()) == 0;
    }

    // ✅ 数字签名（Ed25519）
    static bool generateKeyPair(unsigned char publicKey[32],
                                unsigned char privateKey[64]) {
        return crypto_sign_keypair(publicKey, privateKey) == 0;
    }

    static bool sign(const std::string& message,
                     const unsigned char privateKey[64],
                     std::string& signature) {
        signature.resize(crypto_sign_BYTES);
        unsigned long long sigLen;
        return crypto_sign_detached(
            reinterpret_cast<unsigned char*>(&signature[0]),
            &sigLen,
            reinterpret_cast<const unsigned char*>(message.data()),
            message.size(),
            privateKey
        ) == 0;
    }

    static bool verify(const std::string& message,
                       const std::string& signature,
                       const unsigned char publicKey[32]) {
        return crypto_sign_verify_detached(
            reinterpret_cast<const unsigned char*>(signature.data()),
            reinterpret_cast<const unsigned char*>(message.data()),
            message.size(),
            publicKey
        ) == 0;
    }
};
```

```cpp
// 安全随机数生成 - 密码学中最容易被忽视的环节

class SecureRandom {
public:
    // ✅ 使用密码学安全随机源
    static bool randomBytes(uint8_t* buffer, size_t length) {
        // 方法1：OpenSSL
        return RAND_bytes(buffer, static_cast<int>(length)) == 1;

        // 方法2：libsodium
        // randombytes_buf(buffer, length);
        // return true;

        // 方法3：C++17 std::random_device（需确认实现）
        // std::random_device rd;
        // std::generate(buffer, buffer + length, [&rd]() {
        //     return static_cast<uint8_t>(rd());
        // });
        // return true;
    }

    // ✅ 安全的随机整数
    static uint32_t randomUint32() {
        uint32_t value;
        randomBytes(reinterpret_cast<uint8_t*>(&value), sizeof(value));
        return value;
    }

    // ✅ 安全的随机数范围（无偏）
    static uint32_t randomRange(uint32_t min, uint32_t max) {
        uint32_t range = max - min + 1;
        uint32_t limit = (UINT32_MAX / range) * range;

        uint32_t value;
        do {
            value = randomUint32();
        } while (value >= limit);  // 拒绝有偏的值

        return min + (value % range);
    }

    // ❌ 不安全的随机数
    static void insecureRandom() {
        // ❌ rand() - 不是密码学安全的
        int r = rand();

        // ❌ std::mt19937 - 不是密码学安全的
        std::mt19937 gen(42);  // 可预测的种子
        int r2 = gen();

        // ❌ time()作为种子 - 可预测
        srand(time(nullptr));

        // ❌ /dev/random（可能阻塞）
        // 使用 /dev/urandom 或 getrandom() 代替
    }
};
```

**关键理解**：

1. **不要自己实现密码算法**：使用经过审计的库（OpenSSL、libsodium）
2. **AES-GCM是现代首选**：同时提供加密和认证
3. **RSA加密用OAEP，签名用PSS**：不要使用PKCS#1 v1.5
4. **TLS 1.3是最佳选择**：更快更安全，移除了不安全特性
5. **常量时间比较至关重要**：时序攻击可以泄露HMAC密钥
6. **理解数学原理才能正确使用**：RSA基于大整数分解，ECC基于离散对数
7. **密码学攻击不断演进**：从POODLE到DROWN，旧协议必须淘汰
8. **随机数是密码学的基石**：使用密码学安全的随机源
