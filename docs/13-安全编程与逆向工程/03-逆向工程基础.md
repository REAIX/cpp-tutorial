> 💡 **通俗理解 - 什么是逆向工程？**

想象一下：

- **静态分析** 就像"看图纸"，不运行程序，直接阅读它的二进制代码
- **动态分析** 就像"开车试驾"，实际运行程序，观察它的行为
- **反汇编** 就像"翻译外文"，把0和1翻译成人类可读的汇编语言
- **恶意软件分析** 就像"解剖病毒"，了解恶意程序如何感染和传播

逆向工程是安全研究的核心技能，也是软件保护和漏洞挖掘的基础。

> 🔬 **抽象理解 - 逆向工程方法**：
>
> - **静态分析**：IDA Pro/Ghidra反汇编、控制流图、交叉引用
> - **动态分析**：GDB/x64dbg调试、断点、内存监控
> - **反汇编基础**：x86/ARM指令集、调用约定、栈帧结构
> - **恶意软件分析**：静态特征提取、动态行为监控、沙箱分析

### 1.4 Ghidra脚本分析

```java
// Ghidra脚本示例（Java）

// @category Analysis

import ghidra.app.script.GhidraScript;
import ghidra.program.model.listing.Function;
import ghidra.program.model.listing.FunctionManager;

public class FindCryptoFunctions extends GhidraScript {
    @Override
    public void run() throws Exception {
        FunctionManager funcMgr = currentProgram.getFunctionManager();

        println("=== 寻找加密相关函数 ===");

        // 遍历所有函数
        for (Function func : funcMgr.getFunctions(true)) {
            String name = func.getName().toLowerCase();

            // 检查函数名是否包含加密相关关键词
            if (name.contains("encrypt") || name.contains("decrypt") ||
                name.contains("aes") || name.contains("rsa") ||
                name.contains("sha") || name.contains("md5") ||
                name.contains("hash") || name.contains("cipher") ||
                name.contains("key") || name.contains("base64")) {

                println(String.format("函数: %s 地址: %s 大小: %d",
                    func.getName(),
                    func.getEntryPoint().toString(),
                    func.getBody().getNumAddresses()));
            }
        }
    }
}
```

```python
# IDAPython脚本示例

import idautils
import idc
import idaapi

# 1. 遍历所有函数
def list_all_functions():
    """列出所有函数"""
    for func_ea in idautils.Functions():
        func_name = idc.get_func_name(func_ea)
        func_size = idc.get_func_attr(func_ea, idc.FUNCATTR_END) - func_ea
        print(f"0x{func_ea:08X} {func_name} (size: {func_size})")

# 2. 查找特定字符串的引用
def find_string_references(target_string):
    """查找字符串引用"""
    for s in idautils.Strings():
        if target_string in str(s):
            str_addr = s.ea
            print(f"字符串 '{target_string}' 在 0x{str_addr:08X}")
            # 查找引用此字符串的代码
            for xref in idautils.XrefsTo(str_addr):
                func_name = idc.get_func_name(xref.frm)
                print(f"  被引用于 0x{xref.frm:08X} (函数: {func_name})")

# 3. 查找所有call指令
def find_all_calls():
    """查找所有call指令"""
    for seg_ea in idautils.Segments():
        seg_name = idc.get_segm_name(seg_ea)
        if seg_name in ['.text', '.plt']:
            for head in idautils.Heads(seg_ea, idc.get_segm_end(seg_ea)):
                if idc.print_insn_mnem(head) == 'call':
                    target = idc.get_operand_value(head, 0)
                    target_name = idc.get_func_name(target) or f"0x{target:08X}"
                    print(f"0x{head:08X}: call {target_name}")

# 4. 自动重命名函数（基于字符串引用）
def auto_rename_functions():
    """基于错误字符串自动重命名函数"""
    error_strings = {
        "malloc failed": "allocateMemory",
        "invalid parameter": "validateParams",
        "connection refused": "connectToServer",
        "file not found": "openFile",
        "permission denied": "checkPermission",
    }

    for s in idautils.Strings():
        str_val = str(s)
        for key, new_name in error_strings.items():
            if key in str_val.lower():
                # 找到引用此字符串的函数
                for xref in idautils.XrefsTo(s.ea):
                    func_ea = idc.get_func_attr(xref.frm, idc.FUNCATTR_START)
                    if func_ea != idc.BADADDR:
                        old_name = idc.get_func_name(func_ea)
                        idc.set_name(func_ea, new_name, idc.SN_NOWARN)
                        print(f"重命名: {old_name} -> {new_name}")

# 5. 提取所有加密常量
def find_crypto_constants():
    """查找加密算法特征常量"""
    # AES S-box特征值
    aes_sbox = [0x63, 0x7C, 0x77, 0x7B]
    # MD5初始化值
    md5_init = [0x67452301, 0xEFCDAB89, 0x98BADCFE, 0x10325476]
    # SHA-256初始化值
    sha256_init = [0x6A09E667, 0xBB67AE85, 0x3C6EF372, 0xA54FF53A]

    print("=== 搜索加密常量 ===")

    # 搜索MD5初始化值
    for seg_ea in idautils.Segments():
        for head in idautils.Heads(seg_ea, idc.get_segm_end(seg_ea)):
            value = idc.get_wide_dword(head)
            if value == md5_init[0]:
                # 检查后续值
                if (idc.get_wide_dword(head + 4) == md5_init[1] and
                    idc.get_wide_dword(head + 8) == md5_init[2] and
                    idc.get_wide_dword(head + 12) == md5_init[3]):
                    print(f"MD5初始化值在 0x{head:08X}")
                    for xref in idautils.XrefsTo(head):
                        print(f"  引用于 0x{xref.frm:08X}")
```

### 2.2 GDB Python脚本

```python
# GDB Python脚本

# 保存为gdbscript.py，使用 source gdbscript.py 加载

import gdb

class DumpStackCommand(gdb.Command):
    """转储栈内容：dump_stack <count>"""

    def __init__(self):
        super().__init__("dump_stack", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        count = int(arg) if arg else 20
        rsp = int(gdb.parse_and_eval("$rsp"))
        print(f"栈指针 RSP: {hex(rsp)}")

        for i in range(count):
            addr = rsp + i * 8
            try:
                value = gdb.parse_and_eval(f"*(long*){addr}")
                print(f"  {hex(addr)}: {hex(int(value))}")
            except:
                print(f"  {hex(addr)}: <无法读取>")

DumpStackCommand()

class FindStringCommand(gdb.Command):
    """在内存中搜索字符串：find_string <string>"""

    def __init__(self):
        super().__init__("find_string", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        if not arg:
            print("用法: find_string <string>")
            return

        # 在所有可读段中搜索字符串
        result = gdb.execute(f'find /b 0x400000, 0x7fffffffffff, "{arg}"',
                             to_string=True)
        print(result)

FindStringCommand()
```

### 3.1 x86-64指令集基础

```nasm
; x86-64常用指令

; 数据传送
mov rax, 42          ; 立即数 → 寄存器

; 逻辑运算
and rax, 0xFF        ; 按位与
or rax, 0x80         ; 按位或
```

```nasm
.L_end:

.L_loop_start:
    mov eax, dword ptr [rbp-4]
    cmp eax, dword ptr [rbp-8]    ; i < n
    jge .L_loop_end
    ; 循环体
    add dword ptr [rbp-4], 1      ; i++
    jmp .L_loop_start
.L_loop_end:

; 3. 数组访问
; C代码: arr[i] = value;
    mov eax, dword ptr [rbp-4]    ; i
    cdqe
    lea rdx, [rax*4]              ; i * sizeof(int)

    add rax, rdx
    mov edx, dword ptr [rbp-8]    ; value
    mov dword ptr [rax], edx      ; arr[i] = value

; 4. 函数调用

    mov edx, 3           ; 第3个参数
    mov esi, 2           ; 第2个参数
    mov edi, 1           ; 第1个参数
    call func
    mov dword ptr [rbp-4], eax  ; 保存返回值

; 5. 字符串比较
; C代码: strcmp(s1, s2)
    lea rsi, [rbp-0x20]  ; s2
    lea rdi, [rbp-0x10]  ; s1
    call strcmp
    test eax, eax
    jne .L_not_equal

; 6. switch-case（跳转表）
    mov eax, dword ptr [rbp-4]
    sub eax, 1                   ; case从1开始
    cmp eax, 3                   ; 4个case
    ja .L_default
```

### 5.2 XOR解密脚本

```python
# CTF常用XOR解密脚本

def xor_decrypt(data, key):
    """单字节XOR解密"""
    return bytes([b ^ key for b in data])

def xor_decrypt_multi(data, key):
    """多字节密钥XOR解密"""
    result = bytearray()
    key_len = len(key)
    for i, b in enumerate(data):
        result.append(b ^ key[i % key_len])
    return bytes(result)

def find_xor_key(ciphertext, known_plaintext):
    """已知明文攻击：找出XOR密钥"""
    key = bytearray()
    for i in range(len(known_plaintext)):
        key.append(ciphertext[i] ^ known_plaintext[i])
    return bytes(key)

# 示例：暴力破解单字节XOR
def brute_force_xor(ciphertext):
    """尝试所有256种可能的单字节密钥"""
    results = []
    for key in range(256):
        decrypted = xor_decrypt(ciphertext, key)
        # 检查是否为可打印ASCII
        if all(32 <= b < 127 or b in (10, 13, 9) for b in decrypted):
            results.append((key, decrypted))
    return results

# 示例：频率分析破解XOR
def frequency_analysis_xor(ciphertext):
    """利用英文字母频率分析破解XOR"""
    # 空格是最常见的字符（ASCII 0x20）
    # 找到密文中最常见的字节
    from collections import Counter
    freq = Counter(ciphertext)
    most_common_byte = freq.most_common(1)[0][0]

    # 假设最常见的字节是空格
    key = most_common_byte ^ 0x20
    return xor_decrypt(ciphertext, key)
```

```cpp
class CodeObfuscation {
public:
    // 控制流平坦化（Control Flow Flattening）
    // 将正常的if-else/switch结构打平成状态机
    static int obfuscatedLogic(int input) {
        int state = 0;
        int result = 0;

        while (true) {
            switch (state) {
                case 0:
                    if (input > 10) state = 1;
                    else state = 2;
                    break;
                case 1:
                    result = input * 2;
                    state = 3;
                    break;
                case 2:
                    result = input + 5;
                    state = 3;
                    break;
                case 3:
                    return result;
            }
        }
    }

    // 不透明谓词（Opaque Predicate）
    // 插入永远为真/假的条件，迷惑分析者
    static int withOpaquePredicate(int x) {
        // 不透明谓词：(x * x - x) % 2 == 0 永远为真
        // 但分析者需要花时间验证
        if ((x * x - x) % 2 == 0) {
            return x + 1;  // 真正的逻辑
        } else {
            return deadCode(x);  // 永远不会执行
        }
    }

    // 字符串加密
    static std::string decryptString(const uint8_t* encrypted,
                                      size_t len, uint8_t key) {
        std::string result(len, '\0');
        for (size_t i = 0; i < len; i++) {
            result[i] = encrypted[i] ^ key;
        }
        return result;
    }

private:

};

        // 如果不匹配，说明代码被修改（或被设断点）
        return true;
    }

    // 运行时自修改代码
    static void selfModifyingCode() {
        // 在运行时解密关键代码
        // 执行完后重新加密
        // 使静态分析更困难
    }
};
```

## 6. 逆向分析方法论

### 6.1 系统化逆向分析流程

```
逆向分析方法论 - 系统化分析流程：

阶段1：信息收集（30分钟）
├── 文件类型识别（file命令）
├── 安全特性检查（checksec）
├── 字符串提取（strings）
├── 依赖库分析（ldd）
├── 符号表检查（nm/readelf -s）
└── 版本信息提取

阶段2：高层理解（1-2小时）
├── 识别程序入口点
├── 查找main函数
├── 分析程序整体结构
├── 识别关键数据结构
├── 绘制函数调用图
└── 标注已知库函数

阶段3：深入分析（2-4小时）
├── 反编译关键函数
├── 理解核心算法
├── 识别加密/解密逻辑
├── 追踪数据流
├── 分析输入/输出格式
└── 记录发现

阶段4：验证与利用（1-2小时）
├── 动态调试验证假设
├── 编写PoC
├── 测试边界情况
└── 编写分析报告

关键原则：
1. 从外到内：先理解整体，再深入细节
2. 自顶向下：从main函数开始，逐层深入
3. 关注数据流：追踪数据如何被处理
4. 善用交叉引用：快速定位关键代码
5. 做好笔记：记录分析过程中的发现
```
